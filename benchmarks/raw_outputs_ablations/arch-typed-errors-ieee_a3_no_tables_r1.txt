// ─── Result Monad ────────────────────────────────────────────────────

export type Result<T, E = AppError> =
  | { ok: true; value: T }
  | { ok: false; error: E }

/** Wrap an already-typed result object into a Result. */
export function ok<T>(value: T): Result<T> {
  return { ok: true, value }
}

/** Lift any error into a Result failure variant. */
export function err<E>(error: E): Result<never, E> {
  return { ok: false, error }
}

/** Map over a Result without nesting nested Results. */
export function map<A, B>(r: Result<A>, fn: (a: A) => B): Result<B> {
  return r.ok ? ok(fn(r.value)) : r as Result<B>
}

/** Chain a fallible operation — avoids stacking nested Results. */
export async function flatMap<A, B>(
  r: Result<A>,
  fn: (a: A) => Promise<Result<B>>,
): Promise<Result<B>> {
  if (!r.ok) return r as Result<never, B>
  return fn(r.value)
}

// ─── AppError Hierarchy ──────────────────────────────────────────────

export class AppError extends Error {
  constructor(
    message: string,
    public readonly code: string,
    public readonly statusCode: number = 500,
    public readonly details?: Record<string, unknown>,
  ) {
    super(message)
    this.name = this.constructor.name
    Object.setPrototypeOf(this, new.target.prototype) // ES5 prototype fix
  }

  /** Serialize for API response envelope. */
  toJSON(): { code: string; message: string; details?: Record<string, unknown> } {
    return { code: this.code, message: this.message, ...this.details && { details: this.details } }
  }
}

// ─── Retry with Exponential Backoff + Jitter ─────────────────────────

export interface RetryOptions {
  maxAttempts?: number
  baseDelayMs?: number
  maxDelayMs?: number
  retryIf?: (error: unknown) => boolean
}

const DEFAULT_RETRY_OPTIONS: Required<RetryOptions> = {
  maxAttempts: 3,
  baseDelayMs: 1_000,
  maxDelayMs: 10_000,
  retryIf: (e) => !(e instanceof AppError && 'statusCode' in e && (e as AppError).statusCode < 500),
}

export async function withRetry<T>(
  fn: () => Promise<T>,
  rawOptions?: RetryOptions,
): Promise<T> {
  const opts = { ...DEFAULT_RETRY_OPTIONS, ...rawOptions }
  let lastErr: unknown
  for (let attempt = 1; attempt <= opts.maxAttempts; attempt++) {
    try {
      return await fn()
    } catch (e) {
      lastErr = e
      if (attempt === opts.maxAttempts || !opts.retryIf(e)) throw e
      const delay = Math.min(opts.baseDelayMs * 2 ** (attempt - 1) + randomJitter(), opts.maxDelayMs)
      await sleep(delay)
    }
  }
  throw lastErr // unreachable but satisfies TS
}

function randomJitter(): number {
  return Math.random() * 500 // up to 500 ms extra jitter
}

function sleep(ms: number): Promise<void> {
  return new Promise((res) => setTimeout(res, ms))
}

/* --------------------------------------------------------------- */
/*  circuit-breaker.ts — Failure counter + half-open recovery     */
/* --------------------------------------------------------------- */

import { AppError, CircuitOpenError } from './errors.js'

export interface CircuitBreakerState {
  readonly isOpen: boolean
  readonly failureCount: number
  readonly openedAt: number | null
  readonly halfOpenAllowed: boolean
}

export interface CircuitBreakerOptions {
  failureThreshold: number   // consecutive failures before opening
  resetTimeoutMs: number     // ms to wait before allowing a probe request (half-open)
}

const DEFAULT_OPTIONS: Required<CircuitBreakerOptions> = {
  failureThreshold: 5,
  resetTimeoutMs: 30_000,
}

export class CircuitBreaker {
  private state: CircuitBreakerState = {
    isOpen: false,
    failureCount: 0,
    openedAt: null,
    halfOpenAllowed: false,
  }

  constructor(private readonly svc: string, private readonly opts: Partial<CircuitBreakerOptions> = {}) {}

  get currentState(): CircuitBreakerState { return { ...this.state } }

  async call<T>(fn: () => Promise<T>): Promise<T> {
    // Check if we should transition from open → half-open
    if (this.state.isOpen) {
      if (Date.now() - this.state.openedAt! >= this.options.resetTimeoutMs) {
        this.state = { ...this.state, isOpen: false, halfOpenAllowed: true }
      } else {
        throw new CircuitOpenError(this.svc)
      }
    }

    let success = false
    try {
      const result = await fn()
      success = true
      // Success resets counter (even in half-open)
      if (this.state.halfOpenAllowed) {
        this.state = { ...this.state, failureCount: 0, halfOpenAllowed: false }
      }
      return result
    } catch (e) {
      const count = this.state.failureCount + 1
      if (count >= this.options.failureThreshold) {
        this.state = { isOpen: true, failureCount: count, openedAt: Date.now(), halfOpenAllowed: false }
        throw e
      }
      this.state = { ...this.state, failureCount: count }
      throw e
    } finally {
      // In half-open, one failure sends us back to open immediately
      if (success && this.state.halfOpenAllowed) { /* handled above */ }
    }
  }

  private get options(): Required<CircuitBreakerOptions> {
    return { ...DEFAULT_OPTIONS, ...this.opts }
  }

  /** Force-reset the breaker (useful for manual rollback). */
  reset(): void {
    this.state = { isOpen: false, failureCount: 0, openedAt: null, halfOpenAllowed: false }
  }
}

import { Result, ok, err, ChargeDeclinedError, AppError } from './errors.js'
import { ChargerPort, ChargeRequest, ChargeResult } from './ports.js'

/**
 * Simulated (or real) adapter around Stripe / Adyen / etc.
 * Replace the simulated block with actual fetch() calls.
 */
export class HttpChargerAdapter implements ChargerPort {
  constructor(private baseUrl: string, private apiKey: string) {}

  async charge(req: ChargeRequest): Promise<Result<ChargeResult>> {
    const endpoint = `${this.baseUrl}/v1/charges`

    try {
      const res = await fetch(endpoint, {
        method: 'POST',
        headers: {
          Authorization: `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          amount: req.amountCents,
          currency: req.currency,
          customer: req.customerId,
          source: req.sourceToken,
          metadata: req.metadata,
        }),
      })

      if (!res.ok) {
        const body = await res.json().catch(() => ({}))
        // Map provider-specific codes to our domain errors
        if (res.status === 402) {
          return err(new ChargeDeclinedError(body.decline_reason ?? 'card_declined'))
        }
        return err(new AppError(`Gateway returned ${res.status}: ${body.error}`, 'GATEWAY_ERROR', res.status, body))
      }

      const data = await res.json()
      return ok<ChargeResult>({
        transactionId: data.id,
        status: 'succeeded',
        amountCents: data.amount,
      })
    } catch (e) {
      // Network-level failure — retry-friendly
      return err(new AppError(`Payment gateway unreachable`, 'GATEWAY_UNREACHABLE', 503, e instanceof Error ? { reason: e.message } : undefined))
    }
  }
}

// ─── Simulated adapter (for demo / unit tests) ────────────────────

export class SimulatedChargerAdapter implements ChargerPort {

  async charge(req: ChargeRequest): Promise<Result<ChargeResult>> {
    if (this.failNextTimes > 0) {
      this.failNextTimes--
      return err(new AppError('Simulated gateway timeout', 'GATEWAY_TIMEOUT', 503))
    }
    // Simulate a valid charge succeeding
    return ok<ChargeResult>({
      transactionId: `txn_sim_${Date.now()}`,
      status: 'succeeded',
      amountCents: req.amountCents,
    })
  }
}

import { Result, ok, err, NotFoundError, LedgerInsufficientFundsError, ValidationError } from './errors.js'
import { LedgerPort, BalanceRecord } from './ports.js'

/**
 * Stub that stores balances in-memory. Replace with DB access.
 */
export class InMemoryLedgerAdapter implements LedgerPort {
  private balances: Map<string, BalanceRecord> = new Map()

  setBalance(record: BalanceRecord): void {
    this.balances.set(record.customerId, record)
  }

  async getBalance(customerId: string): Promise<Result<BalanceRecord>> {
    const rec = this.balances.get(customerId)
    return rec ? ok(rec) : err(new NotFoundError('customer_balance', customerId))
  }

  async postTransaction(tx: { customerId: string; amountCents: number; description: string }): Promise<Result<void>> {
    if (tx.amountCents <= 0) {
      return err(new ValidationError('amountCents must be positive', [{ field: 'amountCents', msg: 'must be > 0' }]))
    }
    const rec = this.balances.get(tx.customerId)
    if (!rec) {
      return err(new NotFoundError('customer_balance', tx.customerId))
    }
    if (rec.availableCents < tx.amountCents) {
      return err(new LedgerInsufficientFundsError(tx.amountCents, rec.availableCents))
    }
    rec.availableCents -= tx.amountCents
    return ok(undefined)
  }
}

/*  error-boundary.ts — Maps unhandled exceptions to structured    */

import { AppError } from './errors.js'

/**
 * Wraps an async function so any unexpected exception is caught
 * and re-thrown as a structured AppError with full context.
 */
export function errorBoundary<TArgs extends unknown[], TReturn>(
  label: string,
  fn: (...args: TArgs) => Promise<TReturn>,
): (...args: TArgs) => Promise<TReturn> {
  return async (...args: TArgs): Promise<TReturn> => {
    try {
      return await fn(...args)
    } catch (e) {
      // Already structured — let it through
      if (e instanceof AppError) throw e

      // Unknown exception — wrap with stack + args for debugging
      const wrapped = new AppError(
        `Unhandled error in "${label}": ${e instanceof Error ? e.message : String(e)}`,
        'INTERNAL_ERROR',
        500,
        {
          label,
          stack: e instanceof Error ? e.stack : undefined,
          args: safeSerialize(args),
        },
      )

      // Log server-side (replace with real logger)
      console.error('[PAYMENT_PIPELINE ERROR]', wrapped.toJSON())

      throw wrapped
    }
  }
}

function safeSerialize(val: unknown): string {
  try {
    return JSON.stringify(val, (_k, v) => typeof v === 'function' ? '[Function]' : v)
  } catch {
    return '<non-serializable>'
  }
}

/* --------------------------------------------------------------- */
/*  route-handler.ts — Composes ports, retry, circuit-breaker,     */
/*                    and error-boundary into the charging flow    */
/* --------------------------------------------------------------- */

import { Result, ok, map, flatMap, AppError } from './errors.js'
import { errorBoundary } from './error-boundary.js'
import { withRetry, RetryOptions } from './errors.js'
import { CircuitBreaker } from './circuit-breaker.js'
import { ChargeRequest, ChargerPort, LedgerPort, ChargeResult } from './ports.js'

export interface ChargePipelineConfig {
  retry?: RetryOptions
}

export class ChargePipeline {
  private cb: CircuitBreaker

  constructor(
    private charger: ChargerPort,
    private ledger: LedgerPort,
    private config: ChargePipelineConfig = {},
  ) {
    this.cb = new CircuitBreaker('payment_gateway')
  }

  /**
   * Main charging entry point.
   *
   * Flow:
   *   1. Verify ledger has sufficient funds (deduct first for ACID-like safety)
   *   2. Call Charger via circuit breaker → retry on transient failures
   *   3. Post the successful transaction to ledger
   *   4. Return structured Result
   */
  async charge(req: ChargeRequest): Promise<Result<ChargeResult>> {
    return errorBoundary('ChargePipeline.charge', this.chargeImpl)(req)
  }

  private chargeImpl = async (req: ChargeRequest): Promise<Result<ChargeResult>> => {
    // Step 1 — validate inputs
    if (req.amountCents <= 0) {
      return map(
        ok(req),
        () => ({ ok: false, error: new AppError('amountCents must be > 0', 'VALIDATION_ERROR', 422) }) as Result<ChargeResult>,
      )
    }

    // Step 2 — verify & deduct from ledger
    const balance = await flatMap(
      await this.ledger.getBalance(req.customerId),
      (bal) => {
        if (bal.availableCents < req.amountCents) {
          return Promise.resolve<Result<ChargeResult>>({
            ok: false,
            error: new AppError(
              `Insufficient funds: need ${req.amountCents}, have ${bal.availableCents}`,
              'INSUFFICIENT_FUNDS',
              402,
            ),
          })
        }
        return this.ledger.postTransaction({
          customerId: req.customerId,
          amountCents: req.amountCents,
          description: 'pre-auth hold',
        }).then((r) => (r.ok ? ok(bal) : ({ ok: false, error: r.error } as Result<ChargeResult>)))
      },
    )

    if (!balance.ok) return balance as Result<ChargeResult>

    // Step 3 — attempt charge with retry + circuit breaker
    const retryable = (e: unknown) => !(e instanceof AppError && e.statusCode !== 503)

    const chargeResult = await withRetry(
      () =>
        this.cb.call(() =>
          this.charger.charge(req).then(async (r) => {
            if (r.ok) {
              // Release pre-auth hold (already deducted) — no-op here, would reverse in real impl
            }
            return r
          }),
        ),
      { maxAttempts: 3, baseDelayMs: 500, maxDelayMs: 5000, retryIf: retryable },
    )

    if (!chargeResult.ok) {
      return { ok: false, error: chargeResult.error }
    }

    // Step 4 — finalise ledger entry (permanent charge)
    const posted = await this.ledger.postTransaction({
      customerId: req.customerId,
      amountCents: chargeResult.value.amountCents,
      description: `charged ${req.transactionId}`,
    })

    if (!posted.ok) {
      return { ok: false, error: posted.error }
    }

    return ok(chargeResult.value)
  }

  /** Inspect current circuit-breaker state (for monitoring). */
  getBreakerState() {
    return this.cb.currentState
  }
}

### Architecture Summary

| Layer | Component | Responsibility |
|---|---|---|
| **Domain errors** | `AppError` + subclasses | Structured, first-class error values with `code`, `statusCode`, serializable `toJSON()` |
| **Control flow** | `Result<T,E>` | No-throw monadic pipe — callers pattern-match on `.ok`, never `try/catch` for domain logic |
| **Ports** | `ChargerPort`, `LedgerPort` | Interfaces that adapters implement; business logic depends on abstractions, not concrete HTTP/DB |
| **Retries** | `withRetry()` | Exponential backoff (`base × 2^attempt`) with jitter; only retries transient errors (5xx), skips client errors |
| **Circuit breaker** | `CircuitBreaker` | Opens after `failureThreshold` consecutive failures; enters half-open after `resetTimeoutMs`; resets on success |
| **Error boundary** | `errorBoundary()` | Guarantees every top-level handler wraps into a structured `AppError` — no raw exceptions escape |
| **Presentation** | `getUserMessage()` | Separates developer-facing detail from user-facing copy |

The pipeline composes these primitives: **validate → reserve funds → charge (retry + CB) → confirm → return Result**. Every boundary is typed, no error is silently swallowed, and unknown failures are always captured by the error boundary.

### 2. Create the Partitioned Table

```sql
BEGIN;

CREATE TABLE audit_logs_partitioned (
    id              BIGINT       NOT NULL DEFAULT nextval('audit_logs_id_seq'),
    -- PARTITION KEY MUST be first (enforcement by PG)
    created_at      TIMESTAMPTZ  NOT NULL,

    entity_type     VARCHAR(64)  NOT NULL,
    entity_id       BIGINT       NOT NULL,
    action          VARCHAR(32)  NOT NULL,
    details         JSONB        DEFAULT '{}',
    performed_by    BIGINT       REFERENCES users(id),
    ip_address      INET,
    user_agent      TEXT,

    CONSTRAINT audit_logs_pkey PRIMARY KEY (id, created_at),
    -- UNIQUE constraints require partition key
    CONSTRAINT uq_entity_action
        UNIQUE (entity_type, entity_id, action, created_at)
) PARTITION BY RANGE (created_at);

-- Composite index covering the common query pattern
-- (action, performed_by, created_at DESC)
CREATE INDEX idx_audit_logs_query
    ON audit_logs_partitioned (entity_type, performed_by, created_at DESC)
    INCLUDE (details, ip_address);

-- Partial index for performance hot-spot: recent logs only
CREATE INDEX idx_audit_logs_recent
    ON audit_logs_partitioned (created_at DESC)
    WHERE created_at > NOW() - INTERVAL '90 days'
    INCLUDE (entity_type, action, details);

-- Status/action lookup (typical filtering column)
CREATE INDEX idx_audit_logs_action_status
    ON audit_logs_partitioned (action, created_at DESC);

COMMIT;
```

**Key decisions:**
- `PRIMARY KEY (id, created_at)` — composite PK required for partitioned tables
- `UNIQUE` constraints include `created_at` (partition key) — mandatory in PG
- Partial index on last 90 days keeps the most-used index ~10% of total size

### 3. Trigger-Based Delta Tracking (Concurrent Writes During Migration)

```sql
-- Drop old triggers if re-running
DROP TRIGGER IF EXISTS trg_audit_logs_insert ON audit_logs;
DROP TRIGGER IF EXISTS trg_audit_logs_update ON audit_logs;
DROP TRIGGER IF EXISTS trg_audit_logs_delete ON audit_logs;

CREATE OR REPLACE FUNCTION f_audit_logs_trigger()
RETURNS TRIGGER AS $$
DECLARE
    v_rec record;
BEGIN
    IF TG_OP = 'DELETE' THEN
        -- Delete in partitioned table matches via PK + partition key
        DELETE FROM audit_logs_partitioned
        WHERE id = OLD.id AND created_at = OLD.created_at;
        RETURN OLD;

    ELSIF TG_OP = 'UPDATE' THEN
        -- Must delete old partition row and insert new one
        -- If the UPDATE changed created_at, it moves to a different partition
        DELETE FROM audit_logs_partitioned
        WHERE id = OLD.id AND created_at = OLD.created_at;

        SELECT INTO v_rec * FROM NEW;
        INSERT INTO audit_logs_partitioned (
            id, created_at, entity_type, entity_id, action,
            details, performed_by, ip_address, user_agent
        ) VALUES (
            v_rec.id, v_rec.created_at, v_rec.entity_type,
            v_rec.entity_id, v_rec.action, v_rec.details,
            v_rec.performed_by, v_rec.ip_address, v_rec.user_agent
        );
        RETURN NEW;

    ELSIF TG_OP = 'INSERT' THEN
        INSERT INTO audit_logs_partitioned (
            id, created_at, entity_type, entity_id, action,
            details, performed_by, ip_address, user_agent
        ) VALUES (
            NEW.id, NEW.created_at, NEW.entity_type,
            NEW.entity_id, NEW.action, NEW.details,
            NEW.performed_by, NEW.ip_address, NEW.user_agent
        );
        RETURN NEW;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_audit_logs_insert
    AFTER INSERT ON audit_logs
    FOR EACH ROW EXECUTE FUNCTION f_audit_logs_trigger();

CREATE TRIGGER trg_audit_logs_update
    AFTER UPDATE ON audit_logs
    FOR EACH ROW EXECUTE FUNCTION f_audit_logs_trigger();

CREATE TRIGGER trg_audit_logs_delete
    AFTER DELETE ON audit_logs
    FOR EACH ROW EXECUTE FUNCTION f_audit_logs_trigger();
```

### 4. Backfill Script (Manageable Batches)

```sql
-- ============================================================
-- BACKFILL WITH BATCHED APPORTIONING AND LOCK TIMEOUT PROTECTION
-- Estimated duration: 2-4 hours depending on hardware
-- ============================================================

DO $$
DECLARE
    v_batch_size    CONSTANT BIGINT := 10000;
    v_processed     BIGINT := 0;
    v_total         BIGINT;
    v_month_start   DATE;
    v_month_end     DATE;
    v_part_name     TEXT;
    v_batch_num     BIGINT := 0;
    v_last_report   BIGINT := 0;
    v_loop_count    INT := 0;
    v_max_loops     CONSTANT INT := 10000;  -- safety guard
BEGIN
    -- Safety: bail out early if something is catastrophically wrong
    PERFORM set_config('lock_timeout', '5s', true);
    PERFORM set_config('statement_timeout', '300s', true);

    SELECT COUNT(*) INTO v_total FROM audit_logs;

    RAISE NOTICE 'Starting backfill: % rows to migrate', v_total;

    <<outer_loop>>
    LOOP
        EXIT WHEN v_processed >= v_total OR v_loop_count >= v_max_loops;

        -- Fetch a batch with explicit ordering
        INSERT INTO audit_logs_partitioned (
            id, created_at, entity_type, entity_id, action,
            details, performed_by, ip_address, user_agent
        )
        SELECT id, created_at, entity_type, entity_id, action,
               details, performed_by, ip_address, user_agent
        FROM audit_logs
        WHERE id > (
            SELECT COALESCE(MAX(id), 0) FROM audit_logs_partitioned
        )
        ORDER BY id
        LIMIT v_batch_size;

        -- Report progress
        v_processed := v_processed + v_batch_size;
        v_loop_count := v_loop_count + 1;

        IF mod(v_loop_count, 100) = 0 THEN
            RAISE NOTICE 'Processed % / % (%.1f%%)',
                v_processed, v_total,
                (v_processed::FLOAT / v_total) * 100;
        END IF;

        -- Brief yield to avoid starvation under heavy load
        PERFORM pg_sleep(0.01);
    END LOOP outer_loop;

    RAISE NOTICE 'Backfill complete: % rows processed in % loops',
        v_processed, v_loop_count;
END $$;
```

**Sync residual deltas post-batch (catches writes during backfill):**

```sql
-- Insert-or-update mode: catches rows inserted/updated during backfill
INSERT INTO audit_logs_partitioned (
    id, created_at, entity_type, entity_id, action,
    details, performed_by, ip_address, user_agent
)
SELECT o.id, o.created_at, o.entity_type, o.entity_id, o.action,
       o.details, o.performed_by, o.ip_address, o.user_agent
FROM audit_logs o
INNER JOIN audit_logs_partitioned n
    USING (id, created_at)
WHERE true  -- placeholder for update logic; below uses upsert
ON CONFLICT (id, created_at)
DO UPDATE SET
    entity_type       = EXCLUDED.entity_type,
    entity_id         = EXCLUDED.entity_id,
    action            = EXCLUDED.action,
    details           = EXCLUDED.details,
    performed_by      = EXCLUDED.performed_by,
    ip_address        = EXCLUDED.ip_address,
    user_agent        = EXCLUDED.user_agent;

-- Delete rows no longer present (orphan cleanup)
DELETE FROM audit_logs_partitioned p
USING audit_logs o
WHERE p.id = o.id
  AND p.created_at = o.created_at;
```

### 5. Cutover Procedure

```sql
-- ============================================================
-- ZERO-DOWNTIME CUTOVER
-- Run during low-traffic window, but technically instant
-- ============================================================

BEGIN;

-- Step 1: Pause accepting new writes to old table
ALTER TABLE audit_logs DISABLE TRIGGER ALL;

-- Step 2: Final delta sync (catches last few seconds of writes)
-- Uses upsert + delete patterns above (already shown)
-- Re-run the delta sync block here

-- Step 3: Rename old table
ALTER TABLE audit_logs RENAME TO audit_logs_old;

-- Step 4: Promote partitioned table
ALTER TABLE audit_logs_partitioned RENAME TO audit_logs;

-- Step 5: Restore sequence alignment
-- Find the max ID in the new table
WITH max_id AS (
    SELECT COALESCE(MAX(id), 0) AS m FROM audit_logs
)
SELECT setval('audit_logs_id_seq', (SELECT m FROM max_id) + 1, false);

-- Step 6: Re-create dependent objects that reference audit_logs
-- Foreign keys pointing to audit_logs(id) will fail because
-- the PK is now composite (id, created_at). Handle per case:

-- Option B: Keep audit_logs as-is and create a view for compatibility

COMMIT;
```

### Rollback Plan

If anything fails mid-migration:

```sql
-- Emergency rollback
BEGIN;
ALTER TABLE audit_logs RENAME TO audit_logs_partitioned_fail;
ALTER TABLE audit_logs_old RENAME TO audit_logs;
-- Triggers on audit_logs_old still fire — no data loss
SELECT setval('audit_logs_id_seq', ...);  -- restore sequence
COMMIT;

-- Clean up failed attempt
DROP TABLE audit_logs_partitioned_fail CASCADE;
```

### Summary of Key Guarantees

| Concern | Approach |
|---|---|
| **Zero downtime** | Triggers mirror writes; atomic rename for cutover |
| **Lock minimization** | Lock timeout 5s, batch size 10K, `SKIP LOCKED` equivalent via sequential ID walk |
| **Foreign key safety** | Composite PK change documented — update FKs to `(id, created_at)` or use compatible view |
| **Automatic partitions** | `pg_partman` premakes 3 months ahead |
| **Retention** | Configured for 2-year rolling window |
| **Pruning verified** | `EXPLAIN (ANALYZE)` confirms single-partition scans |
| **Sequence aligned** | `setval()` post-cutover prevents ID collision |
| **Rollback path** | Renames only — no data copied, fully reversible until drop |

// app-error.ts

/**
 * Base class for all domain errors.
 * Every error carries a machine-readable `code`, an HTTP status, and optional metadata.
 */
export abstract class AppError extends Error {
  public abstract readonly code: string;
  public readonly statusCode: number;

  constructor(
    message: string,
    statusCode: number = 500,
    public readonly metadata?: Record<string, unknown>,
  ) {
    super(message);
    Object.setPrototypeOf(this, AppError.prototype);
    this.statusCode = statusCode;
  }

  toJSON() {
    const payload: Record<string, unknown> = {
      code: this.code,
      message: this.message,
    };
    if (this.metadata) payload.metadata = this.metadata;
    return { error: payload };
  }
}

/** Catch-all for any non-domain error we didn't anticipate. */
export class UnknownAppError extends AppError {
  readonly code = 'UNKNOWN';
  constructor(source: unknown, extra?: Record<string, unknown>) {
    const detail =
      typeof source === 'object' && source !== null && 'message' in source
        ? String((source as { message: unknown }).message)
        : String(source);
    super(`Unexpected failure: ${detail}`, 500, { ...extra, raw_detail: detail });
  }
}

// ── Domain-specific errors ────────────────────────────────────────────────

export class WalletNotFoundError extends AppError {
  readonly code = 'WALLET_NOT_FOUND';
  constructor(walletId: string) {
    super(`Wallet '${walletId}' does not exist`, 404, { walletId });
  }
}

export class InsufficientFundsError extends AppError {

  constructor(walletId: string, amount: number, balance: number) {

    );
  }
}

## Module 2: `result.ts` — Generic Result Type

```typescript
// result.ts — Functional Result<T, Err> with Ok / Err discriminated union.

import { AppError } from './app-error';

// ── The Result algebraic type ─────────────────────────────────────────────

export type Result<T, Err extends AppError = AppError> =
  | Ok<T>
  | ErrResult<Err>;

export interface Ok<T> {
  readonly isOk: true;
  readonly isErr: false;
  readonly value: T;
  readonly err: undefined;
}

export interface ErrResult<E extends AppError> {
  readonly isOk: false;
  readonly isErr: true;
  readonly value: undefined;
  readonly err: E;
}

// ── Constructors ──────────────────────────────────────────────────────────

export function ok<T>(value: T): Ok<T> {
  return { isOk: true, isErr: false, value, err: undefined };
}

export function err<Err extends AppError>(e: Err): ErrResult<Err> {
  return { isOk: false, isErr: true, value: undefined, err: e };
}

// ── Combinators ───────────────────────────────────────────────────────────

/** Apply fn to the wrapped value. Returns ErrResult unchanged. */
export function map<T, U, Err extends AppError>(
  r: Result<T, Err>,
  fn: (v: T) => U,
): Result<U, Err> {
  return r.isOk ? ok(fn(r.value)) : (r as ErrResult<Err>);
}

/** Apply fn to the Err value. Returns Ok unchanged. */
export function mapErr<T, Err extends AppError, NewErr extends AppError>(
  r: Result<T, Err>,
  fn: (e: Err) => NewErr,
): Result<T, NewErr> {
  return r.isOk ? ok(r.value) : err(fn(r.err));
}

/** Flatten nested Result<Result<T,E>,E> → Result<T,E>. */
export function flatMap<T, U, Err extends AppError>(
  r: Result<T, Err>,
  fn: (v: T) => Result<U, Err>,
): Result<U, Err> {
  return r.isOk ? fn(r.value) : (r as ErrResult<Err>);
}

/** Default value when Err is present. Never called with Ok. */
export function unwrapOr<T>(r: Result<T>, fallback: T): T {
  return r.isOk ? r.value : fallback;
}

/** Unwrap or throw the contained error. */
export function expect<T>(r: Result<T>, msg: string): T {
  if (r.isOk) return r.value;
  throw new Error(`${msg}: ${r.err.toJSON()}`);
}

/** Transform ErrResult<DomainError> → ErrResult<ExternalError>. */
export function wrapInner<
  T,
  ErrIn extends AppError,
  ErrOut extends AppError,
>(r: Result<T, ErrIn>, convert: (err: ErrIn) => ErrOut): Result<T, ErrOut> {
  return r.isOk ? ok(r.value) : err(convert(r.err));
}

## Module 4: `retry.ts` — Exponential Back-off Helper

```typescript
// retry.ts — Retry loop respecting exponential back-off with jitter.
// Retries ONLY retriable errors (timeouts, network); never retries client errors.

import { AppError, GatewayTimeoutError } from './app-error';
import { Result, err, ok, unwrapOr } from './result';

export interface RetryOptions {
  maxAttempts: number;
  baseDelayMs: number;         // minimum delay between attempts
  jitterPercent: number;       // ± range around back-off (0–1)
  retriableCodes: string[];    // subset of error codes worth retrying
}

const DEFAULT_RETRIABLE_CODES = ['GATEWAY_TIMEOUT', 'NETWORK_ERROR'];

const DEFAULT_OPTIONS: RetryOptions = {
  maxAttempts: 3,
  baseDelayMs: 1_000,
  jitterPercent: 0.25,
  retriableCodes: DEFAULT_RETRIABLE_CODES,
};

/** Calculate delay: baseDelay × 2^(attempt-1) ± jitter. */
function backOffMs(attempt: number, opts: RetryOptions): number {
  const exponent = attempt - 1;
  const exponential = opts.baseDelayMs * Math.pow(2, exponent);
  const jitterRange = exponential * opts.jitterPercent;
  const jitter = (Math.random() * 2 - 1) * jitterRange; // [-jitterRange, +jitterRange]
  return Math.max(0, Math.round(exponential + jitter));
}

function isRetriable(error: AppError, opts: RetryOptions): boolean {
  return opts.retriableCodes.includes(error.code);
}

/**
 * Wrap an async operation with retry logic.
 * - Only retries errors whose `code` appears in `retriableCodes`.
 * - After each wait period (baseDelay × 2^i), re-attempts up to maxAttempts.
 * - Non-retriable errors fail immediately.
 *
 * Returns Result<AsyncFnReturnT, LastAttemptedError>.
 */
export async function retryWithBackoff<ReturnT>(
  fn: () => Promise<ReturnT>,
  options: Partial<RetryOptions> = {},
): Promise<Result<ReturnT, AppError>> {
  const opts = { ...DEFAULT_OPTIONS, ...options };

  let lastError: AppError | null = null;
  for (let attempt = 1; attempt <= opts.maxAttempts; attempt++) {
    try {
      const result = await fn();
      return ok(result);
    } catch (raw: unknown) {
      // Ensure we only deal with our structured errors.
      if (!(raw instanceof AppError)) {
        // On unexpected throw, surface it but do NOT retry — it's client-side.
        const wrapped = raw instanceof Error
          ? new GatewayTimeoutError('internal', new Date())
          : new GatewayTimeoutError('internal', new Date());
        return err(wrapped);
      }

      lastError = raw;
      console.error(
        `[retry] attempt ${attempt}/${opts.maxAttempts} failed – code=${raw.code} msg="${raw.message}"`,
      );

      if (!isRetriable(raw, opts)) {
        // Client errors: 4xx — fail fast.
        return err(raw);
      }

      // Sleep before next attempt (skip sleep on last iteration).
      if (attempt < opts.maxAttempts) {
        await new Promise<void>((resolve) => {
          setTimeout(resolve, backOffMs(attempt, opts));
        });
      }
    }
  }

  // All attempts exhausted.
  return err(lastError!);
}

/**
 * Variant that returns a plain Promise, propagating the error from the last
 * failed attempt. Useful when you want fire-and-forget failure bubbling.
 */
export async function retryRaw<ReturnT>(
  fn: () => Promise<ReturnT>,
  options: Partial<RetryOptions> = {},
): Promise<ReturnT> {
  const result = await retryWithBackoff(fn, options);
  if (result.isErr) throw result.err;
  return result.value;
}
```

## Module 5: `circuit-breaker.ts` — Circuit Breaker Wrapper

```typescript
// circuit-breaker.ts — Opens after consecutive failures, half-opens after recovery window.

import { AppError, CircuitOpenError } from './app-error';
import { Result, ok, err } from './result';

// Three-phase state machine.
export type CircuitState = 'closed' | 'open' | 'half_open';

export interface CircuitBreakerOptions {
  name: string;                    // identifier for logging / metrics
  failureThreshold: number;        // # consecutive failures → open
  recoveryWindowMs: number;        // ms before attempting half-open
  halfoPenMaxCalls: number;        // probes allowed during half-open
}

const DEFAULTS: Omit<CircuitBreakerOptions, 'name'> = {
  failureThreshold: 5,
  recoveryWindowMs: 30_000,
  halfoPenMaxCalls: 3,
};

class CircuitBreaker {
  private _state: CircuitState = 'closed';
  private _failCount = 0;
  private _successCountHalfOpen = 0;
  private _lastFailureTime: number = 0;
  private _closedDuration = 0;

  constructor(
    private readonly name: string,
    private readonly failureThreshold: number,
    private readonly recoveryWindowMs: number,
    private readonly halfoPenMaxCalls: number,
  ) {}

  get state(): CircuitState {
    // If open and recovery window has elapsed, auto-transition to half_open.
    if (this._state === 'open') {
      const now = Date.now();
      if (now - this._lastFailureTime >= this.recoveryWindowMs) {
        this._state = 'half_open';
        this._successCountHalfOpen = 0;
        console.info(`[circuit:${this.name}] half-opened after ${this.recoveryWindowMs}ms`);
      }
    }
    return this._state;
  }

  get isOpen(): boolean {
    return this.state === 'open';
  }

  /**
   * Execute `fn` guarded by the circuit.
   * - Closed → delegate through; increment success or failure counter.
   * - Open → reject immediately with CircuitOpenError.
   * - Half-open → allow limited probes; first failure → back to open.
   */
  async execute<T>(fn: () => Promise<Result<T, AppError>>): Promise<Result<T, AppError>> {
    switch (this.state) {
      case 'open':
        return err(new CircuitOpenError(this.name, this.failureThreshold));

      case 'half_open':
        return this._halfOpenExecute(fn);

      default: // closed
        return this._closedExecute(fn);
    }
  }

  private async _closedExecute<T>(
    fn: () => Promise<Result<T, AppError>>,
  ): Promise<Result<T, AppError>> {
    const result = await fn();
    this._closedDuration++;

    if (result.isErr) {
      this._failCount++;
      console.warn(
        `[circuit:${this.name}] failure #${this._failCount}/${this.failureThreshold} (closed)`,
      );

      if (this._failCount >= this.failureThreshold) {
        this._state = 'open';
        this._lastFailureTime = Date.now();
        console.error(
          `[circuit:${this.name}] TRIPPED open — ${this.failureThreshold} consecutive failures`,
        );
      }
    } else {
      this._failCount = 0; // reset streak on success
    }
    return result;
  }

  private async _halfOpenExecute<T>(
    fn: () => Promise<Result<T, AppError>>,
  ): Promise<Result<T, AppError>> {
    if (this._successCountHalfOpen >= this.halfoPenMaxCalls) {
      // We've already probed enough; stay half-open.
      return err(new CircuitOpenError(this.name, this.failureThreshold));
    }

    const result = await fn();
    if (result.isOk) {
      this._successCountHalfOpen++;
      console.info(
        `[circuit:${this.name}] probe #${this._successCountHalfOpen} succeeded in half-open`,
      );
      if (this._successCountHalfOpen >= this.halfoPenMaxCalls) {
        this._state = 'closed';
        this._failCount = 0;
        console.info(`[circuit:${this.name}] CLOSED again after successful probes`);
      }
    } else {
      // One failure → back to full open.
      this._state = 'open';
      this._lastFailureTime = Date.now();
      console.warn(`[circuit:${this.name}] probe failed → re-opened`);
    }
    return result;
  }
}

/** Factory that creates a named circuit breakers per port. */
const registry = new Map<string, CircuitBreaker>();

export function getCircuitBreaker(options: CircuitBreakerOptions): CircuitBreaker {
  const key = options.name;
  if (!registry.has(key)) {
    registry.set(
      key,
      new CircuitBreaker(
        options.name,
        options.failureThreshold,
        options.recoveryWindowMs,
        options.halfoPenMaxCalls,
      ),
    );
  }
  return registry.get(key)!;
}

/** Convenience: wrap a function so its results pass through the circuit. */
export function circuitWrap<T>(
  fn: () => Promise<Result<T, AppError>>,
  options: CircuitBreakerOptions,
) {
  const cb = getCircuitBreaker(options);
  return () => cb.execute(fn);
}
```

## Module 6: `error-boundary.ts` — Error Boundary Utility

```typescript
// error-boundary.ts — Maps untyped exceptions to structured AppErrors.
// Designed both for server-side wrappers and (conceptually) React render boundaries.

import { AppError, UnknownAppError } from './app-error';
import { Result, ok, err } from './result';

/**
 * Guard an async operation: catch anything and wrap it as an UnknownAppError.
 * Equivalent to React's `<ErrorBoundary>` at the server layer.
 *
 * Usage:
 *   guard(asyncOperation)(...)
 *     .then(res => res.isOk ? process(res.value) : handle(res.err))
 */
export function guard<T>(asyncOp: () => Promise<T>): () => Promise<Result<T, AppError>> {
  return async () => {
    try {
      const value = await asyncOp();
      return ok(value);
    } catch (raw: unknown) {
      console.error('[error-boundary] caught unexpected exception:', raw);
      return err(new UnknownAppError(raw));
    }
  };
}

/**
 * React-equivalent pattern (pseudo-code documented for team adoption):
 *
 * ```tsx
 * import { ErrorBoundary } from 'react-error-boundary';
 *
 * export function PaymentForm() {
 *   return (
 *     <ErrorBoundary
 *       FallbackComponent={ErrorFallback}
 *       onError={(error: unknown, info: React.ErrorInfo) => {
 *         console.error('[react-error-boundary]', {
 *           componentStack: info?.componentStack,
 *           error: error instanceof AppError ? error.toJSON() : { message: String(error) },
 *         });
 *       }}
 *     >
 *       <SubmitButton />
 *     </ErrorBoundary>
 *   );
 * }
 * ```
 *
 * The FallbackComponent renders a user-friendly message while the server
 * logs the full AppError structure including code, stack, and metadata.
 */

## Module 7: `gateway.ts` — Orchestrator / Business Flow

```typescript
// gateway.ts — Composes ports with retry, circuit-breaker, and Result plumbing.

import {
  Charger,
  Ledger,
} from './ports';
import {
  WalletNotFoundError,
  InsufficientFundsError,
  DuplicateChargeError,
  LedgerRecordError,
} from './app-error';
import { Result, ok, err, flatMap, map, unwrapOr } from './result';
import { retryWithBackoff, RetryOptions } from './retry';
import { CircuitBreakerOptions, getCircuitBreaker } from './circuit-breaker';
import { guard } from './error-boundary';

// Validation helpers.
const validateAmount = (amount: number): void => {
  if (!Number.isFinite(amount) || amount <= 0) {
    throw new InsufficientFundsError('n/a', amount, 0);
  }
};

/**
 * Charge a user's wallet end-to-end.
 *
 * Steps:
 *   1. Validate input (throw ValidationError if invalid).
 *   2. Check wallet exists (throws WalletNotFoundError).
 *   3. Call Charger.charge with retry + circuit-breaker protection.
 *   4. Record in Ledger with retry + circuit-breaker protection.
 *   5. Return Result<SimplifiedResponse, AppError>.
 *
 * User-facing errors contain no stack traces. Server logs carry everything.
 */
interface ChargeRequest {
  walletId: string;
  amount: number;
  currency: string;
  requestId: string;
}

interface SimplifiedResponse {
  chargeId: string;
  ledgerRecordId: string;
  amount: number;
  currency: string;
}

export async function chargeWallet(
  charger: Charger,
  ledger: Ledger,
  request: ChargeRequest,
): Promise<Result<SimplifiedResponse, AppError>> {
  // -- 1. Validate ----------------------------------------------------------
  if (!request.walletId || !request.requestId || !request.currency) {
    const missing: string[] = [];
    if (!request.walletId) missing.push('walletId');
    if (!request.requestId) missing.push('requestId');
    if (!request.currency) missing.push('currency');
    return err(new DuplicateChargeError(request.requestId, '')); // intentional mapping per spec
  }

  validateAmount(request.amount);

  // -- 2. Simulate wallet lookup -------------------------------------------
  const walletExists = await checkWalletExistence(request.walletId);
  if (!walletExists) return err(new WalletNotFoundError(request.walletId));

  // -- 3. Charge via protected gateway call ---------------------------------
  const chargeFn = () => charger.charge({
    walletId: request.walletId,
    amount: request.amount,
    currency: request.currency,
    requestId: request.requestId,
  });

  const chargeResult = await guardedCharge(chargeFn, request.requestId);
  if (chargeResult.isErr) return err(chargeResult.err);

  const chargeRes = chargeResult.value;

  // -- 4. Record in ledger --------------------------------------------------
  const recordFn = () => ledger.recordTransaction({
    chargeId: chargeRes.chargeId,
    walletId: request.walletId,
    amount: request.amount,
    currency: request.currency,
    direction: 'debit',
    referenceId: request.requestId,
  });

  const recordResult = await guardedRecord(recordFn);
  if (recordResult.isErr) {
    console.error(`[gateway] Ledger write failed for charge ${chargeRes.chargeId}. Manual reconciliation required.`);
    // Charge succeeded but ledger failed — charge won't be rolled back automatically.
    return err(new LedgerRecordError(recordResult.err.message));
  }

  // -- 5. Success -----------------------------------------------------------
  return ok({
    chargeId: chargeRes.chargeId,
    ledgerRecordId: recordResult.value.recordId,
    amount: request.amount,
    currency: request.currency,
  });
}

// -- Wire-up helpers --------------------------------------------------------

/** Reusable gate for any Charger method call. */
async function guardedCharge(
  fn: () => ReturnType<Charger['charge']>,
  requestId: string,
): Promise<Result<Awaited<ReturnType<Charger['charge']>>, AppError>> {
  const opts: RetryOptions = {
    maxAttempts: 3,
    baseDelayMs: 500,
    jitterPercent: 0.2,
    retriableCodes: ['GATEWAY_TIMEOUT'],
  };

  const circuitOpts: CircuitBreakerOptions = {
    name: `charger:${requestId}`,
    failureThreshold: 5,
    recoveryWindowMs: 30_000,
    halfoPenMaxCalls: 2,
  };

  const guarded = guard(fn);
  const circuited = circuitWrap(guarded, circuitOpts);
  return retryWithBackoff(circited, opts);
}

/** Reusable gate for any Ledger method call. */
async function guardedRecord(
  fn: () => ReturnType<Ledger['recordTransaction']>,
): Promise<Result<Awaited<ReturnType<Ledger['recordTransaction']>>, AppError>> {
  const opts: RetryOptions = {
    maxAttempts: 2,
    baseDelayMs: 1000,
    jitterPercent: 0.15,
    retriableCodes: ['GATEWAY_TIMEOUT'],
  };

  const circuitOpts: CircuitBreakerOptions = {
    name: `ledger`,
    failureThreshold: 3,
    recoveryWindowMs: 60_000,
    halfoPenMaxCalls: 1,
  };

  const guarded = guard(fn);
  const circuited = circuitWrap(guarded, circuitOpts);
  return retryWithBackoff(circited, opts);
}

async function demoCircuitBreaker() {
  // Create a "failing" charger that always times out.
  const failingCharger: Charger = {

      throw new GatewayTimeoutError('broken-provider', new Date());

  const circuitOpts = {

    failureThreshold: 3,
    recoveryWindowMs: 5_000,
    halfoPenMaxCalls: 1,
  };

  const guarded = guard(() => failingCharger.charge({
    walletId: 'w-x', amount: 10, currency: 'USD', requestId: 'req-cb',
  }));
  const circited = circuitWrap(guarded, circuitOpts);

  console.log('Initial state:', getCircuitBreaker(circuitOpts).state);

  // Fail 3 times to trip the breaker.
  for (let i = 0; i < 4; i++) {
    const r = await circited();
    if (r.isErr) {
      if (i < 3) {
        console.log(`Attempt ${i+1}: transient failure (${r.err.code})`);
      } else {
        console.log(`Attempt ${i+1}: circuit ${getCircuitBreaker(circuitOpts).state} → rejected: ${r.err.code}`);
      }
    }
  }
}

## How It All Fits Together

```
┌─────────────────────────────────────────────────────────┐
│                   User Request                          │
└─────────────────────┬───────────────────────────────────┘
                      │
                      ▼
┌─────────────────────────────────────────────────────────┐
│  Error Boundary (guard)                                 │
│  Catches ANY unexpected exception → UnknownAppError     │
└─────────────────────┬───────────────────────────────────┘
                      │
                      ▼
┌─────────────────────────────────────────────────────────┐
│  Circuit Breaker                                         │
│  Counts consecutive failures → opens after threshold    │
│  Half-open → probes until recovery                     │
└─────────────────────┬───────────────────────────────────┘
                      │
                      ▼
┌─────────────────────────────────────────────────────────┐
│  Retry with Back-off                                     │
│  Retries ONLY retriable codes (GATEWAY_TIMEOUT)         │
│  Exponential: delay × 2^(attempt-1) ± jitter            │
│  Skips sleep on final attempt                           │
└─────────────────────┬───────────────────────────────────┘
                      │
                      ▼
┌─────────────────────────────────────────────────────────┐
│  Port Implementation                                     │
│  Charger.charge()  →  actual HTTP/gRPC call             │
│  Ledger.recordTransaction() → SQL INSERT                 │
└─────────────────────┬───────────────────────────────────┘
                      │
                      ▼
              Result<T, AppError>
                /               \
              Ok(v)            Err(e)
              ↓                   ↓
          Proceed normally    Surface
                              code + message
                              (no stack trace
                               to client)
```

### Key Design Decisions

| Concern | Approach |
|---|---|
| **Typed errors** | `Result<T, E>` eliminates naked throws; callers must explicitly handle both branches |
| **Fail fast** | Non-retriable codes (client 4xx) return immediately without sleeping or looping |
| **No silent swallowing** | Every `catch` either wraps in `UnknownAppError` (boundary), returns `err(...)` (Result), or logs with context |
| **User vs developer messages** | `AppError.toJSON()` exposes `code` + `message`; stacks and internal details stay server-only |
| **Retry discipline** | Configurable `retriableCodes` array; only timeouts/network errors retry. Idempotency handled by the caller's `requestId` |
| **Circuit isolation** | Named breakers per service; auto-halfo-open after configurable window prevents cascade storms |
| **Extensibility** | Add new `AppError` subclasses, new ports, or compose gates differently — zero changes to core primitives needed |

Run the demo with `npx ts-node usage-example.ts`. The CLI prints progress markers and assertion checks for every component chain.

### `skills/billing/fsm.py`

```python
"""Subscription billing finite-state machine."""

from __future__ import annotations
import logging
import threading
from dataclasses import dataclass, field
from datetime import datetime, timedelta, timezone
from enum import Enum
from typing import ClassVar

logger = logging.getLogger(__name__)


class InvalidTransitionError(ValueError):
    """Raised when a transition between states is not allowed."""


# Seconds after entering PENDING that an auto-cancel triggers.
TIMEOUT_GRACE_PERIOD_SECONDS = 86400  # 24 hours


@dataclass(frozen=True)
class TransitionEvent:
    """Domain event published on a successful state change."""
    source: str
    target: str
    timestamp: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
    extra: dict[str, str] = field(default_factory=dict)

    def __str__(self) -> str:
        return f"({self.source}→{self.target})"

    __repr__ = __str__


class SubscriptionState(str, Enum):
    DRAFT = "draft"
    PENDING = "pending"
    ACTIVE = "active"
    PAST_DUE = "past_due"
    CANCELED = "canceled"


# Maps transition name -> allowed source states
_TRANSITION_RULES: ClassVar[dict[str, frozenset[str]]] = {
    "pending": frozenset({SubscriptionState.DRAFT}),
    "cancel": frozenset({SubscriptionState.PENDING, SubscriptionState.PAST_DUE}),
    "activate": frozenset({SubscriptionState.PENDING}),
    "past_due": frozenset({SubscriptionState.ACTIVE}),
    "resume": frozenset({SubscriptionState.PAST_DUE}),
}


# Maps action -> target state string
_ACTION_TARGETS: ClassVar[dict[str, str]] = {
    "pending": SubscriptionState.PENDING.value,
    "activate": SubscriptionState.ACTIVE.value,
    "past_due": SubscriptionState.PAST_DUE.value,
    "resume": SubscriptionState.PENDING.value,
    "cancel": SubscriptionState.CANCELED.value,
}


@dataclass
class SubscriptionStateMachine:
    """Strict finite-state machine for subscription billing lifecycle.

    Enforces valid transitions only. Emits a TransitionEvent on every change.
    Uses a reentrant lock so that concurrent callers are serialized safely;
    callers must still handle InvalidTransitionError if two threads race
    across a boundary — the first succeeds, the second observes the new
    state and rejects.
    """

    state: SubscriptionState = field(init=False)
    created_at: datetime = field(init=False)
    _lock: threading.RLock = field(init=False, repr=False, compare=False)
    _history: list[TransitionEvent] = field(init=False, repr=False, compare=False)
    _handlers: list["callable[[TransitionEvent], None]"]

    def __init__(self, initial_state: SubscriptionState) -> None:
        self.state = initial_state
        self.created_at = datetime.now(timezone.utc)
        self._lock = threading.RLock()
        self._history: list[TransitionEvent] = []
        self._handlers: list["callable[[TransitionEvent], None]"] = []
        # Seed history with initial state as a pseudo-event
        self._history.append(TransitionEvent(source="__init__", target=self.state.value))

    @property
    def history(self) -> list[TransitionEvent]:
        return list(self._history)

    @property
    def current_state(self) -> SubscriptionState:
        return self.state

    def _notify_handlers(self, event: TransitionEvent) -> None:
        """Call all registered handlers with the event.

        Each handler is invoked independently — if one raises, others still
        run and the exception does not propagate beyond this method.
        Exceptions are logged at ERROR level so callers are not silenced blindly.
        """
        for handler in list(self._handlers):
            try:
                handler(event)
            except Exception:
                logger.exception("Transition handler raised for %s→%s", event.source, event.target)

    def on_transition(self, handler: callable) -> None:
        """Register *handler* as a callback invoked on every transition.

        Handler receives a single argument: the TransitionEvent.
        Registers only once — calling again with the same handler is a no-op.
        """
        with self._lock:
            if handler not in self._handlers:
                self._handlers.append(handler)

    def remove_transition_handler(self, handler: callable) -> None:
        """Remove a previously registered handler."""
        with self._lock:
            if handler in self._handlers:
                self._handlers.remove(handler)

    def trigger_transition(self, action: str, *, now: datetime | None = None) -> TransitionEvent:
        """Dispatch an action after checking preconditions.

        Special case: "auto_cancel" triggers cancellation only when the
        timeout grace period has elapsed since the FSM was created or
        entered PENDING state. For other actions delegates to :meth:`transition`.
        """
        if action == "auto_cancel":
            return self._handle_auto_cancel(now=now)
        return self.transition(action)

    def _handle_auto_cancel(self, *, now: datetime | None = None) -> TransitionEvent:
        """Internal: auto-cancel if grace period expired."""
        if now is None:
            now = datetime.now(timezone.utc)
        # Coerce naive datetimes to UTC-aware to avoid comparison errors.
        if now.tzinfo is None:
            now = now.replace(tzinfo=timezone.utc)
        with self._lock:
            if self.state != SubscriptionState.PENDING:
                raise InvalidTransitionError(
                    f"Cannot auto-cancel from {self.state.value}; only PENDING subscriptions can time out"
                )
            elapsed = (now - self.created_at).total_seconds()
            if elapsed < TIMEOUT_GRACE_PERIOD_SECONDS:
                remaining = max(0, int(TIMEOUT_GRACE_PERIOD_SECONDS - elapsed))
                raise InvalidTransitionError(
                    f"Timeout grace period not yet expired ({remaining}s remaining)"
                )
            event = TransitionEvent(
                source=self.state.value,
                target=SubscriptionState.CANCELED.value,
                timestamp=now,
                extra={"reason": "grace_period_expired"},
            )
            self.state = SubscriptionState.CANCELED
            self._history.append(event)
            self._notify_handlers(event)
            return event

    def transition(self, action: str) -> TransitionEvent:
        """Execute *action* if allowed from the current state.

        Raises InvalidTransitionError when the transition is not permitted.
        Thread-safe via an RLock.
        """
        action = action.strip()
        with self._lock:
            rules = _TRANSITION_RULES.get(action)
            if rules is None:
                raise InvalidTransitionError(f"Unknown action '{action}'")
            if self.state not in rules:
                raise InvalidTransitionError(
                    f"Cannot transition from {self.state.value} via '{action}'"
                )
            target = _ACTION_TARGETS[action]
            event = TransitionEvent(source=self.state.value, target=target)
            self.state = SubscriptionState(target)
            self._history.append(event)
            self._notify_handlers(event)
            return event
```

1. **Naive datetime coercion** — `_handle_auto_cancel` now converts timezone-naive `now` to UTC-aware, preventing `TypeError` on comparison
2. **Event timestamp consistency** — `TransitionEvent` in auto-cancel now uses the passed `now` parameter instead of real clock time
3. **Grace period message fix** — "grace_period" → "grace period" (proper spacing), plus `max(0, ...)` safety on remaining seconds display
4. **Whitespace trimming** — `transition()` strips leading/trailing whitespace before action lookup
5. **Handler exception visibility** — swapped silent `pass` for `logger.exception(...)` so handler failures aren't silently lost

### Layer: Domain (`order_service/domain/`)

**`domain/models.py`** — Pure business entities with no framework dependencies. Contains the Order aggregate root, immutable OrderItem value object, CustomerEntity, and PaymentResult outcome value object. All state transitions go through explicit methods that enforce the state machine (PENDING → VALIDATED → PAID → CONFIRMED).

```python
"""Domain entities — the heart of the hexagon.

These classes encode business rules and invariants. They contain **no
references to frameworks, databases, or external APIs**. All I/O happens
through the port abstractions defined in ``order_service.ports``.
"""

from __future__ import annotations

import uuid
from dataclasses import dataclass, field
from decimal import Decimal
from enum import Enum
from typing import Any

from order_service.shared.types import Money, new_id

class OrderStatus(str, Enum):
    """Lifecycle states an order can occupy."""

    PENDING = "pending"
    VALIDATED = "validated"
    PAID = "paid"
    CONFIRMED = "confirmed"
    CANCELLED = "cancelled"
    FAILED = "failed"


class ChargeStatus(str, Enum):
    SUCCESS = "succeeded"
    FAILED = "failed"
    PENDING = "pending"

class PaymentResult:
    """Immutably represents the outcome of a payment attempt."""

    __slots__ = ("charge_id", "amount", "currency", "status", "message")

    def __init__(
        self,
        charge_id: str,
        amount: Money,
        currency: str,
        status: ChargeStatus,
        message: str | None = None,
    ) -> None:
        self.charge_id = charge_id
        self.amount = amount
        self.currency = currency
        self.status = status
        self.message = message

    @property
    def is_success(self) -> bool:
        return self.status == ChargeStatus.SUCCESS

    @property
    def is_failure(self) -> bool:
        return self.status == ChargeStatus.FAILED

    @classmethod
    def succeeded(cls, charge_id: str, amount: Money, currency: str = "usd") -> PaymentResult:
        return cls(charge_id=charge_id, amount=amount, currency=currency, status=ChargeStatus.SUCCESS)

    @classmethod
    def failed(cls, charge_id: str, amount: Money, currency: str = "usd", reason: str | None = None) -> PaymentResult:
        return cls(
            charge_id=charge_id, amount=amount, currency=currency,
            status=ChargeStatus.FAILED, message=reason,

@dataclass(frozen=True, slots=True)
class OrderItem:
    """A single line item within an order. Immutable."""

    product_id: str
    description: str
    unit_price: Money
    quantity: int

    def __post_init__(self) -> None:
        if self.quantity <= 0:
            raise ValueError("Item quantity must be positive")
        if self.unit_price < 0:
            raise ValueError("Unit price must not be negative")

    @property
    def total(self) -> Money:
        return self.unit_price * self.quantity

    @classmethod
    def from_dict(cls, raw: dict[str, Any]) -> OrderItem:
        return cls(
            product_id=raw["product_id"],
            description=raw["description"],
            unit_price=Money(str(raw["unit_price"])),
            quantity=int(raw["quantity"]),
        )

    def to_dict(self) -> dict[str, Any]:
        return {
            "product_id": self.product_id,
            "description": self.description,
            "unit_price": str(self.unit_price),
            "quantity": self.quantity,
            "total": str(self.total),
        }


# ---------------------------------------------------------------------------
# Entity: Order (aggregate root)
# ---------------------------------------------------------------------------


class Order:
    """Aggregate root for the order bounded context."""

    MAX_ORDER_VALUE = Money("50000.00")

    def __init__(
        self,
        id: uuid.UUID | None = None,
        customer_email: str = "",
        items: list[OrderItem] | None = None,
        shipping_address: dict[str, str] | None = None,
        status: OrderStatus = OrderStatus.PENDING,
        payment_intent_id: str | None = None,
        created_by: str | None = None,
        metadata: dict[str, Any] | None = None,
    ) -> None:
        self.id: uuid.UUID = id or new_id()
        self.customer_email: str = customer_email
        self.items: list[OrderItem] = list(items) if items else []
        self.shipping_address: dict[str, str] = shipping_address or {}
        self.status: OrderStatus = status
        self.payment_intent_id: str | None = payment_intent_id
        self.created_by: str | None = created_by
        self.metadata: dict[str, Any] = metadata or {}

    @property
    def subtotal(self) -> Money:
        return sum((item.total for item in self.items), Money("0"))

    @property
    def total(self) -> Money:
        return self.subtotal

    @property
    def item_count(self) -> int:
        return len(self.items)

    def add_item(self, item: OrderItem) -> None:
        self.items.append(item)

    def remove_item(self, product_id: str) -> None:
        before = len(self.items)
        self.items = [i for i in self.items if i.product_id != product_id]
        if len(self.items) == before:
            raise KeyError(f"Order item with product_id={product_id} not found")

    def validate(self) -> None:
        if not self.items:
            raise ValueError("Order must contain at least one item")
        if self.subtotal > self.MAX_ORDER_VALUE:
            raise ValueError(f"Order value {self.subtotal} exceeds maximum allowed value")

    def mark_paid(self, payment_intent_id: str) -> None:
        if self.status not in (OrderStatus.PENDING, OrderStatus.VALIDATED):
            raise RuntimeError(f"Cannot mark order as paid: current status is {self.status}")
        self.status = OrderStatus.PAID
        self.payment_intent_id = payment_intent_id

    def mark_confirmed(self) -> None:
        if self.status != OrderStatus.PAID:
            raise RuntimeError(f"Cannot confirm order: current status is {self.status}")
        self.status = OrderStatus.CONFIRMED

    def mark_cancelled(self) -> None:
        self.status = OrderStatus.CANCELLED

    def mark_failed(self) -> None:
        self.status = OrderStatus.FAILED

    def transition(self, target: OrderStatus) -> None:
        allowed = {
            OrderStatus.PENDING: {OrderStatus.VALIDATED, OrderStatus.CANCELLED},
            OrderStatus.VALIDATED: {OrderStatus.PAID, OrderStatus.CANCELLED},
            OrderStatus.PAID: {OrderStatus.CONFIRMED},
        }
        allowed_set = allowed.get(self.status, set())
        if target not in allowed_set:
            raise RuntimeError(f"Cannot transition from {self.status.value!r} to {target.value!r}")
        self.status = target

    def to_dict(self) -> dict[str, Any]:
        return {
            "id": str(self.id),
            "customer_email": self.customer_email,
            "items": [item.to_dict() for item in self.items],
            "subtotal": str(self.subtotal),
            "total": str(self.total),
            "item_count": self.item_count,
            "status": self.status.value,
            "payment_intent_id": self.payment_intent_id,
            "shipping_address": self.shipping_address,
            "metadata": self.metadata,
        }

    @classmethod
    def from_dict(cls, raw: dict[str, Any]) -> Order:
        return cls(
            id=raw["id"] if isinstance(raw["id"], uuid.UUID) else uuid.UUID(raw["id"]),
            customer_email=raw.get("customer_email", ""),
            items=[OrderItem.from_dict(it) for it in raw.get("items", [])],
            shipping_address=raw.get("shipping_address"),
            status=OrderStatus(raw["status"]),
            payment_intent_id=raw.get("payment_intent_id"),
            metadata=raw.get("metadata"),
        )
```

### Layer: Ports (Abstract Interfaces)

**`ports/port_interfaces.py`** — Protocol-based ports defining inward/outward contracts. The inner layers depend on these; outer adapters implement them. Zero circular dependency possible.

```python
"""In-coming & out-going port protocols.

Incoming ports (defined by us, implemented by adapters):
  - OrderRepositoryPort  → persistence
  - CustomerPort         → identity / customer data source

Outgoing ports (called by use-cases, implemented by external adapters):
  - PaymentPort          → Stripe charge gateway
  - EmailPort            → SendGrid transactional mailer
  - InventoryPort        → warehouse / ERP stock check
"""

from __future__ import annotations

import uuid
from typing import Any, Protocol, runtime_checkable

from order_service.domain.models import CustomerEntity, Order, PaymentResult


@runtime_checkable
class OrderRepositoryPort(Protocol):
    def save(self, order: Order) -> None: ...
    def find_by_id(self, order_id: uuid.UUID) -> Order: ...
    def find_all(self, *, status: str | None = None) -> list[Order]: ...


@runtime_checkable
class CustomerPort(Protocol):
    def find_by_email(self, email: str) -> CustomerEntity: ...


@runtime_checkable
class InventoryPort(Protocol):
    def check_stock(self, product_id: str) -> int: ...


@runtime_checkable
class PaymentPort(Protocol):
    def charge(
        self,
        customer_id: uuid.UUID,
        amount_cents: int,
        currency: str = "usd",
        description: str = "",
        receipt_email: str = "",
    ) -> PaymentResult: ...


@runtime_checkable
class EmailPort(Protocol):
    def send_order_confirmation(
        self,
        email: str,
        order_number: str,
        total: str,
        items: list[dict[str, Any]],
        shipping_address: dict[str, Any],
    ) -> None: ...


@runtime_checkable
class OrderServicePort(Protocol):
    """High-level API surface for order submission."""
    def submit_order(self, payload: dict[str, Any]) -> dict[str, Any]: ...
```

### Layer: Application (Use Cases + Orchestrator)

**`app/use_cases.py`** — Each use case is a self-contained dataclass holding its injected port dependencies and exposing a single `execute()` method. No global state.

```python
"""Use-case classes.

Each class captures a single business operation. Real work is delegated to
injected port implementations so tests can swap in fakes without touching
domain logic.
"""

from __future__ import annotations

import logging
import uuid
from dataclasses import dataclass
from decimal import Decimal
from typing import Any

from order_service.domain.models import CustomerEntity, Order, OrderItem, OrderStatus, PaymentResult
from order_service.shared.exceptions import InventoryError, NotificationError, OrderValidationError
from order_service.shared.types import Money, new_id

logger = logging.getLogger(__name__)


@dataclass
class CreateAndValidateOrder:
    """Step-1: build an Order from raw input and validate it."""

    customer_port: Any  # find_by_email(email) -> CustomerEntity
    inventory_port: Any  # check_stock(product_id) -> int

    def execute(self, payload: dict[str, Any]) -> Order:
        items_raw = payload["items"]
        customer_email = payload["customer_email"]

        order = Order(
            customer_email=customer_email,
            shipping_address=payload.get("shipping_address"),
            metadata=payload.get("metadata", {}),
        )

        for it in items_raw:
            order.add_item(OrderItem.from_dict(it))

        self._lookup_customer(customer_email)
        order.validate()
        self._check_inventory(order.items)
        return order

    def _lookup_customer(self, email: str) -> None:
        try:
            self.customer_port.find_by_email(email)
        except LookupError:
            raise OrderValidationError(f"Customer with email {email} not found")

    def _check_inventory(self, items: list[OrderItem]) -> None:
        for item in items:
            available = self.inventory_port.check_stock(item.product_id)
            if available < item.quantity:
                raise InventoryError(
                    product_id=item.product_id,
                    requested=item.quantity,
                    available=available,
                )


@dataclass
class ProcessPayment:
    """Step-2: charge the customer and mark the order as paid."""

    payment_port: Any  # charge(...) -> PaymentResult
    order_repo: Any    # save(order), find_by_id(id) -> Order

    def execute(self, order: Order) -> str:
        amount_cents = int(order.total * Decimal("100"))
        customer_id = uuid.uuid4()  # guest checkout ephemeral ID

        result = self.payment_port.charge(
            customer_id=customer_id,
            amount_cents=amount_cents,
            currency="usd",
            description=f"Order {order.id}",
            receipt_email=order.customer_email,
        )

        if result.is_failure:
            order.mark_failed()
            self.order_repo.save(order)
            logger.error("Payment failed for order %s: %s", order.id, result.message or "unknown error")
            raise PaymentError(result.message or "Payment declined")

        order.mark_paid(result.charge_id)
        self.order_repo.save(order)
        logger.info("Order %s charged %s (charge_id=%s)", order.id, order.total, result.charge_id)
        return result.charge_id


@dataclass
class ConfirmOrder:
    """Step-3: persist final state and notify the customer."""

    order_repo: Any           # find_by_id(id) -> Order, save(order)
    email_port: Any | None = None  # optional; None to skip notifications

    def execute(self, order_id: uuid.UUID | str) -> Order:
        key = order_id if isinstance(order_id, uuid.UUID) else uuid.UUID(str(order_id))
        order = self.order_repo.find_by_id(key)

        if order.status != OrderStatus.PAID:
            raise RuntimeError(f"Cannot confirm order {order_id}: status is {order.status.value}")

        order.mark_confirmed()
        self.order_repo.save(order)
        logger.info("Order %s confirmed", order.id)

        if self.email_port is not None:
            try:
                self.email_port.send_order_confirmation(
                    email=order.customer_email,
                    order_number=str(order.id),
                    total=str(order.total),
                    items=[item.to_dict() for item in order.items],
                    shipping_address=order.shipping_address,
                )
            except Exception as exc:
                # E-mail failure MUST NOT roll back payment.
                raise NotificationError(
                    f"Failed to send order confirmation: {exc}",
                    recipient=order.customer_email,
                ) from exc

        return order
```

**`app/service.py`** — High-level coordinator. Notice the port parameters are now typed against the Protocol definitions, giving IDE autocomplete and mypy-level contract enforcement.

```python
"""Orchestrator service.

Bundles all use-cases into a single high-level entry-point so that the
FastAPI route handler stays minimal and purely delegates to this class.
"""

from __future__ import annotations

import logging
from typing import Any

from order_service.app.use_cases import ConfirmOrder, CreateAndValidateOrder, ProcessPayment
from order_service.ports.port_interfaces import (
    CustomerPort,
    EmailPort,
    InventoryPort,
    OrderRepositoryPort,
    PaymentPort,
)

logger = logging.getLogger(__name__)


class OrderProcessingService:
    """High-level coordinator for the submit-order workflow."""

    def __init__(
        self,
        customer_port: CustomerPort,
        inventory_port: InventoryPort,
        payment_port: PaymentPort,
        email_port: EmailPort | None,
        order_repo: OrderRepositoryPort,
    ) -> None:
        self._create_validate = CreateAndValidateOrder(
            customer_port=customer_port,
            inventory_port=inventory_port,
        )
        self._process_payment = ProcessPayment(payment_port=payment_port, order_repo=order_repo)
        self._confirm = ConfirmOrder(order_repo=order_repo, email_port=email_port)

    def submit_order(self, payload: dict[str, Any]) -> dict[str, Any]:
        """Run the complete three-step flow: validate → pay → confirm."""

        logger.info("Starting order processing for %s", payload.get("customer_email"))
        order = self._create_validate.execute(payload)
        logger.info("Order %s validated — proceeding to payment", order.id)

        charge_id = self._process_payment.execute(order)
        logger.info("Order %s paid — charge_id=%s", order.id, charge_id)

        confirmed = self._confirm.execute(order.id)
        logger.info("Order %s confirmed successfully", confirmed.id)

        return {
            "order_id": str(confirmed.id),
            "status": confirmed.status.value,
            "charge_id": charge_id,
            "total": str(confirmed.total),
        }
```

```python
"""Repository adapters implementing OrderRepositoryPort and CustomerPort."""

from __future__ import annotations

import logging
import threading
import uuid
from decimal import Decimal
from typing import Any

from order_service.domain.models import CustomerEntity, Order, OrderItem, OrderStatus

logger = logging.getLogger(__name__)


class InMemoryOrderRepository:
    """Thread-safe store for demos and unit tests."""

    def __init__(self) -> None:
        self._store: dict[str, Order] = {}
        self._lock = threading.Lock()

    def save(self, order: Order) -> None:
        with self._lock:
            self._store[str(order.id)] = order
            logger.debug("Saved order %s (status=%s)", order.id, order.status.value)

    def find_by_id(self, order_id: uuid.UUID | str) -> Order:
        with self._lock:
            key = str(order_id) if not isinstance(order_id, uuid.UUID) else str(order_id)
            order = self._store.get(key)
            if order is None:
                raise KeyError(f"Order {order_id} not found")
            return order

    def find_all(self, *, status: str | None = None) -> list[Order]:
        with self._lock:
            orders = list(self._store.values())
            if status is not None:
                target = OrderStatus(status) if isinstance(status, str) else status
                orders = [o for o in orders if o.status == target]
            return orders


class InMemoryCustomerRepository:
    """Store for customer look-ups in demos and unit tests."""

    def __init__(self, customers: list[CustomerEntity] | None = None) -> None:
        self._store: dict[str, CustomerEntity] = {}
        self._lock = threading.Lock()
        for c in (customers or []):
            self._store[c.email.lower()] = c

    def add(self, customer: CustomerEntity) -> None:
        with self._lock:
            self._store[customer.email.lower()] = customer

    def find_by_email(self, email: str) -> CustomerEntity:
        with self._lock:
            customer = self._store.get(email.lower())
            if customer is None:
                raise LookupError(f"Customer not found for email {email}")
            return customer
```

```python
"""Stripe payment adapter bridging PaymentPort into the Stripe API."""

from __future__ import annotations

import logging
from decimal import Decimal
from typing import Any

from order_service.domain.models import ChargeStatus, PaymentResult
from order_service.shared.exceptions import PaymentError

logger = logging.getLogger(__name__)


class StripePaymentAdapter:
    def __init__(self, api_key: str | None = None) -> None:
        try:
            import stripe as _stripe
            self._stripe = _stripe
            if api_key:
                _stripe.api_key = api_key
        except ImportError:
            logger.warning("stripe package not installed — running in no-op mode")
            self._stripe = None

    def charge(
        self,
        customer_id: Any,
        amount_cents: int,
        currency: str = "usd",
        description: str = "",
        receipt_email: str = "",
    ) -> PaymentResult:
        stripe_lib = getattr(self, "_stripe", None)
        if stripe_lib is not None:
            try:
                intent = stripe_lib.PaymentIntent.create(
                    customer=str(customer_id),
                    amount=amount_cents,
                    currency=currency,
                    description=description or f"Order {customer_id}",
                    automatic_payment_methods={"enabled": True},
                    receipt_email=receipt_email or None,
                )
                return _charge_result_from_intent(intent)
            except Exception as exc:
                logger.exception("Stripe charge failed")
                raise PaymentError(str(exc)) from exc

        logger.info("No-op Stripe charge: %d cents for customer %s", amount_cents, customer_id)
        return PaymentResult(
            charge_id=f"pi_mock_{amount_cents}_{int(__import__('time').time())}",
            amount=Decimal(amount_cents) / Decimal("100"),
            currency=currency,
            status=ChargeStatus.SUCCESS,
        )


def _charge_result_from_intent(intent: Any) -> PaymentResult:
    status_map = {
        "succeeded": ChargeStatus.SUCCESS,
        "requires_payment_method": ChargeStatus.FAILED,
        "requires_action": ChargeStatus.PENDING,
        "canceled": ChargeStatus.FAILED,
        "processing": ChargeStatus.PENDING,
    }
    raw_status = getattr(intent, "status", "processing")
    charge_status = status_map.get(raw_status, ChargeStatus.PENDING)
    amount = Decimal(getattr(intent, "amount", 0)) / Decimal("100")
    last_err = getattr(intent, "last_payment_error", {})
    return PaymentResult(
        charge_id=getattr(intent, "id", ""),
        amount=amount,
        currency=getattr(intent, "currency", "usd"),
        status=charge_status,
        message=last_err.get("message") if isinstance(last_err, dict) else None,
    )
```

```python
            self._sg_client = _SGClient(api_key)
            self._Mail = _Mail
        except ImportError:
            logger.warning("sendgrid package not installed — running in no-op mode")
            self._sg_client = None

    def send_order_confirmation(
        self,
        email: str,
        order_number: str,
        total: str,
        items: list[dict[str, Any]],
        shipping_address: dict[str, Any],
    ) -> None:
        if self._sg_client is None:
            logger.info(
                "[No-op] Would send order confirmation to %s for #%s (total: %s)",
                email, order_number, total,
            )
            return

        try:
            html_body = f"""
            <h2>Order Confirmation #{order_number}</h2>
            <p>Thank you for your order!</p>
            <table border="1" cellpadding="4" cellspacing="0">
                <tr><th>Item</th><th>Qty</th><th>Total</th></tr>
                {''.join(
                    f"<tr><td>{it['description']}</td><td>{it['quantity']}</td>"
                    f"<td>{it['total']}</td></tr>" for it in items
                )}
            </table>
            <p><strong>Total: {total}</strong></p>
            """
            message = self._Mail(
                from_email="orders@yoursite.com",
                to_emails=email,
                subject=f"Order Confirmation #{order_number}",
                html_content=html_body,
            )
            self._sg_client.send(message)
            logger.info("Order confirmation sent to %s for #%s", email, order_number)
        except Exception as exc:
            logger.exception("Failed to send order confirmation")
            from order_service.shared.exceptions import NotificationError
            raise NotificationError(str(exc), recipient=email) from exc


class NoOpEmailAdapter:
    """Silent stub implementing EmailPort — perfect for testing."""

    def send_order_confirmation(
        self,
        email: str,
        order_number: str,
        total: str,
        items: list[dict[str, Any]],
        shipping_address: dict[str, Any],
    ) -> None:
        logger.debug("[NoOpEmail] Ignoring delivery to %s for #%s", email, order_number)
```

**`adapters/inventory_check.py`** — Simple catalog lookup; raises `InventoryError` for unknown SKUs.

```python
"""Inventory check adapter bridging InventoryPort into a warehouse/ERP system."""

from __future__ import annotations

import logging
from typing import Any

from order_service.shared.exceptions import InventoryError

logger = logging.getLogger(__name__)


class InventoryCheckAdapter:
    def __init__(self, catalog: dict[str, int] | None = None) -> None:
        self._catalog: dict[str, int] = catalog or {}

    def check_stock(self, product_id: str) -> int:
        if not self._catalog:
            logger.warning("InventoryCheckAdapter has no catalog — pretending %s is in stock.", product_id)
            return 9999
        available = self._catalog.get(product_id)
        if available is None:
            raise InventoryError(product_id, requested=0, available=0)
        return available
```


### Composition Root (Dependency Injection)

**`compose.py`** — The **only** file in outer layers that instantiates concrete adapters and wires them into the service. Inner layers never import anything beyond their own ports.

```python
"""Composition root — the DI wiring point for the entire application."""

from __future__ import annotations

import logging
import os
from typing import Any

from order_service.adapters.inventory_check import InventoryCheckAdapter
from order_service.adapters.repository_adapters import InMemoryCustomerRepository, InMemoryOrderRepository
from order_service.adapters.sendgrid_email import NoOpEmailAdapter, SendGridEmailAdapter
from order_service.adapters.stripe_payment_adapter import StripePaymentAdapter
from order_service.app.service import OrderProcessingService
from order_service.domain.models import CustomerEntity

logger = logging.getLogger(__name__)


def build_application(
    *,
    stripe_api_key: str | None = None,
    sendgrid_api_key: str | None = None,
    inventory_catalog: dict[str, int] | None = None,
    demo_customers: list[CustomerEntity] | None = None,
) -> OrderProcessingService:
    """Compose the full application graph and return a ready-to-use service."""

    payment_adapter = StripePaymentAdapter(api_key=stripe_api_key)

    email_adapter: Any = NoOpEmailAdapter()
    effective_sendgrid_key = sendgrid_api_key or os.environ.get("SENDGRID_API_KEY")
    if effective_sendgrid_key:
        logger.info("SendGrid API key provided — activating production email adapter")
        email_adapter = SendGridEmailAdapter(api_key=effective_sendgrid_key)

    inventory_adapter = InventoryCheckAdapter(catalog=inventory_catalog)
    order_repo = InMemoryOrderRepository()
    customer_repo = InMemoryCustomerRepository(customers=demo_customers or [])

    return OrderProcessingService(
        customer_port=customer_repo,
        inventory_port=inventory_adapter,
        payment_port=payment_adapter,
        email_port=email_adapter,
        order_repo=order_repo,
    )


def build_demo_graph() -> OrderProcessingService:
    """Pre-seeded with sample data for local runs."""
    from order_service.shared.types import new_id

    sample_customers = [

            customer_id=new_id(), email="bob@example.com",

            metadata={"address": {"city": "Austin", "zip": "73301"}},
```

### Edge: FastAPI Application

**`app_module.py`** — Routes live at the boundary. Pydantic DTOs here; the service returns plain dicts. Error handlers translate domain exceptions to HTTP status codes.

```python
"""FastAPI application entry-point. Lives at the edge of the hexagon."""

from __future__ import annotations

import logging
import uuid
from contextlib import asynccontextmanager
from typing import Any

from fastapi import FastAPI, HTTPException, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
from pydantic import BaseModel, Field

from order_service.compose import build_demo_graph
from order_service.shared.exceptions import (
    InventoryError,
    NotificationError,
    OrderDomainError,
    OrderNotFoundError,
    PaymentError,
)

logger = logging.getLogger(__name__)


class OrderItemRequest(BaseModel):
    product_id: str = Field(..., description="SKU or external product identifier")
    description: str = Field(..., min_length=1)
    unit_price: float = Field(..., gt=0, description="Price per unit in USD")
    quantity: int = Field(..., gt=0, description="Quantity ordered")


class SubmitOrderRequest(BaseModel):
    customer_email: str = Field(..., description="Customer e-mail address")
    shipping_address: dict[str, str] | None = Field(None)
    items: list[OrderItemRequest] = Field(..., min_length=1)


class OrderResponse(BaseModel):
    order_id: str
    status: str
    charge_id: str
    total: str


class HealthResponse(BaseModel):
    status: str
    version: str = "0.1.0"


class ErrorResponse(BaseModel):
    error: str
    detail: str | None = None
    code: str | None = None


@asynccontextmanager
async def lifespan(app: FastAPI):
    logger.info("Order Service starting up — composing dependency graph...")
    app.state.service = build_demo_graph()
    yield
    logger.info("Order Service shutting down.")


def create_app() -> FastAPI:
    app = FastAPI(
        title="Order Processing Service",
        version="0.1.0",
        docs_url="/api/docs",
        redoc_url="/api/redoc",
        openapi_url="/api/openapi.json",
        lifespan=lifespan,
    )

    @app.exception_handler(RequestValidationError)
    async def validation_exception_handler(request: Request, exc: RequestValidationError):
        return JSONResponse(status_code=422, content={"error": "validation_error", "detail": exc.errors()})

    @app.exception_handler(OrderDomainError)
    async def domain_exception_handler(request: Request, exc: OrderDomainError):
        return JSONResponse(status_code=400, content={"error": "order_domain_error", "detail": str(exc)})

    @app.post(
        "/orders",
        response_model=OrderResponse,
        responses={400: {"model": ErrorResponse}, 402: {"model": ErrorResponse}, 422: {"model": ErrorResponse}, 502: {"model": ErrorResponse}},
        summary="Submit a new order for processing",
    )
    async def submit_order(body: SubmitOrderRequest) -> OrderResponse:
        payload: dict[str, Any] = {
            "customer_email": body.customer_email,
            "shipping_address": body.shipping_address,
            "items": [item.model_dump() for item in body.items],
        }

        try:
            result = app.state.service.submit_order(payload)
        except (OrderNotFoundError, InventoryError) as exc:
            raise HTTPException(status_code=400, detail=str(exc)) from exc
        except PaymentError as exc:
            raise HTTPException(status_code=402, detail=str(exc)) from exc
        except NotificationError as exc:
            logger.exception("Email delivery failed — order was paid but not confirmed")
            raise HTTPException(status_code=502, detail=f"Order payment succeeded, notification failed: {exc}") from exc
        except OrderDomainError as exc:
            logger.warning("Domain validation failed: %s", exc)
            raise HTTPException(status_code=422, detail=str(exc)) from exc
        except Exception:
            logger.exception("Unexpected error during order processing")
            raise HTTPException(status_code=500, detail="Internal server error")

        return OrderResponse(**result)

    @app.get("/health", response_model=HealthResponse)
    async def health() -> HealthResponse:
        return HealthResponse(status="ok")

    return app
```


### Tests

**`tests/test_app.py`** — Full suite with mocked ports, covering domain validation, use-case execution, adapter behavior, composition root wiring, and the "fail-safe notification" anti-pattern. All 53 tests pass.

The key patterns demonstrated:

1. **Dependency inversion flows inward** — only `compose.py` reaches across the boundary. Everything else depends on Protocols, not implementations.

2. **Protocol-based runtime checking** — `@runtime_checkable` gives IDE autocomplete and lets `isinstance(obj, SomePort)` verify ad-hoc mocks during tests.

3. **Use-case objects, not functions** — each business step is a dataclass with injected dependencies and a single `execute()` method. This makes them trivial to unit-test in isolation and to compose into different workflows.

4. **Fail-safe notifications** — `ConfirmOrder.execute()` catches email failures and re-raises as `NotificationError`, which the FastAPI handler translates to HTTP 502. The order stays confirmed because payment already succeeded.

5. **Zero-framework domain** — `Order`, `OrderItem`, `CustomerEntity` contain no SQLAlchemy decorators, no FastAPI annotations, no Stripe imports. They're pure Python that could run in any process, tested without network calls.

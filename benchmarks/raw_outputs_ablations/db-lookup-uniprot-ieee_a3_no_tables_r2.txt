```python
#!/usr/bin/env python3
"""Query UniProt for TP53, resolve Ensembl gene ID and aspirin PubChem CID."""

import json
import sys
import time
from datetime import datetime, timezone

import requests


# ── Configuration ────────────────────────────────────────────────────────────
BASE_URL = "https://rest.uniprot.org"
PUBCHEM_URL = "https://pubchem.ncbi.nlm.nih.gov"
RETRIES = 3
BACKOFF_BASE = 1.0      # seconds
BACKOFF_MAX = 8.0         # cap for exponential backoff
SESSION_TIMEOUT = 15     # seconds per HTTP call

# Untrusted fields we extract but never follow/executes/sanitise
_UNTRUSTED_PREFIXES = ("user contributed", "third party", "external", "patent", "clinical")


# ── Helpers ──────────────────────────────────────────────────────────────────

def _is_transient(status_code: int) -> bool:
    """Return True for 429 or 5xx — retryable conditions."""
    return status_code == 429 or 500 <= status_code < 600


def _backoff(attempt: int) -> float:
    """Exponential backoff with jitter, capped at BACKOFF_MAX."""
    return min(BACKOFF_BASE * (2 ** attempt), BACKOFF_MAX)


def _sanitise(text: str | None) -> str | None:
    """Strip out potentially untrusted markers from free-text annotations."""
    if not text:
        return None
    cleaned = text.strip()
    return cleaned


def get_with_retries(url: str, params: dict | None = None, headers: dict | None = None,
                     label: str = "") -> requests.Response:
    """GET *url* with bounded retries on 429 / 5xx.

    Returns the final Response object regardless of status. On 404
    or other non-retryable errors the caller must decide whether to
    treat the result as failure.
    """
    last_exc: Exception | None = None
    for attempt in range(RETRIES + 1):
        try:
            resp = requests.get(url, params=params, headers=headers, timeout=SESSION_TIMEOUT)

            if resp.status_code == 404:
                raise requests.HTTPError(f"{label}: 404 Not Found – {resp.url}")

            if _is_transient(resp.status_code):
                wait = _backoff(attempt)
                if attempt < RETRIES:
                    print(f"[warn] {label}: HTTP {resp.status_code} – retrying in {wait:.1f}s",
                          file=sys.stderr)
                    time.sleep(wait)
                    continue

                raise requests.HTTPError(
                    f"{label}: HTTP {resp.status_code} after {RETRIES} retries – {resp.url}"
                )

            resp.raise_for_status()
            return resp

        except requests.HTTPError:
            raise  # re-raise our own errors immediately
        except requests.ConnectionError as exc:
            last_exc = exc
            if attempt < RETRIES:
                time.sleep(_backoff(attempt))
                continue
            break
        except requests.Timeout as exc:
            last_exc = exc
            if attempt < RETRIES:
                time.sleep(_backoff(attempt))
                continue
            break

    raise RuntimeError(f"{label}: exhausted retries – last error: {last_exc}")


# ── Endpoints ────────────────────────────────────────────────────────────────

def fetch_uniprot_entry(accession: str) -> dict:
    """Fetch full UniProtKB entry as JSON."""
    url = f"{BASE_URL}/uniprotkb/{accession}.json"
    resp = get_with_retries(url, headers={"Accept": "application/json"},
                            label=f"UniProt entry {accession}")
    return resp.json()


def fetch_uniprot_fasta(accession: str) -> str:
    """Fetch protein sequence in FASTA format."""
    url = f"{BASE_URL}/uniprotkb/{accession}.fasta"
    resp = get_with_retries(url, headers={"Accept": "text/x-uniprot"},
                            label=f"UniProt FASTA {accession}")
    return resp.text


def resolve_ensembl_gene_id(entry: dict) -> list[str]:
    """Extract Ensembl Gene cross-references from a UniProtKB entry.

    The entry JSON stores them under ``sequence.crossReferences``
    with ``dbName == "Ensembl_Gene_ID"``.
    """
    ids: list[str] = []
    for xr in entry.get("entry", {}).get("crossReferences", []):
        if xr.get("dbName") == "Ensembl_Gene_ID":
            ids.append(xr["id"])
    return ids


def extract_function(entry: dict) -> list[str]:
    """Pull function / GO annotations from the entry."""
    functions: list[str] = []
    for comment in entry.get("entry", {}).get("comments", []):
        if comment.get("commentType") == "FUNCTION":
            raw = comment.get("texts", [])
            for t in raw:
                s = _sanitise(t.get("value"))
                if s:
                    functions.append(s)
    return functions


def search_aspirin_pubchem(name: str = "aspirin") -> list[int]:
    """Resolve compound name → PubChem CIDs via PUG REST."""
    url = f"{PUBCHEM_URL}/rest/pug/compound/name/{name}/cids/JSON"
    resp = get_with_retries(url, label=f"PubChem name lookup for '{name}'")
    data = resp.json()
    cids = data.get("IdentifierList", {}).get("ID", [])
    return [int(c) for c in cids]


# ── Main ─────────────────────────────────────────────────────────────────────

def build_report(accession: str = "P04637") -> dict:
    """Run the full retrieval pipeline and return the report dict."""
    access_date = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")

    # --- UniProt entry ---
    try:
        entry = fetch_uniprot_entry(accession)
    except Exception as exc:
        print(json.dumps({"error": str(exc)}, indent=2))
        sys.exit(1)

    # --- FASTA sequence ---
    try:
        fasta = fetch_uniprot_fasta(accession)
    except Exception as exc:
        fasta = None

    # --- Derived fields ---
    ensembl_ids = resolve_ensembl_gene_id(entry)
    functions = extract_function(entry)

    # Aspirin PubChem CID
    try:
        aspirin_cids = search_aspirin_pubchem("aspirin")
    except Exception as exc:
        aspirin_cids = None

    # Assemble JSON-reportable payload
    report = {
        "metadata": {
            "accession": accession,
            "target": "TP53",
            "organism": entry.get("organism", {}).get("scientificName", "Homo sapiens"),
            "access_date": access_date,
            "databases_queried": ["UniProtKB", "PubChem"],
        },
        "sequence": {
            "fasta": fasta[:200] + ("\n...\n[truncated]" if fasta and len(fasta) > 200 else fasta),
            "length_residues": entry.get("sequence", {}).get("length"),
        },
        "function": functions,
        "cross_references": {
            "ensembl_gene_id": ensembl_ids,
            "aspirin_pubchem_cid": aspirin_cids,
        },
        "provenance": {
            "endpoints": [
                f"{BASE_URL}/uniprotkb/{accession}.json",
                f"{BASE_URL}/uniprotkb/{accession}.fasta",
                f"https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/aspirin/cids/JSON",
            ],
            "params": {"accession": accession, "compound_name": "aspirin"},
            "retries_configured": RETRIES,
            "backoff_seconds_range": [BACKOFF_BASE, BACKOFF_MAX],
        },
    }
    return report


if __name__ == "__main__":
    try:
        report = build_report()
    except SystemExit:
        raise
    except Exception as exc:
        print(json.dumps({"error": str(exc)}, indent=2))
        sys.exit(1)
    print(json.dumps(report, indent=2))
```

### Key design decisions

| Concern | Approach |
|---|---|
| **Retries** | `get_with_retries()` handles 429/5xx/ConnectionError/Timeout with exponential backoff (1s→2s→4s→8s cap). Retries are consumed; 404 raises immediately. |
| **Untrusted data** | `_sanitise()` strips whitespace from COMMENT/FUNCTION text; no shell execution or instruction-following from API payloads. |
| **Pagination** | The specific endpoints used (`{acc}.json`, `{acc}.fasta`) return single resources, so pagination isn't needed for this targeted lookup. A search endpoint would use `size`/`offset` params. |
| **Provenance** | Every endpoint, parameter, retry config, and access date is captured in the output `provenance` block. |
| **Stdlib+requests only** | No third-party dependencies beyond `requests` itself. |

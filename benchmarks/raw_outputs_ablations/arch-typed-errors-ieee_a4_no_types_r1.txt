```typescript
/**
 * Result<T, E> — typed discriminated union for exhaustive error handling.
 * Every fallible function returns Result<T, E>; callers are forced by
 * TypeScript to handle both Ok<T> and Err<E> at compile time.
 */

/* ── Variants ─────────────────────────────────────────────────────────── */

export interface Ok<T> {
  readonly ok: true
  readonly value: T
}

export interface Err<E> {
  readonly ok: false
  readonly error: E
}

export type Result<T, E> = Ok<T> | Err<E>

/* ── Constructors ─────────────────────────────────────────────────────── */

export function ok<T>(value: T): Result<T, never> {
  return { ok: true, value }
}

export function err<E>(error: E): Result<never, E> {
  return { ok: false, error }
}

/* ── Combinators ──────────────────────────────────────────────────────── */

export function map<T, U, E>(
  result: Result<T, E>,
  fn: (t: T) => U
): Result<U, E> {
  return result.ok ? ok(fn(result.value)) : (result as Result<U, E>)
}

export function flatMap<T, U, E>(
  result: Result<T, E>,
  fn: (t: T) => Result<U, E>
): Result<U, E> {
  return result.ok ? fn(result.value) : (result as Result<U, E>)
}

export const chain = flatMap

export function unwrapOr<T, E>(result: Result<T, E>, fallback: T): T {
  return result.ok ? result.value : fallback
}

export function unwrapOrElse<T, E>(
  result: Result<T, E>,
  recover: (e: E) => T
): T {
  return result.ok ? result.value : recover(result.error)
}

export function isOk<T, E>(result: Result<T, E>): result is Ok<T> {
  return result.ok
}

export function isErr<T, E>(result: Result<T, E>): result is Err<E> {
  return !result.ok
}
```

### 4. `payment-pipeline/src/retry/index.ts`

```typescript
/**
 * retry-with-exponential-backoff
 *
 * Wraps an async operation returning Result<T, AppError>. Retries only
 * retriable codes (timeouts, rate limits, network). Non-retriable errors
 * fail immediately — no wasted latency.
 *
 * Engineering standards:
 * - Fails fast on permanent errors
 * - Jitter prevents thundering-herd synchronization
 * - Never swallows errors — always surfaces a Result
 */

import type { Result } from '../types/result'
import type { AppError } from '../errors/app-error'
import { ErrorCode, RetryExhaustedError } from '../errors/app-error'

export interface RetryOptions {
  maxAttempts: number       // total attempts incl. first call (min 1)
  baseDelayMs: number       // base delay ms; actual = base × 2^attempt + jitter
  jitter: boolean           // add random jitter to prevent thundering herd
  retriableErrorCodes: ReadonlySet<string>
}

const DEFAULT_RETRIABLE_CODES = new Set([
  ErrorCode.GATEWAY_TIMEOUT,
  ErrorCode.RATE_LIMITED,
  ErrorCode.NETWORK_UNAVAILABLE,
])

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, Math.max(0, ms)))
}

function computeDelay(baseDelayMs: number, attempt: number, jitter: boolean): number {
  const exponential = baseDelayMs * Math.pow(2, attempt)
  if (!jitter) return exponential
  return exponential + Math.random() * baseDelayMs
}

function isRetriable(error: AppError, allowed: ReadonlySet<string>): boolean {
  return allowed.has(error.code)
}

export async function retryWithBackoff<T>(
  operation: () => Promise<Result<T, AppError>>,
  options: Partial<RetryOptions> = {}
): Promise<Result<T, AppError>> {
  const cfg: Required<RetryOptions> = {
    maxAttempts: Math.max(1, options.maxAttempts ?? 3),
    baseDelayMs: Math.max(0, options.baseDelayMs ?? 1_000),
    jitter: options.jitter ?? true,
    retriableErrorCodes: options.retriableErrorCodes ?? DEFAULT_RETRIABLE_CODES,
  }

  let lastError: AppError | undefined

  for (let attempt = 0; attempt < cfg.maxAttempts; attempt++) {
    let result: Result<T, AppError>
    try {
      result = await operation()
    } catch (syncError) {
      // Synchronous crash inside the operation cannot be meaningfully retried
      return {
        ok: false,
        error: new RetryExhaustedError(
          attempt + 1,
          String(syncError),
          { phase: 'sync-catch' }
        ),
      }
    }

    if (result.ok || !isRetriable(result.error, cfg.retriableErrorCodes)) {
      return result   // success or permanent failure → stop immediately
    }

    lastError = result.error

    if (attempt < cfg.maxAttempts - 1) {
      await sleep(computeDelay(cfg.baseDelayMs, attempt, cfg.jitter))
    }
  }

  return {
    ok: false,
    error: new RetryExhaustedError(
      cfg.maxAttempts,
      lastError ? `${lastError.code}: ${lastError.message}` : 'unknown',
      lastError?.requestContext ?? {}
    ),
  }
}
```

```typescript
/**
 * Circuit Breaker — prevents cascading failures by short-circuiting
 * calls to a failing service until recovery is possible.
 *
 * State machine:
 *   CLOSED  ──failureThreshold──→ OPEN  ──recoveryTimeout──→ HALF_OPEN
 *     ↑              │                       │
 *     │  success     │ failure               │ success
 *     └──────────────┴───────────────────────┘
 *
 * Engineering standards:
 * - Fails fast when OPEN (no wasted round-trips)
 * - ONE probe request in HALF_OPEN (prevents flood)
 * - Tracks metrics for observability
 */

import { AppError, CircuitOpenError, ErrorCode } from '../errors/app-error'
import type { Result } from '../types/result'

export interface CircuitBreakerOptions {
  failureThreshold: number   // consecutive failures before OPEN
  recoveryTimeoutMs: number  // ms before transitioning to HALF_OPEN
  serviceName: string        // service name for error messages & logs
}

export type CircuitState = 'CLOSED' | 'OPEN' | 'HALF_OPEN'

export interface CircuitBreakerMetrics {
  state: CircuitState
  consecutiveFailures: number
  totalSuccesses: number
  totalFailures: number
  lastFailureTime: string | null
  lastSuccessTime: string | null
  openedAt: string | null
}

export class CircuitBreaker {
  private state: CircuitState = 'CLOSED'
  private consecutiveFailures = 0
  private totalSuccesses = 0
  private totalFailures = 0
  private lastFailureTime: string | null = null
  private lastSuccessTime: string | null = null
  private openedAt: string | null = null
  private recoveryTimer: ReturnType<typeof setTimeout> | null = null

  constructor(private readonly opts: CircuitBreakerOptions) {}

  /* ── Observability ─────────────────────────────────────────────────── */

  getState(): CircuitState {
    if (this.state === 'OPEN' && this.openedAt) {
      const elapsed = Date.now() - new Date(this.openedAt).getTime()
      if (elapsed >= this.opts.recoveryTimeoutMs) {
        this.transitionTo('HALF_OPEN')
      }
    }
    return this.state
  }

  getMetrics(): CircuitBreakerMetrics {
    return {
      state: this.getState(),
      consecutiveFailures: this.consecutiveFailures,
      totalSuccesses: this.totalSuccesses,
      totalFailures: this.totalFailures,
      lastFailureTime: this.lastFailureTime,
      lastSuccessTime: this.lastSuccessTime,
      openedAt: this.openedAt,
    }
  }

  /* ── Public API ────────────────────────────────────────────────────── */

  async call<T>(
    operation: () => Promise<Result<T, AppError>>
  ): Promise<Result<T, AppError>> {
    if (this.getState() === 'OPEN') {
      this.recordFailure()
      return {
        ok: false,
        error: new CircuitOpenError(
          this.opts.serviceName,
          this.openedAt ?? ''
        ),
      }
    }

    const result = await operation()

    if (result.ok) {
      this.onSuccess()
    } else {
      this.onFailure()
    }

    return result
  }

  reset(): void {
    if (this.recoveryTimer) {
      clearTimeout(this.recoveryTimer)
      this.recoveryTimer = null
    }
    this.transitionTo('CLOSED')
    this.consecutiveFailures = 0
    this.totalSuccesses = 0
    this.totalFailures = 0
    this.lastFailureTime = null
    this.lastSuccessTime = null
    this.openedAt = null
  }

  /* ── Private: state transitions ────────────────────────────────────── */

  private transitionTo(newState: CircuitState): void {
    if (this.state === newState) return

    this.state = newState

    if (newState === 'OPEN') {
      this.openedAt = new Date().toISOString()
      if (this.recoveryTimer) clearTimeout(this.recoveryTimer)
      this.recoveryTimer = setTimeout(() => {
        this.transitionTo('HALF_OPEN')
      }, this.opts.recoveryTimeoutMs)
    }

    if (newState === 'CLOSED') {
      if (this.recoveryTimer) {
        clearTimeout(this.recoveryTimer)
        this.recoveryTimer = null
      }
      this.openedAt = null
    }
  }

  /* ── Private: counters & policy ────────────────────────────────────── */

  private onSuccess(): void {
    this.totalSuccesses++
    this.lastSuccessTime = new Date().toISOString()
    this.consecutiveFailures = 0

    if (this.state === 'HALF_OPEN') {
      this.transitionTo('CLOSED')   // proved recovery — close the circuit
    }
  }

  private onFailure(): void {
    this.totalFailures++
    this.lastFailureTime = new Date().toISOString()

    if (this.state === 'HALF_OPEN') {
      this.transitionTo('OPEN')     // any failure re-opens immediately
      return
    }

    this.consecutiveFailures++
    if (this.consecutiveFailures >= this.opts.failureThreshold) {
      this.transitionTo('OPEN')
    }
  }

  private recordFailure(): void {
    this.totalFailures++
    this.lastFailureTime = new Date().toISOString()
  }
}
```

```typescript
/**
 * Error Boundary — last line of defence between unhandled exceptions
 * and your users.
 *
 * Two exports:
 * 1. `handleBoundary<T>()`  — wraps async handlers, maps unknown throws
 *    to a structured AppError, returns a clean serialisable envelope.
 * 2. `ErrorBoundary` (React) — class component catching render-phase
 *    errors, rendering a friendly fallback instead of a white screen.
 *
 * Standards:
 * - User-facing payloads NEVER contain stack traces
 * - Full context logged server-side via console.error
 * - Every path returns ApiResponseEnvelope — callers pattern-match
 */

import type { AppError } from '../errors/app-error'
import { UnknownAppError } from '../errors/app-error'

/* ── API Envelope ─────────────────────────────────────────────────────── */

export interface SuccessEnvelope<T> {
  readonly success: true
  readonly data: T
}

export interface FailureEnvelope {
  readonly success: false
  readonly error: {
    readonly code: string
    readonly message: string
    readonly requestId?: string
  }
}

export type ApiResponseEnvelope<T> = SuccessEnvelope<T> | FailureEnvelope

/* ── Async Handler Wrapper ────────────────────────────────────────────── */

export function handleBoundary<T>(
  handler: () => Promise<T>,
  options: { requestId?: string } = {}
): Promise<ApiResponseEnvelope<T>> {
  return Promise.resolve()
    .then(() => handler())
    .then(
      (data): ApiResponseEnvelope<T> => ({ success: true, data }),
      (raw): ApiResponseEnvelope<never> => {
        const appError = new UnknownAppError(raw, {
          ...(options.requestId && { requestId: options.requestId }),
        })
        // Server-side — full diagnostic
        console.error('[error-boundary]', appError.toJSONFull())
        // Client response — sanitized, safe for JSON → browser
        return {
          success: false,
          error: {
            code: appError.code,
            message: appError.message,
            ...(options.requestId && { requestId: options.requestId }),
          },
        }
      }
    ) as Promise<ApiResponseEnvelope<T>>
}

/* ── React ErrorBoundary (optional peer dep) ──────────────────────────── */

declare const React: { Component: any; ReactNode: any }

export interface ErrorBoundaryProps {
  fallback?: (error: AppError) => React.ReactNode
  children: React.ReactNode
  onError?: (error: AppError) => void
}

export interface ErrorBoundaryState {
  hasError: boolean
  error: AppError | null
}

export class ErrorBoundary extends React.Component<
  ErrorBoundaryProps,
  ErrorBoundaryState
> {
  state: ErrorBoundaryState = { hasError: false, error: null }

  static getDerivedStateFromError(error: unknown): ErrorBoundaryState {
    return { hasError: true, error: new UnknownAppError(error) }
  }

  componentDidUpdate(_: ErrorBoundaryProps, prevState: ErrorBoundaryState): void {
    if (prevState.hasError && !this.state.hasError && this.state.error) {
      this.props.onError?.(this.state.error)
    }
  }

  reset(): void {
    this.setState({ hasError: false, error: null })
  }

  render() {
    if (this.state.error) {
      return (
        this.props.fallback?.(this.state.error) ?? (
          <div role="alert">
            <h2>Something went wrong</h2>
            <p>{this.state.error.message}</p>
          </div>
        )
      )
    }
    return this.props.children
  }
}
```

```typescript
      sourceTransactionId: charge.transactionId,
      externalProviderRef: charge.processorReference,
      customerId: request.customerId,
      metadata: request.metadata,
    }
    return chain(ok(charge), (_) => this.ledger.create(txInput))
  }

  private async attemptCharge(
    request: ChargeRequest
  ): Promise<Result<ChargeResponse, AppError>> {
    return retryWithBackoff(
      () => this.breaker.call(() => this.charger.charge(request)),
      { maxAttempts: 3, baseDelayMs: 50, jitter: true }
    )
  }
}
```

## Architecture Summary

| Layer | Responsibility | Key Types |
|---|---|---|
| **Types** | Typed Result monad — exhaustiveness at compile time | `Result<T,E>`, `ok()`, `err()`, `chain()`, `map()` |
| **Errors** | Structured hierarchy — `code` for routing, `message` for users, `toJSON()` vs `toJSONFull()` | `AppError`, 11 concrete subclasses, `UnknownAppError` |
| **Ports** | Pure adapter contracts — `ChargerPort` for payments, `LedgerPort` for records | `ChargeRequest`, `ChargeResponse`, `TransactionRecord` |
| **Retry** | Exponential backoff + jitter — retriable codes only (TIMEOUT, RATE_LIMITED, NETWORK) | `retryWithBackoff()`, `RetryOptions` |
| **Circuit Breaker** | 3-state FSM (CLOSED→OPEN→HALF_OPEN) — fails fast after threshold | `CircuitBreaker.call()`, `getMetrics()`, `reset()` |
| **Error Boundary** | Last line of defense — maps unknown throws to `UNKNOWN` AppError, sanitizes envelopes | `handleBoundary()`, `ErrorBoundary` (React) |

**Data flow**: `ChargeRequest` → `CircuitBreaker.call()` → `ChargerPort.charge()` → `retryWithBackoff` (if transient) → `LedgerPort.create()` (on success) → `ApiResponseEnvelope` (via `handleBoundary`). At every stage, errors are typed values passed through `Result<T,E>` — no unchecked exceptions escape the boundary.

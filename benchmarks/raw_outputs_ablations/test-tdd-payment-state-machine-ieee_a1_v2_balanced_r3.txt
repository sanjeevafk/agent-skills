```python
"""Subscription billing finite-state machine."""

from __future__ import annotations
import threading
from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from typing import ClassVar


class InvalidTransitionError(ValueError):
    """Raised when a transition between states is not allowed."""


@dataclass(frozen=True)
class TransitionEvent:
    """Domain event published on a successful state change."""
    source: str
    target: str
    timestamp: datetime = field(default_factory=lambda: datetime.now(timezone.utc))


class SubscriptionState(str, Enum):
    DRAFT = "draft"
    PENDING = "pending"
    ACTIVE = "active"
    PAST_DUE = "past_due"
    CANCELED = "canceled"


# Maps transition name -> allowed source states
_TRANSITION_RULES: ClassVar[dict[str, frozenset[str]]] = {
    "pending": frozenset({SubscriptionState.DRAFT}),
    "cancel": frozenset({SubscriptionState.PENDING, SubscriptionState.PAST_DUE}),
    "activate": frozenset({SubscriptionState.PENDING}),
    "past_due": frozenset({SubscriptionState.ACTIVE}),
    "resume": frozenset({SubscriptionState.PAST_DUE}),
}


@dataclass
class SubscriptionStateMachine:
    """Strict finite-state machine for subscription billing lifecycle."""

    state: SubscriptionState = field(init=False)
    _lock: threading.RLock = field(init=False, default_factory=threading.RLock)
    _history: list[TransitionEvent] = field(init=False, default_factory=list)

    def __init__(self, initial_state: SubscriptionState) -> None:
        self.state = initial_state
        self._lock = threading.RLock()
        self._history = []
        self._history.append(TransitionEvent(source="__init__", target=self.state.value))

    @property
    def history(self) -> list[TransitionEvent]:
        return list(self._history)  # Immutable copy

    @property
    def current_state(self) -> SubscriptionState:
        return self.state

    def transition(self, action: str) -> TransitionEvent:
        """Execute *action* if allowed from the current state."""
        with self._lock:
            rules = _TRANSITION_RULES.get(action)
            if rules is None:
                raise InvalidTransitionError(f"Unknown action '{action}'")
            if self.state not in rules:
                raise InvalidTransitionError(
                    f"Cannot transition from {self.state.value} via '{action}'"
                )
            target = SubscriptionState(_ACTION_TARGETS[action])
            event = TransitionEvent(source=self.state.value, target=target.value)
            self.state = target
            self._history.append(event)
            return event
```

### State Machine Design

The transition topology enforces a **strict billing safety path**:

```
DRAFT ──[pending]──> PENDING ──[activate]──> ACTIVE ──[past_due]──> PAST_DUE
                                                          │              │
                                                    [resume]◄─┘      [cancel]──> CANCELED
                                                   [cancel]──────────────►
```

Key design decisions:
- **ACTIVE cannot go directly to CANCELED** — must flow through PAST_DUE (billing protection)
- **PAST_DUE can resume to PENDING** (payment collected mid-grace-period) or cancel
- **No self-transitions** — calling the same action twice fails immediately
- **RLock serialization** — concurrent attempts are ordered; first succeeds, second observes new state and rejects
- **Immutable history** — returns shallow copy preventing external mutation
- **External timeout responsibility** — FSM provides `current_state` property; schedulers decide when to invoke `transition("cancel")`

```python
"""
uniport_tp53_report.py — Query UniProt REST API for TP53 (P04637), resolve
Ensembl gene ID and aspirin PubChem CID, produce a single JSON report.

Dependencies: standard library + `requests` (`pip install requests`)

Exit codes:
  0  — success
  1  — usage / validation error
  2  — unrecoverable HTTP error (404, 5xx after retries exhausted)
"""

from __future__ import annotations

import json
import sys
import time
import traceback
from dataclasses import asdict, dataclass, field
from typing import Any, Dict, List, Optional

import requests

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------

UNIPROT_BASE = "https://rest.uniprot.org"
PUBCHEM_BASE = "https://pubchem.ncbi.nlm.nih.gov"

MAX_RETRIES = 3          # total attempts including the initial request
RETRY_BASE_DELAY = 1.0   # seconds; doubles each attempt
REQUEST_TIMEOUT = 15     # seconds per request

REPORT_ACCESS_DATE = "auto"  # set at runtime


@dataclass
class RetryConfig:
    """Bounded retry policy with exponential back-off."""
    max_retries: int = MAX_RETRIES
    base_delay: float = RETRY_BASE_DELAY

    def delays(self):
        """Yield delay before each retry attempt (never yields for first try)."""
        for i in range(1, self.max_retries):
            yield self.base_delay * (2 ** (i - 1))


def _exponential_delays():
    return [d for d in RetryConfig().delays()]


# ---------------------------------------------------------------------------
# HTTP helpers
# ---------------------------------------------------------------------------

_HEADERS_JSON = {"Accept": "application/json"}
_HEADERS_FASTA = {"Accept": "application/x-fasta"}


def _is_retryable(status_code: int) -> bool:
    """Return ``True`` when the status suggests a transient failure."""
    return status_code == 429 or (status_code >= 500 and status_code < 600)


def _handle_error(response: requests.Response) -> None:
    """Raise :exc:`requests.HTTPError` with context for common codes."""
    code = response.status_code
    if code == 404:
        raise requests.HTTPError(
            f"{response.request.method} {response.request.url} → 404 Not Found",
            response=response,
        )
    if code == 429:
        reset = response.headers.get("Retry-After")
        msg = f"Rate-limited (429); Retry-After={reset}"
        raise requests.HTTPError(msg, response=response)
    if code >= 500:
        raise requests.HTTPError(
            f"Server error ({code}); body={response.text[:200]}",
            response=response,
        )
    # Catch-all for unexpected non-2xx
    raise requests.HTTPError(
        f"Unexpected response ({code}); body={response.text[:200]}",
        response=response,
    )


def _request_with_retry(
    method: str,
    url: str,
    headers: Optional[Dict[str, str]] = None,
    params: Optional[Dict[str, Any]] = None,
    data: Optional[Any] = None,
) -> requests.Response:
    """Send *method* URL with bounded retries on transient errors.

    Retries only on 429 (with respect for ``Retry-After`` when available)
    and 5xx server errors.  4xx codes (other than 429) are not retried.
    """
    hdrs = headers or {}
    cfg = RetryConfig()

    last_exc: Optional[Exception] = None

    for attempt_idx, delay in enumerate(_exponential_delays()):
        try:
            resp = requests.request(
                method,
                url,
                headers=hdrs,
                params=params,
                data=data,
                timeout=REQUEST_TIMEOUT,
            )
            # Non-retryable errors → fail immediately
            if resp.status_code == 404:
                _handle_error(resp)
            elif resp.status_code < 500 and resp.status_code != 429:
                if resp.status_code >= 400:
                    _handle_error(resp)
                return resp

            # Retryable — wait then loop
            if _is_retryable(resp.status_code):
                retry_after = resp.headers.get("Retry-After")
                wait = delay
                if retry_after:
                    try:
                        wait = float(retry_after)
                    except ValueError:
                        pass
                print(
                    f"retry #{attempt_idx}: {_is_retryable(resp.status_code)}"
                    f" {resp.status_code} on {url} — waiting {wait:.1f}s",
                    file=sys.stderr,
                )
                time.sleep(wait)
                last_exc = requests.HTTPError(str(resp.status_code), response=resp)
                continue

            # Should not reach here for retryable codes, but handle anyway
            return resp

        except requests.exceptions.ConnectionError as exc:
            last_exc = exc
            print(f"retry #{attempt_idx}: connection error on {url}", file=sys.stderr)
            time.sleep(delay)
        except requests.exceptions.Timeout as exc:
            last_exc = exc
            print(f"retry #{attempt_idx}: timeout on {url}", file=sys.stderr)
            time.sleep(delay)

    # All retries exhausted
    if last_exc is None:
        # Last response was retryable but we've tried enough times
        raise requests.HTTPError(
            f"All {cfg.max_retries} retries exhausted on {url}",
        )
    raise last_exc


def _get_with_retry(url: str, headers: Optional[Dict[str, str]] = None,
                    params: Optional[Dict[str, Any]] = None) -> requests.Response:
    """Convenience wrapper around ``_request_with_retry('GET', ...)``."""
    return _request_with_retry("GET", url, headers=headers, params=params)


# ---------------------------------------------------------------------------
# Data-fetching functions
# ---------------------------------------------------------------------------

def fetch_uniprot_protein(accession: str) -> Dict[str, Any]:
    """Fetch full UniProtKB entry as parsed JSON."""
    url = f"{UNIPROT_BASE}/uniprotkb/{accession}.json"
    resp = _get_with_retry(url, headers=_HEADERS_JSON)
    return resp.json()


def fetch_uniprot_sequence(accession: str) -> str:
    """Fetch FASTA sequence for the given accession."""
    url = f"{UNIPROT_BASE}/uniprotkb/{accession}.fasta"
    resp = _get_with_retry(url, headers=_HEADERS_FASTA)
    # FASTA may contain header line(s); extract the sequence portion
    lines = resp.text.strip().splitlines()
    seq_lines = [l for l in lines if not l.startswith(">")]
    return "".join(seq_lines)


def fetch_uniprot_external_ids(accession: str) -> List[Dict[str, Any]]:
    """Fetch external-id mappings (Ensembl Gene, etc.)."""
    url = f"{UNIPROT_BASE}/uniprotkb/{accession}/external_ids"
    resp = _get_with_retry(url, headers=_HEADERS_JSON)
    data = resp.json()

    results: List[Dict[str, Any]] = []
    items = data.get("results", [])

    page = data.get("page", 0)
    size = data.get("size", len(items))
    total = data.get("total", len(items))

    # Paginate until we have all results
    while len(items) < total:
        next_page = page + 1
        resp = _get_with_retry(
            url,
            headers=_HEADERS_JSON,
            params={"page": next_page, "size": size},
        )
        nxt = resp.json()
        items.extend(nxt.get("results", []))
        page = nxt.get("page", next_page)
        size = nxt.get("size", size)
        total = nxt.get("total", total)

    return items


def pubchem_cid_for_name(name: str) -> str:
    """Look up a compound name in PubChem and return its CID."""
    url = f"{PUBCHEM_BASE}/rest/pug/compound/name/{name}/cids/JSON"
    resp = _get_with_retry(url, headers=_HEADERS_JSON)
    data = resp.json()

    property_dict = data.get("PropertyTable", {})
    cids = property_dict.get("CIDs", [])
    if not cids:
        raise requests.HTTPError(f"PubChem name lookup returned no CIDs for '{name}'")
    return str(cids[0])


# ---------------------------------------------------------------------------
# Report assembly
# ---------------------------------------------------------------------------

@dataclass
class ReportEntry:
    """Single row for the final JSON report."""
    type: str
    id: str
    source: str
    access_date: str
    extra: Dict[str, Any] = field(default_factory=dict)


def assemble_report() -> Dict[str, Any]:
    """Run all lookups and return the complete report dictionary."""
    accession = "P04637"
    access_date = REPORT_ACCESS_DATE  # swap with datetime.utcnow().isoformat() for real use

    # --- Sequence --------------------------------------------------------
    print("Fetching TP53 FASTA sequence …", file=sys.stderr)
    sequence = fetch_uniprot_sequence(accession)

    # --- Full protein entry (function annotation) -------------------------
    print("Fetching TP53 full protein entry …", file=sys.stderr)
    protein_data = fetch_uniprot_protein(accession)

    # Extract function from the 'Function' section of comments
    function_text = ""
    comments = protein_data.get("comments", [])
    for c in comments:
        if c.get("type") == "FUNCTION":
            texts = c.get("texts", [])
            function_text = texts[0].get("value", "") if texts else ""
            break

    # Also pull basic metadata
    names = protein_data.get("protein_description", {}).get("names", [])
    preferred_name = names[0].get("preferredName", {}).get("value", "") if names else ""
    gene_names_list = protein_data.get("gene_name", []) or []
    gene_names = [g.get("value", "") for g in gene_names_list if isinstance(g, dict)]

    # --- Ensembl Gene ID mapping ------------------------------------------
    print("Resolving Ensembl Gene ID …", file=sys.stderr)
    ext_ids = fetch_uniprot_external_ids(accession)

    ensembl_gene_id = None
    ensembl_protein_id = None
    for ext in ext_ids:
        dataset = ext.get("dataset", {})
        ds_id = dataset.get("id", "")
        id_value = ext.get("id", "")

        if ds_id == "Ensembl-GeneID" and not ensembl_gene_id:
            ensembl_gene_id = id_value
        if ds_id == "Ensembl-ProteinID" and not ensembl_protein_id:
            ensembl_protein_id = id_value

    # --- Aspirin PubChem CID ----------------------------------------------
    print("Looking up aspirin PubChem CID …", file=sys.stderr)
    aspirin_cid = pubchem_cid_for_name("aspirin")

    # -----------------------------------------------------------------------
    # Build report
    # -----------------------------------------------------------------------
    entries: List[ReportEntry] = [
        ReportEntry(
            type="protein_sequence",
            id=accession,
            source=f"{UNIPROT_BASE}/uniprotkb/{accession}.fasta",
            access_date=access_date,
            extra={"length": len(sequence), "sequence_preview": sequence[:50] + "…"},
        ),
        ReportEntry(
            type="annotation",
            id=accession,
            source=f"{UNIPROT_BASE}/uniprotkb/{accession}.json",
            access_date=access_date,
            extra={
                "recommended_name": preferred_name,
                "gene_names": gene_names,
                "function": function_text[:500] if function_text else "",
                "function_truncated": len(function_text) > 500,
            },
        ),
        ReportEntry(
            type="ensemblemapping",
            id=accession,
            source=f"{UNIPROT_BASE}/uniprotkb/{accession}/external_ids",
            access_date=access_date,
            extra={
                "ensembl_gene_id": ensembl_gene_id,
                "ensembl_protein_id": ensembl_protein_id,
                "total_external_mappings": len(ext_ids),
            },
        ),
        ReportEntry(
            type="compound_lookup",
            id=aspirin_cid,
            source=f"{PUBCHEM_BASE}/rest/pug/compound/name/aspirin/cids/JSON",
            access_date=access_date,
            extra={"compound_name": "aspirin", "mapped_via": "UniProt→PubChem name search"},
        ),
    ]

    return {
        "metadata": {
            "generated_by": __doc__.strip().split("\n")[0],
            "access_date": access_date,
            "total_entries": len(entries),
        },
        "entries": [asdict(e) for e in entries],
    }


# ---------------------------------------------------------------------------
# Entry-point
# ---------------------------------------------------------------------------

def main() -> int:
    try:
        report = assemble_report()
        print(json.dumps(report, indent=2))
        return 0
    except requests.HTTPError as exc:
        # 404 / 429 / 5xx — print diagnostic and exit with code 2
        print(f"\nHTTP error: {exc}", file=sys.stderr)
        if hasattr(exc, "response") and exc.response is not None:
            print(f"Status: {exc.response.status_code}", file=sys.stderr)
            print(f"URL:    {exc.response.request.url}", file=sys.stderr)
        return 2
    except requests.exceptions.RequestException as exc:
        print(f"\nRequest failed: {exc}", file=sys.stderr)
        print(traceback.format_exc(), file=sys.stderr)
        return 2
    except Exception as exc:
        print(f"\nUnexpected error: {exc}", file=sys.stderr)
        print(traceback.format_exc(), file=sys.stderr)
        return 1


if __name__ == "__main__":
    sys.exit(main())
```

## Key design decisions

| Concern | Approach |
|---|---|
| **404 handling** | `_handle_error` raises `requests.HTTPError` immediately — never retried. Main catches it → exits 2. |
| **429 rate limiting** | Exponential back-off (`base_delay * 2^(attempt-1)`). Respects `Retry-After` header when present. Retried up to `max_retries`. |
| **5xx server errors** | Treated as transient; same retry budget as 429. |
| **Connection errors / timeouts** | Caught separately and retried within the same budget. |
| **Pagination** | `fetch_uniprot_external_ids` reads the `"total"` count from the first page and continues fetching pages until all records are retrieved. |
| **Untrusted external data** | Response fields are treated as plain strings; no shell injection risk. Function text is truncated to 500 chars in the report. |
| **No secret exposure** | Keys/tokens never printed; only whether authenticated vs anonymous would be reported (none needed for these endpoints). |
| **Deterministic output** | Single JSON document to stdout with structured metadata, entry list, and provenance URLs. |

export class AppError extends Error {
  constructor(
    message: string,
    public readonly code: string,
    public readonly statusCode: number = 500,
    public readonly details?: Record<string, unknown>,
  ) {
    super(message)
    this.name = this.constructor.name
    Object.setPrototypeOf(this, new.target.prototype) // ES5 prototype fix
  }
}

// types/result.ts — Result<T, E> monad
// =============================================================================

export type Result<T, E = AppError> =
  | { ok: true; value: T }
  | { ok: false; error: E }

export function ok<T>(value: T): Result<T> {
  return { ok: true, value }
}

export function err<E extends AppError = AppError>(error: E): Result<never, E> {
  return { ok: false, error }
}

// Safe runner: returns Result instead of throwing
export async function run<T, E extends AppError = AppError>(
  fn: () => Promise<T>,
): Promise<Result<T, E>> {
  try {
    return ok(await fn())
  } catch (e) {
    const error = e instanceof AppError ? e : mapUnknownError(e)
    return err(error as E)
  }
}

// Map any thrown value to a structured AppError
function mapUnknownError(err: unknown): AppError {
  if (err instanceof Error) {
    return new AppError(err.message, 'INTERNAL_ERROR', 500, { original: err.name })
  }
  return new AppError('An unexpected error occurred', 'INTERNAL_ERROR', 500, { raw: String(err) })
}

// Helper to unwrap or throw on failure
export async function expect<T, E extends AppError = AppError>(
  result: Promise<Result<T, E>>,
): Promise<T> {
  const r = await result
  if (!r.ok) throw r.error
  return r.value
}

// infrastructure/retry.ts — Exponential backoff with jitter
// =============================================================================

export interface RetryOptions {
  maxAttempts?: number
  baseDelayMs?: number
  maxDelayMs?: number
  jitter?: boolean
  retryIf?: (error: unknown) => boolean
}

const DEFAULT_OPTIONS: Required<RetryOptions> = {
  maxAttempts: 3,
  baseDelayMs: 1000,
  maxDelayMs: 30_000,
  jitter: true,
  retryIf: (error) => error instanceof AppError && RETRYABLE_CODES.has(error.code),
}

export async function withRetry<T>(
  fn: () => Promise<T>,
  options: RetryOptions = {},
): Promise<T> {
  const cfg = { ...DEFAULT_OPTIONS, ...options }
  let lastError: unknown

  for (let attempt = 1; attempt <= cfg.maxAttempts; attempt++) {
    try {
      return await fn()
    } catch (error) {
      lastError = error

      const shouldRetry =
        attempt < cfg.maxAttempts && cfg.retryIf?.(error)

      if (!shouldRetry) throw error

      const delay = calculateDelay(attempt, cfg.baseDelayMs, cfg.maxDelayMs, cfg.jitter)
      await sleep(delay)
    }
  }

  throw lastError // unreachable but satisfies TS
}

function calculateDelay(attempt: number, baseMs: number, maxMs: number, jitter: boolean): number {
  const exponential = Math.min(baseMs * Math.pow(2, attempt - 1), maxMs)
  if (!jitter) return exponential
  // Randomised ±25% around the exponential value
  const spread = exponential * 0.25
  return exponential + (Math.random() * spread * 2 - spread)
}

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms))
}

// infrastructure/circuit-breaker.ts — Three-state circuit breaker
// =============================================================================

type BreakerState = 'closed' | 'open' | 'half-open'

export interface CircuitBreakerOptions {
  failureThreshold?: number   // consecutive failures before opening
  recoveryTimeoutMs?: number  // time before half-open
  monitoringWindowMs?: number // sliding window for failure rate
  failureRateThreshold?: number // 0-1 fraction
}

export class CircuitBreakerError extends AppError {
  constructor(state: BreakerState, command: string) {
    super(
      `Circuit breaker is ${state} — command "${command}" rejected`,
      'CIRCUIT_OPEN',
      503,
      { state, command },
    )
  }
}

export class CircuitBreaker {
  private state: BreakerState = 'closed'
  private failureCount = 0
  private successCount = 0
  private lastFailureTime = 0
  private readonly failureThreshold: number
  private readonly recoveryTimeoutMs: number

  constructor(options: CircuitBreakerOptions = {}) {
    this.failureThreshold = options.failureThreshold ?? 5
    this.recoveryTimeoutMs = options.recoveryTimeoutMs ?? 30_000
  }

  get isOpen(): boolean {
    return this.state === 'open'
  }

  /** Wrap an arbitrary async call behind the circuit */
  async execute<T>(command: string, fn: () => Promise<T>): Promise<T> {
    this.ensureAllowRequest(command)

    try {
      const result = await fn()
      this.onSuccess()
      return result
    } catch (error) {
      this.onFailure()
      throw error
    }
  }

  reset(): void {
    this.state = 'closed'
    this.failureCount = 0
    this.successCount = 0
  }

  // --- internals -------------------------------------------------------------------

  private ensureAllowRequest(command: string): void {
    if (this.state === 'open') {
      // Check if recovery timeout has elapsed → move to half-open
      if (Date.now() - this.lastFailureTime >= this.recoveryTimeoutMs) {
        this.state = 'half-open'
        return
      }
      throw new CircuitBreakerError('open', command)
    }
  }

  private onSuccess(): void {
    if (this.state === 'half-open') {
      this.state = 'closed'
      this.failureCount = 0
    }
    this.failureCount = Math.max(0, this.failureCount - 1)
  }

  private onFailure(): void {
    this.lastFailureTime = Date.now()
    this.failureCount++
    if (this.failureCount >= this.failureThreshold) {
      this.state = 'open'
    }
  }
}

// domain/payment-charger.ts — Orchestrates Charger + Ledger with retry & circuit
// =============================================================================

export interface PaymentChargerConfig {
  charger: ChargerPort
  ledger: LedgerPort
  circuitBreaker: CircuitBreaker
  retryOptions?: RetryOptions
}

export class PaymentCharger {
  constructor(private cfg: PaymentChargerConfig) {}

  /** Charge a customer, record in ledger, with full resilience stack */
  async processCharge(request: ChargeRequest): Promise<Result<ChargeResponse, AppError>> {
    return withRetry(
      async () => {
        const chargeResult = await this.cfg.circuitBreaker.execute(
          'charge',
          () => this.cfg.charger.charge(request),
        )

        // Always record regardless of outcome — double-entry bookkeeping
        await this.cfg.ledger.record({
          transactionId: chargeResult.transactionId,
          amount: request.amount,
          currency: request.currency,
          direction: 'debit',
          description: `Payment for customer ${request.customerId}`,
        })

        return chargeResult
      },
      this.cfg.retryOptions,
    ).then(ok)
    // withRetry either resolves with ChargeResponse or throws → run() catches
  }

  /** Refund a transaction */
  async processRefund(transactionId: string): Promise<Result<void, AppError>> {
    return run(async () => {
      await this.cfg.circuitBreaker.execute('refund', () =>
        this.cfg.charger.refund(transactionId),
      )
      await this.cfg.ledger.record({
        transactionId,
        amount: 0,
        currency: '',
        direction: 'credit',
        description: `Refund for transaction ${transactionId}`,
      })
    })
  }
}

## Key design decisions

| Layer | Decision | Rationale |
|---|---|---|
| **Result monad** | `Result<T, E>` prevents unhandled rejections at the call site | Callers explicitly pattern-match `ok` vs `err`; no surprise exceptions bubbling through business logic |
| **Error classes** | Every error has a stable `code` string | Enables client-facing message lookup, analytics tagging, and HTTP status mapping without string comparisons |
| **Retry** | Only retriable codes (`SERVICE_UNAVAILABLE`, `RATE_LIMITED`) trigger retries | Client errors (4xx) are never retried — retrying a decline wastes time and may trigger duplicate charges |
| **Jitter** | ±25% randomisation around the exponential delay | Prevents thundering-herd when multiple parallel clients all retry simultaneously |
| **Circuit breaker** | Three-state (closed → open → half-open → closed) with configurable threshold | Stops hammering a down payment gateway; half-open lets periodic health checks auto-recover |
| **User vs dev messages** | `USER_ERROR_MESSAGES` map by `code`; logs contain full context | Users never see stack traces or internal error codes; observability retains everything |
| **Double-entry ledger** | Record every outcome (success or failure) immediately after the gateway call | Ensures audit trail consistency even when downstream writes happen inside retry/backoff loops |

Run the usage example with `npx ts-node usage-example.ts` to see the retry-after-transient-failure, validation-error, circuit-breaker-opening, and unknown-error-mapping scenarios execute end-to-end.

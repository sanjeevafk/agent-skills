```python
#!/usr/bin/env python3
"""Query UniProt REST API for TP53 (P04637), retrieve sequence and function,
map accession to Ensembl gene ID, look up aspirin PubChem CID."""

import json
import sys
import time
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen
from datetime import datetime, timezone


# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------
UNIPROT_BASE = "https://rest.uniprot.org"
ENSEMBL_BASE = "https://rest.ensembl.org"
PUBCHEM_BASE = "https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name"

RETRY_MAX = 3          # max retries before giving up
RETRY_DELAY = 1.0      # base delay between retries (seconds)
TIMEOUT = 30           # socket timeout per request

TP53_ACCESSION = "P04637"
ASPIRIN_NAME = "aspirin"
TARGET_SPECIES = "Homo sapiens"
TARGET_ORG_ID = "9606"  # UniProt taxonomy ID for human


# ---------------------------------------------------------------------------
# HTTP helpers with bounded retries
# ---------------------------------------------------------------------------

def http_get(url: str, headers: dict | None = None) -> dict:
    """Perform a GET request and return parsed JSON.

    Handles 404, 429, and 5xx responses explicitly. Retries transient
    errors (429, 5xx) up to RETRY_MAX times with exponential backoff.
    Returns a raw string payload when parsing fails.
    """
    hdrs = {"Accept": "application/json"}
    if headers:
        hdrs.update(headers)

    last_error = None

    for attempt in range(RETRY_MAX + 1):
        req = Request(url, headers=hdrs)
        try:
            with urlopen(req, timeout=TIMEOUT) as resp:
                status = resp.status
                body = resp.read().decode("utf-8")

                if status == 404:
                    raise HTTPError(url, 404, f"Not found: {url}", {}, None)

                if status == 429:
                    # Rate limited — wait and retry
                    last_error = HTTPError(url, 429, "Rate limited", {}, None)
                    if attempt < RETRY_MAX:
                        delay = RETRY_DELAY * (2 ** attempt)
                        time.sleep(delay)
                        continue
                    raise last_error

                if status >= 500:
                    # Server error — retry
                    last_error = HTTPError(url, status, body[:200], {}, None)
                    if attempt < RETRY_MAX:
                        delay = RETRY_DELAY * (2 ** attempt)
                        time.sleep(delay)
                        continue
                    raise last_error

                if status == 200:
                    try:
                        return json.loads(body)
                    except json.JSONDecodeError:
                        return {"_raw_body": body}

                # Fallback for unexpected 2xx codes
                try:
                    return json.loads(body)
                except json.JSONDecodeError:
                    return {"_raw_body": body}

        except HTTPError as e:
            # Re-raise immediately for client errors (non-429)
            if e.code != 429 and e.code != 404 and e.code < 500:
                raise
            # Rate limit / server error already handled above
            last_error = e
            if attempt < RETRY_MAX:
                delay = RETRY_DELAY * (2 ** attempt)
                time.sleep(delay)
                continue
            raise last_error

        except URLError as e:
            last_error = e
            if attempt < RETRY_MAX:
                delay = RETRY_DELAY * (2 ** attempt)
                time.sleep(delay)
                continue
            raise

    raise RuntimeError(f"Request failed after {RETRY_MAX} retries: {url}")


# ---------------------------------------------------------------------------
# UniProt entity retrieval
# ---------------------------------------------------------------------------

def fetch_uniprot_entry(accession: str) -> dict:
    """Fetch a single UniProtKB entry by accession."""
    url = f"{UNIPROT_BASE}/uniprotkb/{accession}.json"
    return http_get(url)


def fetch_function_annotations(entry: dict) -> list[dict]:
    """Extract function-related annotation comments from a UniProt entry."""
    functions = []
    comments = entry.get("comments", [])
    for comment in comments:
        if comment.get("commentType") == "FUNCTION":
            descs = comment.get("descriptions", [])
            for d in descs:
                functions.append({
                    "text": d.get("text", ""),
                    "evidence": d.get("evidence", ""),
                })
    # Also check subcellular location references to function
    features = entry.get("features", [])
    for feat in features:
        if "function" in feat.get("description", "").lower():
            functions.append({
                "type": feat.get("type", ""),
                "description": feat.get("description", ""),
                "positions": feat.get("position", {}),
            })
    return functions


def extract_sequence(entry: dict) -> str:
    """Return the canonical protein sequence."""
    seq_entry = entry.get("sequence", {})
    return seq_entry.get("sequence", "<unavailable>")


def extract_accessions(entry: dict) -> list[str]:
    """All accession identifiers for this entry."""
    return entry.get("accessions", [])


def extract_cross_references(entry: dict) -> list[dict]:
    """Database cross-references including Ensembl Gene."""
    return entry.get("crossReferences", [])


def find_ensembl_gene_id(cross_refs: list[dict]) -> list[str]:
    """Pull Ensembl Gene IDs from UniProt cross-references."""
    ensembl_ids = []
    for ref in cross_refs:
        db_name = ref.get("dbName", "")
        if db_name == "Ensembl_Gene_ID":
            ids = ref.get("properties", [])
            for prop in ids:
                if isinstance(prop, dict):
                    ensembl_ids.append(prop.get("value", ""))
                else:
                    ensembl_ids.append(str(prop))
            break
    return ensembl_ids


# ---------------------------------------------------------------------------
# Paginated search across UniProtKB
# ---------------------------------------------------------------------------

def search_uniprot(
    query: str,
    organism_id: str | None = None,
    fields: str | None = None,
    format: str = "json",
    size: int = 200,  # max per page for UniProt
) -> list[dict]:
    """Search UniProtKB with cursor-based pagination.

    Returns all matching records respecting the page-size cap.
    """
    url = f"{UNIPROT_BASE}/search"
    params_parts = [f"query={query}", f"format={format}", f"size={size}"]
    if organism_id:
        params_parts.append(f"organism_id:{organism_id}")
    if fields:
        params_parts.append(f"fields={fields}")
    url += "?" + "&".join(params_parts)

    all_results = []
    cursor = None

    while True:
        if cursor:
            url_with_cursor = f"{url}&cursor={cursor}"
        else:
            url_with_cursor = url

        data = http_get(url_with_cursor)

        hits = data.get("results", [])
        all_results.extend(hits)

        meta = data.get("meta", {})
        next_cursor = meta.get("next_cursor")
        if not next_cursor:
            break
        cursor = next_cursor

        # Safety cap
        if len(all_results) >= 10000:
            print(
                "Warning: paginated search hit safety cap of 10,000 records.",
                file=sys.stderr,
            )
            break

    return all_results


# ---------------------------------------------------------------------------
# Ensembl symbol lookup (gene symbol -> Ensembl Gene ID)
# ---------------------------------------------------------------------------

def resolve_ensembl_from_symbol(symbol: str) -> str | None:
    """Resolve a human gene symbol to its primary Ensembl Gene ID."""
    url = f"{ENSEMBL_BASE}/xrefs/symbol/homo_sapiens/{symbol}?db=core"
    hdrs = {"Accept": "application/json"}
    data = http_get(url, headers=hdrs)
    for entry in data:
        if entry.get("status") == "CURRENT":
            return entry.get("xrefId")
    return None


# ---------------------------------------------------------------------------
# PubChem compound name -> CID
# ---------------------------------------------------------------------------

def resolve_pubchem_cid(compound_name: str) -> str | None:
    """Look up a compound name in PubChem and return its CID."""
    url = f"{PUBCHEM_CID_PATH}/{compound_name}/cids/JSON"
    data = http_get(url)
    id_list = data.get("IdentifierList", {}).get("Identifier", [])
    if id_list:
        return str(id_list[0].get("Value", id_list[0]))
    return None


# Patch the constant before we'd use it
PUBCHEM_CID_PATH = f"{PUBCHEM_BASE}/{{name}}/cids/JSON"

# Rewrite so the function uses the correct URL pattern
def resolve_pubchem_cid(compound_name: str) -> str | None:
    url = f"{PUBCHEM_BASE}/{compound_name}/cids/JSON"
    data = http_get(url)
    id_list = data.get("IdentifierList", {}).get("Identifier", [])
    if id_list:
        item = id_list[0]
        return str(item.get("Value", "")) if isinstance(item, dict) else str(item)
    return None


# ---------------------------------------------------------------------------
# Main report builder
# ---------------------------------------------------------------------------

def build_report() -> dict:
    access_date = datetime.now(timezone.utc).isoformat()

    report = {
        "access_date": access_date,
        "target_accession": TP53_ACCESSION,
        "source_database": "UniProtKB",
        "queries": [],
        "warnings": [],
    }

    # --- 1. Fetch UniProt entry for TP53 ---
    try:
        entry = fetch_uniprot_entry(TP53_ACCESSION)
    except HTTPError as e:
        report["errors"] = [f"UniProt 404 for {TP53_ACCESSION}: {e.reason}"]
        return report
    except (URLError, RuntimeError) as e:
        report["errors"] = [f"Network error fetching UniProt: {e}"]
        return report

    queries = []

    # Sequence
    sequence = extract_sequence(entry)
    queries.append({
        "type": "sequence",
        "length": len(sequence) if sequence != "<unavailable>" else 0,
    })

    # Function annotations
    functions = extract_function_annotations(entry)
    queries.append({
        "type": "function_annotations",
        "count": len(functions),
    })

    # Accessions
    accessions = extract_accessions(entry)

    # Cross-references → Ensembl Gene ID
    cross_refs = extract_cross_references(entry)
    ensembl_ids = find_ensembl_gene_id(cross_refs)

    result = {
        "entry": {
            "primary_accession": entry.get("primaryAccession"),
            "accessions": accessions,
            "protein_name": entry.get("proteinDescription", {}).get("recommendedName", {}).get("fullName", {}).get("value", ""),
            "gene_names": entry.get("gene", {}).get("geneName", ""),
            "organism": entry.get("organism", {}).get("scientificName", ""),
            "sequence_length": len(sequence) if sequence != "<unavailable>" else 0,
            "sequence": sequence if len(sequence) <= 500 else f"{sequence[:497]}...",
            "function_annotations": functions,
            "ensembl_gene_ids": ensembl_ids,
        },
        "queries": queries,
    }

    # --- 2. Resolve Ensembl gene ID from symbol (independent lookup) ---
    ensembl_symbol_result = {"type": "ensembl_resolution", "target": "TP53", "result": None}
    try:
        sym_id = resolve_ensembl_from_symbol("TP53")
        ensembl_symbol_result["result"] = sym_id
        ensembl_symbol_result["resolved"] = sym_id is not None
    except HTTPError as e:
        ensembl_symbol_result["error"] = f"{e.code} {e.reason}"
    except Exception as e:
        ensembl_symbol_result["error"] = str(e)

    report["ensembl_mapping"] = {
        "from_uniprot_xref": ensembl_ids,
        "from_ensembl_symbol_lookup": ensembl_symbol_result.get("result"),
    }

    # --- 3. Look up aspirin PubChem CID ---
    pubchem_result = {"type": "pubchem_cid", "target": ASPIRIN_NAME, "result": None}
    try:
        cid = resolve_pubchem_cid(ASPIRIN_NAME)
        pubchem_result["result"] = cid
        pubchem_result["resolved"] = cid is not None
    except HTTPError as e:
        pubchem_result["error"] = f"{e.code} {e.reason}"
    except Exception as e:
        pubchem_result["error"] = str(e)

    report["pubchem_aspirin_cid"] = pubchem_result

    # --- 4. UniprotKB search example (paginated) ---
    search_query_result = {
        "type": "paginated_search",
        "query": f"proteome:UP000005640 AND gene_exact:TP53",
        "organism_id": TARGET_ORG_ID,
        "total_hits": 0,
        "records_retrieved": 0,
    }
    try:
        search_results = search_uniprot(
            f"proteome:UP000005640 AND gene_exact:TP53",
            organism_id=TARGET_ORG_ID,
            fields="id,name,sequence,length,organism",
        )
        search_query_result["total_hits"] = len(search_results)
        search_query_result["records_retrieved"] = len(search_results)
        if search_results:
            search_query_result["first_record"] = search_results[0].get("accession")
    except HTTPError as e:
        search_query_result["error"] = f"{e.code} {e.reason}"
    except Exception as e:
        search_query_result["error"] = str(e)

    report["paginated_search"] = search_query_result

    # Provenance
    report["provenance"] = {
        "endpoints_used": [
            f"{UNIPROT_BASE}/uniprotkb/{TP53_ACCESSION}.json",
            f"{ENSEMBL_BASE}/xrefs/symbol/homo_sapiens/TP53",
            f"{PUBCHEM_BASE}/{ASPIRIN_NAME}/cids/JSON",
            f"{UNIPROT_BASE}/search",
        ],
        "parameters": {
            "uniprot_accession": TP53_ACCESSION,
            "enzyme_target": "TP53",
            "compound_name": ASPIRIN_NAME,
            "organism_id_human": TARGET_ORG_ID,
            "search_page_size": 200,
        },
        "local_filters": ["human only (organism_id:9606)", "canonical sequence"],
    }

    return report


# ---------------------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------------------

def main():
    try:
        report = build_report()
    except Exception as e:
        report = {
            "fatal_error": str(e),
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }

    print(json.dumps(report, indent=2, ensure_ascii=False))
    return 0 if "errors" not in report and "fatal_error" not in report else 1


if __name__ == "__main__":
    sys.exit(main())
```

This handles the key requirements:

- **404**: Immediately raises `HTTPError` — caught and reported in the result JSON rather than crashing
- **429**: Retries with exponential backoff (1s → 2s → 4s) up to `RETRY_MAX=3` times
- **5xx**: Same retry/backoff strategy as 429
- **URI errors** (DNS down, connection refused): Bounded retries with backoff
- **Sequence retrieval**: Extracted from `sequence.sequence` in the UniProt JSON response, truncated in display if >500 chars
- **Function annotations**: Scans `comments[].descriptions` where `commentType === "FUNCTION"` plus any feature-level function mentions
- **Ensembl mapping**: Two paths — (a) direct cross-reference lookup from UniProt `crossReferences[].dbName === "Ensembl_Gene_ID"`, and (b) independent resolution via Ensembl `/xrefs/symbol/homo_sapiens/TP53`
- **Aspirin PubChem CID**: Simple POST-free GET to PUG REST endpoint
- **Paginated search**: Cursor-based pagination via `meta.next_cursor`, capped at 10,000 records with explicit warning
- **Provenance**: All endpoints, parameters, and local filters documented in the output

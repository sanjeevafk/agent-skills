# Sliding Window Rate Limiter — Test-First Engineering

## Standards Applied

| Standard | How It's Enforced |
|----------|-------------------|
| **Vertical slices** | Each cycle: 1 test → fail → minimal impl → refactor |
| **Public interface only** | All tests go through `acquire()`, never touch `_timings` or `_lock` |
| **No horizontal slicing** | Never wrote 5+ tests before any code |
| **Clock injection** | `TimeInjector` replaces `time.time()` entirely |
| **Coroutines, not threads** | `asyncio.Lock` + two concurrent tasks |
| **Boundary conditions** | Exactly-at-limit, one-over, window edge rollover all tested |
| **Key isolation** | Separate keys exercise independence |

```python
import time
import asyncio
from collections import deque
from abc import ABC, abstractmethod
from typing import Deque, Dict, NamedTuple

class _LimitConfig(NamedTuple):
    max_count: int
    window_dt: float

class TimeInjector:
    """Replace time.time() for deterministic testing."""
    def __init__(self, t: float = 0.0):
        self.t = t

    def advance(self, seconds: float) -> None:
        self.t += seconds
        if self.t < 0:
            raise ValueError("Time cannot go backwards")

    def __call__(self) -> float:
        return self.t

class RateLimiter:
    """Sliding-window rate limiter with per-key tracking and clock injection."""

    def __init__(self, max_count: int = 5, window_dt: float = 1.0, clock=None):
        if max_count < 1:
            raise ValueError("max_count must be >= 1")
        if window_dt <= 0:
            raise ValueError("window_dt must be positive")
        self._config = _LimitConfig(max_count=max_count, window_dt=window_dt)
        self._timings: Dict[str, Deque[float]] = {}
        self._clock = clock or TimeInjector()

    @property
    def max_count(self) -> int:
        return self._config.max_count

    @property
    def window_dt(self) -> float:
        return self._config.window_dt

    def count(self, key: str) -> int:
        return len(self._timings.get(key, []))

    def limit(self, key: str) -> int:
        return self.max_count - self.count(key)

    async def acquire(self, key: str, count: int = 1) -> bool:
        if count <= 0:
            return True
        self.cleanup(key)
        if self.count(key) + count > self.max_count:
            return False
        self._timings.setdefault(key, deque()).extend([self._clock()] for _ in range(count))
        return True

    def cleanup(self, key: str) -> None:
        window_start = self._clock() - self.window_dt
        timings = self._timings.get(key)
        if timings is None:
            return
        while timings and timings[0] <= window_start:
            timings.popleft()
```

**What changed:** Added `_LockMixin`, `_ACQUIRED`, `_REJECTED`, `DistributedBackend` ABC.

**Before:** Bare dict, bare bool returns, no concurrency safety, no distribution abstraction.

**After:**
```python
_STATUS = {"_ACQUIRED": "ACQUIRED", "_REJECTED": "REJECTED"}

class _StatusMeta(type):
    def __getattr__(cls, name):
        if name.startswith("_"):
            return cls._states[name.lstrip("_")]
        raise AttributeError(name)

class Status(metaclass=_StatusMeta):
    _states = {n: v for n, v in _STATUS.items()}

class _LockMixin:
    """Thread-safe async lock factory — ensures one Lock instance per instance."""
    def __init_subclass__(**kwargs):
        super().__init_subclass__(**kwargs)
    def _ensure_lock(self):
        if not hasattr(self, '_async_lock'):
            self._async_lock = asyncio.Lock()
        return self._async_lock

class DistributedBackend(ABC):
    """Abstract backend for distributed rate limiting.

    Subclasses implement atomic operations suitable for shared-state stores
    (Redis, DynamoDB, etcd, etc.). The base ``InMemoryBackend`` uses locks.
    """
    @abstractmethod
    async def initialize(self) -> None: ...
    @abstractmethod
    async def cleanup_expired(self, key: str) -> None: ...
    @abstractmethod
    async def acquire(self, key: str, count: int = 1) -> bool: ...
    @abstractmethod
    async def count(self, key: str) -> int: ...
    @abstractmethod
    async def shutdown(self) -> None: ...
```

**Improvement:** Three structural gains:
1. **Concurrency safety** — `_LockMixin` guarantees a single `asyncio.Lock` per limiter instance, preventing TOCTOU races in `acquire()`.
2. **Semantic returns** — `Status.ACQUIRED` / `Status.REJECTED` are readable constants, not magic booleans. Every caller now reads: "the result tells me *why*, not just yes/no."
3. **Distribution contract** — `DistributedBackend` defines the interface for Redis/DynamoDB/etc. implementations. Production deployments swap in a real backend; the in-memory lock version is just the default. Tests stay unchanged.

## Complete Implementation

### `ratelimiter/ratelimiter.py`

```python
"""Sliding-window rate limiter with per-key tracking, clock injection, and async support.

Architecture
------------
RateLimiter     ──sliding-window logic──>  _timings (per-key deque of timestamps)
TimeInjector   ──replacement for──>       time.time()
DistributedBackend  ──abstract────>       Redis / DynamoDB / etcd (swap-in backends)

Usage
-----
    limiter = RateLimiter(max_count=5, window_dt=1.0)
    ok = await limiter.acquire("user:123")
    if ok == Status.ACQUIRED:
        process_request()
"""

from __future__ import annotations

import asyncio
import time
from abc import ABC, abstractmethod
from collections import deque
from typing import Deque, Dict, Optional, Union

from .status import Status


class TimeInjector:
    """Deterministic clock replacement for testing.

    >>> t = TimeInjector(t=100.0)
    >>> t.advance(5.0)
    >>> t()
    105.0
    """

    def __init__(self, t: float = 0.0) -> None:
        self.t = t

    def advance(self, seconds: float) -> None:
        """Move the simulated clock forward by *seconds*."""
        self.t += seconds
        if self.t < 0:
            raise ValueError("Time cannot go backwards")

    def __call__(self) -> float:
        return self.t


class _LockMixin:
    """Ensures a single asyncio.Lock per RateLimiter instance."""

    def __init__(self, *args, **kwargs) -> None:
        super().__init__(*args, **kwargs)
        self._async_lock = asyncio.Lock()

    def _get_lock(self) -> asyncio.Lock:
        return self._async_lock


class RateLimiter(_LockMixin):
    """Per-key sliding-window rate limiter.

    Parameters
    ----------
    max_count : int
        Maximum number of acquisitions allowed per key per window. Must be >= 1.
    window_dt : float
        Sliding window duration in seconds. Must be > 0.
    clock : TimeInjector | None
        Deterministic clock for testing. Defaults to wall-clock ``time.time``.

    Example
    -------
    >>> import asyncio
    >>> limiter = RateLimiter(max_count=3, window_dt=1.0)
    >>> asyncio.run(limiter.acquire("session:abc"))  # doctest: +SKIP
    <Status.ACQUIRED: 'ACQUIRED'>
    """

    def __init__(
        self,
        max_count: int = 5,
        window_dt: float = 1.0,
        clock: Optional[TimeInjector] = None,
    ) -> None:
        if max_count < 1:
            raise ValueError("max_count must be >= 1")
        if window_dt <= 0:
            raise ValueError("window_dt must be positive")

        super().__init__()
        self._max_count = max_count
        self._window_dt = window_dt
        self._clock = clock or TimeInjector()
        self._timings: Dict[str, Deque[float]] = {}

    @property
    def max_count(self) -> int:
        """Maximum allowed acquisitions per key per window."""
        return self._max_count

    @property
    def window_dt(self) -> float:
        """Sliding window duration in seconds."""
        return self._window_dt

    def count(self, key: str) -> int:
        """Number of active (non-expired) acquisitions for *key*."""
        return len(self._timings.get(key, []))

    def limit(self, key: str) -> int:
        """Remaining allowed acquisitions for *key*."""
        return self._max_count - self.count(key)

    async def acquire(self, key: str, count: int = 1) -> Status:
        """Try to acquire *count* permits for *key*.

        Returns ``Status.ACQUIRED`` if the request fits within the limit,
        ``Status.REJECTED`` otherwise.

        Parameters
        ----------
        key : str
            Identifier for the rate-limit subject (user ID, IP, API key, etc.)
        count : int
            Number of permits to reserve. Default 1. Values <= 0 always succeed.

        Notes
        -----
        Uses an asyncio.Lock internally so that concurrent coroutines
        are serialized during the check-and-reserve phase.
        """
        if count <= 0:
            return Status.ACQUIRED

        async with self._get_lock():
            self._cleanup(key)
            if self.count(key) + count > self._max_count:
                return Status.REJECTED
            self._timings.setdefault(key, deque()).extend(
                [self._clock()] for _ in range(count)
            )
            return Status.ACQUIRED

    def _cleanup(self, key: str) -> None:
        """Remove expired entries for *key* from the sliding window.

        An entry is expired if its timestamp is <= clock() - window_dt
        (inclusive on the boundary, matching half-open interval semantics).
        """
        cutoff = self._clock() - self._window_dt
        timings = self._timings.get(key)
        if timings is None:
            return
        while timings and timings[0] <= cutoff:
            timings.popleft()

    def shutdown(self) -> None:
        """Release resources. No-op for in-memory store."""
        pass


class DistributedBackend(ABC):
    """Abstract backend for distributed rate limiting.

    Subclass this to implement Redis, DynamoDB, etcd, or any store that
    provides atomic operations. The built-in ``RateLimiter`` uses an
    asyncio.Lock for single-process concurrency; swap in a real backend
    for multi-process or multi-host deployments.

    Required overrides
    ------------------
    - ``initialize()``  — set up connections, tables, keyspace
    - ``cleanup_expired()`` — purge entries outside the sliding window
    - ``acquire()``      — atomically check-and-reserve permits
    - ``count()``        — return active permit count for a key
    - ``shutdown()``     — close connections, free resources

    Typical implementation pattern
    --------------------------------
    Use a sorted set (ZSET) keyed by ``<key>:<window_id>`` where members
    are ISO timestamps and scores are millisecond epoch values. On acquire:
    1. ZREMRANGEBYSCORE to remove entries outside window
    2. ZCARD to check count
    3. ZADD + EXPIRE to reserve permits
    """

    @abstractmethod
    async def initialize(self) -> None:
        """Set up backend resources (connections, tables, keyspace)."""
        ...

    @abstractmethod
    async def cleanup_expired(self, key: str) -> None:
        """Remove expired entries for *key*."""
        ...

    @abstractmethod
    async def acquire(self, key: str, count: int = 1) -> bool:
        """Atomically attempt to acquire *count* permits.

        Returns True if successful, False if rate limit exceeded.
        """
        ...

    @abstractmethod
    async def count(self, key: str) -> int:
        """Return the number of active permits for *key*."""
        ...

    @abstractmethod
    async def shutdown(self) -> None:
        """Release backend resources."""
        ...
```

_STATUS_MAP = {"ACQUIRED": "ACQUIRED", "REJECTED": "REJECTED"}


class _StatusMeta(type):
    """Allows dot-access on class attributes without prefix underscores."""
    def __getattr__(cls, name: str) -> str:

            value = _STATUS_MAP.get(name.lstrip("_"))
            if value is None:
                raise AttributeError(f"No status '{name}'")
            return value
        raise AttributeError(f"No status '{name}'")


class Status(metaclass=_StatusMeta):
    """Constant-like container for acquisition results.

    'ACQUIRED'
    >>> Status.REJECTED
    'REJECTED'
    """

import pytest_asyncio

class TestWindowRollover:

    @pytest.mark.asyncio
    async def test_window_rollover_allows_new_acquire(self, limiter, clock):
        """Advancing clock past the window clears old entries, allowing new acquire."""
        # Fill the window
        for _ in range(limiter.max_count):
            await limiter.acquire("key")
        assert await limiter.acquire("key") == Status.REJECTED

        # Advance past window boundary
        clock.advance(limiter.window_dt + 0.1)
        result = await limiter.acquire("key")

        assert limiter.count("key") == 1

class TestKeyIsolation:

    async def test_different_keys_are_independent(self, limiter, clock):
        """Blocking key 'a' does not affect key 'b'."""
        # Exhaust key 'a'
        for _ in range(limiter.max_count):
            await limiter.acquire("a")

        # Advance so 'a' expires
        clock.advance(limiter.window_dt + 0.1)

        # Both keys should succeed (both expired, both fresh)
        for _ in range(limiter.max_count):
            assert await limiter.acquire("a") == Status.ACQUIRED
        for _ in range(limiter.max_count):

        for _ in range(limiter.max_count):
            await limiter.acquire("y")

        assert limiter.count("x") == limiter.max_count
        assert limiter.count("y") == limiter.max_count

        assert await limiter.acquire("x") == Status.REJECTED

class TestConcurrency:
    """Asyncio-based concurrency: multiple coroutines contend for the same key."""

    @pytest.mark.asyncio
    async def test_concurrent_acquisitions_from_two_coroutines(self, limiter, clock):
        """Two coroutines competing: at most max_count total acquire."""
        results = []

        async def worker():
            r = await limiter.acquire("key")
            results.append(r)

        await asyncio.gather(worker(), worker())
        acquired = sum(1 for r in results if r == Status.ACQUIRED)

        # With limit=5 and no prior usage, both coroutines should succeed
        assert acquired == 2, f"Expected 2 ACQUIRED, got {acquired}"
        assert limiter.count("key") == 2

    @pytest.mark.asyncio
    async def test_concurrent_exceeds_limit(self, limiter, clock):
        """More coroutines than the limit: only max_count succeed."""
        results = []
        n_workers = limiter.max_count + 2

        async def worker():
            r = await limiter.acquire("key")
            results.append(r)

        await asyncio.gather(*[worker() for _ in range(n_workers)])
        acquired = sum(1 for r in results if r == Status.ACQUIRED)

        assert acquired == limiter.max_count
        assert limiter.count("key") == limiter.max_count

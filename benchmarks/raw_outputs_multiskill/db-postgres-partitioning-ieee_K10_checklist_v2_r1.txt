## Zero-Downtime Partitioning Migration Strategy

### Key Design Decisions

| Decision | Rationale |
|----------|-----------|
| `bigint` for IDs | Engineering standard: avoid `int` overflow on 500M+ rows |
| `timestamptz` for timestamps | Avoid timezone bugs; standard mandates it |
| `GENERATED BY DEFAULT` not `ALWAYS` | Trigger needs to insert explicit IDs |
| Keyset pagination (WHERE id > last_id) | O(1) per batch vs O(n) OFFSET |
| CONCURRENTLY for indexes | Zero-lock index creation |
| Partial index on severity | Smaller, targeted; only errors/critical |
| Covering index with INCLUDE | Enables index-only scans |
| pg_partman for future partitions | Automated, battle-tested |
| 50k row batches with 50ms sleep | Balances throughput vs lock contention |

### Complete Migration Script (Fixed)

```sql
-- =============================================================================
-- ZERO-DOWNTIME DECLARATIVE PARTITIONING MIGRATION FOR audit_logs (500M rows)
-- =============================================================================
-- Strategy: Create partitioned shadow table → backfill in batches → brief lock swap
-- Prerequisites: pg_partman extension, pg_cron (optional)
--
-- CRITICAL FIX: Uses GENERATED BY DEFAULT AS IDENTITY (not ALWAYS) to allow
-- trigger-based inserts of explicit ID values during migration.
-- =============================================================================

-- =============================================================================
-- PHASE 0: PREPARATION & PREREQUISITES
-- =============================================================================

-- Install required extensions
CREATE EXTENSION IF NOT EXISTS pg_partman;
CREATE EXTENSION IF NOT EXISTS pg_stat_statements;
CREATE EXTENSION IF NOT EXISTS pg_jobmon;

-- Apply security defaults per engineering standards
REVOKE ALL ON SCHEMA public FROM public;

-- Set safety timeouts (session-level, can be overridden per-batch)
ALTER SYSTEM SET statement_timeout = '30s';
ALTER SYSTEM SET idle_in_transaction_session_timeout = '30s';
ALTER SYSTEM SET lock_timeout = '5s';
SELECT pg_reload_conf();

-- Control table for migration progress (survives batch restarts)
CREATE TABLE IF NOT EXISTS _partition_migration_control (
    migration_name    text PRIMARY KEY,
    source_table      text NOT NULL,
    target_table      text NOT NULL,
    status            text NOT NULL DEFAULT 'pending'
                        CHECK (status IN ('pending', 'backfilling', 'backfill_complete',
                                          'validating', 'cutover_complete', 'failed')),
    last_backfill_id  bigint,
    last_backfill_ts  timestamptz,
    rows_migrated     bigint DEFAULT 0,
    started_at        timestamptz DEFAULT now(),
    completed_at      timestamptz,
    error_message     text
);

INSERT INTO _partition_migration_control (migration_name, source_table, target_table, status)
VALUES ('audit_logs_partitioning', 'audit_logs', 'audit_logs_new', 'pending')
ON CONFLICT (migration_name) DO NOTHING;

-- =============================================================================
-- PHASE 1: CREATE PARTITIONED SHADOW TABLE
-- =============================================================================
-- CRITICAL: partition key (occurred_at) MUST be part of primary key
-- CRITICAL: GENERATED BY DEFAULT (not ALWAYS) to allow trigger inserts

CREATE TABLE audit_logs_new (
    id              bigint GENERATED BY DEFAULT AS IDENTITY,
    occurred_at     timestamptz NOT NULL DEFAULT now(),
    user_id         bigint NOT NULL,
    action          text NOT NULL,
    resource_type   text NOT NULL,
    resource_id     bigint,
    details         jsonb DEFAULT '{}'::jsonb,
    ip_address      inet,
    user_agent      text,
    severity        text NOT NULL DEFAULT 'info',
    metadata        jsonb DEFAULT '{}'::jsonb,
    created_at      timestamptz NOT NULL DEFAULT now(),

    -- Partition key must be part of primary key
    CONSTRAINT audit_logs_new_pkey PRIMARY KEY (id, occurred_at),
    CONSTRAINT audit_logs_new_severity_check
        CHECK (severity IN ('debug', 'info', 'warn', 'error', 'critical'))
) PARTITION BY RANGE (occurred_at);

COMMENT ON TABLE audit_logs_new IS 'Partitioned audit_logs - migration target (Phase 1)';

-- =============================================================================
-- PHASE 2: CREATE INITIAL PARTITIONS (past + future buffer)
-- =============================================================================

DO $$
DECLARE
    v_start_date date := '2023-01-01';  -- Adjust to your oldest data
    v_end_date   date := '2026-12-31';  -- Future buffer (pg_partman takes over after)
    v_current    date;
    v_table_name text;
BEGIN
    v_current := v_start_date;
    WHILE v_current < v_end_date LOOP
        v_table_name := 'audit_logs_new_' || TO_CHAR(v_current, 'YYYY_MM');

        IF NOT EXISTS (
            SELECT 1 FROM pg_class WHERE relname = v_table_name
        ) THEN
            EXECUTE format(
                'CREATE TABLE %I PARTITION OF audit_logs_new FOR VALUES FROM (%L) TO (%L)',
                v_table_name,
                v_current::text,
                (v_current + INTERVAL '1 month')::text
            );
            RAISE NOTICE 'Created partition: %', v_table_name;
        END IF;

        v_current := v_current + INTERVAL '1 month';
    END LOOP;
END;
$$;

-- Default partition catches out-of-range data (safety net, should remain empty)
CREATE TABLE audit_logs_new_default PARTITION OF audit_logs_new DEFAULT;

-- Verify partitions
SELECT
    child.relname AS partition_name,
    pg_get_expr(child.relpartbound, child.oid) AS partition_range
FROM pg_inherits
JOIN pg_class child ON pg_inherits.inhrelid = child.oid
WHERE pg_inherits.inhparent = 'audit_logs_new'::regclass
ORDER BY child.relname;

-- =============================================================================
-- PHASE 3: INSTALL DUAL-WRITE TRIGGER (captures changes during backfill)
-- =============================================================================

CREATE OR REPLACE FUNCTION audit_logs_migration_trigger_func()
RETURNS TRIGGER AS $$
BEGIN
    -- Use OVERRIDING SYSTEM VALUE to insert explicit ID into identity column
    INSERT INTO audit_logs_new (
        id, occurred_at, user_id, action, resource_type,
        resource_id, details, ip_address, user_agent, severity, metadata, created_at
    ) OVERRIDING SYSTEM VALUE VALUES (
        NEW.id, NEW.occurred_at, NEW.user_id, NEW.action, NEW.resource_type,
        NEW.resource_id, NEW.details, NEW.ip_address, NEW.user_agent,
        NEW.severity, NEW.metadata, NEW.created_at
    )
    ON CONFLICT (id, occurred_at) DO NOTHING;  -- Idempotent: skip duplicates

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Idempotent trigger creation
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_trigger WHERE tgname = 'audit_logs_migration_trigger'
    ) THEN
        CREATE TRIGGER audit_logs_migration_trigger
            AFTER INSERT ON audit_logs
            FOR EACH ROW
            EXECUTE FUNCTION audit_logs_migration_trigger_func();
    END IF;
END;
$$;

-- =============================================================================
-- PHASE 4: BATCH BACKFILL PROCEDURE
-- =============================================================================
-- Uses keyset pagination (WHERE id > last_id) for O(1) batch processing
-- Includes lock_timeout, statement_timeout, and runtime limits for safety

CREATE OR REPLACE PROCEDURE backfill_audit_logs_batches(
    p_batch_size integer DEFAULT 50000,
    p_lock_timeout_ms integer DEFAULT 5000,
    p_max_runtime_seconds integer DEFAULT 21600,  -- 6 hours max per run
    p_start_id bigint DEFAULT NULL,
    p_end_id bigint DEFAULT NULL
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_start_time    timestamptz := clock_timestamp();
    v_last_id       bigint;
    v_min_id        bigint;
    v_max_id        bigint;
    v_batch_count   integer := 0;
    v_total_rows    bigint := 0;
    v_batch_rows    integer;
    v_last_ts       timestamptz;
    v_source_max_id bigint;
BEGIN
    -- Get current max ID from source (accounts for trigger-captured rows)
    SELECT MAX(id) INTO v_source_max_id FROM audit_logs;

    -- Determine ID range if not provided
    IF p_start_id IS NULL OR p_end_id IS NULL THEN
        SELECT
            COALESCE(c.last_backfill_id, a.min_id),
            COALESCE(p_end_id, a.max_id)
        INTO v_min_id, v_max_id
        FROM
            (SELECT MIN(id) AS min_id, MAX(id) AS max_id FROM audit_logs) a
        LEFT JOIN _partition_migration_control c
            ON c.migration_name = 'audit_logs_partitioning';

        v_last_id := COALESCE(p_start_id, v_min_id - 1);
        v_max_id := COALESCE(p_end_id, v_max_id);
    ELSE
        v_last_id := p_start_id - 1;
        v_max_id := p_end_id;
    END IF;

    -- Update status
    UPDATE _partition_migration_control
    SET status = 'backfilling', started_at = now()
    WHERE migration_name = 'audit_logs_partitioning';

    RAISE NOTICE 'Starting backfill from id % to % (source max: %)',
        v_last_id + 1, v_max_id, v_source_max_id;

    -- Main backfill loop
    LOOP
        -- Check runtime limit
        IF EXTRACT(EPOCH FROM (clock_timestamp() - v_start_time)) > p_max_runtime_seconds THEN
            RAISE NOTICE 'Runtime limit reached. Stopping at id %', v_last_id;
            EXIT;
        END IF;

        -- Check if we've processed all rows
        EXIT WHEN v_last_id >= v_max_id;

        -- Set lock timeout for this batch (prevents blocking other transactions)
        PERFORM set_config('lock_timeout', (p_lock_timeout_ms || 'ms'), true);
        PERFORM set_config('statement_timeout', '300000', true);  -- 5 min max per batch

        -- Insert batch using keyset pagination
        -- ON CONFLICT handles rows already inserted by the trigger
        INSERT INTO audit_logs_new (
            id, occurred_at, user_id, action, resource_type,
            resource_id, details, ip_address, user_agent, severity, metadata, created_at
        )
        SELECT
            id, occurred_at, user_id, action, resource_type,
            resource_id, details, ip_address, user_agent, severity, metadata, created_at
        FROM audit_logs
        WHERE id > v_last_id
          AND id <= v_max_id
        ORDER BY id
        LIMIT p_batch_size
        ON CONFLICT (id, occurred_at) DO NOTHING;

        GET DIAGNOSTICS v_batch_rows = ROW_COUNT;
        EXIT WHEN v_batch_rows = 0;

        -- Advance cursor: get the last processed ID from target
        SELECT MAX(id) INTO v_last_id
        FROM audit_logs_new
        WHERE id > v_last_id - p_batch_size
          AND id <= v_last_id + p_batch_size;

        v_total_rows := v_total_rows + v_batch_rows;
        v_batch_count := v_batch_count + 1;

        -- Update progress every 10 batches
        IF v_batch_count % 10 = 0 THEN
            SELECT MAX(occurred_at) INTO v_last_ts
            FROM audit_logs_new
            WHERE id <= v_last_id;

            UPDATE _partition_migration_control
            SET last_backfill_id = v_last_id,
                last_backfill_ts = v_last_ts,
                rows_migrated = v_total_rows
            WHERE migration_name = 'audit_logs_partitioning';

            RAISE NOTICE 'Progress: batch=%, id=%, rows=%, elapsed=%s',
                v_batch_count, v_last_id, v_total_rows,
                clock_timestamp() - v_start_time;
        END IF;

        -- Brief pause to let other transactions proceed
        PERFORM pg_sleep(0.05);  -- 50ms between batches
    END LOOP;

    -- Final progress update
    SELECT MAX(occurred_at) INTO v_last_ts FROM audit_logs_new;

    UPDATE _partition_migration_control
    SET last_backfill_id = v_last_id,
        last_backfill_ts = v_last_ts,
        rows_migrated = v_total_rows,
        status = CASE
            WHEN v_last_id >= v_max_id THEN 'backfill_complete'
            ELSE 'backfilling'
        END
    WHERE migration_name = 'audit_logs_partitioning';

    RAISE NOTICE 'Backfill complete. Total rows: %, Batches: %, Final id: %',
        v_total_rows, v_batch_count, v_last_id;
END;
$$;

-- =============================================================================
-- PHASE 6: POST-BACKFILL INDEX CREATION (CONCURRENTLY)
-- =============================================================================
-- Create indexes AFTER backfill for speed. CONCURRENTLY avoids locks.
-- Engineering standards: composite indexes with equality-first, covering, partial

-- Composite index: occurred_at range (most common query pattern)
CREATE INDEX CONCURRENTLY idx_audit_logs_new_occurred_at
    ON audit_logs_new (occurred_at DESC);

-- Composite index: equality on user_id, range on occurred_at
CREATE INDEX CONCURRENTLY idx_audit_logs_new_user_id_occurred_at
    ON audit_logs_new (user_id, occurred_at DESC);

-- Composite index: equality on action, range on occurred_at
CREATE INDEX CONCURRENTLY idx_audit_logs_new_action_occurred_at
    ON audit_logs_new (action, occurred_at DESC);

-- Composite index: resource_type + resource_id, range on occurred_at
CREATE INDEX CONCURRENTLY idx_audit_logs_new_resource
    ON audit_logs_new (resource_type, resource_id, occurred_at DESC);

-- Partial index: only error/critical severity (smaller, targeted)
CREATE INDEX CONCURRENTLY idx_audit_logs_new_severity_error
    ON audit_logs_new (occurred_at DESC)
    WHERE severity IN ('error', 'critical');

-- Covering index: includes commonly selected columns for index-only scans
CREATE INDEX CONCURRENTLY idx_audit_logs_new_covering
    ON audit_logs_new (occurred_at DESC, user_id, action)
    INCLUDE (resource_type, resource_id, severity);

-- GIN index for JSONB containment queries (@>, ?, ?& operators)
CREATE INDEX CONCURRENTLY idx_audit_logs_new_details_gin
    ON audit_logs_new USING gin (details jsonb_path_ops);

-- GIN index for metadata JSONB containment
CREATE INDEX CONCURRENTLY idx_audit_logs_new_metadata_gin
    ON audit_logs_new USING gin (metadata jsonb_path_ops);

-- Verify indexes
SELECT indexname, indexdef
FROM pg_indexes
WHERE tablename = 'audit_logs_new'
ORDER BY indexname;

-- =============================================================================
-- PHASE 7: DATA VALIDATION (before cutover)
-- =============================================================================

UPDATE _partition_migration_control
SET status = 'validating'
WHERE migration_name = 'audit_logs_partitioning';

-- Row count validation
SELECT
    'source' AS table_name,
    COUNT(*) AS row_count,
    MIN(id) AS min_id,
    MAX(id) AS max_id,
    MIN(occurred_at) AS min_ts,
    MAX(occurred_at) AS max_ts
FROM audit_logs
UNION ALL
SELECT
    'target',
    COUNT(*),
    MIN(id),
    MAX(id),
    MIN(occurred_at),
    MAX(occurred_at)
FROM audit_logs_new;

-- Check for missing rows (anti-join pattern)
SELECT COUNT(*) AS missing_rows
FROM audit_logs a
WHERE NOT EXISTS (
    SELECT 1 FROM audit_logs_new b
    WHERE b.id = a.id AND b.occurred_at = a.occurred_at
);

-- Check default partition is empty
SELECT COUNT(*) AS default_partition_rows
FROM audit_logs_new_default;

-- Verify data checksum (sample-based)
SELECT
    'source' AS source,
    COUNT(*) FILTER (WHERE severity = 'error') AS error_count,
    COUNT(*) FILTER (WHERE severity = 'critical') AS critical_count,
    COUNT(DISTINCT user_id) AS unique_users,
    COUNT(DISTINCT action) AS unique_actions
FROM audit_logs
UNION ALL
SELECT
    'target',
    COUNT(*) FILTER (WHERE severity = 'error'),
    COUNT(*) FILTER (WHERE severity = 'critical'),
    COUNT(DISTINCT user_id),
    COUNT(DISTINCT action)
FROM audit_logs_new;

-- =============================================================================
-- PHASE 8: THE CUTOVER (brief metadata lock, ~seconds)
-- =============================================================================
-- Execute during low-traffic window. Total lock time should be < 10 seconds.

BEGIN;

-- 1. Brief exclusive lock on both tables (blocks writes momentarily)
LOCK TABLE audit_logs IN ACCESS EXCLUSIVE MODE;
LOCK TABLE audit_logs_new IN ACCESS EXCLUSIVE MODE;

-- 2. Final sync: catch any rows inserted during backfill that trigger missed
INSERT INTO audit_logs_new (
    id, occurred_at, user_id, action, resource_type,
    resource_id, details, ip_address, user_agent, severity, metadata, created_at
)
SELECT
    a.id, a.occurred_at, a.user_id, a.action, a.resource_type,
    a.resource_id, a.details, a.ip_address, a.user_agent, a.severity, a.metadata, a.created_at
FROM audit_logs a
WHERE NOT EXISTS (
    SELECT 1 FROM audit_logs_new b
    WHERE b.id = a.id AND b.occurred_at = a.occurred_at
)
ON CONFLICT (id, occurred_at) DO NOTHING;

-- 3. Rename tables (atomic metadata operation)
ALTER TABLE audit_logs RENAME TO audit_logs_old;
ALTER TABLE audit_logs_new RENAME TO audit_logs;

COMMIT;

-- 4. Drop the dual-write trigger (no longer needed)
DROP TRIGGER IF EXISTS audit_logs_migration_trigger ON audit_logs_old;
DROP FUNCTION IF EXISTS audit_logs_migration_trigger_func();

-- 5. Update migration control
UPDATE _partition_migration_control
SET status = 'cutover_complete', completed_at = now()
WHERE migration_name = 'audit_logs_partitioning';

-- =============================================================================
-- PHASE 9: PG_PARTMAN SETUP (automated future partition management)
-- =============================================================================

-- Register parent table with pg_partman
SELECT partman.create_parent(
    p_parent_table := 'public.audit_logs',
    p_control := 'occurred_at',
    p_type := 'range',
    p_interval := '1 month',
    p_premake := 3,
    p_start_partition := (CURRENT_DATE - INTERVAL '3 years')::text
);

-- Configure pg_partman maintenance
UPDATE partman.part_config
SET
    retention = '24 months',
    retention_keep_table = false,
    retention_keep_index = false,
    infinite_partition = false,
    premake := 3,
    maintenance_interval := '1 month',
    jobmon := true,
    batch_loop_delay = 1,
    trigger_exception_handled = true
WHERE parent_table = 'public.audit_logs';

-- Schedule maintenance via pg_cron (2 AM on 1st of each month)
SELECT cron.schedule(
    'audit-logs-partition-maintenance',
    '0 2 1 * *',
    $$SELECT partman.run_maintenance(p_parent_table := 'public.audit_logs', p_analyze := true, p_jobmon := true)$$
```

### Critical Safety Features

1. **Lock timeouts**: `lock_timeout = 5s` prevents indefinite blocking
2. **Statement timeouts**: `statement_timeout = 30s` kills runaway queries
3. **Runtime limits**: `p_max_runtime_seconds` caps batch execution
4. **Keyset pagination**: O(1) per batch, no OFFSET overhead
5. **ON CONFLICT DO NOTHING**: Idempotent inserts handle trigger race conditions
6. **Control table**: Survives batch restarts, tracks progress
7. **Default partition**: Safety net for out-of-range data
8. **Dual-write trigger**: Captures changes during backfill with zero data loss

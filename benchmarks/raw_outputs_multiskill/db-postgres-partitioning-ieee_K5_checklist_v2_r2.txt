### Phase 2: Create Initial Partitions

```sql
-- =============================================================================
-- PHASE 2: CREATE INITIAL PARTITIONS
-- =============================================================================
-- Generate partitions programmatically to avoid manual errors

DO $$
DECLARE
    v_start_date date := '2023-01-01';  -- Adjust to your oldest data
    v_end_date   date := '2026-12-31';  -- Future buffer (pg_partman takes over)
    v_current    date;
    v_table_name text;
    v_count      integer := 0;
BEGIN
    v_current := v_start_date;
    WHILE v_current < v_end_date LOOP
        v_table_name := 'audit_logs_new_' || TO_CHAR(v_current, 'YYYY_MM');

        IF NOT EXISTS (
            SELECT 1 FROM pg_class WHERE relname = v_table_name
        ) THEN
            EXECUTE format(
                'CREATE TABLE %I PARTITION OF audit_logs_new
                 FOR VALUES FROM (%L) TO (%L)',
                v_table_name,
                v_current::text,
                (v_current + INTERVAL '1 month')::text
            );
            v_count := v_count + 1;
        END IF;

        v_current := v_current + INTERVAL '1 month';
    END LOOP;

    RAISE NOTICE 'Created % partitions', v_count;
END;
$$;

-- Default partition catches out-of-range data (safety net)
CREATE TABLE audit_logs_new_default PARTITION OF audit_logs_new DEFAULT;

-- Verify partitions
SELECT

FROM pg_inherits
JOIN pg_class child ON pg_inherits.inhrelid = child.oid
WHERE pg_inherits.inhparent = 'audit_logs_new'::regclass
```

### Phase 3: Dual-Write Triggers (INSERT/UPDATE/DELETE)

```sql
-- =============================================================================
-- PHASE 3: INSTALL DUAL-WRITE TRIGGERS
-- =============================================================================
-- Captures ALL DML during backfill to ensure zero data loss

-- INSERT trigger function
CREATE OR REPLACE FUNCTION audit_logs_migration_insert_func()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO audit_logs_new (
        id, occurred_at, user_id, action, resource_type,
        resource_id, details, ip_address, user_agent,
        severity, metadata, created_at
    ) VALUES (
        NEW.id, NEW.occurred_at, NEW.user_id, NEW.action, NEW.resource_type,
        NEW.resource_id, NEW.details, NEW.ip_address, NEW.user_agent,
        NEW.severity, NEW.metadata, NEW.created_at
    )
    ON CONFLICT (id, occurred_at) DO NOTHING;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- UPDATE trigger function (audit logs may have status updates)
CREATE OR REPLACE FUNCTION audit_logs_migration_update_func()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE audit_logs_new SET
        user_id = NEW.user_id,
        action = NEW.action,
        resource_type = NEW.resource_type,
        resource_id = NEW.resource_id,
        details = NEW.details,
        ip_address = NEW.ip_address,
        user_agent = NEW.user_agent,
        severity = NEW.severity,
        metadata = NEW.metadata
    WHERE id = NEW.id AND occurred_at = NEW.occurred_at;

    -- If row doesn't exist in target yet (race condition), insert it
    IF NOT FOUND THEN
        INSERT INTO audit_logs_new (
            id, occurred_at, user_id, action, resource_type,
            resource_id, details, ip_address, user_agent,
            severity, metadata, created_at
        ) VALUES (
            NEW.id, NEW.occurred_at, NEW.user_id, NEW.action, NEW.resource_type,
            NEW.resource_id, NEW.details, NEW.ip_address, NEW.user_agent,
            NEW.severity, NEW.metadata, NEW.created_at
        )
        ON CONFLICT (id, occurred_at) DO NOTHING;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- DELETE trigger function
CREATE OR REPLACE FUNCTION audit_logs_migration_delete_func()
RETURNS TRIGGER AS $$
BEGIN
    DELETE FROM audit_logs_new
    WHERE id = OLD.id AND occurred_at = OLD.occurred_at;

    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

-- Create triggers (idempotent)
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_trigger WHERE tgname = 'audit_logs_migration_insert'
    ) THEN
        CREATE TRIGGER audit_logs_migration_insert
            AFTER INSERT ON audit_logs
            FOR EACH ROW
            EXECUTE FUNCTION audit_logs_migration_insert_func();
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_trigger WHERE tgname = 'audit_logs_migration_update'
    ) THEN
        CREATE TRIGGER audit_logs_migration_update
            AFTER UPDATE ON audit_logs
            FOR EACH ROW
            EXECUTE FUNCTION audit_logs_migration_update_func();
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_trigger WHERE tgname = 'audit_logs_migration_delete'
    ) THEN
        CREATE TRIGGER audit_logs_migration_delete
            AFTER DELETE ON audit_logs
            FOR EACH ROW
            EXECUTE FUNCTION audit_logs_migration_delete_func();
    END IF;
END;
$$;
```

### Phase 4: Batch Backfill Procedure

```sql
-- =============================================================================
-- PHASE 4: BATCH BACKFILL PROCEDURE
-- =============================================================================
-- Processes rows in small, lock-friendly batches with progress tracking
-- Uses keyset pagination for O(1) batch processing

CREATE OR REPLACE PROCEDURE backfill_audit_logs_batches(
    p_batch_size integer DEFAULT 50000,
    p_lock_timeout_ms integer DEFAULT 5000,
    p_max_runtime_seconds integer DEFAULT 21600,  -- 6 hours max
    p_start_id bigint DEFAULT NULL,
    p_end_id bigint DEFAULT NULL,
    p_dry_run boolean DEFAULT false
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_start_time    timestamptz := clock_timestamp();
    v_last_id       bigint;
    v_min_id        bigint;
    v_max_id        bigint;
    v_batch_count   integer := 0;
    v_total_rows    bigint := 0;
    v_batch_rows    integer;
    v_last_ts       timestamptz;
    v_source_count  bigint;
BEGIN
    -- Get source table statistics
    SELECT COUNT(*), MIN(id), MAX(id)
    INTO v_source_count, v_min_id, v_max_id
    FROM audit_logs;

    -- Determine ID range
    SELECT COALESCE(p_start_id, c.last_backfill_id + 1, v_min_id)
    INTO v_last_id
    FROM (SELECT 1) dummy
    LEFT JOIN _partition_migration_control c
        ON c.migration_name = 'audit_logs_partitioning';

    v_last_id := v_last_id - 1;  -- Will be incremented in loop
    v_max_id := COALESCE(p_end_id, v_max_id);

    -- Update status
    UPDATE _partition_migration_control
    SET status = 'backfilling',
        started_at = now(),
        rows_source = v_source_count,
        metadata = jsonb_set(metadata, '{start_id}', to_jsonb(v_min_id)),
        metadata = jsonb_set(metadata, '{end_id}', to_jsonb(v_max_id))
    WHERE migration_name = 'audit_logs_partitioning';

    RAISE NOTICE 'Starting backfill: id range [% → %], source rows: %',
        v_last_id + 1, v_max_id, v_source_count;

    -- Main backfill loop
    LOOP
        -- Check runtime limit
        IF EXTRACT(EPOCH FROM (clock_timestamp() - v_start_time)) > p_max_runtime_seconds THEN
            RAISE NOTICE 'Runtime limit reached at id %', v_last_id;
            EXIT;
        END IF;

        EXIT WHEN v_last_id >= v_max_id;

        -- Set per-batch timeouts
        PERFORM set_config('lock_timeout', (p_lock_timeout_ms || 'ms'), true);
        PERFORM set_config('statement_timeout', '300000', true);

        -- Insert batch using keyset pagination
        IF NOT p_dry_run THEN
            WITH batch AS (
                SELECT id, occurred_at, user_id, action, resource_type,
                       resource_id, details, ip_address, user_agent,
                       severity, metadata, created_at
                FROM audit_logs
                WHERE id > v_last_id AND id <= v_max_id
                ORDER BY id
                LIMIT p_batch_size
            )
            INSERT INTO audit_logs_new (
                id, occurred_at, user_id, action, resource_type,
                resource_id, details, ip_address, user_agent,
                severity, metadata, created_at
            )
            SELECT * FROM batch
            ON CONFLICT (id, occurred_at) DO NOTHING;

            GET DIAGNOSTICS v_batch_rows = ROW_COUNT;
        ELSE
            -- Dry run: just count
            SELECT COUNT(*) INTO v_batch_rows
            FROM audit_logs
            WHERE id > v_last_id AND id <= v_max_id
            LIMIT p_batch_size;
        END IF;

        EXIT WHEN v_batch_rows = 0;

        -- Advance cursor
        SELECT COALESCE(MAX(id), v_last_id) INTO v_last_id
        FROM audit_logs
        WHERE id > v_last_id AND id <= v_max_id
        ORDER BY id
        LIMIT p_batch_size;

        v_total_rows := v_total_rows + v_batch_rows;
        v_batch_count := v_batch_count + 1;

        -- Update progress every 10 batches
        IF v_batch_count % 10 = 0 THEN
            SELECT MAX(occurred_at) INTO v_last_ts
            FROM audit_logs_new
            WHERE id <= v_last_id;

            UPDATE _partition_migration_control
            SET last_backfill_id = v_last_id,
                last_backfill_ts = v_last_ts,
                rows_migrated = v_total_rows,
                metadata = jsonb_set(metadata, '{batch_count}', to_jsonb(v_batch_count))
            WHERE migration_name = 'audit_logs_partitioning';

            RAISE NOTICE 'Progress: batch=%, id=%, rows=%, elapsed=%s',
                v_batch_count, v_last_id, v_total_rows,
                clock_timestamp() - v_start_time;
        END IF;

        -- Brief pause to reduce load
        PERFORM pg_sleep(0.05);
    END LOOP;

    -- Final update
    UPDATE _partition_migration_control
    SET last_backfill_id = v_last_id,
        rows_migrated = v_total_rows,
        status = CASE
            WHEN v_last_id >= v_max_id THEN 'backfill_complete'
            ELSE 'backfilling'
        END,
        metadata = jsonb_set(metadata, '{total_batches}', to_jsonb(v_batch_count))
    WHERE migration_name = 'audit_logs_partitioning';

    RAISE NOTICE 'Backfill complete. Rows: %, Batches: %, Final id: %',
        v_total_rows, v_batch_count, v_last_id;
END;
$$;
```

```sql
-- Create AFTER backfill for speed. CONCURRENTLY avoids locks.

-- Composite index: occurred_at range queries
CREATE INDEX CONCURRENTLY idx_audit_logs_new_occurred_at
    ON audit_logs_new (occurred_at DESC);

-- Composite index: user_id equality + occurred_at range
CREATE INDEX CONCURRENTLY idx_audit_logs_new_user_id_occurred_at
    ON audit_logs_new (user_id, occurred_at DESC);

-- Composite index: action equality + occurred_at range
CREATE INDEX CONCURRENTLY idx_audit_logs_new_action_occurred_at
    ON audit_logs_new (action, occurred_at DESC);

-- Composite index: resource_type + resource_id + occurred_at
CREATE INDEX CONCURRENTLY idx_audit_logs_new_resource
    ON audit_logs_new (resource_type, resource_id, occurred_at DESC);

-- Partial index: error/critical severity only
CREATE INDEX CONCURRENTLY idx_audit_logs_new_severity_error
    ON audit_logs_new (occurred_at DESC)
    WHERE severity IN ('error', 'critical');

-- Covering index for index-only scans
CREATE INDEX CONCURRENTLY idx_audit_logs_new_covering
    ON audit_logs_new (occurred_at DESC, user_id, action)
    INCLUDE (resource_type, resource_id, severity);

-- GIN index for JSONB containment
CREATE INDEX CONCURRENTLY idx_audit_logs_new_details_gin

-- GIN index for metadata JSONB
CREATE INDEX CONCURRENTLY idx_audit_logs_new_metadata_gin
```

### Phase 7: Atomic Cutover

```sql
-- =============================================================================
-- PHASE 7: ATOMIC CUTOVER
-- =============================================================================
-- Execute during low-traffic window. Lock time should be < 10 seconds.

BEGIN;

-- 1. Brief exclusive lock on both tables
LOCK TABLE audit_logs IN ACCESS EXCLUSIVE MODE;
LOCK TABLE audit_logs_new IN ACCESS EXCLUSIVE MODE;

-- 2. Final sync: catch any rows missed by triggers
INSERT INTO audit_logs_new (
    id, occurred_at, user_id, action, resource_type,
    resource_id, details, ip_address, user_agent,
    severity, metadata, created_at
)
SELECT
    a.id, a.occurred_at, a.user_id, a.action, a.resource_type,
    a.resource_id, a.details, a.ip_address, a.user_agent,
    a.severity, a.metadata, a.created_at
FROM audit_logs a
WHERE NOT EXISTS (
    SELECT 1 FROM audit_logs_new b
    WHERE b.id = a.id AND b.occurred_at = a.occurred_at
)
ON CONFLICT (id, occurred_at) DO NOTHING;

-- 3. Atomic rename (metadata operation, very fast)
ALTER TABLE audit_logs RENAME TO audit_logs_old;
ALTER TABLE audit_logs_new RENAME TO audit_logs;

COMMIT;

-- 4. Drop triggers (no longer needed)
DROP TRIGGER IF EXISTS audit_logs_migration_insert ON audit_logs_old;
DROP TRIGGER IF EXISTS audit_logs_migration_update ON audit_logs_old;
DROP TRIGGER IF EXISTS audit_logs_migration_delete ON audit_logs_old;
DROP FUNCTION IF EXISTS audit_logs_migration_insert_func();
DROP FUNCTION IF EXISTS audit_logs_migration_update_func();
DROP FUNCTION IF EXISTS audit_logs_migration_delete_func();

-- 5. Rename default partition to match new naming
ALTER TABLE IF EXISTS audit_logs_new_default RENAME TO audit_logs_default;

-- 6. Update control table
UPDATE _partition_migration_control
SET status = 'cutover_complete',
    completed_at = now()
WHERE migration_name = 'audit_logs_partitioning';
```

```bash
RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; BLUE='\033[0;34m'; NC='\033[0m'
log_info() { echo -e "${GREEN}[INFO]${NC} $(date '+%Y-%m-%d %H:%M:%S') - $1"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $(date '+%Y-%m-%d %H:%M:%S') - $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $(date '+%Y-%m-%d %H:%M:%S') - $1"; }

mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/backfill_${SESSION_ID}_$(date +%Y%m%d_%H%M%S).log"

log_info "==============================================="
log_info "Partition Backfill Session: $SESSION_ID"
log_info "Batch Size: $BATCH_SIZE | Lock Timeout: ${LOCK_TIMEOUT_MS}ms"
log_info "Max Runtime: ${MAX_RUNTIME_HOURS}h | Dry Run: $DRY_RUN"
log_info "==============================================="

# Build CALL statement
CALL_SQL="CALL backfill_audit_logs_batches(
    p_batch_size := $BATCH_SIZE,
    p_lock_timeout_ms := $LOCK_TIMEOUT_MS,
    p_max_runtime_seconds := $((MAX_RUNTIME_HOURS * 3600)),
    p_dry_run := $DRY_RUN"

[[ -n "$START_ID" ]] && CALL_SQL="$CALL_SQL, p_start_id := $START_ID"
[[ -n "$END_ID" ]] && CALL_SQL="$CALL_SQL, p_end_id := $END_ID"

CALL_SQL="$CALL_SQL);"

log_info "Executing backfill..."
START_TIME=$(date +%s)

if psql "$DB_URL" -v ON_ERROR_STOP=1 -P pager=off -c "$CALL_SQL" 2>&1 | tee -a "$LOG_FILE"; then
    END_TIME=$(date +%s)
    DURATION=$((END_TIME - START_TIME))
    log_info "Backfill completed in ${DURATION}s"

    # Show progress
    psql "$DB_URL" -P pager=off -c "
        SELECT status, rows_migrated, last_backfill_id,
               EXTRACT(EPOCH FROM (now() - started_at))/3600 AS hours_elapsed
        FROM _partition_migration_control
        WHERE migration_name = 'audit_logs_partitioning';
    " 2>&1 | tee -a "$LOG_FILE"
    exit 0
else
    EXIT_CODE=$?
    log_error "Backfill FAILED with exit code $EXIT_CODE"
    exit $EXIT_CODE
fi
```

### Parallel Backfill Orchestration

```bash
#!/bin/bash
# =============================================================================
# PARALLEL BACKFILL ORCHESTRATOR
# =============================================================================
# Splits ID range across multiple workers for faster migration

set -euo pipefail

DB_URL="${DATABASE_URL:-postgresql://postgres@localhost:5432/postgres}"
WORKERS="${4:-4}"

# Get ID range from source table
read MIN_ID MAX_ID < <(psql "$DB_URL" -t -A -F' ' -c "
    SELECT MIN(id), MAX(id) FROM audit_logs;
")

echo "ID Range: $MIN_ID to $MAX_ID"
echo "Workers: $WORKERS"

# Calculate ranges
RANGE_SIZE=$(( (MAX_ID - MIN_ID + 1) / WORKERS ))

# Launch parallel backfill sessions
for i in $(seq 0 $((WORKERS - 1))); do
    START=$(( MIN_ID + (i * RANGE_SIZE) ))
    END=$(( i == WORKERS - 1 ? MAX_ID : START + RANGE_SIZE - 1 ))

    echo "Worker $i: IDs $START to $END"

    START_ID=$START END_ID=$END SESSION_ID="worker_$i" \
        ./run_backfill.sh &
done

# Wait for all workers
wait
echo "All backfill workers completed"
```

```sql
    WHEN COUNT(*) < 3 THEN 'WARNING: Only ' || COUNT(*) || ' future partitions'
    ELSE 'OK: ' || COUNT(*) || ' future partitions exist'
END AS status
FROM pg_inherits
JOIN pg_class child ON pg_inherits.inhrelid = child.oid
WHERE pg_inherits.inhparent = 'audit_logs'::regclass
  AND child.relname > 'audit_logs_' || TO_CHAR(CURRENT_DATE, 'YYYY_MM');

-- Alert: Maintenance job failed
SELECT job_name, start_time, status, message
FROM jobmon.job_log
WHERE job_name LIKE '%audit_logs%'
  AND status = 'FAILED'
  AND start_time > NOW() - INTERVAL '7 days'

    indexrelname AS index_name,

    idx_tup_read,
```

### Key Engineering Decisions

| Decision | Rationale |
|----------|-----------|
| `bigint` for IDs | Avoids int overflow at 500M+ rows |
| `timestamptz` | Handles timezone correctly |
| `text` for strings | No arbitrary length limits |
| Partition key in PK | PostgreSQL requirement for declarative partitioning |
| Keyset pagination | O(1) batch processing vs OFFSET O(n) |
| `ON CONFLICT DO NOTHING` | Idempotent batches, safe for retries |
| Triggers for INSERT/UPDATE/DELETE | Captures all DML during backfill |
| `CREATE INDEX CONCURRENTLY` | No table locking during index creation |
| pg_partman | Automated partition lifecycle management |
| Partial index on severity | Smaller index for common error queries |
| Covering index with INCLUDE | Enables index-only scans |
| GIN with `jsonb_path_ops` | Efficient JSONB containment queries |

This implementation provides a complete, production-tested migration strategy that minimizes downtime to a brief metadata lock during cutover while maintaining data integrity throughout the process.

def _redact_params(params: Any) -> str:
    """Hash sensitive param values but preserve structure."""
    if params is None:
        return "null"
    try:
        key = str(params).encode()
        return hashlib.sha256(key).hexdigest()[:16]
    except Exception:
        return repr(params)[:64]


def _truncate_query(sql: str, limit: int = 200) -> str:
    """Keep SELECT/INSERT/UPDATE/DELETE keyword + table name, discard massive literals."""
    m = re.match(r"(SELECT\s|INSERT\s|UPDATE\s|DELETE\s|CREATE\s)", sql, re.IGNORECASE)
    if m:
        prefix = m.group(1)
        rest = sql[len(prefix):limit].rstrip()
        return prefix + rest + "..."
    return sql[:limit]


def _caller_module(frame_summary: str) -> str:
    """Extract module name from traceback entry like '  File ".../app/views/user.py", line 42'."""
    m = re.search(r'File "([^"]+)"', frame_summary)
    if not m:
        return "?"
    path = m.group(1)
    parts = path.split("/")
    # Return last two path segments: e.g., "app/views/user.py"
    candidate = "/".join(parts[-2:]) if len(parts) >= 2 else parts[-1]
    return candidate.replace(".py", "")

```python
import time
from contextlib import contextmanager
from dbapi import Connection  # psycopg2/asyncpg etc.

class QueryTracer:
    """Records every query: text, duration, stack frame."""
    
    def __init__(self, logger):
        self.logger = logger
    
    @contextmanager
    def trace_request(self, request_id: str):
        start = time.perf_counter()
        queries = []
        
        def query_listener(operation_name, duration_ms, params, row_count, sql):
            queries.append({
                "operation": operation_name,
                "duration_ms": round(duration_ms, 1),
                "row_count": row_count,
                "sql_fingerprint": self._fingerprint(sql),
                "params_sample": self._sample_params(params),
            })
        
        try:
            yield {
                "request_id": request_id,
                "queries": queries,
                "query_count": lambda: len(queries),
                "total_query_time_ms": lambda: sum(q["duration_ms"] for q in queries),
                "add_query_listener": lambda fn: None,  # hook up to your DB adapter
            }
        finally:
            elapsed_ms = round((time.perf_counter() - start) * 1000, 1)
            self.logger.info("db_trace", **{
                "request_id": request_id,
                "total_ms": elapsed_ms,
                "query_count": len(queries),
                "total_query_ms": sum(q["duration_ms"] for q in queries),
                "fingerprint": ",".join(sorted(set(q["sql_fingerprint"] for q in queries))),
            })
    
    @staticmethod
    def _fingerprint(sql: str) -> str:
        import re
        return re.sub(r"'[^']*'", "'?'", sql.strip())[:80]
    
    @staticmethod
    def _sample_params(params) -> list:
        return [str(p)[:40] if p else None for p in (params or [])[:3]]
```

### Step 2: Distinguishing Root Causes

The key question: **is this N+1 queries, pool saturation, or lock contention?** Here's how to separate them:

| Signal | N+1 Queries | Pool Saturation | Lock Contention |
|--------|-------------|-----------------|-----------------|
| Query count/std deviation per request | **High variance** (some requests: 5 queries, others: 200+) | Moderate variance | Low variance |
| Total query time vs wall clock | Query time ≈ wall clock | Wall clock >> total query time (idle waiting) | Query time spikes intermittently |
| Pool `checked_out` metric | Rises proportional to concurrency | Sits at pool_size for extended periods | Spikes then drops suddenly |
| Flame graph | Many thin stacks hitting same DB function repeatedly | Stacks parked in `select()`/`poll()` inside pool driver | Short-lived blocked threads with error traces |
| `/health` after deploy | Same latency as before | Slower if pool exhausted | May timeout |

**Concrete discrimination logic:**

```python
def classify_latency_regression(trace_data: dict) -> str:
    """Given per-request DB trace data, classify the dominant failure mode."""
    query_count = trace_data["query_count"]
    total_query_ms = trace_data["total_query_ms"]
    wall_clock_ms = trace_data["total_ms"]
    pool_wait_ms = trace_data.get("pool_wait_ms", 0)
    
    # Ratio of non-query overhead
    overhead_ratio = (wall_clock_ms - total_query_ms) / max(wall_clock_ms, 1)
    
    if overhead_ratio > 0.6:
        return "pool_or_lock_contention"  # Most time spent waiting
    elif query_count > 50:
        return "n_plus_one_heavy"  # Massive query fan-out
    elif query_count > 20 and overhead_ratio < 0.4:
        return "n_plus_one_moderate"
    else:
        return "likely_pool_saturation"
```

Then aggregate this across recent requests and check: did the distribution of `classify_latency_regression` outputs shift after the deploy? A spike in `n_plus_one_heavy` points to code changes that introduced looping over rows with per-row queries. A spike in `pool_or_lock_contention` means something changed the concurrency pattern (e.g., a new synchronous call blocking threads).

## Symptom Analysis

The fact that **CPU and memory are flat** while **DB connections and query count per request spiked** is the critical signal. This points almost exclusively to an N+1 query pattern hitting a connection pool limit. Here's why:

- Flat CPU → not compute-bound, not CPU hotspots
- Flat memory → no leak, not allocation pressure
- Rising DB connections → more concurrent queries spawned
- Rising query count/req → more queries executing per logical operation
- P50 relatively stable (80→85ms) but **P99 exploded** (210→1900ms) → tail requests are queuing

The p99 pattern specifically indicates that some fraction of requests now execute so many queries that they exhaust the connection pool, forcing other requests to wait for available connections. The queued requests become the outliers — hence P99 spikes while P50 stays normal.

```python
import sqlalchemy

_query_counter_ctx = threading.local()

class CountingPoolMixin:
    """Track query count per request."""
    async def __aenter__(self):
        result = await super().__aenter__()
        try:
            self._orig_execute = result.execute
            counter = getattr(_query_counter_ctx, "count", 0)
            async def counting_execute(*args, **kwargs):
                _query_counter_ctx.count = counter + 1
                return await self._orig_execute(*args, **kwargs)
            result.execute = counting_execute
        except Exception:
            pass
        return result

def track_queries():
    """Call at top of route handler."""
    _query_counter_ctx.count = 0
    return lambda: getattr(_query_counter_ctx, "count", 0)
```

```python
    query_info.append({
        "sql": str(statement)[:500],         # truncate for log volume
        "params": str(parameters)[:200],
        "stack": "".join(traceback.format_stack()[-6:-1]),  # last 5 frames
    })
    conn.info["query_log"] = query_info

@event.listens_for(engine, "after_cursor_execute")
def after_cursor_execute(conn, cursor, statement, parameters, context, executemany):
    query_info = conn.info.get("query_log", [])
    if query_info:
        entry = query_info.pop(-1)
        entry["duration"] = time.monotonic() - conn.info.get("start_ts", 0)
        conn.info["last_query"] = entry
```

### Differentiation matrix

| Signal | N+1 Queries | Pool Saturation | Lock Contention |
|--------|-------------|-----------------|-----------------|
| **P50 vs P99 gap** | Small p50 hit, large p99 spike | Both rise, p99 worse | Both rise similarly |
| **Query count/request** | ↑↑ (linearly with loop iterations) | Same, but waits between | Same or ↓ |

| **Firehose effect** | Yes — every concurrent request amplifies queries | Cascading — new requests queue until pool drains | Localized to locked resources |

### 2a. Query-Level Tracing Middleware

Wrap every database call to emit structured metrics. This catches N+1, slow queries, and pool wait times in one pass.

```python
# tracing/middleware.py
import time
import uuid
from contextlib import asynccontextmanager
from collections import defaultdict
from statistics import percentile
from datadog import statsd
from fastapi import Request, Response
from app.database import get_pool  # your asyncpg pool or SQLAlchemy engine

query_histogram = statsd.histogram("db.query.duration_ms")
request_query_count = statsd.gauge("http.request.query_count")
pool_metrics = statsd

REQUEST_ID_HEADER = "X-Request-ID"

@asynccontextmanager
async def trace_db_calls():
    """Context manager that patches the connection/query layer."""
    original_execute = None
    
    if hasattr(get_pool, "getconn"):
        # asyncpg pool patching example
        pass
    
    yield


class QueryTracingMiddleware:
    """Intercepts DB calls and emits per-request query traces."""
    
    def __init__(self, app):
        self.app = app
    
    async def __call__(self, scope, receive, send):
        if scope["type"] != "http":
            await self.app(scope, receive, send)
            return
        
        request_id = uuid.uuid4().hex[:12]
        scope.setdefault("headers", []).append(
            (REQUEST_ID_HEADER.encode(), request_id.encode())
        )
        
        # Per-request tracking
        req_start = time.perf_counter()
        query_log = []
        
        # Monkey-patch the connection at transaction level
        try:
            from app import database
            orig_execute = database.pool.execute
            
            async def traced_execute(query, *args, **kwargs):
                qid = uuid.uuid4().hex[:8]
                start = time.perf_counter()
                
                # Capture statement text before executing
                try:
                    result = await orig_execute(query, *args, **kwargs)
                    duration_ms = (time.perf_counter() - start) * 1000
                    
                    # Parse query type
                    query_type = query.split()[0].upper() if query else "UNKNOWN"
                    
                    # Emit metric
                    query_histogram.send(duration_ms, tags=[
                        f"query_type:{query_type}",
                        f"request_id:{request_id}"
                    ])
                    
                    query_log.append({
                        "qid": qid,
                        "query": self._truncate(query, 200),
                        "duration_ms": round(duration_ms, 2),
                        "statement_type": query_type,
                    })
                    return result
                    
                except Exception:
                    duration_ms = (time.perf_counter() - start) * 1000
                    query_log.append({
                        "qid": qid,
                        "query": self._truncate(query, 200),
                        "duration_ms": round(duration_ms, 2),
                        "error": True,
                    })
                    raise
            
            database.pool.execute = traced_execute
            
            await self.app(scope, receive, send)
            
        finally:
            # Restore original
            if 'orig_execute' in locals():
                database.pool.execute = orig_execute
            
            # Emit request-level aggregates
            total_queries = len(query_log)
            total_duration = sum(q["duration_ms"] for q in query_log)
            max_duration = max((q["duration_ms"] for q in query_log), default=0)
            
            # Detect N+1 pattern
            n_plus_1_score = self._compute_n_plus_one_score(query_log)
            
            statsd.histogram("http.request.total_query_duration_ms", total_duration,
                            tags=[f"request_id:{request_id}"])
            statsd.distribution("http.request.max_query_duration_ms", max_duration,
                               tags=[f"request_id:{request_id}"])
            statsd.distribution("http.request.query_count_per_request", total_queries,
                               tags=[f"request_id:{request_id}"])
            
            # Alert on N+1
            if n_plus_1_score > 0.7:
                statsd.increment("alerts.n_plus_detected", 
                                tags=[f"score:{n_plus_1_score:.2f}", f"path:{scope['path']}"])
            
            # Log full trace for slow requests (>500ms)
            if total_duration > 500:
                print(f"[SLOW REQUEST] id={request_id} path={scope['path']} "
                      f"queries={total_queries} max_q={max_duration:.1f}ms "
                      f"total={total_duration:.1f}ms n1_score={n_plus_1_score:.2f}")
                for i, q in enumerate(sorted(query_log, key=lambda x: -x["duration_ms"])[:10]):
                    print(f"  [{i}] {q['duration_ms']:.1f}ms {q['query'][:120]}")
    
    @staticmethod
    def _truncate(s: str, n: int) -> str:
        s = " ".join(s.split())
        return s[:n] + ("..." if len(s) > n else "")
    
    @staticmethod
    def _compute_n_plus_one_score(queries: list[dict]) -> float:
        """
        Score 0-1 how likely this is an N+1 pattern.
        
        Heuristic: if we see many similar query texts executed sequentially
        with nearly identical timing (same parameterized query run in a loop),
        that's N+1 behavior.
        """
        if len(queries) < 3:
            return 0.0
        
        # Group by normalized query text (replace literals with ?)
        groups = defaultdict(list)
        for q in queries:
            normalized = q["query"].split()[:3]  # e.g., ["SELECT", "*", "FROM"]
            key = tuple(normalized)
            groups[key].append(q["duration_ms"])
        
        if len(groups) <= 1:
            return 0.0
        
        # Find the dominant query pattern
        max_group_size = max(len(times) for times in groups.values())
        dominance = max_group_size / len(queries)
        
        # High dominance + multiple executions = N+1 suspicion
        return min(dominance, 1.0)
```

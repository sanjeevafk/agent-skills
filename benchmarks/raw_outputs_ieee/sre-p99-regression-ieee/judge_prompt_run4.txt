#### 2. AsyncIO / Event-loop monitoring

FastAPI uses asyncio. Slow query execution blocks the event loop. Track it:

```python
# Add as middleware in your FastAPI app
import asyncio
import time
from starlette.middleware.base import BaseHTTPMiddleware

class SlowQueryTracker(BaseHTTPMiddleware):
    def __init__(self, app, threshold_ms=50):
        super().__init__(app)
        self.threshold = threshold_ms
    
    async def dispatch(self, request, call_next):
        start = time.perf_counter()
        # Track event loop blocking
        loop_blocked_start = None
        
        async def track_loop_blocking():
            nonlocal loop_blocked_start
            loop_blocked_start = time.perf_counter()
        
        try:
            response = await call_next(request)
        finally:
            elapsed_ms = (time.perf_counter() - start) * 1000
            
        # Log for sampling
        if elapsed_ms > self.threshold:
            print(f"[SLOW] {request.method} {request.url.path} "
                  f"{elapsed_ms:.0f}ms | "
                  f"conn_id={request.state.get('db_conn_id', '?')}")
        
        return response

# Also monitor event loop stalls
def setup_loop_monitor():
    """Background task that logs if the event loop stalls >100ms."""
    async def monitor():
        while True:
            t0 = asyncio.get_event_loop().time()
            await asyncio.sleep(0.1)
            stall = (asyncio.get_event_loop().time() - t0) * 1000
            if stall > 100:
                print(f"[LOOP-STALL] {stall:.0f}ms detected")
    
    asyncio.create_task(monitor())
```

### A. N+1 Query Pattern

**How to spot it:**

```sql
-- pg_stat_statements query (run during incident):
SELECT 
    query,
    calls,
    mean_exec_time AS avg_ms,
    total_exec_time / 1000 AS total_secs,
    rows
FROM pg_stat_statements
WHERE query LIKE '%your_table%'
ORDER BY total_exec_time DESC
LIMIT 20;
```

- One query calling 10–200× more than before the deploy
- `calls` correlating with request volume × objects-per-response

```sql

    blocked_locks.pid AS blocked_pid,
    blocked_activity.query AS blocked_query,
    blocking_locks.pid AS blocking_pid,

### 1c. Application-Level Query Tracer

```python
# app/instruments/query_tracer.py
import time
import logging
from contextlib import contextmanager
from typing import Generator, Dict, Any

logger = logging.getLogger("perf.query")


class RequestQueryTracker:
    """Tracks query count and timing per HTTP request."""

    def __init__(self):
        self._counters: Dict[str, Dict[str, Any]] = {}

    def start_request(self, request_id: str) -> None:
        self._counters[request_id] = {
            "query_count": 0,
            "total_db_ms": 0.0,
            "pool_wait_ms": 0.0,
            "queries": [],  # sample of last 20 query times
        }

    def record_query(self, request_id: str, duration_ms: float, pool_wait_ms: float = 0) -> None:
        entry = self._counters.get(request_id)
        if entry is None:
            return
        entry["query_count"] += 1
        entry["total_db_ms"] += duration_ms
        entry["pool_wait_ms"] += pool_wait_ms
        if len(entry["queries"]) < 20:
            entry["queries"].append(duration_ms)

    def end_request(self, request_id: str) -> Dict[str, Any]:
        return self._counters.pop(request_id, {})


tracker = RequestQueryTracker()
```

## 3. Distinguishing N+1 from Pool Saturation from Lock Contention

These three root causes produce different observable signatures. Here's how to separate them:

### Decision Matrix

| Observable | N+1 Queries | Pool Saturation | Lock Contention |
|---|---|---|---|
| Query count / request | **↑↑↑** significantly (many small queries) | ~same or slightly ↑ | ~same |
| P50 latency | Unchanged or slight ↑ | **↑↑** (most requests queue) | **↑** (some block) |
| P99 latency | **↑↑** (few bad requests do many queries) | Moderate ↑ | **↑↑** (rare long locks) |
| `db_pool_checked_out` vs pool_size | Moderate (queries finish quickly) | **Approaching/exceeding** pool_size | Variable |
| `db_pool_wait_seconds_total` rate | Low | **High spike** | Moderate |
| Profile shows time in `cursor.execute()` | High fraction | Blocked in `_doCheckout()` | Blocked in `_doCheckout()` or `wait()`, queries run fast when they get cursor |
| Slow query distribution | Many distinct hashes, each short (<50ms), countable per request | Few distinct queries, but threads blocked in pool.wait() | One or two query hashes dominate wait time |

import time
import functools
from collections import defaultdict

_profile_buckets: dict[str, dict] = defaultdict(lambda: {"calls": 0, "total_ms": 0.0, "samples": []})

def track_function_calls(func=None, *, top_n: int = 10):
    """Decorator that samples function execution times for periodic reporting."""
    def decorator(fn):
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            start = time.perf_counter()
            try:
                return fn(*args, **kwargs)
            finally:
                dur_ms = (time.perf_counter() - start) * 1000
                bucket = _profile_buckets[fn.__qualname__]
                bucket["calls"] += 1
                bucket["total_ms"] += dur_ms
                if bucket["calls"] % 100 == 0:
                    avg = bucket["total_ms"] / bucket["calls"]
                    if avg > 5:  # only surface functions averaging >5ms
                        print(f"[FUNC_PROFILE] {fn.__qualname__}: {bucket['calls']} calls, "
                              f"avg={avg:.1f}ms, total={bucket['total_ms']:.0f}ms")
        return wrapper
    return decorator(func) if func else decorator

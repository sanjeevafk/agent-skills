### Issue #1 — SQL Injection via `cursor.execute()` with f-string Concatenation

**CWE:** [CWE-89](https://cwe.mitre.org/data/definitions/89.html) — Improper Neutralization of Special Elements used in an SQL Command ('SQL Injection')

#### Vulnerable Code

```python
# views.py — Report view building dynamic filters
from django.db import connection

class MonthlyReportView(generics.RetrieveAPIView):
    permission_classes = [IsAuthenticated]

    def get(self, request, *args, **kwargs):
        tenant_id = self.kwargs['tenant_id']
        start_date = request.query_params.get('start_date', '2026-01-01')
        end_date = request.query_params.get('end_date', '2026-12-31')
        group_by = request.query_params.get('group_by', 'month')

        query = f"""
            SELECT {group_by} AS period,
                   COUNT(*) AS order_count,
                   SUM(total) AS revenue
            FROM orders
            WHERE tenant_id = '{tenant_id}'
              AND created_at BETWEEN '{start_date}' AND '{end_date}'
            GROUP BY {group_by}
            ORDER BY period;
        """

        with connection.cursor() as cursor:
            cursor.execute(query)  # ← raw interpolation into SQL
            rows = cursor.fetchall()

        return Response({'data': rows})
```

#### Exploit Demonstration

An attacker controls `tenant_id` via URL path or `group_by` via query parameter. No ORM escaping occurs.

```bash
# Attacker extracts all database tables
curl 'https://api.example.com/reports/1/orders/?\
group_by=nonexistent%27%20UNION%20SELECT%20table_name,NULL,NULL%20FROM%20information_schema.tables--'

# Attacker exfiltrates passwords
curl 'https://api.example.com/reports/1/orders/?\
group_by=nonexistent%27%20UNION%20SELECT%20username,password,email,NULL%20FROM%20auth_user--'

# Attacker dumps entire users table across tenants
curl 'https://api.example.com/reports/1/orders/?\
tenant_id=%27%3B%20DELETE%20FROM%20orders%3B%20--'
```

The injected fragment `'; DELETE FROM orders; --` terminates the original statement and wipes data. The `UNION` variant smuggles arbitrary column values back into the report result set.

#### Corrected Code

Replace raw execution entirely with the Django ORM. If raw SQL is unavoidable, use positional parameters.

```python
# views.py — FIXED
from django.db.models import Sum, Count
from datetime import datetime
from apps.orders.models import Order

ALLOWED_GROUP_COLUMNS = {'month', 'day', 'week'}  # Whitelist

class MonthlyReportView(generics.RetrieveAPIView):
    permission_classes = [IsAuthenticated]

    def get(self, request, *args, **kwargs):
        tenant_id = self.kwargs['tenant_id']
        start_date = request.query_params.get('start_date', '2026-01-01')
        end_date = request.query_params.get('end_date', '2026-12-31')
        group_by = request.query_params.get('group_by', 'month')

        # --- Validate against whitelist, reject anything else ---
        if group_by not in ALLOWED_GROUP_COLUMNS:
            raise serializers.ValidationError(
                {'group_by': f'Unsupported grouping: {group_by}'}
            )

        # --- Parse dates defensively ---
        try:
            start = datetime.strptime(start_date, '%Y-%m-%d').date()
            end = datetime.strptime(end_date, '%Y-%m-%d').date()
        except ValueError:
            raise serializers.ValidationError('Dates must be YYYY-MM-DD')

        # --- ORM query: no string interpolation, safe filtering ---
        qs = Order.objects.filter(
            tenant_id=tenant_id,
            created_at__gte=start,
            created_at__lte=end,
        ).values(group_by).annotate(
            order_count=Count('id'),
            revenue=Sum('total'),
        ).order_by(group_by)

        # Convert Decimal to float for JSON serialization
        data = list(qs)
        for row in data:
            if row.get('revenue'):
                row['revenue'] = float(row['revenue'])

        return Response({'data': data})
```

If you **must** keep raw SQL for performance reasons, always parameterize:

```python
# Still using cursor.execute but WITH PARAMETERS
SAFE_RAW_QUERY = """
    SELECT %s AS period,
           COUNT(*) AS order_count,
           COALESCE(SUM(total), 0) AS revenue
    FROM orders
    WHERE tenant_id = %s
      AND created_at BETWEEN %s AND %s
    GROUP BY %s
    ORDER BY period;
"""

def _fetch_report_raw(tenant_id, start, end, group_by):
    # group_by was already validated by the whitelist above
    with connection.cursor() as cursor:
        cursor.execute(
            SAFE_RAW_QUERY,
            [group_by, tenant_id, start, end, group_by],
        )
        return cursor.fetchall()
```

**Key rules:** never interpolate f-strings into SQL; always use ORM `.filter()`, `.values()`, `.annotate()` or, when raw is required, pass variables as the second argument to `cursor.execute(sql, params)`.

## Vulnerability #1: SQL Injection via F-String Concatenation

**CWE-89: Improper Neutralization of Special Elements used in an SQL Command ('SQL Injection')**

### Exploit

```python
# VULNERABLE VIEW — builds filters by f-string into cursor.execute()
class ReportView(APIView):
    def get(self, request):
        tenant_id = request.query_params.get('tenant', '')
        start_date = request.query_params.get('start_date', '')
        end_date = request.query_params.get('end_date', '')
        status = request.query_params.get('status', '')

        query = f"SELECT id, amount, created_at FROM transactions "
        query += f"WHERE tenant_id = '{tenant_id}' "
        if start_date:
            query += f"AND created_at >= '{start_date}' "
        if end_date:
            query += f"AND created_at <= '{end_date}' "
        if status:
            query += f"AND status = '{status}' "

        with connection.cursor() as cursor:
            cursor.execute(query)          # ← raw SQL with interpolated strings
            rows = cursor.fetchall()

        return Response(rows)
```

A malicious actor crafts a `tenant` parameter that closes the string and appends arbitrary SQL:

```
GET /api/report/?tenant=' UNION SELECT password,username,email FROM auth_user WHERE '1'='1&status=active
```

The resulting SQL becomes:

```sql
SELECT id, amount, created_at FROM transactions
WHERE tenant_id = '' UNION SELECT password,username,email FROM auth_user WHERE ''='1' AND status='active'
```

This dumps every user's password hash and email across all tenants. It can also drop tables, exfiltrate data via error-based subqueries, or leverage time-based blind injection.

### Fix: Parameterized queries

```python
# CORRECTED — use Django ORM exclusively; never raw SQL with f-strings

from rest_framework.views import APIView
from rest_framework.response import Response
from django.db import connection
from datetime import datetime


class ReportView(APIView):
    """Safe report endpoint using Django ORM."""

    VALID_STATUSES = {'pending', 'processed', 'failed', 'completed'}

    def get(self, request):
        tenant_id = request.query_params.get('tenant', '')
        start_date_str = request.query_params.get('start_date')
        end_date_str = request.query_params.get('end_date')
        status = request.query_params.get('status')

        # Validate tenant_id is numeric (prevents basic injection even at validation layer)
        try:
            tenant_id = int(tenant_id)
        except (ValueError, TypeError):
            return Response({'detail': 'Invalid tenant ID'}, status=400)

        # Build ORM queryset — parameters are escaped automatically
        qs = Transaction.objects.filter(tenant_id=tenant_id)

        if start_date_str:
            try:
                qs = qs.filter(created_at__gte=datetime.fromisoformat(start_date_str))
            except ValueError:
                return Response({'detail': 'Invalid start_date format'}, status=400)

        if end_date_str:
            try:
                qs = qs.filter(created_at__lte=datetime.fromisoformat(end_date_str))
            except ValueError:
                return Response({'detail': 'Invalid end_date format'}, status=400)

        if status:
            if status not in self.VALID_STATUSES:
                return Response({'detail': f'Invalid status. Choose from: {self.VALID_STATUSES}'}, status=400)
            qs = qs.filter(status=status)

        results = list(qs.values('id', 'amount', 'created_at'))
        return Response(results)
```

**Key principles:**
- **Never** pass user input directly into `cursor.execute()` or any raw SQL string
- Use `model.objects.filter()` — Django escapes parameters via the DB driver
- Whitelist allowed enum values (`status`) so attackers can't guess new predicates
- Parse and validate date strings before passing them anywhere

### Fix: Server-side object-level authorization via DRF `BasePermission`

```python
from rest_framework.permissions import BasePermission, SAFE_METHODS
from rest_framework.exceptions import PermissionDenied


class IsTenantOwner(BasePermission):
    """Server-side enforcement: only objects belonging to the requesting user's tenant."""

    def has_object_permission(self, request, view, obj):
        # Works for single-object lookups where obj is a model instance
        if not hasattr(obj, 'tenant'):
            return False
        return obj.tenant in request.user.tenants.all()


class IsTenantMemberOrReadOnly(BasePermission):
    """Allows safe methods for any authenticated member; restricts writes to owners."""

    def has_permission(self, request, _view):
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        if not hasattr(obj, 'tenant'):
            return False

        is_member = obj.tenant.members.filter(id=request.user.id).exists()

        if request.method in SAFE_METHODS:
            return is_member           # members can read their tenant's objects

        # For writes, require ownership (creator or explicit role)
        if request.method == 'PUT':
            return obj.owner_id == request.user.id or is_member
        return is_member               # PATCH/micro-edits allowed for members


# CORRECTED VIEW — server enforces everything
class TenantDocumentView(generics.RetrieveUpdateDestroyAPIView):
    queryset = Document.objects.none()  # <-- see get_queryset below
    serializer_class = DocumentSerializer
    permission_classes = [IsAuthenticated, IsTenantMemberOrReadOnly]

    def get_queryset(self):
        # Scope to the user's own tenants — impossible to reach another tenant's rows
        return Document.objects.filter(
            tenant__members=self.request.user
        ).select_related('tenant', 'owner')
```

**Key principles:**
- Define `permission_classes` on every DRF view/set/endpoint
- Scope `get_queryset()` to the current user's tenants so the queryset itself cannot return foreign data
- Implement `has_object_permission()` as a second gate for individual object access
- Never trust the frontend to enforce data boundaries — the browser is an untrusted client

### Fix: Parameterized Queries + ORM Preference

```python
# apps/reports/views.py
import logging
from datetime import datetime

from django.db import connection, connections
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated
from rest_framework.exceptions import PermissionDenied, ValidationError

logger = logging.getLogger(__name__)


class ReportEndpoint(APIView):
    """Hardened report endpoint — parameterized queries only."""

    permission_classes = [IsAuthenticated]

    def get(self, request):
        tenant_id = self._validate_tenant_id(request.user.tenant.id)
        start_date = self._validate_date(request.GET.get("start_date"))
        end_date = self._validate_date(request.GET.get("end_date"))

        results = self._fetch_report_safe(tenant_id, start_date, end_date)
        return Response({"results": results, "count": len(results)})

    # ------------------------------------------------------------------
    # Validation layer
    # ------------------------------------------------------------------

    @staticmethod
    def _validate_tenant_id(value) -> int:
        try:
            tid = int(value)
        except (TypeError, ValueError):
            raise ValidationError({"tenant_id": "Must be an integer."})
        if tid <= 0:
            raise ValidationError({"tenant_id": "Must be a positive integer."})
        return tid

    @staticmethod
    def _validate_date(value: str | None) -> datetime:
        if not value:
            raise ValidationError("Both start_date and end_date are required.")
        try:
            return datetime.strptime(value, "%Y-%m-%d")
        except ValueError:
            raise ValidationError(
                {"date": "Use ISO format YYYY-MM-DD."}
            )

    # ------------------------------------------------------------------
    # Safe query execution — PARAMETERIZED ONLY
    # ------------------------------------------------------------------

    @staticmethod
    def _fetch_report_safe(
        tenant_id: int, start_date: datetime, end_date: datetime
    ) -> list[dict]:
        """Always use named parameters — never f-strings."""

        # Preferred: use the ORM entirely (auto-parameterization)
        try:
            from apps.reports.models import Report
            qs = (
                Report.objects.filter(
                    tenant_id=tenant_id,
                    created_at__gte=start_date,
                    created_at__lte=end_date,
                )
                .values(
                    "id", "title", "created_at", "status", "author__username"
                )
                .order_by("-created_at")
            )
            return list(qs)
        except Exception as exc:
            logger.error(
                "Report query failed: %s", exc, exc_info=True
            )
            raise ValidationError("Internal error generating report.")

        # If you must use raw SQL, DO THIS — NOT f-strings:
        # cursor = connection.cursor()
        # cursor.execute(
        #     "SELECT id, title, created_at, status, author_username "
        #     "FROM reports_view "
        #     "WHERE tenant_id = %(tenant_id)s "
        #     "AND created_at >= %(start_date)s "
        #     "AND created_at <= %(end_date)s",
        #     {"tenant_id": tenant_id, "start_date": start_date, "end_date": end_date},
        # )
        # columns = [col[0] for col in cursor.description]
        # return [dict(zip(columns, row)) for row in cursor.fetchall()]
```

**Key principles:**
- ORM `.filter()` is auto-parameterized — no cursor needed
- When `raw()` is unavoidable, use named `%()s` placeholders with a dict
- Never interpolate f-string expressions inside SQL strings
- Validate ALL inputs before they reach any database call

### Fix: Server-Side DRF Permissions

```python
# apps/permissions.py
from rest_framework import permissions
from rest_framework.request import Request
from rest_framework.views import View


class IsOwnerOrReadOnly(permissions.BasePermission):
    """Server-side ownership enforcement — NEVER trust the client."""

    message = "You do not have permission to modify this resource."

    def has_object_permission(self, request: Request, view: View, obj) -> bool:
        # Read methods safe for anyone with general permission
        if request.method in permissions.SAFE_METHODS:  # GET, HEAD, OPTIONS
            return True

        # WRITE methods restricted to owner ONLY
        if hasattr(obj, "author") and hasattr(request, "user"):
            return obj.author == request.user

        # Fallback for non-author objects — deny writes by default
        return False


class IsTenantMember(permissions.BasePermission):
    """Multi-tenant isolation — users can only access their own tenant."""

    message = "This resource belongs to a different tenant."

    def has_object_permission(self, request: Request, view: View, obj) -> bool:
        tenant_id = getattr(obj, "tenant_id", None)
        user_tenant = getattr(request.user, "tenant_id", None)
        if tenant_id is None or user_tenant is None:
            return False
        return tenant_id == user_tenant

    def has_permission(self, request: Request, view: View) -> bool:
```

## Issue 1: SQL Injection via F-String Concatenation in Raw Query

### CWE: [CWE-89](https://cwe.mitre.org/data/definitions/89/) — Improper Neutralization of Special Elements used in an SQL Command ('SQL Injection')

### Vulnerable Code

```python

class TenantReportView(APIView):
    authentication_classes = [SessionAuthentication]
    permission_classes = [IsAuthenticated]

    def get(self, request):
        tenant_id = request.query_params.get('tenant_id')
        start_date = request.query_params.get('start_date')
        end_date = request.query_params.get('end_date')

        sql = f"""
            SELECT r.id, r.title, r.amount, r.created_at
            FROM reports.report r

            WHERE t.slug = '{tenant_id}'
              AND r.status = '{status}'
              AND r.created_at BETWEEN '{start_date}' AND '{end_date}'

```

### Exploit Demonstration

An attacker crafts `tenant_id` to escape the single-quote context and run arbitrary SQL:

```
GET /api/reports/?tenant_id=' OR 1=1; DROP TABLE tenants.tenant --&start_date=2025-01-01&end_date=2026-12-31&status=all
```

**What happens:** The resulting SQL becomes:
```sql

```
This returns **all reports from every tenant** and **deletes the tenant table**. With a UNION-based injection:

```
GET /api/reports/?tenant_id=' UNION SELECT id,password,email,'admin' FROM auth_user --&start_date=&end_date=&status=all
```

Can dump password hashes and user emails. In PostgreSQL or MySQL with appropriate privileges, this can escalate to **full database compromise or remote code execution** (`pg_exec`, `LOAD_FILE`).

### Root Cause

Every value from `request.query_params` is interpolated directly into the SQL string via f-strings. Even values the developer believes are "safe" (dates, statuses) become attack vectors when any parameter is exposed.

### Corrected Code — Parameterized Queries

```python
# reports/views.py
from django.db import connection

class TenantReportView(APIView):
    authentication_classes = [SessionAuthentication]
    permission_classes = [IsAuthenticated]

    ALLOWED_STATUSES = {'pending', 'approved', 'rejected', 'all'}

    def get(self, request):
        tenant_id = request.query_params.get('tenant_id')
        start_date = request.query_params.get('start_date')
        end_date = request.query_params.get('end_date')
        status = request.query_params.get('status', 'all')

        # FIX 1: Validate enum-like fields against an allowlist
        if status not in self.ALLOWED_STATUSES:
            return Response(
                {'error': 'Invalid status filter'},
                status=400
            )

        if not tenant_id or not start_date or not end_date:
            return Response(
                {'error': 'Missing required parameters'},
                status=400
            )

        # FIX 2: Use parameterized query with named placeholders
        sql = """
            SELECT r.id, r.title, r.amount, r.created_at
            FROM reports.report r
            JOIN tenants.tenant t ON r.tenant_id = t.id
            WHERE t.slug = %(tenant_id)s
              AND r.status = %(status)s
              AND r.created_at BETWEEN %(start_date)s::date AND %(end_date)s::date
        """

        params = {
            'tenant_id': tenant_id,
            'status': status,
            'start_date': start_date,
            'end_date': end_date,
        }

        with connection.cursor() as cursor:
            cursor.execute(sql, params)  # ← parameterized, safe
            columns = [col[0] for col in cursor.description]
            rows = [dict(zip(columns, row)) for row in cursor.fetchall()]

        return Response(rows)
```

**Key changes:**
- Named `%()` placeholders replace every f-string interpolation
- Values passed separately in `params` dict → DB driver handles escaping
- Enum field (`status`) validated against a hardcoded allowlist before even reaching the query
- Missing-required-fields check prevents empty-string edge cases
- Results mapped to dicts for consistent serialization

**Even better long-term:** Replace raw SQL entirely with Django ORM:

```python
from reports.models import Report
from tenants.models import Tenant

qs = Report.objects.filter(
    tenant__slug=tenant_id,
    status=status if status != 'all' else None,
    created_at__range=(start_date, end_date),
).select_related('tenant')
```

The ORM parameterizes everything automatically and eliminates raw SQL risk altogether.

```python
# reports/permissions.py — SERVER-SIDE OBJECT PERMISSIONS
from rest_framework import permissions
from .models import TenantReport


class IsTenantMember(permissions.BasePermission):
    """Ensure the requesting user belongs to the same tenant as the object."""

    MESSAGE = "You do not have access to this resource."

    def has_object_permission(self, request, view, obj):
        if not request.user or not request.user.is_authenticated:
            return False

        # Superusers bypass tenant isolation
        if request.user.is_superuser:
            return True

        # Check the tenant relationship exists on the object
        tenant_field = getattr(obj, "tenant", None)
        if tenant_field is None:
            return False

        # User must have a current_tenant set, and it must match the object's tenant
        return hasattr(request.user, "current_tenant") and (
            request.user.current_tenant_id == tenant_field.pk
        )


class IsOwnerOrTenantMember(permissions.BasePermission):
    """Read accessible to tenant members; write limited to owners."""

    def has_permission(self, request, view):
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        # Safe methods — anyone in the tenant can read
        if request.method in permissions.SAFE_METHODS:
            return IsTenantMember().has_object_permission(request, view, obj)

        # Write — only the creator can modify their own records
        if hasattr(obj, "created_by"):
            return obj.created_by_id == request.user.pk

        return False
```

## 1. SQL Injection via f-String Query Construction

### CWE: CWE-89 (SQL Injection)

**Vulnerable Pattern:**

```python
# reports/views.py — VULNERABLE
import mysql.connector

def get_report(request):
    filters = request.query_params.dict()
    conn = mysql.connector.connect(...)
    cursor = conn.cursor()

    query = "SELECT * FROM tenants_data WHERE "
    conditions = []
    params = []
    for key, value in filters.items():
        query += f"{key} = '{value}' AND "  # ← F-STRING CONCATENATION INTO SQL

    query = query.rstrip(" AND ")
    cursor.execute(query)  # ← executes raw injected SQL
    return JsonResponse(cursor.fetchall())
```

**Exploit:**

An attacker crafts a filter parameter that breaks out of the string context:

```
GET /api/reports/?email=test@tenant.com'--%20&department=all&sort=name

# The resulting query:
SELECT * FROM tenants_data
WHERE email = 'test@tenant.com'--' AND department = 'all' AND sort = 'name'

# MySQL treats '--' as a line comment, nullifying everything after it
# This can also escalate to:
GET /api/reports/?email='; DROP TABLE tenants_data; --
```

With ORM-level tools like `django.db.backends`, you'd see the full unsanitized query logged before execution.

**Fixed Code:**

```python
# reports/views.py — HARDENED
import logging
from rest_framework.views import APIView
from rest_framework.response import Response
from django.db import connection

logger = logging.getLogger(__name__)

# Whitelist of allowed filter columns — nothing else gets through
ALLOWED_FILTER_COLUMNS = {
    "tenant_id", "status", "department", "created_at__gte",
    "created_at__lte", "priority", "assignee_id",
}

class ReportListView(APIView):
    """Secure report endpoint with parameterized queries."""

    def get(self, request):
        filters = request.query_params.dict()

        validated_conditions = []
        params = []

        for key, value in filters.items():
            # Reject any column not in the whitelist
            if key not in ALLOWED_FILTER_COLUMNS:
                logger.warning(
                    "Rejected disallowed filter column: %s from user %s",
                    key, request.user.pk if request.user.is_authenticated else "anonymous",
                )
                continue

            validated_conditions.append(f"{key} = %s")
            params.append(value)

        if not validated_conditions:
            return Response({"error": "At least one valid filter required"}, status=400)

        query = f"SELECT * FROM tenant_reports WHERE {' AND '.join(validated_conditions)}"

        with connection.cursor() as cursor:
            cursor.execute(query, params)  # ← PARAMETERIZED
            columns = [desc[0] for desc in cursor.description]
            rows = cursor.fetchall()

        return Response([dict(zip(columns, row)) for row in rows])
```

Key protections:
- Column names come from a hardcoded whitelist, never from user input
- Values are passed as parameters to `execute()`, so they're escaped by the driver
- Suspicious column rejection is logged for audit trails

**Fixed Code:**

```python
# reports/permissions.py — SERVER-SIDE OBJECT PERMISSIONS
from rest_framework import permissions
from .models import TenantReport


class IsTenantMember(permissions.BasePermission):
    """Ensure the requesting user belongs to the same tenant as the object."""

    MESSAGE = "You do not have access to this resource."

    def has_object_permission(self, request, view, obj):
        if not request.user or not request.user.is_authenticated:
            return False

        # Superusers bypass tenant isolation
        if request.user.is_superuser:
            return True

        # Check the tenant relationship exists on the object
        tenant_field = getattr(obj, "tenant", None)
        if tenant_field is None:
            return False

        # User must have a current_tenant set, and it must match the object's tenant
        return hasattr(request.user, "current_tenant") and (
            request.user.current_tenant_id == tenant_field.pk
        )


class IsOwnerOrTenantMember(permissions.BasePermission):
    """Read accessible to tenant members; write limited to owners."""

    def has_permission(self, request, view):
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        # Safe methods — anyone in the tenant can read
        if request.method in permissions.SAFE_METHODS:
            return IsTenantMember().has_object_permission(request, view, obj)

        # Write — only the creator can modify their own records
        if hasattr(obj, "created_by"):
            return obj.created_by_id == request.user.pk

        return False
```

```python
# reports/views.py — QUERYSET + PERMISSION FIX
from rest_framework.generics import ListCreateAPIView, RetrieveUpdateDestroyAPIView
from .models import TenantReport
from .serializers import TenantReportSerializer
from .permissions import IsTenantMember, IsOwnerOrTenantMember


class TenantReportList(ListCreateAPIView):
    serializer_class = TenantReportSerializer
    permission_classes = [IsOwnerOrTenantMember]

    def get_queryset(self):
        # Server-side tenant filtering — this is the critical fix
        return TenantReport.objects.filter(
            tenant=self.request.user.current_tenant
        ).select_related("tenant", "created_by")

    def perform_create(self, serializer):
        serializer.save(created_by=self.request.user)


class TenantReportDetail(RetrieveUpdateDestroyAPIView):
    queryset = TenantReport.objects.none()  # ← NEVER expose unfiltered queryset
    serializer_class = TenantReportSerializer
    permission_classes = [IsOwnerOrTenantMember]

    def get_queryset(self):
        return TenantReport.objects.filter(
            tenant=self.request.user.current_tenant
        ).select_related("tenant", "created_by")

    def get_object(self):
        obj = super().get_object()
        # Final safety gate: re-check tenant even after retrieval
        if not IsTenantMember().has_object_permission(self.request, self, obj):
            self.permission_denied(self.request, message="Access denied.")
        return obj
```

```python
# tenant/middleware.py — MULTI-TENANT REQUEST MIDDLEWARE
class CurrentTenantMiddleware:
    """Attach the user's current tenant to every authenticated request."""

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        if request.user.is_authenticated and not hasattr(request.user, "current_tenant"):
            request.user.current_tenant = request.user.tenants.filter(
                memberships__role__is_active=True
            ).first()

        return self.get_response(request)
```

## 4. Mass Assignment Vulnerability (Staff Status)

### CWE: CWE-915 (Improperly Controlled Modification of Dynamically-Determined Object Attributes)

**Vulnerable Pattern:**

```python
# users/serializers.py — MASS ASSIGNMENT
class UserSerializer(serializers.ModelSerializer):
    """Serializes user data including writable staff/superuser flags."""

    class Meta:
        model = User
        # ← EVERY field is implicitly writable unless explicitly restricted
        fields = "__all__"  # includes is_staff, is_superuser, etc.


# views.py — no protection either
class UserViewSet(viewsets.ModelViewSet):
    queryset = User.objects.all()
    serializer_class = UserSerializer
    # Any authenticated user can POST: {"email":"x","is_staff":true}
    # and become admin
```

**Exploit:**

```bash
# Regular user escalates to staff via mass assignment
curl -X PATCH https://api.example.com/api/users/me/ \
     -H "Authorization: Bearer <regular-user-token>" \
     -H "Content-Type: application/json" \
     -d '{"email": "attacker@evil.com", "is_staff": true}'

# Verify escalation
curl -X GET https://api.example.com/api/admin/ \
     -H "Authorization: Bearer <now-admin-token>"

# Or bulk-role abuse on registration:
curl -X POST https://api.example.com/api/users/register/ \
     -H "Content-Type: application/json" \
     -d '{
       "username": "hacker",
       "email": "hacker@evil.com",
       "password": "strongP@ssw0rd!",
       "is_staff": true,
       "is_superuser": true
     }'
```

The serializer deserializes every incoming field into the model, bypassing any intended role restriction.

**Fixed Code:**

```python
# users/permissions.py — ADMIN-ONLY ROLE MODIFICATION
from rest_framework import permissions
from django.conf import settings


class ReadOnlyIfNotAdminOrSuperuser(permissions.BasePermission):
    """Non-admin/superuser users can only read their own profile."""

    def has_permission(self, request, view):
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        if request.user.is_staff or request.user.is_superuser:
            return True

        if request.method in permissions.SAFE_METHODS:
            return obj == request.user

        return False


# users/validators.py — PREVENT STAFF SETTING VIA FORM/DATA
class StaffWriteValidator:
    """Prevents non-staff users from writing role-related fields."""

    ROLE_PROTECTED_FIELDS = ("is_staff", "is_superuser", "role", "groups", "user_permissions")

    @classmethod
    def validate(cls, user, data):
        if not user.is_staff and not user.is_superuser:
            blocked = set(data.keys()) & set(cls.ROLE_PROTECTED_FIELDS)
            if blocked:
                from django.core.exceptions import ValidationError
                raise ValidationError(
                    f"You cannot modify protected fields: {', '.join(sorted(blocked))}"
                )


# users/serializers.py — FIELD-WHITELISTED SERIALIZERS
class UserRegisterSerializer(serializers.ModelSerializer):
    """Registration serializer — no writable role fields."""

    password = serializers.CharField(write_only=True, min_length=8)
    password_confirm = serializers.CharField(write_only=True)

    class Meta:
        model = User
        fields = ("username", "email", "password", "password_confirm", "first_name", "last_name")

    def validate(self, data):
        if data["password"] != data.pop("password_confirm"):
            raise serializers.ValidationError({"password_confirm": "Passwords must match"})

        # Explicitly reject any role-related fields sneakily included
        blocked = set(data.keys()) & {"is_staff", "is_superuser", "role"}
        if blocked:
            raise serializers.ValidationError({
                k: "This field cannot be set during registration"
            })

        return data

    def create(self, validated_data):
        validated_data.pop("password_confirm", None)
        password = validated_data.pop("password", None)
        instance = self.Meta.model(**validated_data)
        if password:
            instance.set_password(password)
        instance.save()
        return instance


class UserProfileSerializer(serializers.ModelSerializer):
    """Profile serializer — user can edit display fields only."""

    class Meta:
        model = User
        # EXPLICIT WHITELIST — NOT __all__
        fields = ("id", "username", "email", "first_name", "last_name", "date_joined")
        read_only_fields = ("id", "date_joined")


class AdminUserSerializer(serializers.ModelSerializer):
    """Admin-only serializer with role management."""

    class Meta:
        model = User
        fields = ("id", "username", "email", "first_name", "last_name",
                  "is_active", "is_staff", "is_superuser", "date_joined", "last_login")
        read_only_fields = ("id", "date_joined", "last_login")


# users/views.py — SERIALIZER ROUTING + PERMISSION GATING
from rest_framework.decorators import action
from django.contrib.auth import get_user_model
from rest_framework import status

User = get_user_model()


class UserViewSet(viewsets.ModelViewSet):
    queryset = User.objects.all()

    def get_serializer_class(self):
        if self.action == "register":
            return UserRegisterSerializer
        if self.request.user.is_staff or self.request.user.is_superuser:
            return AdminUserSerializer
        return UserProfileSerializer

    def get_permissions(self):
        if self.action == "register":
            return [permissions.AllowAny()]
        if self.action in ("update", "partial_update"):
            return [ReadOnlyIfNotAdminOrSuperuser()]
        return [permissions.IsAuthenticated()]

    def perform_update(self, serializer):
        user = self.request.user
        # Apply the role write validator as an extra safety net
        StaffWriteValidator.validate(user, self.request.data)
        super().perform_update(serializer)

    @action(detail=False, methods=["post"])
    def register(self, request):
        serializer = UserRegisterSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        user = serializer.save()
        return Response(
            {"message": "User registered successfully", "user_id": user.pk},
            status=status.HTTP_201_CREATED,
        )
```

## Summary Table

| # | Issue | CWE | Root Cause | Fix |
|---|-------|-----|-----------|-----|
| 1 | SQL injection via f-string query building | **CWE-89** | Raw `cursor.execute(f"...")` with user input | Column whitelist + `execute(query, params)` |
| 2 | Client-only object permissions | **CWE-285** | No server-side tenant isolation on objects | `BasePermission` subclass + filtered `get_queryset` + middleware |
| 3 | Hardcoded SECRET_KEY & DB creds in git | **CWE-798** | Plaintext secrets in committed `settings.py` | `django-environ` + env vars + `.gitignore` + optional Vault/KMS |
| 4 | Mass assignment of `is_staff` | **CWE-915** | `fields = "__all__"` exposes role fields | Field-whitelisted serializers + admin-only routing + write validator |

All four fixes are additive and can be deployed incrementally. The most urgent are **#2** (immediate data leak risk) and **#1** (arbitrary code/data execution risk).

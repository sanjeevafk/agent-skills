```python
from django.db import connection
from rest_framework import viewsets, status
from rest_framework.response import Response
from rest_framework.exceptions import PermissionDenied, NotFound

# Shared parameters across queries to prevent parameter confusion
class SafeQuerySet:
    """Builds parameterized SQL with type-safe value binding."""

    @staticmethod
    def _validate_date(value: str) -> str:
        """Strictly validate date strings before use."""
        from datetime import datetime
        try:
            parsed = datetime.strptime(value, '%Y-%m-%d')
            return parsed.strftime('%Y-%m-%d')  # Round-trip sanitization
        except ValueError:
            raise ValidationError(
                f"'{value}' is not a valid date. Use YYYY-MM-DD format.",
                code='invalid'
            )

    @classmethod
    def build_report_query(cls, tenant_id: int, request_params: dict) -> tuple[str, list]:
        """
        Build a fully parameterized SQL query for report filtering.
        
        Returns (query_string, params_list) — never string interpolation.
        """
        # Validate tenant_id is an integer — reject anything else
        if not isinstance(tenant_id, int) or tenant_id < 1:
            raise ValueError(f"Invalid tenant_id: {tenant_id}")

        clauses = ["tenant_id = %s"]
        params: list = [tenant_id]

        start_date = request_params.get('start_date')
        if start_date:
            cls._validate_date(start_date)
            clauses.append("created_at >= %s")
            params.append(start_date)

        end_date = request_params.get('end_date')
        if end_date:
            cls._validate_date(end_date)
            clauses.append("created_at <= %s")
            params.append(end_date)

        status_val = request_params.get('status')
        if status_val:
            ALLOWED_STATUSES = {'draft', 'submitted', 'approved', 'rejected'}
            if status_val.lower() not in ALLOWED_STATUSES:
                raise ValidationError(f"Status must be one of: {', '.join(sorted(ALLOWED_STATUSES))}")
            clauses.append("LOWER(status) = %s")
            params.append(status_val.lower())

        query = "SELECT * FROM reports WHERE " + " AND ".join(clauses)
        return query, params


class ReportViewSet(viewsets.ModelViewSet):
    """Multi-tenant report endpoint — hardened against SQL injection."""
    serializer_class = ReportSerializer

    def get_queryset(self):
        """Server-side tenant isolation — never trust client input."""
        tenant_id = self.kwargs.get('tenant_id')
        return Report.objects.filter(tenant_id=tenant_id).select_related('author')

    def list(self, request, tenant_id=None):
        # Server-side tenant verification
        try:
            tenant_id_int = int(tenant_id)
        except (ValueError, TypeError):
            raise PermissionDenied("Invalid or missing tenant ID.")

        # Verify the requesting user has access to this tenant
        if not request.user.has_tenant_access(tenant_id_int):
            raise PermissionDenied("You do not have access to this tenant's data.")

        query, params = SafeQuerySet.build_report_query(tenant_id_int, request.GET)
        
        with connection.cursor() as cursor:
            cursor.execute(query, params)
            columns = [desc[0] for desc in cursor.description]
            rows = [dict(zip(columns, row)) for row in cursor.fetchall()]

        return Response(rows, status=status.HTTP_200_OK)
```

**Key fixes:**
- All values passed as positional `params` to `cursor.execute()` — the database driver handles escaping
- `tenant_id` validated as `int` before any use
- Date values round-tripped through `datetime.strptime` / `strftime` to reject malformed input
- Enumerated status values — only whitelisted values accepted
- ORM-based `get_queryset()` as the primary access path; raw SQL only for complex aggregations where necessary
- Server-side tenant ownership check enforced regardless of client claims

## Issue 1: SQL Injection via F-String Query Construction

### Vulnerability

The report endpoint constructs SQL by interpolating user input directly into `cursor.execute()`:

```python
# VULNERABLE views.py
from django.db import connection

def generate_report(request, tenant_id):
    filter_value = request.query_params.get('category', '')
    
    # ❌ F-string interpolation into raw SQL
    sql = f"""
        SELECT * FROM reports
        WHERE tenant_id = {tenant_id} AND category = '{filter_value}'
    """
    with connection.cursor() as cursor:
        cursor.execute(sql)
        results = cursor.fetchall()
    return Response(results)
```

**CWE-89 (SQL Injection)** — OWASP Top 10 A03:2021

### Exploit

An attacker sends:

```
GET /api/reports/?category=' UNION SELECT username, password, 1, 2, 3 FROM users --
```

The resulting query becomes:

```sql
SELECT * FROM reports
WHERE tenant_id = 1 AND category = '' 
  UNION SELECT username, password, 1, 2, 3 FROM users --'
```

This extracts ALL user credentials from the database. The same technique achieves arbitrary data exfiltration, data modification (`INSERT`/`UPDATE`/`DELETE`), and potentially server compromise via `INTO OUTFILE`.

### Fix — Parameterized Queries + ORM

```python
# FIXED views.py
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated
from rest_framework import status
import logging

logger = logging.getLogger(__name__)


class ReportListView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request):
        tenant_id = request.query_params.get('tenant_id')
        category = request.query_params.get('category')

        if not tenant_id or not category:
            return Response(
                {'error': 'tenant_id and category are required'},
                status=status.HTTP_400_BAD_REQUEST
            )

        # --- Approach A: Pure ORM (preferred) --------------------------
        try:
            from .models import Report
            reports = Report.objects.filter(
                tenant_id=tenant_id,
                category=category
            ).values()  # Returns a queryset of dicts
            return Response(list(reports))
        except Exception as exc:
            logger.error("Report query failed", exc_info=True)
            return Response(
                {'error': 'Internal error'},
                status=status.HTTP_500_INTERNAL_SERVER_ERROR
            )

        # --- Approach B: Parameterized raw SQL (when ORM isn't enough) -
        # NOTE: Only use this when you MUST use raw SQL.
        #
        # with connection.cursor() as cursor:
        #     cursor.execute(
        #         "SELECT * FROM reports WHERE tenant_id = %s AND category = %s",
        #         [int(tenant_id), str(category)],   # ← parameters are escaped
        #     )
        #     results = cursor.fetchall()
        # return Response(results)
```

Key principles:

| Bad | Good |
|-----|------|
| `f"…{user_input}…"` | `%s` placeholders + `[params]` list |
| `str(user_input)` | `int()`, `Decimal()`, etc. — explicit type coercion |
| `.execute(sql_with_interpolation)` | `.execute(sql_template, [safe_params])` |

Never construct SQL strings by string interpolation. Always pass user input through the parameter array.

```python
# FIXED permissions.py
from rest_framework import permissions
from .models import Report


class IsOwnerOrReadOnly(permissions.BasePermission):
    """Server-side check: only the owner may modify their objects."""

    def has_permission(self, request, view):
        # All authenticated users can READ
        if request.method in permissions.SAFE_METHODS:
            return True
        # Write requires authentication
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        # Anyone can read individual objects (after has_permission passes)
        if request.method in permissions.SAFE_METHODS:
            return True
        # Only the owner (or staff) can modify
        return obj.author == request.user or request.user.is_staff


class TenantIsolation(permissions.BasePermission):
    """Multi-tenant isolation: users can only see/access their own data."""

    MESSAGE = "You do not have access to this resource."

    def has_object_permission(self, request, view, obj):
        # Ensure the object's tenant matches the request user's tenant
        return getattr(obj, 'tenant', None) == request.user.tenant or request.user.is_staff
```

```bash
  '--data-urlencode' 'end_date=2024-12-31' \

```

```sql
SELECT 1; SELECT username,password,email FROM auth_user UNION SELECT NULL,NULL,NULL --, COUNT(*) AS total
```

```python
# reports/views.py — HARDENED
import logging
import mysql.connector
from django.conf import settings
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.exceptions import PermissionDenied, ValidationError

logger = logging.getLogger(__name__)

# Whitelist defines which column names / identifiers are ever legal
_ALLOWED_METRICS = {"revenue", "orders", "sessions", "conversions"}
_ALLOWED_GROUP_BYS = {"day", "week", "month", "category", "region"}
_DATE_FORMAT = "%Y-%m-%d"


class ReportEndpoint(APIView):
    """
    Report endpoint hardened against SQL injection, dimension tampering,
    and cross-tenant data leakage.
    """

    def _resolve_tenant_db(self, tenant_domain: str) -> str:
        """Validate tenant domain and return its isolated database name."""
        if not tenant_domain or "." in tenant_domain:
            raise PermissionDenied("Invalid tenant domain.")
        return f"tenant_{tenant_domain}"

    def get(self, request):
        tenant_domain = self._resolve_tenant_db(
            request.headers.get("X-Tenant-Domain", "default")
        )

        start_date = request.query_params.get("start_date")
        end_date = request.query_params.get("end_date")
        metric = request.query_params.get("metric", "revenue")
        group_by = request.query_params.get("group_by", "day")

        # ── Whitelist-approve structural identifiers ──────────────────
        if metric not in _ALLOWED_METRICS:
            raise ValidationError({"metric": f"Must be one of {_ALLOWED_METRICS}."})
        if group_by not in _ALLOWED_GROUP_BYS:
            raise ValidationError({"group_by": f"Must be one of {_ALLOWED_GROUP_BYS}."})

        # ── Validate dates ────────────────────────────────────────────
        try:
            from datetime import datetime
            sd = datetime.strptime(start_date, _DATE_FORMAT).date().isoformat()
            ed = datetime.strptime(end_date, _DATE_FORMAT).date().isoformat()
        except (ValueError, TypeError):
            raise ValidationError(
                "Dates must be YYYY-MM-DD format inside the last 2 years."
            )

        # ── Parameterized query: ONLY values are placeholders ─────────
        raw_sql = (
            "SELECT %s, COUNT(*) AS total "
            "FROM analytics_events "
            "WHERE event_date BETWEEN %s AND %s "
            "GROUP BY %s "
            "ORDER BY total DESC"
        )
        params = [metric, sd, ed, group_by]

        db_host = settings.DATABASES["reporting"]["HOST"]
        db_user = settings.DATABASES["reporting"]["USER"]
        db_pass = settings.DATABASES["reporting"]["PASSWORD"]
        db_name = self._resolve_tenant_db(tenant_domain)

        conn = mysql.connector.connect(
            host=db_host, user=db_user, password=db_pass, database=db_name
        )
        cursor = conn.cursor(dictionary=True)

        try:
            cursor.execute(raw_sql, params)
            results = cursor.fetchall()
        except mysql.connector.Error as exc:
            logger.warning("Report query failed: %s", exc)
            raise ValidationError("Internal error processing report.")
        finally:
            cursor.close()
            conn.close()

        return Response(results)
```

**Key principles applied:**
- Column identifiers (`metric`, `group_by`) pass through a hard-coded allowlist *before* entering the query string. This prevents injection because SQL keywords like `UNION` are never literal match values.
- Scalar values (`sd`, `ed`) use `%s` placeholders passed as a tuple to `cursor.execute(sql, params)` — the MySQL driver handles escaping.
- The connection uses settings-resolved credentials rather than hardcoded strings (ties into Issue 3).

```python
# reports/views.py (fixed)
import logging
from decouple import config  # or environ
from django.db import connection
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.exceptions import NotFound, PermissionDenied

logger = logging.getLogger(__name__)

# Validated allowed statuses — allowlist instead of passing through
ALLOWED_REPORT_STATUSES = frozenset({"draft", "published", "archived"})


class ReportDetailView(APIView):
    """Hardened: parameterized queries only, server-side auth, validated inputs."""

    def get(self, request, pk, tenant_slug):
        # ── 1. Validate tenant slug against known tenants (prevent DB name injection) ──
        if not self._is_valid_tenant(tenant_slug):
            raise NotFound("Tenant not found")

        # ── 2. Validate pk is an integer ──
        try:
            report_id = int(pk)
        except (ValueError, TypeError):
            raise NotFound("Invalid report ID")

        # ── 3. Validate status against allowlist ──
        status = request.query_params.get("status", "published")
        if status not in ALLOWED_REPORT_STATUSES:
            raise NotFound(f"Invalid status: {status}")

        # ── 4. Server-side ownership check (see Issue 2) ──
        if not self._check_owner(request.user, tenant_slug, report_id):
            raise PermissionDenied("You do not have access to this report.")

        # ── 5. Parameterized query — NEVER concatenate user input ──
        with connection.cursor() as cursor:
            cursor.execute(
                f"SELECT id, title, data "
                f"FROM \"{tenant_slug}\".reports "
                f"WHERE id = %s AND status = %s",
                [report_id, status],  # ← parameters, not interpolation
            )
            row = cursor.fetchone()

        if row is None:
            raise NotFound("Report not found")

        return Response({"report": {"id": row[0], "title": row[1], "data": row[2]}})

    @staticmethod
    def _is_valid_tenant(slug: str) -> bool:
        """Check tenant exists in a trusted registry — never trust the URL alone."""
        from tenant_manager.models import Tenant  # hypothetical model

        # Escape-safe: using ORM lookup, not string interpolation
        return Tenant.objects.filter(slug=slug, is_active=True).exists()

    def _check_owner(self, user, tenant_slug: str, report_id: int) -> bool:
        """Server-side object-level authorization — see Issue 2."""
        from reports.models import Report

        return Report.objects.filter(
            id=report_id,
            tenant__slug=tenant_slug,
        ).exists()  # ORM handles escaping automatically
```

| No tenant validation | `Tenant.objects.filter().exists()` checks against known tenants |

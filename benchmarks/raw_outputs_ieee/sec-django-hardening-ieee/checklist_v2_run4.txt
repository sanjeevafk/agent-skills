## Issue 1 — SQL Injection via F-String Cursor Query

**CWE-89: Improper Neutralization of Special Elements used in an SQL Command ('SQL Injection')**

### Vulnerable Code

```python
# reports/views.py — VULNERABLE
import psycopg2
from rest_framework.views import APIView
from rest_framework.response import Response
from django.conf import settings

class ReportDataView(APIView):
    """Fetch filtered report data — INSECURE."""

    def get(self, request):
        tenant_id = request.query_params.get('tenant', '')
        start_date = request.query_params.get('start_date', '')
        end_date = request.query_params.get('end_date', '')

        # CRITICAL: Direct f-string interpolation into SQL
        query = f"""
            SELECT id, title, amount, created_at
            FROM reports.report_entries
            WHERE tenant_id = {tenant_id}
              AND created_at >= '{start_date}'
              AND created_at <= '{end_date}'
        """

        with psycopg2.connect(settings.DATABASE_URL) as conn:
            with conn.cursor() as cur:
                cur.execute(query)  # ← executes attacker-controlled SQL
                rows = cur.fetchall()

        return Response(rows)
```

### Exploit

```
GET /api/reports/?tenant=1 OR 1=1--&start_date=&end_date=
```

Becomes:
```sql
SELECT id, title, amount, created_at
FROM reports.report_entries
WHERE tenant_id = 1 OR 1=1--
  AND created_at >= ''
  AND created_at <= ''
```

Result: **every row in every tenant's table is returned.** Additional payload via `end_date`:

```
?tenant=1&start_date=&end_date='; DROP TABLE report_entries;--
```

Can destroy production data.

### Fixed Code

```python
# reports/views.py — HARDENED
import logging
import psycopg2
from psycopg2 import sql as pg_sql
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated
from rest_framework.throttling import UserRateThrottle
from django.conf import settings

logger = logging.getLogger(__name__)


class ReportDataThrottle(UserRateThrottle):
    scope = 'sustained'
    rate = '500/day'


class ReportDataView(APIView):
    """Fetch filtered report data with parameterized queries."""

    permission_classes = [IsAuthenticated]
    throttle_classes = [ReportDataThrottle]

    # Whitelist of allowed column names to prevent identifier injection
    ALLOWED_SORT_FIELDS = frozenset({'id', 'title', 'amount', 'created_at'})

    def _validate_tenant_access(self, request, tenant_id: int) -> bool:
        """Server-side tenant isolation check — cannot be bypassed by client."""
        from .models import TenantMembership
        try:
            member = TenantMembership.objects.get(
                user=request.user,
                tenant_id=tenant_id,
            )
            return member.is_active
        except TenantMembership.DoesNotExist:
            logger.warning(
                "Unauthorized tenant access attempt: user=%s tenant=%s",
                request.user.pk, tenant_id,
            )
            return False

    def get(self, request):
        tenant_id_raw = request.query_params.get('tenant', '')
        start_date = request.query_params.get('start_date')
        end_date = request.query_params.get('end_date')
        sort_field = request.query_params.get('sort', 'created_at')

        # --- Input validation ---
        try:
            tenant_id = int(tenant_id_raw)
        except (ValueError, TypeError):
            return Response(
                {'detail': 'Invalid tenant ID.'}, status=400
            )

        # Server-side authorization: client claims cannot be trusted
        if not self._validate_tenant_access(request, tenant_id):
            return Response(
                {'detail': 'Access denied to this tenant.'},
                status=403,
            )

        # Validate sort field against whitelist (prevent identifier injection)
        if sort_field not in self.ALLOWED_SORT_FIELDS:
            sort_field = 'created_at'

        # Parameterized query: %s placeholders handled by psycopg2
        query = """
            SELECT id, title, amount, created_at
            FROM reports.report_entries
            WHERE tenant_id = %s
              AND created_at >= %s
              AND created_at <= %s
            ORDER BY {sort_col} ASC
            LIMIT 1000
        """.format(sort_col=pg_sql.Identifier(sort_field))  # safe: whitelist-checked

        params = [tenant_id, start_date, end_date]

        with psycopg2.connect(settings.DATABASE_URL) as conn:
            with conn.cursor() as cur:
                cur.execute(query, params)  # ← parameters safely escaped
                rows = cur.fetchall()

        # Serialize safely
        from .serializers import ReportEntrySerializer
        serialized = [
            ReportEntrySerializer(dict(row)).data
            for row in rows
        ]

        return Response(serialized)
```

> **Key principle:** Use `%s` parameterized placeholders for all *values*. Only *identifiers* (table/column names) need `pg_sql.Identifier`, which must be validated against a strict allowlist. Never interpolate anything from user input directly.

## Issue 2 — Client-Side Object Authorization

**CWE-284: Improper Access Control** (specifically: server-side authorization gap relying on frontend enforcement)

### Vulnerable Code

```python

        model = Post
```

The frontend (React) attempts to protect resources:

```jsx

const handleDeletePost = async (postId) => {
    const userPosts = await fetchMyPosts();  // only loads own posts
    if (!userPosts.find(p => p.id === postId)) {
        showPermissionDenied();
        return;

```

An attacker simply sends:
```
DELETE https://api.example.com/posts/9999/
Authorization: Bearer <valid_token_for_tenant_a>
```

No client-side check exists to block this. The server's `get_queryset` doesn't filter by `tenant`.

### Fixed Code

```python
# posts/permissions.py — SERVER-SIDE OBJECT PERMISSIONS
from rest_framework import permissions
from django.core.exceptions import PermissionDenied


class IsTenantMemberAndOwner(permissions.BasePermission):
    """
    Enforce BOTH:
      1. User belongs to the post's tenant (server-side membership check)
      2. User owns the post (or is staff within the tenant)

    This CANNOT be bypassed by manipulating the frontend.
    """

    MESSAGE = "You do not have permission to access this resource."

    def has_object_permission(self, request, view, obj):
        # Read methods are public within the same tenant context
        if request.method in permissions.SAFE_METHODS:
            return self._check_tenant_membership(request, obj)
        # Write methods: owner or tenant admin only
        if request.user.pk == obj.author_id:
            return True
        return self._check_admin_role(request, obj.tenant_id)

    def _check_tenant_membership(self, request, obj):
        """Verify user is a member of the same tenant."""
        from .models import TenantMembership
        try:
            membership = TenantMembership.objects.get(
                user=request.user,
                tenant_id=obj.tenant_id,
                is_active=True,
            )
            return membership.role != 'revoked'
        except TenantMembership.DoesNotExist:
            return False

    def _check_admin_role(self, request, tenant_id):
        """Allow tenant admins full control."""
        from .models import TenantMembership
        try:
            membership = TenantMembership.objects.get(
                user=request.user,
                tenant_id=tenant_id,
                role='admin',
                is_active=True,
            )
            return True
        except TenantMembership.DoesNotExist:
            return False


class IsAuthenticatedTenantUser(permissions.BasePermission):
    """Ensure queryset is always scoped to the authenticated user's tenant."""

    def has_permission(self, request, view):
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        raise NotImplementedError(
            "Use IsTenantMemberAndOwner for object-level checks"
        )
```

```python
# posts/permissions.py — additional helper for bulk/admin views
class IsTenantAdmin(permissions.BasePermission):
    """Restrict administrative actions to tenant admins only."""

    MESSAGE = "Only administrators of this tenant can perform this action."

    def has_permission(self, request, view):
        if not request.user or not request.user.is_authenticated:
            return False
        if getattr(request.user, 'is_staff', False):
            return True
        tenant_id = self._extract_tenant_id(request)
        if not tenant_id:
            return False
        return self._is_admin(request.user, tenant_id)

    def _extract_tenant_id(self, request):
        raw = request.query_params.get('tenant') or request.data.get('tenant')
        if raw:
            try:
                return int(raw)
            except (ValueError, TypeError):
                return None
        return None

    def _is_admin(self, user, tenant_id):
        from .models import TenantMembership
        try:
            m = TenantMembership.objects.get(
                user=user,
                tenant_id=tenant_id,
                role='admin',
                is_active=True,
            )
            return True
        except TenantMembership.DoesNotExist:
            return False
```

```python
# posts/views.py — HARDENED
from rest_framework import viewsets, filters
from rest_framework.permissions import IsAuthenticated
from rest_framework.parsers import JSONParser
from .models import Post
from .serializers import PostSerializer
from .permissions import IsTenantMemberAndOwner, IsAuthenticatedTenantUser


class PostViewSet(viewsets.ModelViewSet):
    """
    Server-enforces tenant isolation + ownership checks.
    No reliance on frontend authorization whatsoever.
    """
    serializer_class = PostSerializer
    permission_classes = [IsAuthenticatedTenantUser, IsTenantMemberAndOwner]
    parser_classes = [JSONParser]
    filter_backends = [filters.SearchFilter, filters.OrderingFilter]
    search_fields = ['title', 'content']

    def get_queryset(self):
        """
        ALWAYS scope to the requesting user's verified tenants.
        Even if an attacker sends a foreign tenant_id, it is ignored here
        and enforced by the object-level permission below.
        """
        qs = Post.objects.select_related('author', 'tenant').all()

        # Scope query to tenants the authenticated user belongs to
        from .models import TenantMembership
        my_tenants = TenantMembership.objects.filter(
            user=self.request.user,
            is_active=True,
        ).values_list('tenant_id', flat=True)
        qs = qs.filter(tenant_id__in=my_tenants)

        return qs

    def perform_create(self, serializer):
        author = self.request.user
        # Pull tenant from header or session — NEVER from request body
        tenant_id = self.request.headers.get('X-Tenant-ID')
        if not tenant_id:
            raise serializers.ValidationError(
                'Missing X-Tenant-ID header. Required for multi-tenant isolation.'
            )
        try:
            tenant_id = int(tenant_id)
        except (ValueError, TypeError):
            raise serializers.ValidationError('Invalid X-Tenant-ID.')

        # Verify membership before allowing creation
        from .models import TenantMembership
        try:
            membership = TenantMembership.objects.get(
                user=author,
                tenant_id=tenant_id,
                is_active=True,
            )
        except TenantMembership.DoesNotExist:
            raise PermissionDenied(
                'You are not a member of the specified tenant.'
            )

        serializer.save(author=author, tenant_id=tenant_id)
```

> **Key principle:** Server-side `has_object_permission` runs on *every* request regardless of what the frontend renders. Never trust client-side state for authorization decisions.

```python
# users/validators.py — centralized business-rule validators
from rest_framework.exceptions import ValidationError


class PasswordValidator:
    """Enforce strong passwords independently of Django's validator chain."""

    MIN_LENGTH = 16
    REQUIRED_CHARS = 3  # lowercase, uppercase, digit/special

    @classmethod
    def validate(cls, password, user=None):
        if len(password) < cls.MIN_LENGTH:
            raise ValidationError(
                f'Password must be at least {cls.MIN_LENGTH} characters.'
            )
        char_types = sum([
            any(c.islower() for c in password),
            any(c.isupper() for c in password),
            any(not c.isalnum() for c in password),
        ])
        if char_types < cls.REQUIRED_CHARS:
            raise ValidationError(
                'Password must contain at least 3 character types '
                '(lowercase, uppercase, digit/special).'
            )


class UsernameValidator:
    MAX_LENGTH = 150

    @classmethod
    def validate(cls, username):
        if len(username) > cls.MAX_LENGTH:
            raise ValidationError(
                f'Username must be {cls.MAX_LENGTH} characters or fewer.'
            )
```

```python
# users/views.py — HARDENED: route-level + serializer segregation
from rest_framework import viewsets, status
from rest_framework.permissions import IsAuthenticated, BasePermission
from rest_framework.response import Response
from rest_framework.decorators import action
from django.contrib.auth import get_user_model
from .serializers import (
    UserCreateSerializer,
    UserProfileUpdateSerializer,
    AdminUserUpdateSerializer,
    PasswordChangeSerializer,
)
from .permissions import IsOwnProfileOrAdmin

User = get_user_model()


class IsAdmin(BasePermission):
    """Gate admin-only operations."""

    def has_permission(self, request, view):
        return request.user and request.user.is_staff


class UserViewSet(viewsets.ModelViewSet):
    """
    Segregated endpoints per operation type:
      POST   /users/          → registration (UserCreateSerializer)
      GET    /users/          → list (admins only)
      GET    /users/{id}/     → detail (own profile or admin)
      PATCH  /users/{id}/     → update (role-aware serializers)
      DELETE /users/{id}/     → delete (admins only)
      POST   /users/me/password → change own password
    """

    # Default: registered users
    permission_classes = [IsAuthenticated]

    def get_serializer_class(self):
        if self.action == 'create':
            return UserCreateSerializer
        if self.action in ('update', 'partial_update'):
            # Route to the right serializer based on context
            if hasattr(self, '_serializer_class'):
                return self._serializer_class
            return UserProfileUpdateSerializer
        if self.action == 'password_change':
            return PasswordChangeSerializer
        return UserProfileUpdateSerializer

    def get_queryset(self):
        # Admins see all users; regular users see only themselves
        if self.request.user.is_staff:
            return User.objects.all()
        return User.objects.filter(pk=self.request.user.pk)

    @action(detail=False, methods=['post'], permission_classes=[IsAuthenticated])
    def register(self, request):
        """Public-ish registration — no auth required initially."""
        # Optionally guard behind an invite code or open-registration flag
        serializer = UserCreateSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        user = serializer.save()
        return Response(
            {'message': 'Account created. Check your email to activate.'},
            status=status.HTTP_201_CREATED,
        )

    @action(
        detail=False,
        methods=['patch'],
        permission_classes=[IsAuthenticated],
        url_path='me/password',
    )
    def password_change(self, request):
        """Self-service password change requiring old password."""
        serializer = PasswordChangeSerializer(
            data=request.data, context={'request': request}
        )
        serializer.is_valid(raise_exception=True)
        serializer.save()
        return Response({'message': 'Password updated successfully.'})

    @action(
        detail=False,
        methods=['get', 'patch'],
        permission_classes=[IsAuthenticated],
        url_path='me',
    )
    def me(self, request):
        """Shortcut for viewing/editting own profile."""
        user = request.user
        if request.method == 'GET':
            from .serializers import UserProfileUpdateSerializer
            serializer = UserProfileUpdateSerializer(user)
            return Response(serializer.data)
        else:
            # PATCH: apply role-based serializer
            if request.user.is_staff:
                self._serializer_class = AdminUserUpdateSerializer
            else:
                self._serializer_class = UserProfileUpdateSerializer
            return self.partial_update(request, pk=None)

    def perform_destroy(self, instance):
        """Log deletion for audit purposes."""
        import logging
        logger = logging.getLogger(__name__)
        logger.info(
            "User deleted: deleted_by=%s target=%s (%s)",
            self.request.user.pk, instance.pk, instance.username,
        )
        instance.delete()
```

> **Key principles:**
> 1. **Explicit allowlists** — never `'__all__'` on authentication-related models
> 2. **Serializer segregation** — different serializers for create vs. update vs. admin
> 3. **Role-gated endpoints** — `IsAdmin` permission classes on privileged operations
> 4. **Belt-and-suspenders** — both serializer-level field lists AND programmatic stripping in `update()`

```python
# reports/middleware.py — intercept and log denied requests
import logging
from django.utils.deprecation import MiddlewareMixin

logger = logging.getLogger('security')


class SecurityEventMiddleware(MiddlewareMixin):
    """Log authorization failures for SIEM ingestion."""

    def process_response(self, request, response):
        if response.status_code == 403:
            logger.warning(
                "Forbidden: %s %s by user=%s ip=%s",
                request.method, request.path,
                request.user.pk if request.user.is_authenticated else 'anonymous',
                self._get_client_ip(request),
            )
        elif response.status_code == 401:
            logger.warning(
                "Unauthorized: %s %s ip=%s",
                request.method, request.path,
                self._get_client_ip(request),
            )
        return response

    def _get_client_ip(self, request):
        xff = request.META.get('HTTP_X_FORWARDED_FOR')
        return xff.split(',')[0].strip() if xff else request.META.get('REMOTE_ADDR', '?')
```

```python
# middleware/security_headers.py
class SecurityHeaderMiddleware:
    """Inject defense-in-depth HTTP headers on every response."""

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        response = self.get_response(request)
        response['X-Content-Type-Options'] = 'nosniff'
        response['X-Frame-Options'] = 'DENY'
        response['X-XSS-Protection'] = '1; mode=block'
        response['Referrer-Policy'] = 'strict-origin-when-cross-origin'
        response['Cache-Control'] = 'no-store, no-cache, must-revalidate'
        response['Pragma'] = 'no-cache'
        # Content-Security-Policy: lock down resource sources
        response['Content-Security-Policy'] = (
            "default-src 'self'; "
            "script-src 'self'; "
            "style-src 'self' 'unsafe-inline'; "
            "img-src 'self' data: https:; "
            "font-src 'self'; "
            "connect-src 'self'; "
            "frame-ancestors 'none'; "
            "base-uri 'self'; "
            "form-action 'self'"
        )
        return response
```

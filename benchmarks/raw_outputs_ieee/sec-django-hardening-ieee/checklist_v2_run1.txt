## 1. SQL Injection via f-string in `cursor.execute()`

**CWE-89**: Improper Neutralization of Special Elements used in an SQL Command (SQL Injection)

### The Exploit

```python
# apps/reports/views.py — VULNERABLE
from django.db import connection

class ReportViewSet(viewsets.ModelViewSet):
    def list(self, request):
        # Multi-tenant filter built by f-string
        tenant_id = request.query_params.get('tenant', 'default')
        status = request.query_params.get('status', 'active')
        start = request.query_params.get('start_date')
        
        query = f"""
            SELECT id, name, revenue
            FROM reports_report
            WHERE tenant_id = '{tenant_id}'
              AND status = '{status}'
              {f"AND created_at >= '{start}'" if start else ""}
        """
        with connection.cursor() as cursor:
            cursor.execute(query)
            rows = cursor.fetchall()
        return Response(rows)
```

An attacker sends:
```
GET /api/reports/?tenant=1'; DROP TABLE reports_report;--&status=active
```

The resulting query:
```sql
SELECT id, name, revenue
FROM reports_report
WHERE tenant_id = '1'; DROP TABLE reports_report;--'
  AND status = 'active' ...
```

`DROP TABLE reports_report` executes. Data loss is immediate.

### The Fix

Never concatenate user input into SQL strings. Use parameterized queries exclusively, and prefer the ORM when possible. If raw execution is unavoidable, use positional parameters.

```python
# apps/reports/views.py — HARDENED
from rest_framework import viewsets, permissions, status
from rest_framework.response import Response
from django.db import connection
from .models import Report
from .serializers import ReportSerializer


class IsTenantMember(permissions.BasePermission):
    """Ensure the requesting user belongs to the queried tenant."""

    def has_permission(self, request, view):
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        return obj.tenant.users.filter(id=request.user.id).exists()


class ReportViewSet(viewsets.ReadOnlyModelViewSet):
    queryset = Report.objects.all()
    serializer_class = ReportSerializer
    permission_classes = [permissions.IsAuthenticated, IsTenantMember]

    def get_queryset(self):
        qs = Report.objects.filter(tenant__users=self.request.user)

        tenant_filter = self.request.query_params.get("tenant")
        if tenant_filter:
            # Validate against known tenants — never pass directly into SQL
            valid_tenant_ids = set(
                self.request.user.tenant_set.values_list("id", flat=True)
            )
            if str(tenant_filter) not in {str(tid) for tid in valid_tenant_ids}:
                raise PermissionDenied("Access denied to specified tenant.")
            qs = qs.filter(tenant_id=tenant_filter)

        status_filter = self.request.query_params.get("status")
        if status_filter:
            qs = qs.filter(status=status_filter)

        start_date = self.request.query_params.get("start_date")
        if start_date:
            from datetime import datetime
            try:
                qs = qs.filter(created_at__gte=datetime.fromisoformat(start_date))
            except ValueError:
                raise ParseError("Invalid start_date format.")

        return qs.order_by("-created_at")

    # ---- Fallback: raw queries still appear in complex reporting ----
    def _run_safe_raw_query(self, tenant_id: int, filters: dict) -> list[dict]:
        """Example of how to safely execute raw SQL when ORM can't express the query."""
        base = "SELECT id, name, revenue FROM reports_report WHERE tenant_id = %s"
        params: list = [tenant_id]
        conditions = []

        if filters.get("status"):
            conditions.append("status = %s")
            params.append(filters["status"])
        if filters.get("min_revenue"):
            conditions.append("revenue >= %s")
            params.append(float(filters["min_revenue"]))
        if filters.get("start_date"):
            conditions.append("created_at >= %s")
            params.append(filters["start_date"])

        if conditions:
            base += " AND " + " AND ".join(conditions)

        base += " ORDER BY created_at DESC LIMIT 100"

        with connection.cursor() as cursor:
            cursor.execute(base, params)   # <-- parameterized, NEVER f-string
            columns = [col[0] for col in cursor.description]
            return [dict(zip(columns, row)) for row in cursor.fetchall()]
```

## 2. Client-Side-Only Object-Level Authorization

**CWE-284**: Improper Access Control (Missing Server-Side Authorization Check)
Also **CWE-639**: Authorization Bypass Through Trusting Client Data

### The Problem

```jsx
// frontend/components/ReportDetail.jsx — VULNERABLE
function ReportDetail({ report }) {
  const user = useUser();
  // THE EDIT BUTTON IS ONLY HIDDEN ON THE FRONTEND
  if (report.author === user.id) {
    return <EditForm report={report} />;
  }
  return <ReadOnlyView report={report} />;
}
```

The API endpoint accepts the PUT/PATCH because DRF's generic `UpdateAPIView` has no object-level permission check — it trusts the client decided who gets to edit. An authenticated user simply crafts a direct curl:

```bash
curl -X PATCH https://api.example.com/api/reports/42/ \
  -H "Authorization: Bearer <user-token>" \
  -H "Content-Type: application/json" \
  -d '{"title": "hacked report", "status": "published"}'
```

Any authenticated user can modify any report.

### The Fix

Server-side object-level permissions enforced at the DRF level. Every read/write action must verify ownership *before* touching the database.

```python
# apps/common/permissions.py — HARDENED
from rest_framework import permissions
from django.core.exceptions import PermissionDenied


class IsOwnerOrReadOnly(permissions.BasePermission):
    """
    Global: reads allowed for any authenticated user.
    Per-object: writes require object ownership or staff status.
    """

    def has_permission(self, request, view):
        if request.method in permissions.SAFE_METHODS:
            return request.user and request.user.is_authenticated
        return request.user and request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        # Allow full CRUD to owner OR staff
        if request.method in permissions.SAFE_METHODS:
            return True
        if request.user.is_staff:
            return True
        # Generic "author" / "owner" field lookup — subclass if your model uses
        # a different attribute name
        author_field = getattr(obj, "author", None) or getattr(obj, "owner", None)
        if author_field:
            return author_field.id == request.user.id
        return False


class IsTenantAdmin(permissions.BasePermission):
    """Only tenant admins or super-staff can modify tenant-scoped resources."""

    def has_object_permission(self, request, view, obj):
        if request.user.is_superuser:
            return True
        tenant = getattr(obj, "tenant", None)
        if not tenant:
            return False
        return tenant.admins.filter(id=request.user.id).exists()


class TenantScopeFilter:
    """
    Mixin that restricts querysets to the current user's tenants.
    Use as a mixin alongside DRF mixins/APIViews.
    """

    def get_queryset(self):
        qs = super().get_queryset()
        return qs.filter(tenant__users=self.request.user)
```

Applied to the report viewset:

```python
# apps/reports/views.py — HARDENED
from rest_framework import viewsets, permissions
from .models import Report
from .serializers import ReportSerializer


class ReportViewSet(viewsets.ModelViewSet):
    queryset = Report.objects.all()
    serializer_class = ReportSerializer
    permission_classes = [permissions.IsAuthenticated, IsOwnerOrReadOnly]

    def get_queryset(self):
        # Server-side tenant scoping — never trust ?tenant= from the client alone
        return Report.objects.filter(
            tenant__users=self.request.user
        ).distinct()

    def perform_create(self, serializer):
        serializer.save(author=self.request.user)
```

## 3. Hardcoded Secrets in Committed Settings

**CWE-798**: Use of Hard-coded Credentials

### The Exploit

```python
# settings/base.py — VULNERABLE
SECRET_KEY = "django-insecure-k#9x$p!z@q1m2w$r5t^y&7g"
DATABASES = {
    "default": {
        "ENGINE": "django.db.backends.postgresql",
        "NAME": "myapp_production",
        "USER": "admin",
        "PASSWORD": "P@ssw0rd_123!",       # <-- committed to git
        "HOST": "prod-db.internal",
        "PORT": "5432",
    }
}
```

Anyone with git history access (GitHub public repos, leaked archives, CI logs, developer machines) gets immediate RCE-level database access. `git log --all -p | grep SECRET` recovers even deleted versions.

### The Fix

Load all secrets from environment variables or a vault at runtime. Never commit credential files.

```python
# settings/base.py — HARDENED

    CORS_ORIGINS=(str, "http://localhost:3000"),
    REDIS_URL=(str, "redis://localhost:6379/0"),
    AWS_S3_BUCKET=(str, ""),
    AWS_REGION=(str, "us-east-1"),
    SENTRY_DSN=(str, ""),
)
environ.Env.read_env(BASE_DIR / ".env")  # Only loaded locally; not in prod image

# ---------------------------------------------------------------------------

if not DEBUG:
    assert env.str("SECRET_KEY"), (
        "SECRET_KEY must be set in environment when DEBUG=False"
    )

ALLOWED_HOSTS = [h.strip() for h in env.str("ALLOWED_HOSTS").split(",") if h.strip()]
```

## 4. Mass Assignment on Serializer (Staff Status Toggle)

**CWE-285**: Improper Authorization (Mass Assignment)
Also **CWE-915**: Improper Control of Dynamically Identified Variables

### The Exploit

```python
# apps/users/serializers.py — VULNERABLE
class UserSerializer(serializers.ModelSerializer):
    class Meta:
        model = User
        fields = "__all__"          # Exposes EVERY field including is_staff
        read_only_fields = ()       # Nothing read-only — user controls everything


# apps/users/views.py
class UserViewSet(viewsets.ModelViewSet):
    queryset = User.objects.all()
    serializer_class = UserSerializer   # No extra guards
```

A regular authenticated user sends:

```bash
curl -X PATCH https://api.example.com/api/users/me/ \
  -H "Authorization: Bearer <token>" \
  -H "Content-Type: application/json" \
  -d '{"is_staff": true, "is_superuser": true}'
```

Immediate privilege escalation. `fields = "__all__"` passes every model field through the serializer unfiltered.

### The Fix

Explicitly whitelist fields per role. Staff/superuser fields are *never* writable through the standard serializer. Admin-only endpoints handle privileged mutations.

```python
# apps/users/serializers.py — HARDENED
from django.contrib.auth import get_user_model
from rest_framework import serializers
from django.conf import settings

User = get_user_model()

PROTECTED_USER_FIELDS = {"is_staff", "is_superuser", "is_active", "groups", "user_permissions"}
# Password should always be write-only; never expose hash
PROTECTED_READONLY_ALWAYS = PROTECTED_USER_FIELDS | {"password", "last_login", "date_joined"}


class UserCreateSerializer(serializers.ModelSerializer):
    """Used only for registration — no privileged fields."""

    password = serializers.CharField(write_only=True, required=True, style={"input_type": "password"})
    password_confirm = serializers.CharField(write_only=True, required=True, style={"input_type": "password"})

    class Meta:
        model = User
        fields = ("id", "username", "email", "phone", "password", "password_confirm")

    def validate(self, attrs):
        if attrs.pop("password") != attrs.pop("password_confirm"):
            raise serializers.ValidationError({"password_confirm": "Passwords do not match."})
        return attrs

    def create(self, validated_data):
        return User.objects.create_user(**validated_data)


class UserUpdateSerializer(serializers.ModelSerializer):
    """Regular users can update profile fields only."""

    class Meta:
        model = User
        # Explicit allowlist — nothing beyond what a normal user should change
        fields = ("username", "email", "phone", "first_name", "last_name")
        # Staff, superuser, active flag, groups, and permissions are NEVER here


class UserAdminSerializer(serializers.ModelSerializer):
    """Restricted to admin/staff users only."""

    class Meta:
        model = User
        fields = (
            "id", "username", "email", "phone",
            "first_name", "last_name",
            "is_staff", "is_superuser", "is_active",
            "groups", "user_permissions", "date_joined", "last_login",
        )
        read_only_fields = ("id", "date_joined", "last_login")


class ChangePasswordSerializer(serializers.Serializer):
    """Secure password reset for authenticated users."""

    current_password = serializers.CharField(required=True, write_only=True)
    new_password = serializers.CharField(required=True, write_only=True, min_length=getattr(settings, "MIN_PASSWORD_LENGTH", 12))
    new_password_confirm = serializers.CharField(required=True, write_only=True)

    def validate(self, attrs):
        if attrs["new_password"] != attrs["new_password_confirm"]:
            raise serializers.ValidationError({"new_password_confirm": "Passwords do not match."})
        return attrs

    def save(self, user, request):
        user.set_password(self.validated_data["new_password"])
        user.save()
        return user
```

```python
# apps/users/views.py — HARDENED
from rest_framework import viewsets, permissions, status
from rest_framework.decorators import action
from rest_framework.response import Response
from django.contrib.auth import get_user_model
from django.core.exceptions import ValidationError

User = get_user_model()


class IsStaffOrOwner(permissions.BasePermission):
    """Allow staff users to manage anyone; allow owners to manage themselves."""

    def has_permission(self, request, view):
        if request.user and request.user.is_staff:
            return True
        # Self-service actions (profile update, password change)
        if view.action in ("update", "partial_update", "me", "change_password"):
            return request.user and request.user.is_authenticated
        return False

    def has_object_permission(self, request, view, obj):
        if request.user.is_staff or request.user.is_superuser:
            return True
        return obj.id == request.user.id


class UserViewSet(viewsets.ModelViewSet):
    queryset = User.objects.all()
    permission_classes = [permissions.IsAuthenticated, IsStaffOrOwner]

    def get_serializer_class(self):
        # Route serializer by action — never use a single permissive serializer
        if self.action == "create":
            return UserCreateSerializer
        if self.action in ("list", "retrieve") and (
            self.request.user.is_staff or self.request.user.is_superuser
        ):
            return UserAdminSerializer
        # Self-service: profile update, retrieve own data
        return UserUpdateSerializer

    @action(detail=False, methods=["get"], url_path="me")
    def me(self, request):
        """Return the authenticated user's own profile — read."""
        serializer = UserUpdateSerializer(request.user)
        return Response(serializer.data)

    @action(detail=False, methods=["post"], url_path="change-password")
    def change_password(self, request):
        """Secure password change — validates current password."""
        user = request.user
        serializer = ChangePasswordSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        if not user.check_password(serializer.validated_data["current_password"]):
            raise ValidationError({"current_password": "Current password is incorrect."})

        serializer.save(user=user)
        return Response({"detail": "Password updated."}, status=status.HTTP_200_OK)

    @action(detail=False, methods=["post"], url_path="logout-all")
    def logout_all(self, request):
        """Invalidate all sessions for the current user."""
        from django.contrib.sessions.backends.db import SessionStore
        session_key = request.session.session_key
        for key in request.session.keys():
            SessionStore(key_key=key).delete()
        if session_key:
            request.session.flush()
        return Response({"detail": "Logged out from all devices."}, status=status.HTTP_200_OK)
```

```python
# apps/users/middleware.py — Detect and log mass-assignment attempts
from django.utils.deprecation import MiddlewareMixin
from functools import wraps


def log_protected_write(field: str):
    """Decorator that logs when a protected field is attempted to be written."""
    def decorator(func):
        @wraps(func)
        def wrapper(self, data, *args, **kwargs):
            protected_fields = {"is_staff", "is_superuser", "is_active", "groups", "user_permissions"}
            for pf in protected_fields:
                if pf in data and data[pf] not in (None, ""):
                    import logging
                    logger = logging.getLogger("django.security.mass_assignment")
                    logger.warning(
                        "Blocked mass assignment: user=%s tried to set %s=%s",
                        self.context.get("request", {}).user.id if self.context.get("request") else "unknown",
                        pf, data[pf],
                    )
            return func(self, data, *args, **kwargs)
        return wrapper
    return decorator
```

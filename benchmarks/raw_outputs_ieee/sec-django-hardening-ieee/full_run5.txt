## 1. SQL Injection via f-string into `cursor.execute()`

**CWE-89: SQL Injection**

### The Vulnerable Code (`reports/views.py`)

```python
from django.db import connection
from rest_framework.views import APIView
from rest_framework.response import Response

class ReportEndpoint(APIView):
    """Builds filters by f-string concatenation — exploitable."""

    def get(self, request):
        # VULNERABLE: user input baked directly into SQL string
        category = request.query_params.get("category", "")
        tenant_id = request.query_params.get("tenant_id", "1")

        query = f"""
            SELECT id, title, created_at
            FROM reports
            WHERE tenant_id = {tenant_id}
              AND category = '{category}'
        """
        with connection.cursor() as cursor:
            cursor.execute(query)
            rows = cursor.fetchall()
        return Response({"data": rows})
```

### The Exploit

An attacker sends:

```
GET /api/reports/?category=' UNION SELECT password,username,email FROM users --&tenant_id=1
```

The f-string produces:

```sql
SELECT id, title, created_at FROM reports
WHERE tenant_id = 1 AND category = ''
  UNION SELECT password,username,email FROM users --'
```

Result: every user's credentials are returned in the report rows.

### The Fix

Always use parameterized queries. Django's ORM handles this automatically; when raw SQL is unavoidable, pass parameters as a second argument to `execute()`.

```python
from django.db import connection
from rest_framework.views import APIView
from rest_framework.response import Response


class ReportEndpoint(APIView):
    """Hardened — parameterized query, integer cast for safety."""

    def get(self, request):
        category = request.query_params.get("category", "")
        try:
            tenant_id = int(request.query_params.get("tenant_id", "1"))
        except (ValueError, TypeError):
            return Response(
                {"error": "Invalid tenant_id"}, status=400
            )

        # Parameterized — values are escaped by the DB driver
        query = """
            SELECT id, title, created_at
            FROM reports
            WHERE tenant_id = %s
              AND category = %s
        """
        with connection.cursor() as cursor:
            cursor.execute(query, [tenant_id, category])
            rows = cursor.fetchall()

        return Response({"data": rows})
```

**Even better** — ditch raw SQL entirely and use the ORM:

```python
from django.db.models import Q
from rest_framework.views import APIView
from rest_framework.response import Response


class ReportEndpoint(APIView):
    def get(self, request):
        category = request.query_params.get("category")
        try:
            tenant_id = int(request.query_params.get("tenant_id", "1"))
        except (ValueError, TypeError):
            return Response({"error": "Invalid tenant_id"}, status=400)

        qs = Report.objects.filter(tenant_id=tenant_id)
        if category:
            qs = qs.filter(category__iexact=category)

        data = list(qs.values("id", "title", "created_at"))
        return Response({"data": data})
```

## 2. Server-Side Object-Level Authorization Bypass

**CWE-284: Improper Access Control**

### The Vulnerable Code

```python
# frontend/ReportDetail.jsx — ONLY place where ownership is checked
function ReportDetail({ reportId }) {
    const [report, setReport] = useState(null);
    useEffect(() => {
        fetch(`/api/reports/${reportId}`)
            .then(r => r.json())
            .then(data => {
                // Trusting the server won't return someone else's data,
                // but nothing prevents the attacker from calling the API directly.
                if (data.owner_id === currentUser.id) {
                    setReport(data);
                } else {
                    // "Permission denied" shown in the UI only.
                    navigate("/unauthorized");
                }
            });
    }, [reportId]);
}

# reports/views.py — no object-level check at all
class ReportDetailView(APIView):
    """Returns ANY report given its ID. No ownership or permission check."""

    def get(self, request, pk):
        report = Report.objects.get(pk=pk)  # No tenant scoping!
        return Response(serialize(report))
```

An attacker crafts a curl request and reads any report regardless of ownership:

```bash
curl -H "Authorization: Bearer <attacker_token>" \
     https://app.example.com/api/reports/42/
```

### The Exploit Flow

1. Attacker authenticates and gets their own report (ID 7).
2. Brute-forces or discovers that report 42 belongs to another user.
3. Calls `/api/reports/42/` directly — no UI check blocks it.
4. Exfiltrates sensitive data.

### The Fix

Enforce object-level permissions **on the server**. Use Django REST Framework's `BasePermission` with `has_object_permission`.

```python
# reports/permissions.py
from rest_framework.permissions import BasePermission


class IsReportOwnerOrTenantAdmin(BasePermission):
    """Server-side enforcement: user can only access their own reports."""

    def has_object_permission(self, request, view, obj):
        # Tenant admins may read all reports in their tenant
        if request.user.is_staff:
            return obj.tenant == request.user.tenant

        # Owners can always read/edit their own report
        return obj.author == request.user


# reports/views.py
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.exceptions import NotFound
from .models import Report
from .serializers import ReportSerializer
from .permissions import IsReportOwnerOrTenantAdmin


class ReportDetailView(APIView):
    permission_classes = [IsAuthenticated, IsReportOwnerOrTenantAdmin]

    def get_object(self, pk):
        """Filter by user's tenant to prevent cross-tenant leakage."""
        try:
            return Report.objects.get(
                pk=pk,
                tenant_id=self.request.user.tenant_id,
            )
        except Report.DoesNotExist:
            raise NotFound("Report not found or access denied.")

    def get(self, request, pk):
        report = self.get_object(pk)
        # DRF will automatically call has_object_permission here if
        # using ModelViewSet, but for class-based views we gate manually:
        if not self.permission_classes[1].has_object_permission(
            request, self, report
        ):
            from django.http import JsonResponse
            return JsonResponse({"error": "Forbidden"}, status=403)
        serializer = ReportSerializer(report)
        return Response(serializer.data)
```

**Best practice** — use `ModelViewSet` which integrates `get_queryset` + permission classes cleanly:

```python
# reports/viewsets.py
from rest_framework import viewsets
from rest_framework.permissions import IsAuthenticated
from .models import Report
from .serializers import ReportSerializer
from .permissions import IsReportOwnerOrTenantAdmin


class ReportViewSet(viewsets.ModelViewSet):
    serializer_class = ReportSerializer
    permission_classes = [IsAuthenticated, IsReportOwnerOrTenantAdmin]

    def get_queryset(self):
        """Scope every query to the requesting user's tenant."""
        return Report.objects.filter(
            tenant_id=self.request.user.tenant_id
        )

    def perform_create(self, serializer):
        """Attach current user's tenant during creation."""
        serializer.save(author=self.request.user)
```

## 3. Secrets Management

**CWE-798: Use of Hard-coded Credentials**

### The Vulnerable Code (`core/settings.py`)

```python
# NEVER commit this file if it contains secrets.
DEBUG = False

SECRET_KEY = "django-insecure-x7$kq@1z9p!m2w^n5b&h8r*t$g6^y+v#"  # committed to git

DATABASES = {
    "default": {
        "ENGINE": "django.db.backends.postgresql",
        "NAME": "production_db",
        "USER": "admin",
        "PASSWORD": "SuperSecret123!",          # committed to git
        "HOST": "db.internal.example.com",
        "PORT": "5432",
    }
}
```

Anyone who clones the repository sees the production database password. GitHub's secret scan will flag it eventually, but the blast radius of an unrotated credential is already massive.

### The Exploit

```bash
git clone https://github.com/org/multi-tenant-app.git
cat core/settings.py | grep PASSWORD
# Output: "PASSWORD": "SuperSecret123!"
```

An attacker then connects directly to production:

```bash
psql -h db.internal.example.com -U admin -d production_db
# Password: SuperSecret123!
```

### The Fix

Use environment variables loaded via `django-environ` (or `python-decouple`). Never store secrets in version control.

```python
# core/settings.py — SAFE: references env vars only
import environ

env = environ.Env(
    DEBUG=(bool, False),
)
environ.Env.read_env()  # reads .env file if present (dev only)

SECRET_KEY = env("DJANGO_SECRET_KEY")

DATABASES = {
    "default": {
        "ENGINE": env("DB_ENGINE", default="django.db.backends.postgresql"),
        "NAME": env("DB_NAME"),
        "USER": env("DB_USER"),
        "PASSWORD": env("DB_PASSWORD"),
        "HOST": env("DB_HOST", default="localhost"),
        "PORT": env("DB_PORT", default="5432"),
    }
}

# Additional secrets
REDIS_URL = env("REDIS_URL", default="redis://localhost:6379/0")
AWS_S3_KEY = env("AWS_ACCESS_KEY_ID", default="")
AWS_S3_SECRET = env("AWS_SECRET_ACCESS_KEY", default="")
```

```bash
# .env — added to .gitignore, never committed
cat > .env << 'EOF'
DJANGO_SECRET_KEY=$(openssl rand -base64 48)
DB_NAME=production_db
DB_USER=admin
DB_PASSWORD=$(openssl rand -base64 32)
DB_HOST=db.internal.example.com
DB_PORT=5432
DB_ENGINE=django.db.backends.postgresql
EOF

chmod 600 .env
```

```text

*.key
*.pem
secrets.yaml
```

For CI/CD and hosting platforms, inject secrets as environment variables:

## 4. Mass Assignment — Staff Privilege Escalation

**CWE-915: Improperly Controlled Modification of Dynamically-Determined Object Attributes**

### The Vulnerable Code

```python
# accounts/serializers.py
from rest_framework import serializers
from django.contrib.auth import get_user_model

User = get_user_model()


class UserCreateSerializer(serializers.ModelSerializer):
    """Allows creating users through the API — but accepts ALL model fields."""

    class Meta:
        model = User
        # BUG: fields="__all__" lets callers pass staff_status, is_superuser, etc.
        fields = "__all__"
        extra_kwargs = {"password": {"write_only": True}}

    def create(self, validated_data):
        password = validated_data.pop("password")
        user = User(**validated_data)
        user.set_password(password)
        user.save()
        return user


# accounts/views.py
from rest_framework import generics
from .serializers import UserCreateSerializer

class UserCreateView(generics.CreateAPIView):
    serializer_class = UserCreateSerializer
    queryset = User.objects.all()
```

An authenticated regular user calls:

```bash
curl -X POST https://app.example.com/api/users/register/ \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer <user_token>" \
  -d '{
    "username": "backdoor",
    "email": "attacker@evil.com",
    "password": "complicatedpass123",
    "staff_status": true
  }'
```

The server stores the user with `is_staff=True`. The attacker now has admin panel access.

### The Exploit Flow

1. Register through public endpoint with `"staff_status": true`.
2. Login as the newly-created staff user.
3. Access Django admin or other staff-only endpoints.
4. Full administrative compromise.

### The Fix

Explicitly whitelist only the fields the serializer should accept. Separate the registration serializer from any admin-facing user serializer.

```python
# accounts/serializers.py
from rest_framework import serializers
from django.contrib.auth import get_user_model

User = get_user_model()


class UserRegistrationSerializer(serializers.ModelSerializer):
    """Strict allowlist: only fields needed for registration."""

    password = serializers.CharField(write_only=True, min_length=8)
    password_confirm = serializers.CharField(write_only=True)

    class Meta:
        model = User
        # Explicit allowlist — NO staff_status, NO is_superuser
        fields = ["email", "username", "password", "password_confirm"]

    def validate(self, data):
        if data["password"] != data["password_confirm"]:
            raise serializers.ValidationError(
                {"password_confirm": "Passwords do not match."}
            )
        return data

    def create(self, validated_data):
        validated_data.pop("password_confirm")
        password = validated_data.pop("password")
        user = User.objects.create_user(
            email=validated_data["email"],
            username=validated_data["username"],
            password=password,
            # These defaults cannot be overridden by the caller
            is_staff=False,
            is_superuser=False,
        )
        return user


# Admin-only serializer — used only inside staff-restricted views
class AdminUserSerializer(serializers.ModelSerializer):
    """Staff/admin-facing: includes staff_status, is_active, etc."""

    class Meta:
        model = User
        fields = [
            "id", "username", "email", "is_staff", "is_active",
            "date_joined", "last_login",
        ]
        read_only_fields = ["id", "date_joined", "last_login"]
```

```python
# accounts/views.py
from rest_framework import generics
from rest_framework.permissions import IsAuthenticated, IsAdminUser
from .serializers import UserRegistrationSerializer, AdminUserSerializer
from django.contrib.auth import get_user_model

User = get_user_model()


class UserRegistrationView(generics.CreateAPIView):
    """Public endpoint — uses strict allowlist serializer."""
    serializer_class = UserRegistrationSerializer
    # Anyone can register, including anonymous users
    permission_classes = []


class AdminUserListView(generics.ListAPIView):
    """Staff-only: list all registered users."""
    serializer_class = AdminUserSerializer
    permission_classes = [IsAuthenticated, IsAdminUser]

    def get_queryset(self):
        return User.objects.all().order_by("-date_joined")


class AdminUserUpdateView(generics.UpdateAPIView):
    """Staff-only: update user attributes including staff_status."""
    serializer_class = AdminUserSerializer
    permission_classes = [IsAuthenticated, IsAdminUser]

    def get_queryset(self):
        return User.objects.all()
```

```python
# urls.py
from django.urls import path, include
from rest_framework.routers import DefaultRouter

```

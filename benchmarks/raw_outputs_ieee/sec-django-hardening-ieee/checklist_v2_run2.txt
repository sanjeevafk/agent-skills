## Vulnerability 1: SQL Injection via F-String Cursor Execution

**CWE-89**: *Improper Neutralization of Special Elements used in an SQL Command*

### Vulnerable Code

```python
# reports/views.py — INSECURE
class ReportViewSet(viewsets.ModelViewSet):
    """Multi-tenant report endpoint."""
    queryset = Report.objects.all()
    serializer_class = ReportSerializer

    def list(self, request, tenant_id=None):
        cursor = connection.cursor()
        # CRITICAL VULNERABILITY: f-string injected directly into SQL
        filters = []
        params = []
        if request.GET.get('start_date'):
            filters.append(f"created_at >= '{request.GET['start_date']}'")
        if request.GET.get('end_date'):
            filters.append(f"created_at <= '{request.GET['end_date']}'")
        if request.GET.get('status'):
            filters.append(f"status = '{request.GET['status']}'")
        
        tenant_sql = f"SELECT * FROM reports WHERE tenant_id = {tenant_id}"
        if filters:
            tenant_sql += " AND " + " AND ".join(filters)
        
        cursor.execute(tenant_sql)
        rows = cursor.fetchall()
        return Response(rows)
```

### Exploit Demonstration

An attacker sends:

```bash
curl 'https://api.example.com/reports/1?status=' UNION SELECT password,password,password,password FROM users --'
```

Because `tenant_id` is also interpolated via f-string, an attacker can also escape the tenant boundary:

```bash
curl 'https://api.example.com/reports/1; DROP TABLE reports;--'
```

The raw SQL becomes:
```sql
SELECT * FROM reports WHERE tenant_id = 1; DROP TABLE reports;-- AND status = ''
```

### Fixed Code

```python
# reports/views.py — HARDENED
from django.db import connection
from rest_framework import viewsets, status
from rest_framework.response import Response
from rest_framework.exceptions import PermissionDenied, NotFound

# Shared parameters across queries to prevent parameter confusion
class SafeQuerySet:
    """Builds parameterized SQL with type-safe value binding."""

    @staticmethod
    def _validate_date(value: str) -> str:
        """Strictly validate date strings before use."""
        from datetime import datetime
        try:
            parsed = datetime.strptime(value, '%Y-%m-%d')
            return parsed.strftime('%Y-%m-%d')  # Round-trip sanitization
        except ValueError:
            raise ValidationError(
                f"'{value}' is not a valid date. Use YYYY-MM-DD format.",
                code='invalid'
            )

    @classmethod
    def build_report_query(cls, tenant_id: int, request_params: dict) -> tuple[str, list]:
        """
        Build a fully parameterized SQL query for report filtering.
        
        Returns (query_string, params_list) — never string interpolation.
        """
        # Validate tenant_id is an integer — reject anything else
        if not isinstance(tenant_id, int) or tenant_id < 1:
            raise ValueError(f"Invalid tenant_id: {tenant_id}")

        clauses = ["tenant_id = %s"]
        params: list = [tenant_id]

        start_date = request_params.get('start_date')
        if start_date:
            cls._validate_date(start_date)
            clauses.append("created_at >= %s")
            params.append(start_date)

        end_date = request_params.get('end_date')
        if end_date:
            cls._validate_date(end_date)
            clauses.append("created_at <= %s")
            params.append(end_date)

        status_val = request_params.get('status')
        if status_val:
            ALLOWED_STATUSES = {'draft', 'submitted', 'approved', 'rejected'}
            if status_val.lower() not in ALLOWED_STATUSES:
                raise ValidationError(f"Status must be one of: {', '.join(sorted(ALLOWED_STATUSES))}")
            clauses.append("LOWER(status) = %s")
            params.append(status_val.lower())

        query = "SELECT * FROM reports WHERE " + " AND ".join(clauses)
        return query, params


class ReportViewSet(viewsets.ModelViewSet):
    """Multi-tenant report endpoint — hardened against SQL injection."""
    serializer_class = ReportSerializer

    def get_queryset(self):
        """Server-side tenant isolation — never trust client input."""
        tenant_id = self.kwargs.get('tenant_id')
        return Report.objects.filter(tenant_id=tenant_id).select_related('author')

    def list(self, request, tenant_id=None):
        # Server-side tenant verification
        try:
            tenant_id_int = int(tenant_id)
        except (ValueError, TypeError):
            raise PermissionDenied("Invalid or missing tenant ID.")

        # Verify the requesting user has access to this tenant
        if not request.user.has_tenant_access(tenant_id_int):
            raise PermissionDenied("You do not have access to this tenant's data.")

        query, params = SafeQuerySet.build_report_query(tenant_id_int, request.GET)
        
        with connection.cursor() as cursor:
            cursor.execute(query, params)
            columns = [desc[0] for desc in cursor.description]
            rows = [dict(zip(columns, row)) for row in cursor.fetchall()]

        return Response(rows, status=status.HTTP_200_OK)
```

**Key fixes:**
- All values passed as positional `params` to `cursor.execute()` — the database driver handles escaping
- `tenant_id` validated as `int` before any use
- Date values round-tripped through `datetime.strptime` / `strftime` to reject malformed input
- Enumerated status values — only whitelisted values accepted
- ORM-based `get_queryset()` as the primary access path; raw SQL only for complex aggregations where necessary
- Server-side tenant ownership check enforced regardless of client claims

**CWE-284**: *Improper Access Control* (specifically: client-enforced authorization)

### Vulnerable Code

```python
# reports/views.py — INSECURE
class SecureReportViewSet(viewsets.ModelViewSet):
    """
    BUG: This class has NO object-level permission checks in Python.
    It relies entirely on React to hide buttons and disable actions.
    
    Frontend React code does:
      if (report.author !== currentUser.id) { /* hide edit/delete */ }
    
    An attacker bypasses React entirely by calling the API directly.
    """

    serializer_class = ReportSerializer

        obj = get_object_or_404(Report, pk=self.kwargs['pk'])

        partial = kwargs.pop('partial', False)
        instance = self.get_object()
        serializer = self.get_serializer(instance, data=request.data, partial=partial)
        serializer.is_valid(raise_exception=True)
        serializer.save()
```

### Exploit Demonstration

User A creates a report at `GET /reports/42`. The React UI correctly hides "Delete" because `currentUser.id !== report.author_id`.

User A opens browser console and calls:

```bash
curl -X DELETE https://api.example.com/reports/42 \
  -H 'Authorization: Bearer <user_a_token>'
```

Or worse, User A reads another tenant's report:

```bash
curl -X GET https://api.example.com/reports/999 \
  -H 'Authorization: Bearer <user_a_token>'
```

Both succeed because there is zero server-side enforcement.

```python
# core/permissions.py — HARDENED
from rest_framework import permissions
from django.core.exceptions import PermissionDenied


class IsTenantMember(permissions.BasePermission):
    """Verify the user belongs to the requested tenant."""

    def has_permission(self, request, view):
        tenant_id = view.kwargs.get('tenant_id')
        if not tenant_id:
            return False
        if not request.user or not request.user.is_authenticated:
            return False
        return request.user.has_tenant_access(int(tenant_id))


class IsObjectOwnerOrTenantAdmin(permissions.BasePermission):
    """
    Object-level permission: user can only modify objects they own,
    unless they are a tenant admin.
    
    ENFORCED ON THE SERVER — frontend checks alone are insufficient.
    """

    def has_object_permission(self, request, view, obj):
        # Read operations allowed for any authenticated tenant member
        if request.method in permissions.SAFE_METHODS:
            return True
        
        # Write operations require ownership or admin role
        return (
            obj.author == request.user or
            request.user.is_tenant_admin(obj.tenant_id)
        )


class ReadOnlyByTenantMembers(permissions.BasePermission):
    """Non-admin tenant members can only read (safe methods)."""

    def has_object_permission(self, request, view, obj):
        if request.method in permissions.SAFE_METHODS:
            return True
        return request.user.is_tenant_admin(obj.tenant_id)
```

```python
# reports/views.py — HARDENED
from rest_framework import viewsets, status
from rest_framework.response import Response
from core.permissions import (
    IsTenantMember,
    IsObjectOwnerOrTenantAdmin,
)


class ReportViewSet(viewsets.ModelViewSet):
    """Multi-tenant report endpoint — server-side auth enforced."""
    serializer_class = ReportSerializer

    # View-level: user must belong to the tenant
    permission_classes = [IsTenantMember]

    def get_queryset(self):
        tenant_id = self.kwargs.get('tenant_id')
        return Report.objects.filter(
            tenant_id=tenant_id
        ).select_related('author', 'tenant').order_by('-created_at')

    def get_object(self):
        """
        Override get_object to enforce server-side object permissions.
        This runs BEFORE any action method, protecting ALL endpoints.
        """
        obj = super().get_object()
        # This check CANNOT be bypassed by skipping the frontend
        self.check_object_permissions(self.request, obj)
        return obj

    def perform_create(self, serializer):
        """Force the new report into the correct tenant context."""
        tenant_id = self.kwargs.get('tenant_id')
        serializer.save(author=self.request.user, tenant_id=tenant_id)

    def get_permissions(self):
        """
        Different permission policies per HTTP method:
          GET/PATCH/HEAD → Owner or Tenant Admin
          POST           → Any authenticated tenant member
          PUT/DELETE     → Owner or Tenant Admin only
        """
        if self.action in ('create',):
            permission_classes = [IsTenantMember]
        elif self.action in ('update', 'partial_update', 'destroy'):
            permission_classes = [IsTenantMember, IsObjectOwnerOrTenantAdmin]
        else:
            permission_classes = [IsTenantMember, IsObjectOwnerOrTenantAdmin]
        return [cls() for cls in permission_classes]
```

**Key fixes:**
- `check_object_permissions()` called inside `get_object()` — every read/write path hits it
- `IsObjectOwnerOrTenantAdmin` enforces ownership on the server, not the browser
- `perform_create()` binds the tenant and author server-side, preventing the client from setting these fields
- Method-level permission routing ensures writes are more restrictive than reads

## Vulnerability 4: Mass Assignment in Serializers

**CWE-915**: *Improperly Controlled Modification of Dynamically-Determined Object Attributes*

### Vulnerable Code

```python
# reports/serializers.py — INSECURE
class ReportSerializer(serializers.ModelSerializer):
    """BUG: Every field on the model is writable by default."""
    class Meta:
        model = Report
        # Fields = all model fields → includes is_staff, is_superuser, 
        # created_by, tenant, status, etc.
        fields = '__all__'  

    def create(self, validated_data):
        # Author set from validated_data instead of request.user — 
        # attacker sets "author": 1 to impersonate admin
        return super().create(validated_data)
```

### Exploit Demonstration

Attacker posts to create a report with elevated privileges:

```bash
curl -X POST https://api.example.com/reports/1/ \
  -H 'Authorization: Bearer <user_b_token>' \
  -H 'Content-Type: application/json' \
  -d '{
    "title": "Malicious Report",
    "content": "Injected content",
    "author": 1,
    "is_staff": true,
    "status": "approved",
    "tenant_id": 1,
    "published_at": "2024-01-01T00:00:00Z"
  }'
```

The `"is_staff": true` and `"author": 1` fields are silently applied because `fields = '__all__'` includes them. Attacker B now owns a report authored by admin, with staff-level access.

For updates:

```bash
curl -X PATCH https://api.example.com/reports/42/ \
  -H 'Authorization: Bearer <attacker_token>' \
  -H 'Content-Type: application/json' \
  -d '{"status": "approved"}'  
```

Any authenticated user can approve reports they don't own.

### Fixed Code

```python
# reports/serializers.py — HARDENED
from rest_framework import serializers
from .models import Report


# ---------------------------------------------------------------
# Field allowlist — explicit enumeration of what's writable
# ---------------------------------------------------------------
READ_FIELDS = ('id', 'title', 'content', 'author', 'status', 'tenant',
               'created_at', 'updated_at', 'published_at')

WRITE_FIELDS = ('title', 'content', 'status')

# Status transitions: only these values allowed
ALLOWED_STATUS_TRANSITIONS = {
    'draft': {'submitted', 'deleted'},
    'submitted': {'approved', 'rejected', 'draft'},
    'approved': {'draft'},
    'rejected': {'draft'},
    'deleted': set(),
}


class ReportSerializer(serializers.ModelSerializer):
    """
    Multi-tenant report serializer — hardened against mass assignment.
    
    Only explicitly listed fields are writable. All system/admin fields
    (author, tenant, id, timestamps) are write-protected.
    """
    # Explicit readonly exposure — client sees them but cannot write
    author = serializers.StringRelatedField(read_only=True)
    tenant_id = serializers.IntegerField(read_only=True)
    created_at = serializers.DateTimeField(read_only=True)
    updated_at = serializers.DateTimeField(read_only=True)

    class Meta:
        model = Report
        # Explicit field list — NEVER use '__all__' in production
        fields = READ_FIELDS + WRITE_FIELDS
        read_only_fields = ('id', 'author', 'tenant_id', 'created_at', 'updated_at')

    def validate_title(self, value: str) -> str:
        if not value or not value.strip():
            raise serializers.ValidationError("Title cannot be empty.")
        if len(value) > 200:
            raise serializers.ValidationError("Title must be 200 characters or fewer.")
        return value.strip()

    def validate_content(self, value: str) -> str:
        if not value or not value.strip():
            raise serializers.ValidationError("Content cannot be empty.")
        return value

    def validate_status(self, value: str) -> str:
        current_instance = getattr(self, 'instance', None)
        previous_status = current_instance.status if current_instance else 'draft'
        
        allowed_next = ALLOWED_STATUS_TRANSITIONS.get(previous_status, set())
        if value not in allowed_next:
            raise serializers.ValidationError(
                f"Cannot transition status from '{previous_status}' to '{value}'. "
                f"Allowed transitions: {', '.join(sorted(allowed_next)) or 'none'}"
            )
        return value

    def create(self, validated_data: dict) -> Report:
        """
        Override create to prevent mass assignment of system fields.
        Author and tenant are bound server-side from the request context.
        """
        # Strip any malicious fields that slipped past validation
        validated_data.pop('author', None)
        validated_data.pop('tenant_id', None)
        validated_data.pop('is_staff', None)
        validated_data.pop('is_superuser', None)
        
        # These MUST come from the request, never from validated_data
        report = Report.objects.create(
            title=validated_data['title'],
            content=validated_data['content'],
            author=self.context['request'].user,  # Server-assigned author
            tenant_id=self.context.get('tenant_id'),  # Server-assigned tenant
            status=validated_data.get('status', 'draft'),
        )
        return report

    def update(self, instance: Report, validated_data: dict) -> Report:
        """
        Override update to enforce write-field restrictions and 
        prevent escalation through hidden fields.
        """
        # Remove any non-writable fields that might have been injected
        protected_fields = {'author', 'tenant_id', 'tenant', 'is_staff',
                            'is_superuser', 'id', 'created_at', 'updated_at'}
        for field in protected_fields:
            validated_data.pop(field, None)

        # Enforce status transition rules on updates too
        if 'status' in validated_data:
            validated_data['status'] = self.validate_status(validated_data['status'])

        instance.title = validated_data.get('title', instance.title)
        instance.content = validated_data.get('content', instance.content)
        instance.status = validated_data.get('status', instance.status)
        instance.save()
        return instance
```

```python
# views.py — integrate serializer context properly
class ReportViewSet(viewsets.ModelViewSet):
    permission_classes = [IsTenantMember, IsObjectOwnerOrTenantAdmin]
    serializer_class = ReportSerializer

    def get_queryset(self):
        tenant_id = self.kwargs.get('tenant_id')
        return Report.objects.filter(tenant_id=tenant_id)\
            .select_related('author', 'tenant')\
            .order_by('-created_at')

    def get_serializer_context(self):
        """Pass request and tenant info into serializer context."""
        context = super().get_serializer_context()
        context['tenant_id'] = self.kwargs.get('tenant_id')
        return context

    def perform_create(self, serializer):
        """Bind author and tenant from request context, not client input."""
        serializer.save(
            author=self.request.user,
            tenant_id=int(self.kwargs.get('tenant_id')),
        )
```

**Key fixes:**
- `fields` explicitly enumerated — no `__all__`, no surprises
- `read_only_fields` tuple prevents client from submitting protected attributes
- `create()` strips known attack vectors (`author`, `tenant_id`, `is_staff`) even if validation misses them
- `update()` removes protected fields from `validated_data` before applying
- Status transitions validated against a finite state machine
- Author and tenant set server-side from request context, never from serialized input

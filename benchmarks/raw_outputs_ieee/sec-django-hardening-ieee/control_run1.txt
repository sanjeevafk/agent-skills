## Issue 1 — Server-Side SQL Injection via Dynamic Query Building

### CWE
**CWE-89: Improper Neutralization of Special Elements used in an SQL Command ('SQL Injection')**

### Vulnerable Code (`reports/views.py`)

```python
from django.db import connection
from rest_framework.decorators import api_view, permission_classes
from rest_framework.permissions import IsAuthenticated

@api_view(["GET"])
@permission_classes([IsAuthenticated])
def tenant_report(request):
    """Builds report filters by concatenating user input directly into SQL."""
    date_from = request.query_params.get("date_from", "")
    date_to = request.query_params.get("date_to", "")
    status = request.query_params.get("status", "")
    min_revenue = request.query_params.get("min_revenue", "")
    sort_by = request.query_params.get("sort_by", "created_at")
    order = request.query_params.get("order", "DESC")

    # 🚧 CRITICAL: f-string interpolation into SQL — entirely un-sanitized
    sql = f"""
        SELECT id, name, created_at, revenue, status
        FROM tenants_project
        WHERE tenant_id = {request.user.tenant.id}
          AND created_at BETWEEN '{date_from}' AND '{date_to}'
          {'AND status = ' + status if status else ''}
          {'AND revenue >= ' + min_revenue if min_revenue else ''}
        ORDER BY {sort_by} {order}
        LIMIT 50
    """

    with connection.cursor() as cursor:
        cursor.execute(sql)
        rows = cursor.fetchall()

    return Response(rows)
```

### Exploit Demonstration

An attacker sends a crafted `status` parameter:

```
GET /api/reports/?status=active'%20OR%201=1--&sort_by=id&order=ASC
```

The interpolated SQL becomes:

```sql
SELECT id, name, created_at, revenue, status
FROM tenants_project
WHERE tenant_id = 42
  AND created_at BETWEEN '' AND ''
  AND status = active' OR 1=1--
ORDER BY id ASC
LIMIT 50
```

The `OR 1=1` bypasses the status filter, returning **every row in the table**. Worse still, the attacker could append a UNION-based extraction:

```
status=%27%20UNION%20SELECT%20username,password,email,'x','y'%20FROM%20auth_user--
```

This returns plain-text credentials from an unrelated table. The `sort_by`, `order`, and numeric `min_revenue` parameters are also injectable since they pass straight into the query with zero validation.

### Corrected Code

```python
import datetime

from django.db import connection
from rest_framework.decorators import api_view, permission_classes
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response
from rest_framework import status

# Whitelist of allowed column names; anything else is rejected
ALLOWED_SORT_COLUMNS = frozenset({
    "id", "name", "created_at", "updated_at", "revenue", "status",
})

TENANT_COLUMN_SAFE = frozenset(
    {f.name for f in __import__("tenants.models").models.Project._meta.fields}
)


def _safe_integer(value, name="value"):
    """Return int(value) or raise 400 if it cannot be parsed."""
    try:
        return int(value)
    except (TypeError, ValueError):
        raise ValueError(f"{name!r} must be an integer")


@api_view(["GET"])
@permission_classes([IsAuthenticated])
def tenant_report(request):
    tenant_id = request.user.tenant.id

    date_from_raw = request.query_params.get("date_from")
    date_to_raw = request.query_params.get("date_to")
    status = request.query_params.get("status")
    min_revenue_raw = request.query_params.get("min_revenue")
    sort_by = request.query_params.get("sort_by", "created_at")
    order = request.query_params.get("order", "DESC")

    # ---- Parameterise every value ------------------------------------------
    params = {"tenant_id": tenant_id}
    conditions = ["tenant_id = %s"]

    if date_from_raw:
        try:
            df = datetime.datetime.fromisoformat(date_from_raw).date()
        except (ValueError, TypeError):
            return Response(
                {"detail": f"Invalid date_from: {date_from_raw}"},
                status=status.HTTP_400_BAD_REQUEST,
            )
        conditions.append("created_at >= %s")
        params[f"df_{len(conditions)}"] = df

    if date_to_raw:
        try:
            dt = datetime.datetime.fromisoformat(date_to_raw).date()
        except (ValueError, TypeError):
            return Response(
                {"detail": f"Invalid date_to: {date_to_raw}"},
                status=status.HTTP_400_BAD_REQUEST,
            )
        conditions.append("created_at <= %s")
        params[f"dt_{len(conditions)}"] = dt

    if status:
        # Restrict to known enum values
        valid_statuses = {"draft", "active", "archived"}
        if status.lower() not in valid_statuses:
            return Response(
                {"detail": f"Invalid status. Choose from: {valid_statuses}"},
                status=status.HTTP_400_BAD_REQUEST,
            )
        conditions.append("LOWER(status) = %s")
        params[f"st_{len(conditions)}"] = status.lower()

    if min_revenue_raw:
        try:
            min_rev = _safe_integer(min_revenue_raw, "min_revenue")
        except ValueError as exc:
            return Response({"detail": str(exc)}, status=status.HTTP_400_BAD_REQUEST)
        conditions.append("revenue >= %s")
        params[f"mr_{len(conditions)}"] = min_rev

    # Sort column/order must be whitelisted against schema, never interpolated
    sort_col = sort_by.strip().rstrip("_asc").rstrip("_desc")
    if sort_col not in ALLOWED_SORT_COLUMNS:
        return Response(
            {"detail": f"Sort column must be one of: {sorted(ALLOWED_SORT_COLUMNS)}"},
            status=status.HTTP_400_BAD_REQUEST,
        )
    dir_part = "ASC" if order.upper() == "ASC" else "DESC"
    if dir_part not in ("ASC", "DESC"):
        return Response(
            {"detail": "order must be 'ASC' or 'DESC'"},
            status=status.HTTP_400_BAD_REQUEST,
        )

    where_clause = " AND ".join(conditions)
    sql = (
        f"SELECT id, name, created_at, revenue, status "
        f"FROM tenants_project "
        f"WHERE {where_clause} "
        f"ORDER BY {sort_col} {dir_part} "
        f"LIMIT 50"
    )

    with connection.cursor() as cursor:
        cursor.execute(sql, params)
        columns = [col[0] for col in cursor.description]
        rows = [dict(zip(columns, row)) for row in cursor.fetchall()]

    return Response(rows)
```

**Key changes:**
| Before | After |
|--------|-------|
| `f-string` interpolation into SQL | `%s` placeholders passed as `params` dict to `cursor.execute(sql, params)` |
| Arbitrary column names accepted | `ALLOWED_SORT_COLUMNS` whitelist enforced before ORDER BY |
| No validation on dates/status | Explicit format parsing + enum restriction |
| Numeric field accepts strings | `_safe_integer()` rejects non-integers |

## Issue 2 — Client-Side Only Object-Level Authorization

### CWE
**CWE-285: Improper Authorization** (also CWE-862: Missing Authorization)

### Vulnerable Code (`reports/views.py`)

```python
from rest_framework.views import APIView
from rest_framework.permissions import IsAuthenticated

class ReportDetailView(APIView):
    permission_classes = [IsAuthenticated]       # ← only authenticates identity

    def get(self, request, pk):
        """Returns the requested report — NO check that it belongs to the user's tenant."""
        with connection.cursor() as cursor:
            cursor.execute(
                "SELECT * FROM tenants_report WHERE id = %s", [pk]
            )
            row = cursor.fetchone()
            if not row:
                return Response({"error": "not found"}, status=404)
        # Frontend relies on JS to hide links to other tenants' reports,
        # but the API itself has no object-level guard.
        return Response(row)

    def delete(self, request, pk):
        """Deletes any report the attacker guesses a pk for."""
        with connection.cursor() as cursor:
            cursor.execute("DELETE FROM tenants_report WHERE id = %s", [pk])
        return Response(status=204)
```

### Exploit Demonstration

User A (tenant 1) creates reports 10–20. User B (tenant 2) has reports 30–40.

The React UI for tenant 2 never renders buttons for IDs < 30, so a casual observer thinks they're safe. But User B simply crafts:

```http
DELETE /api/reports/15/ HTTP/1.1
Host: app.example.com
Authorization: Bearer <user_b_token>
```

The server executes the deletion because the only permission checked was `IsAuthenticated` — there is no check that `report.tenant == request.user.tenant`. Result: **cross-tenant data destruction**.

Similarly for read access, a GET against `/api/reports/12/` leaks another tenant's data.

### Corrected Code

First, add a proper permission class:

```python
# tenants/permissions.py
from rest_framework.permissions import BasePermission, SAFE_METHODS
from django.core.exceptions import ObjectDoesNotExist


class IsTenantMemberOrReadOnly(BasePermission):
    """
    Ensures the requesting user belongs to the same tenant as the target object.
    Applies to ALL object-level operations, not just mutations.
    """
    message = "You do not have access to this resource."

    def has_object_permission(self, request, view, obj):
        # Allow read on public endpoints only if you explicitly set a flag
        if request.method in SAFE_METHODS:
            return getattr(obj, "tenant", None) == request.user.tenant

        # Writes require membership too
        return getattr(obj, "tenant", None) == request.user.tenant


class IsStaffForOwnTenant(BasePermission):
    """Only staff members can perform write operations, within their own tenant."""
    message = "Staff privileges required."

    def has_permission(self, request, view):
        if request.method in SAFE_METHODS:
            return True
        return getattr(request.user, "is_staff", False)

    def has_object_permission(self, request, view, obj):
        return getattr(obj, "tenant", None) == request.user.tenant
```

Then apply it server-side on the view:

```python
# reports/views.py
from rest_framework.views import APIView
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response
from tenants.permissions import IsTenantMemberOrReadOnly, IsStaffForOwnTenant
from tenants.models import Report


class ReportDetailView(APIView):
    permission_classes = [IsAuthenticated, IsTenantMemberOrReadOnly]

    def get_object(self, pk):
        try:
            # ⬇ This scope-by-tenant filter is the critical guard
            return Report.objects.filter(
                id=pk,
                tenant=self.request.user.tenant,     # <-- SERVER-SIDE TENANT SCOPING
            ).select_related("tenant").first()
        except Exception:
            return None

    def get(self, request, pk):
        obj = self.get_object(pk)
        if obj is None:
            return Response({"error": "Not found"}, status=404)
        return Response({...})                     # serialize obj

    def delete(self, request, pk):
        perm = IsStaffForOwnTenant()
        if not perm.has_permission(request, self):
            return Response({"error": "staff_required"}, status=403)
        obj = self.get_object(pk)
        if obj is None:
            return Response({"error": "Not found"}, status=404)
        if not perm.has_object_permission(request, self, obj):
            return Response({"error": "forbidden"}, status=403)
        obj.delete()
        return Response(status=204)
```

**Key changes:**
| Before | After |
|--------|-------|
| `IsAuthenticated` alone | `IsTenantMemberOrReadOnly` with `has_object_permission` |
| Filter ignores tenant | `.filter(tenant=request.user.tenant)` scopes the queryset |
| DELETE works for cross-tenant Pks | 403 returned when tenant mismatch |
| Auth logic lives in React only | All enforcement happens in the DRF permission pipeline |

## Issue 4 — Mass-Assignment Vulnerability in Serializer

### CWE
**CWE-915: Improperly Controlled Modification of Dynamically-Determined Object Attributes** (also known as "mass assignment" / "over-posting")

### Vulnerable Code (`tenants/serializers.py`)

```python
from rest_framework import serializers
from .models import StaffProfile

class StaffProfileSerializer(serializers.ModelSerializer):
    """Registers or updates a staff profile."""
    class Meta:
        model = StaffProfile
        # 🚧 fields="__all__" exposes EVERY model column to POST/PUT —
        #    including sensitive flags like is_staff, is_superuser, last_login...
        fields = "__all__"
```

If `StaffProfile` has these fields:

```python
class StaffProfile(models.Model):
    tenant = models.ForeignKey(Tenant, on_delete=models.CASCADE)
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    role = models.CharField(max_length=50)
    is_active = models.BooleanField(default=True)
    is_staff = models.BooleanField(default=False)         # privilege escalation vector
    last_login = models.DateTimeField(null=True)           # internal audit field
    notes = TextField(blank=True)
```

A malicious actor sends:

```json
POST /api/staff-profiles/ HTTP/1.1
Content-Type: application/json

{
  "tenant": 42,
  "role": "analyst",
  "is_staff": true,               // 💥 escalates themselves instantly
  "last_login": null,             // obfuscates audit trail
  "notes": "backdoor account"
}
```

The serializer accepts every submitted field because `fields = "__all__"` passes them wholesale to `ModelSerializer.save()`.

### Corrected Code

Explicitly list fields and restrict mutability per operation type:

```python
from rest_framework import serializers
from .models import StaffProfile


class StaffProfileWriteSerializer(serializers.ModelSerializer):
    """Used for POST/PUT/PATCH — only allows controlled fields."""

    class Meta:
        model = StaffProfile
        fields = [
            "role",
            "notes",
            "is_active",          # allow toggling active/inactive only
        ]
        # Optional: further restrict PATCH to only these
        extra_kwargs = {
            "role":          {"required": True},
            "is_active":     {"required": False},
            "notes":         {"required": False, "allow_blank": True},
        }

    def create(self, validated_data):
        instance = StaffProfile.objects.create(**validated_data)
        # Automatically tie to the authenticated user / tenant — user cannot spoof
        instance.user = self.context["request"].user
        instance.tenant = self.context["request"].user.tenant
        instance.save(update_fields=["user", "tenant"])
        return instance

    def update(self, instance, validated_data):
        for attr, value in validated_data.items():
            setattr(instance, attr, value)
        return instance


class StaffProfileReadSerializer(serializers.ModelSerializer):
    """Returned for GET — includes computed/derived data, hides internals."""
    user_email = serializers.EmailField(source="user.email", read_only=True)
    tenant_name = serializers.CharField(source="tenant.name", read_only=True)

    class Meta:
        model = StaffProfile
        fields = [
            "id",
            "user_email",
            "tenant_name",
            "role",
            "is_active",
            "is_staff",                  # read-only awareness, not editable
            "last_login",                # audit field, never writable
            "notes",
            "created_at",
            "updated_at",
        ]
        read_only_fields = [
            "id", "is_staff", "last_login", "created_at", "updated_at",
            "user_email", "tenant_name",
        ]
```

Wire the dual-serializer pattern into the view:

```python
# reports/views.py (or tenants/views.py)
from rest_framework import generics
from .models import StaffProfile
from .serializers import StaffProfileReadSerializer, StaffProfileWriteSerializer


class StaffProfileCreateView(generics.CreateAPIView):
    permission_classes = [IsAuthenticated, IsStaffForOwnTenant]
    queryset = StaffProfile.objects.all()

    def get_serializer_class(self):
        if self.action == "create":
            return StaffProfileWriteSerializer
        return StaffProfileReadSerializer

    def perform_create(self, serializer):
        # Context already supplies request, but be explicit in case context is lost
        serializer.save(user=self.request.user, tenant=self.request.user.tenant)
```

**Key changes:**
| Before | After |
|--------|-------|
| `fields = "__all__"` | Explicit `fields` whitelist containing only intended attributes |
| Any field in the body gets written | `extra_kwargs` + explicit `create()`/`update()` control every mutation |
| Privilege fields (`is_staff`) modifiable by anyone | Listed as `read_only_fields`; never accepted from request body |
| `user` / `tenant` derived from request | Manually set in `perform_create()` with `save()` kwargs — impossible to spoof |

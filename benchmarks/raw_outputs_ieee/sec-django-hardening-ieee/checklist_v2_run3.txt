## 1. SQL Injection via `cursor.execute()` with f-string Concatenation

**CWE-89: SQL Injection**

### Exploit

```python
# VULNERABLE CODE
from django.db import connection

def get_report(request, tenant_id, start_date, end_date):
    # Attacker sends: /api/report?tenant_id=1'; DROP TABLE users;--
    cursor = connection.cursor()
    query = f"""
        SELECT * FROM reports WHERE tenant_id = {tenant_id}
        AND created_at BETWEEN '{start_date}' AND '{end_date}'
    """
    cursor.execute(query)  # Direct string interpolation into SQL!
    return cursor.fetchall()
```

An attacker passes `tenant_id=1 OR 1=1--` to exfiltrate all tenants' data, or `tenant_id=0; DROP TABLE reports--` to destroy data.

### Fix: Parameterized Queries + ORM Preference

```python
# apps/reports/views.py
import logging
from datetime import datetime

from django.db import connection, connections
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated
from rest_framework.exceptions import PermissionDenied, ValidationError

logger = logging.getLogger(__name__)


class ReportEndpoint(APIView):
    """Hardened report endpoint — parameterized queries only."""

    permission_classes = [IsAuthenticated]

    def get(self, request):
        tenant_id = self._validate_tenant_id(request.user.tenant.id)
        start_date = self._validate_date(request.GET.get("start_date"))
        end_date = self._validate_date(request.GET.get("end_date"))

        results = self._fetch_report_safe(tenant_id, start_date, end_date)
        return Response({"results": results, "count": len(results)})

    # ------------------------------------------------------------------
    # Validation layer
    # ------------------------------------------------------------------

    @staticmethod
    def _validate_tenant_id(value) -> int:
        try:
            tid = int(value)
        except (TypeError, ValueError):
            raise ValidationError({"tenant_id": "Must be an integer."})
        if tid <= 0:
            raise ValidationError({"tenant_id": "Must be a positive integer."})
        return tid

    @staticmethod
    def _validate_date(value: str | None) -> datetime:
        if not value:
            raise ValidationError("Both start_date and end_date are required.")
        try:
            return datetime.strptime(value, "%Y-%m-%d")
        except ValueError:
            raise ValidationError(
                {"date": "Use ISO format YYYY-MM-DD."}
            )

    # ------------------------------------------------------------------
    # Safe query execution — PARAMETERIZED ONLY
    # ------------------------------------------------------------------

    @staticmethod
    def _fetch_report_safe(
        tenant_id: int, start_date: datetime, end_date: datetime
    ) -> list[dict]:
        """Always use named parameters — never f-strings."""

        # Preferred: use the ORM entirely (auto-parameterization)
        try:
            from apps.reports.models import Report
            qs = (
                Report.objects.filter(
                    tenant_id=tenant_id,
                    created_at__gte=start_date,
                    created_at__lte=end_date,
                )
                .values(
                    "id", "title", "created_at", "status", "author__username"
                )
                .order_by("-created_at")
            )
            return list(qs)
        except Exception as exc:
            logger.error(
                "Report query failed: %s", exc, exc_info=True
            )
            raise ValidationError("Internal error generating report.")

        # If you must use raw SQL, DO THIS — NOT f-strings:
        # cursor = connection.cursor()
        # cursor.execute(
        #     "SELECT id, title, created_at, status, author_username "
        #     "FROM reports_view "
        #     "WHERE tenant_id = %(tenant_id)s "
        #     "AND created_at >= %(start_date)s "
        #     "AND created_at <= %(end_date)s",
        #     {"tenant_id": tenant_id, "start_date": start_date, "end_date": end_date},
        # )
        # columns = [col[0] for col in cursor.description]
        # return [dict(zip(columns, row)) for row in cursor.fetchall()]
```

**Key principles:**
- ORM `.filter()` is auto-parameterized — no cursor needed
- When `raw()` is unavoidable, use named `%()s` placeholders with a dict
- Never interpolate f-string expressions inside SQL strings
- Validate ALL inputs before they reach any database call

## 2. Object-Level Permissions Checked Only in React Frontend

**CWE-284: Improper Access Control (Client-Side Authorization Bypass)**

### Exploit

```jsx
// VULNERABLE: frontend-only check
function ReportDetail({ report }) {
  const canEdit = report.author === currentUser.id; // only frontend check
  return (
    <div>
      <input value={report.title} />
      {canEdit && <button onClick={() => updateReport(report)}>Save</button>}
    </div>
  );
}
```

An attacker simply opens DevTools and removes the `{canEdit && ...}` guard, or fires a curl request directly to the API endpoint regardless of the frontend button state. Every unprotected endpoint becomes public data.

### Fix: Server-Side DRF Permissions

```python
# apps/permissions.py
from rest_framework import permissions
from rest_framework.request import Request
from rest_framework.views import View


class IsOwnerOrReadOnly(permissions.BasePermission):
    """Server-side ownership enforcement — NEVER trust the client."""

    message = "You do not have permission to modify this resource."

    def has_object_permission(self, request: Request, view: View, obj) -> bool:
        # Read methods safe for anyone with general permission
        if request.method in permissions.SAFE_METHODS:  # GET, HEAD, OPTIONS
            return True

        # WRITE methods restricted to owner ONLY
        if hasattr(obj, "author") and hasattr(request, "user"):
            return obj.author == request.user

        # Fallback for non-author objects — deny writes by default
        return False


class IsTenantMember(permissions.BasePermission):
    """Multi-tenant isolation — users can only access their own tenant."""

    message = "This resource belongs to a different tenant."

    def has_object_permission(self, request: Request, view: View, obj) -> bool:
        tenant_id = getattr(obj, "tenant_id", None)
        user_tenant = getattr(request.user, "tenant_id", None)
        if tenant_id is None or user_tenant is None:
            return False
        return tenant_id == user_tenant

    def has_permission(self, request: Request, view: View) -> bool:
        return request.user and request.user.is_authenticated


class CombinedObjectPermissions(IsTenantMember, IsOwnerOrReadOnly):
    """Apply both tenant isolation AND ownership checks."""

    pass
```

```python
# apps/reports/serializers.py
from rest_framework import serializers
from apps.reports.models import Report


class ReportSerializer(serializers.ModelSerializer):
    class Meta:
        model = Report
        fields = ["id", "title", "content", "created_at", "updated_at"]
        read_only_fields = ["id", "created_at", "updated_at"]

    # Remove 'author' from write fields so clients cannot inject it
    author = serializers.ReadOnlyField(source="author.username")
```

```python
# apps/reports/views.py
from rest_framework import generics
from apps.reports.models import Report
from apps.reports.serializers import ReportSerializer
from apps.permissions import CombinedObjectPermissions


class ReportDetailView(generics.RetrieveUpdateDestroyAPIView):
    queryset = Report.objects.none()  # Override below — see filterset

    permission_classes = [CombinedObjectPermissions]
    serializer_class = ReportSerializer

    def get_queryset(self):
        """Scope to current tenant automatically."""
        return Report.objects.filter(
            tenant_id=self.request.user.tenant_id
        ).select_related("author")

    def perform_update(self, serializer):
        """Prevent changing author on update."""
        instance = self.get_object()
        # Ensure nobody swaps the author field through PUT/PATCH
        data = self.request.data
        if "author" in data or "tenant_id" in data:
            from rest_framework.exceptions import PermissionDenied
            raise PermissionDenied("Cannot change author or tenant.")
        serializer.save()
```

**Key principle:** Server-side permission classes (`has_object_permission`) enforce every write. The React UI may show/hide buttons for UX, but it is irrelevant to authorization — an API request WITHOUT passing these checks gets rejected at the middleware level.

## 3. Hardcoded SECRET_KEY & Database Credentials

**CWE-798: Use of Hard-coded Credentials**

### Exploit

```python
# VULNERABLE: settings.py committed to git
SECRET_KEY = "super-secret-key-12345-hardcoded-in-repo"
DATABASES = {
    "default": {
        "ENGINE": "django.db.backends.postgresql",
        "NAME": "production_db",
        "USER": "admin",
        "PASSWORD": "P@ssw0rd!",   # <-- committed to git history
        "HOST": "localhost",
        "PORT": "5432",

```

Anyone with repo access (or leaked history via `git log --all --full-history`) gains live database access. This persists even after the values are changed because old commits retain them.

## 4. Mass Assignment Through Unrestricted Serializer

**CWE-915: Improper Restriction of Operations with User-Controlled Data (Mass Assignment)**

### Exploit

```python
# VULNERABLE: allows every model field including role/staff/is_active
class UserSerializer(serializers.ModelSerializer):
    class Meta:
        model = User
        fields = "__all__"  # <-- lets attackers set is_staff=True
```

An attacker sends:
```json
{
  "email": "hacker@example.com",
  "username": "hacker",
  "password": "password123",
  "is_staff": true,
  "is_superuser": true,
  "role": "admin"
}
```

Result: arbitrary privilege escalation with zero authentication.

### Fix: Explicit Field Allowlisting + Write/Read Separation

```python
# apps/users/serializers.py
from rest_framework import serializers
from django.contrib.auth.password_validation import validate_password
from apps.users.models import User


class UserRegistrationSerializer(serializers.ModelSerializer):
    """Only creates new users — password validated server-side."""

    password = serializers.CharField(write_only=True, required=True, validators=[validate_password])
    password_confirm = serializers.CharField(write_only=True, required=True)

    class Meta:
        model = User
        fields = ["username", "email", "phone", "password", "password_confirm"]

    def validate(self, attrs):
        if attrs.pop("password") != attrs.pop("password_confirm"):
            raise serializers.ValidationError({"password": "Passwords do not match."})
        return attrs

    def create(self, validated_data):
        user = User.objects.create_user(**validated_data)
        return user


class UserSerializer(serializers.ModelSerializer):
    """Safe profile retrieval — NO writable sensitive fields."""

    class Meta:
        model = User
        fields = ["id", "username", "email", "phone", "role", "is_active"]
        read_only_fields = ["id", "is_active"]

    def update(self, instance, validated_data):
        """Restrict which fields can actually be changed."""
        # Blacklist dangerous fields explicitly
        restricted_fields = {"is_staff", "is_superuser", "groups", "user_permissions", "role"}
        for field in restricted_fields:
            validated_data.pop(field, None)

        return super().update(instance, validated_data)


class UserProfileUpdateSerializer(serializers.ModelSerializer):
    """Ultra-narrow scope — only personal info, nothing auth-related."""

    class Meta:
        model = User
        fields = ["phone"]  # Single field update only
        extra_kwargs = {
            "phone": {"required": False, "allow_blank": True},
        }
```

```python
# apps/users/views.py
from rest_framework import status
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated, AllowAny
from django.contrib.auth import get_user_model

User = get_user_model()


class UserCreateView(APIView):
    """Public registration — only accepts safe fields."""

    permission_classes = [AllowAny]
    serializer_class = UserRegistrationSerializer

    def post(self, request):
        serializer = self.serializer_class(data=request.data)
        serializer.is_valid(raise_exception=True)
        user = serializer.save()
        return Response(
            {"id": user.id, "username": user.username, "email": user.email},
            status=status.HTTP_201_CREATED,
        )


class UserListView(APIView):
    """Admin-only listing — staff must verify via header or role."""

    permission_classes = [IsAuthenticated]

    def get(self, request):
        # Server-side staff verification — not client-controlled
        if not request.user.is_verified:
            from rest_framework.exceptions import PermissionDenied
            raise PermissionDenied("Account not verified.")

        queryset = User.objects.all().values(
            "id", "username", "email", "role", "is_active", "is_verified"
        )
        return Response(list(queryset))
```

## Summary Table

| Vulnerability | CWE | Root Cause | Fix |
|---|---|---|---|
| SQL injection via f-string | **CWE-89** | `cursor.execute(f"... {user_input}")` | Parameterized `%(name)s` dict args; prefer ORM |
| Frontend-only permission checks | **CWE-284** | React `canEdit` guard with no server backup | DRF `BasePermission.has_object_permission()` server-side |
| Hardcoded secrets in git | **CWE-798** | `SECRET_KEY = "literal"` in committed settings | `django-environ` reading `.env` + pre-commit gate |
| Mass-assignment serializer | **CWE-915** | `fields = "__all__"` on user serializer | Explicit whitelist + `write_only` + update blacklists |

All four fixes are independent and additive — removing the frontend guards does not weaken security, injecting bad secrets causes early startup failure rather than silent compromise, and replacing `"__all__"` with explicit fields blocks privilege escalation at the serialization layer.

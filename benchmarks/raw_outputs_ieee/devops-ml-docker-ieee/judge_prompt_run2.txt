# ============================================================

# --no-deps: we do NOT need system libraries here.
#            Torch wheel includes its own libtorch; extra deps
#            go into stage 2 where they are installed once.
RUN pip install --no-cache-dir \
        --dest /wheels \
        --platform manylinux2014_x86_64 \
        --python-version 312 \
        --only-binary=:all: \

    torch==2.5.1+cpu torchaudio==2.5.1+cpu torchvision==0.20.1+cpu

# Pin exact hashes for reproducible installs in stage 2.
# `pip hash` output; update when torch version bumps:
#   $ docker run --rm -v $(pwd):/out wheels-image \
#       pip hash /wheels/torch-*.whl /wheels/torchaudio-*.whl /wheels/torchvision-*.whl | tee /out/hash-manifest.txt
COPY --from=wheels /wheels /app/wheels

# Freeze the resolved dependency tree so stage 2 reuses it
# without re-resolving every transitive dep.
RUN pip freeze --all > /app/requirements-frozen.txt


# ============================================================
# Stage 2: Runtime — minimal surface area
# ============================================================
FROM python:3.12-slim AS runtime

LABEL org.opencontainers.image.source="https://github.com/your-org/inference-service" \
      org.opencontainers.image.description="CPU-only PyTorch inference service"

# --- Locale & timezone (often needed by ML pipelines) ---
ENV LANG=C.UTF-8 \
    LC_ALL=C.UTF-8 \
    TZ=UTC

# --- PyTorch environment knobs (CPU-only, no GPU probes) ---
ENV TORCH_CUDA_ARCH_LIST="" \
    PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    HF_HUB_ENABLE_HF_TRANSFER=0

# ----------------------------------------------------------
# 2a. System packages — kept small; no build toolchain
# ----------------------------------------------------------
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
        libgomp1 \           # required by libomp shipped with libtorch
        libgl1-mesa-glx \    # fallback for models that query GL
        curl \               # used by HEALTHCHECK
    && rm -rf /var/lib/apt/lists/*

# ----------------------------------------------------------
# 2b. Non-root user (UID/GID are arbitrary primes to avoid
#     collisions with host-mounted volumes)
# ----------------------------------------------------------
RUN groupadd --gid 9999 appgroup && \
    useradd --uid 9999 --gid appgroup --create-home --shell /bin/false appuser

# ----------------------------------------------------------
# 2c. Project directory with correct ownership so the appuser
#     can write logs/artifacts but NOT modify installed code.
# ----------------------------------------------------------
RUN mkdir -p /app/{data,output,logs} && \
    chown -R appuser:appgroup /app

# ----------------------------------------------------------
# 2d. Install dependencies FROM THE WHEELS STAGE.
#     Strategy for cache optimization:
#       1. Copy requirements list FIRST (changes rarely)
#          → pip cache hit unless requirements change.
#       2. Copy wheel files SECOND (more stable than source).
#       3. COPY application code LAST (changes every commit).
#     This means if you touch src/*.py the wheel layer stays
#     cached; only the final COPY re-runs.
# ----------------------------------------------------------

# 2d-i. Requirements list (hash-verified)
COPY --chown=appuser:appgroup \
     ./requirements.txt /app/requirements.txt

# 2d-ii. Wheel artifacts from stage 1

USER appuser

RUN pip install --no-cache-dir \
        --no-index \
        --find-links=file:///app/wheels \
        --require-hashes \
        -r /app/requirements.txt

# ----------------------------------------------------------
# 2e. Application source — copied last for maximum cache.
# ----------------------------------------------------------
COPY --chown=appuser:appgroup ./src /app/src

# Set working directory after all copies so relative paths
# inside the entrypoint resolve correctly.
WORKDIR /app


# ----------------------------------------------------------
# 2f. Healthcheck endpoint — lightweight; does NOT trigger
#     model warm-up (that happens on first real request).
# ----------------------------------------------------------
HEALTHCHECK --interval=30s --timeout=10s --start-period=60s --retries=3 \
    CMD curl -f http://localhost:8000/healthz || exit 1


# ----------------------------------------------------------
# 2g. Entry point — passes secrets via stdin or mount, NEVER
#     baked as ENV.
# ----------------------------------------------------------
EXPOSE 8000

CMD ["python", "-m", "uvicorn", "src.api:app", \
     "--host", "0.0.0.0", \
     "--port", "8000"]

## Layer-by-layer justification

### Why multi-stage?

A single-stage image ends up at ~9 GB because the builder pulls gcc, g++, make, kernel headers, CUDA toolchains, and keeps intermediate build artifacts. By separating **wheels** (stage 1) from **runtime** (stage 2), the final image contains only:

- A slim Python base (~150 MB vs ~800 MB for full Ubuntu-based builder)
- The `.whl` binaries (already compiled, no linker needed)
- A few shared libraries (`libgomp`, `libgl`)
- Your application code

This cuts the image roughly 9 GB → 2–2.3 GB depending on what model weights are mounted at runtime (they shouldn't be baked in).

### Why `python:3.12-slim` for both stages?

Alpine-based images (`python:3.12-alpine`) use musl libc which has known ABI incompatibilities with pre-built PyTorch wheels (manylinux uses glibc). Using `slim` (Debian-based, glibc) ensures binary compatibility while still keeping the base small.

### Why `--dest /wheels` + `--no-deps` in stage 1?

- `--dest` downloads everything into one directory, making a single `COPY --from` into stage 2. No piecemeal fetching.
- `--no-deps` skips system-level transitive dependencies here since stage 2 handles installing them once from the same frozen resolver output.

### Why `pip freeze` into `requirements-frozen.txt`?

It captures the exact resolved transitive tree (numpy, safetensors, pydantic core, etc.) so stage 2 doesn't re-run the dependency resolver—a potentially network-heavy and flaky step.

### Why copy order matters (requirements → wheels → code)?

| Layer | Changes | Cache benefit |
|-------|---------|---------------|
| `requirements.txt` | Rarely (version bumps only) | High—stays cached through most deploys |
| `/app/wheels` | Very rare (new torch release) | Higher—unchanged across daily commits |
| `src/` | Every code change | Low—but this is the cheapest layer to rebuild |

If you swapped the order so `src/` came before `wheels/`, every code edit would bust the wheel installation layer too, wasting time downloading/reinstalling torch.

### Why `libgomp1` and `libgl1-mesa-glx`?

PyTorch's CPU builds link against `libgomp` (OpenMP runtime). Without it, you get `cannot open shared object file` errors at import time. `libgl` is occasionally queried by vision models even on CPU. Both are marked `--no-install-recommends` to stay lean, and `apt lists` are deleted in the same `RUN` to avoid caching metadata.

### Why UID/GID 9999?

Arbitrary high-numbered UIDs reduce the chance of collision with any host-mounted volume (e.g., Docker-in-Docker, bind mounts for debugging). This isn't security-hardening by itself—it avoids permission headaches when hosts happen to use non-standard UIDs.

### Why `USER appuser` before `pip install`?

Installing packages as non-root prevents accidental world-writable permissions and ensures the resulting site-packages are owned by `appuser`. If you `USER` *after* pip install, the installed files end up owned by root in a non-root container, which can cause subtle permission errors.

## Design Decisions (Layer by Layer)

| Decision | Rationale |
|---|---|
| **Base: `python:3.12-slim-bookworm`** | Official Python 3.12, slim variant (~150MB base vs ~400MB full). Bookworm = Debian 12 = long LTS support and widely available CPU-only `torch-cpu` wheel. |
| **Stage 1 `deps` → Stage 2 `runtime` (no `build` stage)** | PyTorch wheels are pre-built. We don't compile anything locally — we download compiled `.whl`s. The "build" isolation means: dependencies installed in one layer, copied into runtime via multi-stage COPY so no pip/cache metadata leaks into final image. This gives us the isolated-wheel guarantee without needing a separate compilation stage. |
| **Dependency order: system → static assets → deps → app code** | System packages change least often. Static assets (model weights excluded via .dockerignore) next. Then `requirements.txt` which changes less frequently than source code. Finally app code. If you edit a `.py` file, layers above re-hit nothing. Only the last layer rebuilds. |
| **`pip install --no-cache-dir --prefer-binary`** | No pip cache bloat. `--prefer-binary` avoids compiling from source where possible. Pinned hashes in `requirements.txt` mean reproducible installs with no network-based tampering. |
| **Non-root user via `RUN addgroup... && adduser...` then `USER`** | Containers running as root = privilege escalation risk. We create `appgroup`/`appuser` matching the UID/GID convention (1001:1001) and set ownership before switching. `security_opt: no-new-privileges` in compose adds another defense layer. |
| **HEALTHCHECK every 30s, timeout 5s, 3 retries** | `/healthz` should respond in <1s for an idle inference server. 30s interval avoids overwhelming the health endpoint. Timeout=5s gives headroom for slow cold-start responses. Retries=3 allows one transient blip without marking unhealthy. |
| **`.dockerignore` excludes `.venv`, `datasets/`, `*.ckpt`, `data/`, test dirs** | Keeps image tiny. Model checkpoints can be mounted at runtime or pulled by the entrypoint — never baked into the image. Venv excluded because we use pip-install in the container, not a host-side venv. |
| **Runtime secrets via `--secret` flag or env_file** | Nothing marked as `ENV` in the Dockerfile. Any API keys, model download tokens, or DB credentials pass through `docker run --env-file` or `docker compose env_file`. In Swarm/K8s they'd use Docker/Kubernetes secrets. |
| **Final size budget** | Base (~150MB) + torch-cpu (~600MB) + torch-models-transforms (~300MB) + uvicorn/gunicorn (~5MB) + app code (~50MB) + system libs (~100MB) ≈ ~1.2GB comfortably under 2.5GB. Leaving ~1.3GB headroom for any additional inference libraries. |

## `Dockerfile`

```dockerfile
###############################################################################
# Stage 1: deps — resolve & download all Python dependencies (pinned, hash-verified)
###############################################################################
FROM python:3.12-slim-bookworm AS deps

WORKDIR /build

# 1) System-level build/runtime dependencies needed for C extensions or native libs.
#    These change rarely; layer will be cached unless modified.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
        libssl3 \
        libomp5 \
        ca-certificates \
        fonts-liberation \
    && rm -rf /var/lib/apt/lists/*

# 2) Copy dependency manifests first — most stable layer boundary.
COPY requirements.txt pyproject.toml ./

# 3) Install deps with hash verification, into a known output directory.
#    --no-cache-dir prevents pip cache from inflating the layer.
#    --prefix isolates wheels to /build/.local so we can COPY them selectively.
RUN pip install --no-cache-dir \
        --prefix=/build/.local \
        --require-hashes \
        -r requirements.txt

# 4) Export an index of installed packages for traceability (optional but useful).
RUN pip freeze > /build/requirements.txt.pinned


###############################################################################
# Stage 2: runtime — minimal final image with only what's needed
###############################################################################
FROM python:3.12-slim-bookworm AS runtime

LABEL maintainer="team-infra@example.com"
LABEL org.opencontainers.image.source="https://github.com/example/inference-service"

# 1) System runtime deps (minimal subset of deps stage — no build tools).
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
        libssl3 \
        libomp5 \
        ca-certificates \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean

# 2) Create non-root user/group with standard conventions.
RUN groupadd -g 1001 appgroup && \
    useradd -u 1001 -g appgroup -m -s /bin/bash appuser

# 3) Working directory for the application.
WORKDIR /app

# 4) Set Python environment variables before installing Python packages.
ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1 \
    PATH="/app/.local/bin:$PATH"

# 5) Copy only the installed site-packages from the deps stage.
#    --from=deps restricts to artifacts built in the previous stage.
#    --chown ensures the non-root user owns everything at copy time.
COPY --from=deps --chown=appuser:appgroup /build/.local /app/.local
COPY --from=deps --chown=appuser:appgroup /build/requirements.txt.pinned /app/

# 6) Copy application source code AFTER dependencies so source-only changes
#    skip the expensive pip install layer.
COPY --chown=appuser:appgroup src/ ./src/
COPY --chown=appuser:appgroup pyproject.toml ./

# 7) Healthcheck — probes /healthz; should respond in <1s for idle service.
HEALTHCHECK --interval=30s --timeout=5s --start-period=10s --retries=3 \
    CMD python -c "import urllib.request; urllib.request.urlopen('http://localhost:8000/healthz')" || exit 1

# 8) Switch to non-root user BEFORE exposing port or setting CMD.
USER appuser

EXPOSE 8000

# 9) Entrypoint uses exec form to receive signals properly (SIGTERM forwarding).
CMD ["uvicorn", "src.main:app", "--host", "0.0.0.0", "--port", "8000"]
```

## Why This Meets Every Requirement

| Requirement | How It's Met |
|---|---|
| **Multi-stage build** | Two stages: `deps` resolves pins, `runtime` copies only compiled artifacts. Build metadata stays in stage 1. |
| **Wheels isolated from runtime** | `--prefix=/build/.local` in deps stage; runtime COPY pulls only `/app/.local` (site-packages + binaries), never pip's internal state. |
| **CPU-only torch with pinned hashes** | `--require-hashes` in `pip install`; `requirements.txt` uses `--index-url https://download.pytorch.org/whl/cpu` and pins every package including `torch==2.5.1+cpu` with its SHA-256. Hashes verified at install time; no supply-chain injection. |
| **Non-root user with proper ownership** | `appuser:appgroup` created at UID/GID 1001. `COPY --chown=appuser:appgroup` transfers ownership during artifact copy. `USER appuser` switches before EXPOSE/CMD. |
| **HEALTHCHECK on /healthz** | 30s interval, 5s timeout, 10s start-period (for model warm-up), 3 retries. Uses inline Python curl to avoid adding `curl` to the final image. |
| **Layer ordering for cache hits** | System deps → static manifests → pip install → source code. Editing `.py` files only invalidates the final layer. Changing only `requirements.txt` skips all downstream layers except pip install. |
| **.dockerignore excludes datasets & .venv** | Both explicitly listed. Also excludes checkpoints (`*.ckpt`, `*.pt`, etc.) so model weights are always mounted at runtime. |
| **Secrets at runtime, never ENV** | No `ENV` directives for sensitive values. `.env` loaded via `env_file` at compose/runtime launch. Models mounted via bind volume. |
| **Final image < 2.5GB** | Estimated ~1.2GB: Python base 150MB + torch-cpu 600MB + transformers/sentence-transformers 300MB + app code + system libs ~150MB. Leaves 1.3GB margin. Without `--no-cache-dir` pip would add ~200MB of downloaded tarballs still present in the layer. |

## Dockerfile

```dockerfile
# =============================================================================
# Stage 1: Build — resolve dependencies, extract wheels, strip artifacts
# =============================================================================
FROM python:3.12-bookworm AS builder

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=0 \
    PIP_DISABLE_PIP_VERSION_CHECK=1

# 1. Install build-time tools needed only for compiling C extensions.
#    These are discarded entirely when the runtime stage starts.
RUN apt-get update && apt-get install -y --no-install-recommends \
        build-essential gcc g++ libopenblas-dev \
    && rm -rf /var/lib/apt/lists/*

# 2. Fetch all dependency wheels into a known directory so we can copy them
#    into the final stage without re-downloading or re-resolving.
WORKDIR /build
COPY requirements.txt ./

# torch-cpu+cpu must be installed before any numpy/pillow/etc that may depend
# on it; pip resolves transitive deps in order here.
RUN set -ex \
    && pip wheel --no-build-isolation --wheel-dir /build/wheels \
         "torch==2.5.1+cpu" \
         --extra-index-url https://download.pytorch.org/whl/cpu \
    && pip wheel --no-build-isolation --wheel-dir /build/wheels \
         -r requirements.txt \
    # Remove downloaded source archives from wheel cache (pip caches .tar.gz)
    && find /root/.cache/pip -name "*.tar.gz" -delete \
    # Strip debug symbols from compiled C/C++ extensions (~100 MB saved)
    && find /build/wheels -name "*.so" -exec strip --strip-debug {} + || true

# =============================================================================
# Stage 2: Runtime — minimal image with extracted wheels, non-root user
# =============================================================================
FROM python:3.12-slim-bookworm AS runtime

# Preserve environment conventions across stages
ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    TORCH_CUDA_ARCH_LIST="" \
    PYTORCH_PRECOMPILE_KERNELS="0" \
    LD_LIBRARY_PATH="/usr/lib/x86_64-linux-gnu:/lib/x86_64-linux-gnu"

# Alpine would need libc6-compat overhead; bookworm-slim has native glibc,
# which PyTorch C++ extensions require. No compat layer needed.
RUN groupadd --gid 1001 app && \
    useradd --uid 1001 --gid app --shell /usr/sbin/nologin --create-home app

WORKDIR /app

# 1. Copy only pre-built wheels — no network access at install time.
#    The hash pin ensures each package is cryptographically verified.
COPY --from=builder /build/wheels /wheels

# pip install with --hash rejects tampered packages. Values shown are SHA256
# digests of the published .whl blobs. Replace with actual hashes from your
# `pip hash` run; these examples use the current stable release.
RUN set -ex \
    && pip install --no-cache-dir --no-index \
         --find-links=/wheels \
         --hash=sha256:a72bdfdf ...torch... \
         torch-cpu \
    && pip install --no-cache-dir --no-index \
         --find-links=/wheels \
         numpy pillow \
    # Project deps (if requirements.txt lists more packages)
    # && pip install --no-cache-dir --no-index \
    #      --find-links=/wheels \
    #      -r requirements.txt

# 2. Application code copied last so dependency layers remain cached across
#    deployments where only source changes.
COPY . .

RUN chown -R app:app /app \
    && find /app -type d -name "__pycache__" -exec rm -rf {} + 2>/dev/null || true

# Switch away from root before any runtime work.
USER app

HEALTHCHECK --interval=30s --timeout=5s --start-period=10s --retries=3 \
    CMD python -c "import urllib.request; r = urllib.request.urlopen('http://localhost:8000/healthz'); exit(0 if r.status == 200 else 1)" || exit 1

EXPOSE 8000

ENTRYPOINT ["python", "-m", "uvicorn"]
CMD ["app.main:create_app", "--host", "0.0.0.0", "--port", "8000"]
```

### Layer-by-layer justification

| Layer | Decision | Why |
|-------|----------|-----|
| `python:3.12-bookworm` (builder) | Debian-based distro for full dev toolchain | Compiling C extensions for numpy, etc., needs `gcc`, `g++`, OpenBLAS headers. Bookworm provides everything without extra packaging overhead. |
| `apt-get install ... && rm -rf /var/lib/apt/lists/*` | Inline cleanup in same `RUN` | Removes APT metadata from the same layer; prevents bloating the image with 200+ MB of package index files. If you split this into two `RUN` statements, the apt list data persists in an intermediate layer even if deleted in the next. |
| `pip wheel --wheel-dir /build/wheels torch-cpu` | Torch wheels fetched first | CPU-only torch (~1 GB) is the single largest dependency. Resolving its transitive deps (numpy, typing-extensions, sympy, filelock) early means subsequent `pip wheel -r requirements.txt` runs against an already-resolved graph — faster and deterministic. |
| `find .../pip -name "*.tar.gz" -delete` | Remove source archives | Pip stores both `.whl` and `.tar.gz` sources in its cache under `--no-build-isolation`. The `.tar.gz` files add ~400 MB with zero runtime value — they're only useful for recompilation. Deleting them inside the builder stage shrinks the builder layer and keeps the `/build/wheels` directory lean. |
| `strip --strip-debug *.so` | Strip debug symbols | PyTorch C++ extensions ship with `.so` files containing `.debug_info` sections. `strip --strip-debug` removes these without breaking functionality, saving ~80-150 MB in the final image. |
| `python:3.12-slim-bookworm` (runtime) | Debian slim, not Alpine | PyTorch loads `libtorch_python.so` and other C++ shared libs at runtime. Those libraries were compiled against glibc, not musl. Using Alpine requires installing `libc6-compat` (~30 MB) as a compatibility shim that adds complexity and potential ABI mismatches. Slim-bookworm has native glibc — clean, correct, and smaller total footprint than alpine + compat. |
| `groupadd / useradd --shell nologin` | Non-root user via system account | `--shell /usr/sbin/nologin` prevents interactive login; `--create-home` gives a clean home dir. UID 1001 / GID 1001 is conventional for containerized apps and avoids conflicts with host-level UIDs. This satisfies the security requirement of never running as root. |
| `COPY --from=builder /build/wheels /wheels` | Extracted wheels from previous stage | The entire point of multi-stage: nothing from the builder (compiled binaries, debug info, apt lists, Python cache) leaks into the runtime. Only `.whl` files travel. |
| `pip install --hash=...` | Cryptographic hash verification | Each `.whl` blob is verified against a known SHA256 digest before installation. If the wheel was corrupted in transit or tampered with (supply-chain attack), pip refuses to install. Run `pip hash /wheels/torch-*.whl` once and paste the result. |
| **Final COPY** | Source code copied last | Docker caches layers by content hash. Since the application source changes far more frequently than dependencies, placing `COPY . .` last means that for typical commits (code changes only), the pip install layer above is reused instantly. You only rebuild the heavy dependency layer when `requirements.txt` actually changes. |

## Justification for Every Layer Decision

### Multi-stage Architecture (Stage 1 → Stage 2)

**Why:** PyTorch's CUDA build alone is ~2.5 GB. CPU-only on `slim` is ~1.1 GB. A single-stage image with all build tools (`gcc`, `make`, kernel headers) easily exceeds 9 GB because `build-essential` and its transitive dependencies bloat `/usr`. By splitting into a builder that has compilers and a runtime that only has compiled binaries, we eliminate ~4 GB of toolchain noise.

### Layer Ordering — Dependencies Before Code

**Order used:**
1. Base image + system libs
2. PyTorch wheels (rarely change)
3. App wheels (change infrequently)
4. Application code (changes constantly)

**Why it matters:** When you edit a single Python file in your service, layers 1–3 hit cache and only layer 4 rebuilds. This reduces typical rebuilds from ~120s to ~8s. If code were copied before requirements, every commit would re-download and reinstall PyTorch.

### Pinned Hashes via `--require-hashes`

In `requirements.txt`:
```text
--require-hashes
torch==2.6.0+cpu \
    --hash sha256:<actual_hash_from_pypi>
```

**Why:** Prevents supply-chain attacks where a compromised PyPI mirror serves poisoned wheels. Pip verifies the hash against the known value before installation — if even one byte differs, the build fails. The alternative (no pinning) lets a bad actor inject arbitrary code into your image on every rebuild.

### Non-Root User

**Pattern:** `groupadd` → `useradd` → `chown` → `USER`

**Why:** Running as root inside a container means a breakout gives root access to the host. With `USER appuser`, a vulnerability in the inference API or PyTorch exploitation only yields a limited-privilege process. The `--system` flag creates the user with UID < 1000, which is conventional for services.

### `tini` as Init Process

**Why:** Containers without an init process leak orphaned child processes when their parent dies, and cannot receive `SIGTERM` gracefully. `tini` sits at PID 1, reaps zombies, and forwards signals. Without it, your FastAPI/Flask app receives no `SIGTERM` during graceful shutdown, causing in-flight requests to be killed abruptly.

### HEALTHCHECK Configuration

| Flag | Value | Rationale |
|------|-------|-----------|
| `--interval` | 10s | Checks every 10s — fast enough to detect hangs without excessive overhead |
| `--timeout` | 5s | Kills the health-check probe if it hangs for 5s |
| `--start-period` | 30s | Gives slow model-loading startup time before failures count |
| `--retries` | 3 | Ignores transient blips; marks unhealthy after 3 consecutive failures |

### Secrets at Runtime, Not Build Time

**Anti-pattern avoided:** Putting secrets in `ENV` statements in the Dockerfile:
```dockerfile
# BAD — secret lives in image history, visible via docker history
ENV DB_PASSWORD=supersecret123
```

**Correct pattern:** Mount secrets as files via Docker Compose. They appear in `/run/secrets/` inside the container at runtime and are never baked into any image layer. `docker history` shows nothing. Even `docker inspect` only masks the content.

### `.dockerignore` — What Gets Excluded and Why

| Pattern | Size Impact | Reason |
|---------|-------------|--------|
| `.venv/` | 2–4 GB | Duplicate `site-packages`, cached wheels, symlinks |
| `datasets/` | 10–100 GB | Model weights/training data, always downloaded or mounted at runtime |
| `__pycache__/` | 50–500 MB | Recompile happens anyway; adds no value |
| `*.log` | Variable | Log files belong in volumes/logs, not image layers |
| `Dockerfile*` | Small | Prevents recursive inclusion of other Docker configs |

The most impactful exclusions are `.venv` and `datasets` — together they typically account for 80%+ of unnecessary bytes being staged for the build context transfer.

### Final Image Size Estimate

| Component | Approximate Size |
|-----------|-----------------|
| `python:3.12-slim` base | ~120 MB |
| `libgomp1` + `libglib2.0-0` + `libmagic1` + `tini` | ~30 MB |
| `torch` (CPU-only) | ~950 MB |
| `torchvision` (CPU-only) | ~15 MB |
| `torchaudio` (CPU-only) | ~10 MB |
| uvicorn + app code + remaining deps | ~100 MB |
| **Total** | **~1.2 GB** |

Well under the 2.5 GB target. The original 9 GB was inflated by: full Ubuntu base (~500 MB), CUDA toolchain (~4 GB), build-essential (~500 MB), cached pip downloads (~1 GB), kernel headers (~300 MB), and uncompressed layer duplication (~2 GB).

## Justification — Why Each Layer Decision Was Made

| Layer Decision | Rationale |
|---|---|
| **Multi-stage build** | Build-time deps (`build-essential`, compilers) live only in `wheel-builder`. The final image has *zero* compilers, debug symbols, or C headers. That alone shaves 1–2 GB off the previous 9 GB monolith. |
| **`pip wheel --only-binary=:all:`** | Forces pre-built wheels only. No `gcc` installed, no compilation step. Faster builds, reproducible artifacts, and avoids pulling in header packages that bloat the image. |
| **CPU-only torch** | CUDA runtime + cuDNN + NCCL = ~1.5 GB of wasted space when you have no GPU. `torch+cpu` variant is ~400 MB. Total savings: ~50% of what would be consumed by PyTorch alone. |
| **Hash-pinned `requirements.txt`** | `--require-hashes` blocks supply-chain tampering. An attacker cannot swap a malicious wheel because the hash check fails. Deterministic installs mean rebuilds are bit-identical. |
| **Venv + `pip install --no-index --find-links`** | Installing from the local `/wheels` directory inside the runtime stage is faster than reaching out to PyPI and isolates the runtime build from the internet. The `--no-index` flag guarantees offline installs. |
| **Dependency layers BEFORE source copy** | `COPY requirements.txt` → `pip install` happens *before* `COPY . .`. If you edit only your app code, Docker reuses the cached wheel-install layer. No unnecessary recompilation. |
| **Non-root `USER appuser`** | Running as UID 1000 eliminates entire classes of vulnerabilities (container escape gains you a shell, not root on the host). Also prevents accidental writes outside the workdir. |
| **`tini` as ENTRYPOINT** | Without an init process, signals like SIGTERM get lost and containers hang during shutdown. `tini` reaps zombies and forwards signals properly. |
| **`HEALTHCHECK` hitting `/healthz`** | Orchestrators (Docker Compose, Kubernetes) probe this endpoint to auto-recover hung containers. 30 s interval / 5 s timeout / 3 retries is a sensible default that won't hammer the service. |
| **Secrets via `docker-compose secrets`** | `ENV SECRET_KEY=xxx` in the Dockerfile persists in image history and `docker inspect`. Secrets are injected at runtime via Docker Secret or Swarm, exist only in memory, and are never stored in the image layers. |
| **`.dockerignore` excludes `datasets/`** | Models and training data are massive. Bind-mount them instead so images stay lean and data stays on disk where it belongs. |
| **`.apt-get` + `rm -rf /var/lib/apt/lists/*`** | Debian-slim ships with an empty apt cache, but `apt-get update` populates `/var/lib/apt/lists`. Cleaning it removes metadata after install (~200 MB saved). |

| **Cache mount for pip** | `--mount=type=cache,target=/root/.cache/pip` (if using BuildKit) caches downloaded wheels between builds. Subsequent installs of unchanged dependencies skip the network entirely. |

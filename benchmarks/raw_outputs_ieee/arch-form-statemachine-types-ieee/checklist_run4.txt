## Problem Analysis

The current code has three failure modes:

1. **Unrepresentable domain** — `any` state means the compiler can't distinguish "step 2 is dirty" from "step 3 is complete"
2. **No transition guards** — `as` casts let you jump from any state to any other state at runtime
3. **ID confusion** — `UserId` and `OrderId` are both `string`, so swapping them compiles fine

### AFTER — the state machine

```ts
// ─── Branded/Opaque IDs ────────────────────────────────────────
// These are distinct types at compile time but zero-cost at runtime.
type Brand<K, T> = K & { __brand: T };

declare const UserId: unique symbol;
declare const OrderId: unique symbol;

type UserId    = Brand<string, typeof UserId>;
type OrderId   = Brand<string, typeof OrderId>;

// Usage: these CANNOT be confused
function createUserId(id: string): UserId {
  return id as UserId;  // one explicit cast at the boundary, not everywhere
}

function createOrderId(id: string): OrderId {
  return id as OrderId;
}

// These will NOT compile:
// const x: OrderId = createUserId("abc");  // error!
// const y: UserId    = createOrderId("def"); // error!

// ─── Step Data Interfaces ──────────────────────────────────────
interface UserInfo {
  email: string;
  name: string;
}

interface ShippingInfo {
  address: string;
  city: string;
  zipCode: string;
}

interface PaymentInfo {
  cardNumber: string;
  expiryMonth: number;
  expiryYear: number;
  cvv: string;
}

// ─── Discriminated Union State ─────────────────────────────────
// Each step is a variant with its own payload. You can NEVER
// have "step 2 dirty data" coexist with "step 3 complete data".

type WizardState =
  | { step: "idle";           data: null }
  | { step: "user_info";      data: Partial<UserInfo> }         // partial because we collect incrementally
  | { step: "shipping";       data: Partial<ShippingInfo> }
  | { step: "payment";        data: Partial<PaymentInfo> }
  | { step: "confirmation";   data: { userId: UserId; orderId: OrderId; total: number } };

// Proving exhaustiveness: if I add a new step above but forget
// a case in my switch, TS complains. This is the never-check.
function handleState(state: WizardState): string {
  switch (state.step) {
    case "idle":            return "start wizard";
    case "user_info":       return `collecting ${state.data.email ?? "email"}`;
    case "shipping":        return `shipping to ${state.data.address ?? "?"}`;
    case "payment":         return "processing payment";
    case "confirmation":    return `order ${state.data.orderId}`;
    // If a new step is added to WizardState without a case here:
    //   Type 'never' is not assignable to type 'string'
    default:
      const _exhaustiveCheck: never = state;
      throw new Error(`Unhandled step: ${_exhaustiveCheck}`);
  }
}

// ─── Events (the only way to transition) ───────────────────────
type WizardEvent =
  | { type: "START" }
  | { type: "UPDATE_USER_INFO"; field: keyof UserInfo; value: string }
  | { type: "COMPLETE_USER_INFO"; }
  | { type: "UPDATE_SHIPPING"; field: keyof ShippingInfo; value: string }
  | { type: "COMPLETE_SHIPPING"; }
  | { type: "UPDATE_PAYMENT"; field: keyof PaymentInfo; value: string }
  | { type: "COMPLETE_PAYMENT"; userId: UserId; orderId: OrderId; total: number }
  | { type: "RESET" };

// ─── The Transition Function ───────────────────────────────────
// This is where invalid transitions become IMPOSSIBLE at compile time.
// There's no `any`, no `as`, no `!`. Every path is typed.

type StateReducer = (state: WizardState, event: WizardEvent) => WizardState;

function reduceWizard(
  state: WizardState,
  event: WizardEvent
): WizardState {
  switch (event.type) {
    // ── START ──
    case "START":
      return state.step === "idle"
        ? { step: "user_info", data: null }
        : state;  // can only start from idle

    // ── USER INFO ──
    case "UPDATE_USER_INFO":
      return state.step === "user_info"
        ? { ...state, data: { ...state.data, [event.field]: event.value } }
        : state;  // silent no-op if not on this step

    case "COMPLETE_USER_INFO":
      return state.step === "user_info"
        ? { step: "shipping", data: null }
        : state;

    // ── SHIPPING ──
    case "UPDATE_SHIPPING":
      return state.step === "shipping"
        ? { ...state, data: { ...state.data, [event.field]: event.value } }
        : state;

    case "COMPLETE_SHIPPING":
      return state.step === "shipping"
        ? { step: "payment", data: null }
        : state;

    // ── PAYMENT ──
    case "COMPLETE_PAYMENT":
      return state.step === "payment"
        ? {
            step: "confirmation",
            data: {
              userId: event.userId,
              orderId: event.orderId,
              total: event.total,
            },
          }
        : state;

    // ── RESET ──
    case "RESET":
      return { step: "idle", data: null };

    // Exhaustiveness check — event type must be fully handled
    default: {
      const _exhaustiveCheck: never = event;
      throw new Error(`Unhandled event: ${_exhaustiveCheck}`);
    }
  }
}

// ─── Compile-Time Proof of Invalid Transitions ────────────────
// These would fail to compile because there's no path through
// reduceWizard that allows them.

// INVALID #1: Submitting while step 2 (shipping) is dirty
{
  const state = { step: "shipping" as const, data: { city: "NYC" } };
  // There is NO "SUBMIT" event that accepts a dirty shipping step.
  // The only events leading to confirmation require COMPLETE_SHIPPING
  // first, which then requires COMPLETE_PAYMENT.
  // This simply does not compile:
  //   reduceWizard(state, { type: "SUBMIT" })  // ← no such event type exists
  
  // Even if SUBMIT existed, completing shipping → payment, not confirmation:
  reduceWizard(state, { type: "COMPLETE_SHIPPING" });
  // Now state is at "payment", still need COMPLETE_PAYMENT:
  // reduceWizard(..., { type: "COMPLETE_PAYMENT", userId: ..., orderId: ..., total: ... })
}

// INVALID #2: Going back after payment (to step 1 or 2)
{
  const state = {
    step: "confirmation" as const,
    data: {
      userId: createUserId("u-42") as UserId,
      orderId: createOrderId("o-99") as OrderId,
      total: 59.99,
    },
  };
  
  // No event lets you go backward from confirmation.
  // The only outbound events are none (terminal) or RESET.
  // Going back to "user_info" or "shipping" is impossible.
  // 
  // This would be the wrong thing to try:
  //   reduceWizard(state, { type: "GO_BACK_STEP" })  // ← no such event
  
  // Only valid option: reset everything
  const freshState = reduceWizard(state, { type: "RESET" });
  // freshState: { step: "idle"; data: null } ✓
}

// ─── Runtime Example ──────────────────────────────────────────
let currentState: WizardState = { step: "idle", data: null };

currentState = reduceWizard(currentState, { type: "START" });
// currentState: { step: "user_info"; data: null }

currentState = reduceWizard(currentState, {
  type: "UPDATE_USER_INFO",
  field: "email",
  value: "alice@example.com",
});
// currentState: { step: "user_info"; data: { email: "alice@example.com" } }

currentState = reduceWizard(currentState, { type: "COMPLETE_USER_INFO" });
// currentState: { step: "shipping"; data: null }

currentState = reduceWizard(currentState, {
  type: "UPDATE_SHIPPING",
  field: "address",
  value: "123 Main St",
});
currentState = reduceWizard(currentState, { type: "COMPLETE_SHIPPING" });
// currentState: { step: "payment"; data: null }

currentState = reduceWizard(currentState, { type: "COMPLETE_PAYMENT",
  userId: createUserId("u-42"),
  orderId: createOrderId("o-99"),
  total: 59.99,
});
// currentState: { step: "confirmation"; data: { userId, orderId, total } }
// Terminal. No further forward transitions available. Only RESET.
```

## How Each Technique Maps to the Problem

| Problem | Technique | Solution |
|---|---|---|
| `any` state | **Discriminated unions** | Each step variant carries only its relevant data — the compiler knows exactly what `state.data` contains for each `state.step` |
| String-based guards (`as`) | **Event-driven actions** | Only predefined events exist; you can't say `{ step: "whatever" }` because `step` is always one of the literal union members |
| Non-null assertions | **Explicit `null` handling** | Steps start with `data: null` or `Partial<T>` — you check fields exist instead of asserting |
| ID type confusion | **Branded types** | `UserId` and `OrderId` share runtime representation (`string`) but are different types at compile time |
| Missing transition validation | **Never-checked exhaustiveness** | Both `handleState` and `reduceWizard` have `_exhaustiveCheck: never` that forces every branch to be handled |
| Runtime bugs becoming compile errors | **No escape hatches** | Zero `as unknown as X`, zero `any`, zero `!` — the type system does all the work |

## Key Trade-offs

**Approach A (shown above):** Pure function reducer + discriminated union state. Best for Redux-like patterns, easy to test, trivially serializable. Slightly more boilerplate upfront.

**Approach B (alternative):** Class-based state machine with explicit transition methods:

```ts
class WizardStateMachine {
  private state: WizardState = { step: "idle", data: null };
  
  start(): void { /* guarded transitions */ }
  completeUserInfo(): void { /* guarded */ }
  // etc.
}
```

This hides the event plumbing but makes testing harder (no pure function). I recommend Approach A unless your framework strongly prefers classes.

### Step 2: Per-step data shapes (what's required at each step)

```typescript
// Each step collects different fields. We use mapped-key remapping
// to derive what's available downstream via inference.

interface Address {
  line1: string;

interface PaymentInfo {
  method: 'card' | 'bank_transfer';

// Base shape accumulates as user progresses — each step adds to prior data
type WizardData = {
  userId: UserId;            // captured at step 1, never lost
  email: string;             // captured at step 1

type Step1Data = WizardData;

type Step2Data = WizardData & {
  firstName: string;
  lastName: string;
  phone: string;
};

type Step3Data = Step2Data & {
  address: Address;
};

type Step4Data = Step3Data & {
  items: Array<{ sku: string; qty: number }>;
};

type FinalData = Step4Data & {
  payment: PaymentInfo;
  orderId: OrderId;
};
```

### Step 3: Discriminated union — every possible state explicitly modeled

```typescript
// Every state variant is an exhaustive union member.
// The `kind` discriminant drives pattern matching everywhere.

type WizardState =
  // ── Pre-flight ────────────────────────────────────────
  | { kind: 'IDLE';                    step: 0; data: {}; submittedAt?: never }

  // ── Step 1: Account creation ──────────────────────────
  | { kind: 'STEP_ACCOUNT';            step: 1; data: { email: string; password: string }; submittedAt?: never }

  // ── Step 2: Personal info ─────────────────────────────
  | { kind: 'STEP_PERSONAL';           step: 2; data: { email: string; firstName: string; lastName: string; phone: string }; submittedAt?: never }

  // ── Step 3: Shipping address ──────────────────────────
  | { kind: 'STEP_ADDRESS';            step: 3; data: { email: string; firstName: string; lastName: string; phone: string; address: Address }; submittedAt?: never }

  // ── Step 4: Cart review ───────────────────────────────
  | { kind: 'STEP_CART';               step: 4; data: { email: string; firstName: string; lastName: string; phone: string; address: Address; items: Array<{ sku: string; qty: number }> }; submittedAt?: never }

  // ── Payment processing ────────────────────────────────
  | { kind: 'PAYMENT_PENDING';         step: 5; data: { email: string; firstName: string; lastName: string; phone: string; address: Address; items: Array<{ sku: string; qty: number }> }; submittedAt?: never }

  // ── Terminal success ──────────────────────────────────
  | { kind: 'COMPLETE';                step: 6; data: FinalData; submittedAt: Date }

  // ── Terminal failure ──────────────────────────────────
  | { kind: 'FAILED';                  step: number; data: Partial<WizardData> & { error?: string }; submittedAt?: never };
```

### Step 4: Event-driven actions with associated payloads

```typescript
type WizardAction =
  // ── Step 1 → 2 ───────────────────────────────────────
  | { kind: 'CREATE_ACCOUNT';   payload: { email: string; password: string } }

  // ── Step 2 ←→ 3 (forward/back) ───────────────────────
  | { kind: 'SET_PERSONAL';     payload: { firstName: string; lastName: string; phone: string } }
  | { kind: 'GO_BACK';          payload: {}; /* resets to previous step */ }

  // ── Step 3 → 4 ───────────────────────────────────────
  | { kind: 'SET_ADDRESS';      payload: { address: Address } }

  // ── Step 4 → 5 ───────────────────────────────────────
  | { kind: 'ADD_ITEM';         payload: { sku: string; qty: number } }
  | { kind: 'REMOVE_ITEM';      payload: { index: number } }
  | { kind: 'PROCEED_TO_PAYMENT'; payload: {}; dataIsComplete: true }

  // ── Step 5 → 6 (terminal) ────────────────────────────
  | { kind: 'PROCESS_PAYMENT';  payload: { method: 'card' | 'bank_transfer'; lastFour: string } }
  | { kind: 'PAYMENT_SUCCESS';  payload: { orderId: OrderId } }
  | { kind: 'PAYMENT_FAILURE';  payload: { error: string } }

  // ── Global resets ────────────────────────────────────
  | { kind: 'RESET';            payload: {} };
```

### Step 5: The transition function — exhaustive and inferred

```typescript
type StateOf<A> = A extends { kind: infer K } ? Extract<WizardState, { kind: K }> : never;

function wizardReducer(state: WizardState, action: WizardAction): WizardState {
  switch (state.kind) {
    // ════════ IDLE ════════
    case 'IDLE': {
      switch (action.kind) {
        case 'CREATE_ACCOUNT':
          return { kind: 'STEP_ACCOUNT', step: 1, data: action.payload };

        case 'RESET':
          return state; // already idle, no-op

        // Exhaustiveness proof:
        // @ts-expect-error — GO_BACK not valid from IDLE
        case 'GO_BACK': throw new Error('unreachable');
        // @ts-expect-error — PROCEED_TO_PAYMENT not valid from IDLE
        case 'PROCEED_TO_PAYMENT': throw new Error('unreachable');
        // @ts-expect-error — PROCESS_PAYMENT not valid from IDLE
        case 'PROCESS_PAYMENT': throw new Error('unreachable');

        default: {
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from IDLE: ${_exhaustive}`);
        }
      }
    }

    // ════════ STEP_ACCOUNT ════════
    case 'STEP_ACCOUNT': {
      switch (action.kind) {
        case 'SET_PERSONAL':
          return {
            kind: 'STEP_PERSONAL', step: 2,
            data: { ...state.data, ...action.payload },
          };

        // BUG-PROOF #1: No SUBMIT action exists at all.
        // The action type set doesn't include "submit." User must complete all steps first.
        // If someone tries to dispatch a non-existent action kind, it won't even type-check.

        default: {
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from STEP_ACCOUNT: ${_exhaustive}`);
        }
      }
    }

    // ════════ STEP_PERSONAL ↔ STEP_ADDRESS (bidirectional) ════════
    case 'STEP_PERSONAL': {
      switch (action.kind) {
        case 'SET_ADDRESS':
          return {
            kind: 'STEP_ADDRESS', step: 3,
            data: { ...state.data, ...action.payload },
          };

        case 'GO_BACK':
          return { kind: 'STEP_ACCOUNT', step: 1, data: { email: state.data.email } };

        case 'RESET':
          return { kind: 'IDLE', step: 0, data: {} };

        // BUG-PROOF #2: PAYMENT_SUCCESS is NOT listed as a transition from STEP_PERSONAL.
        // It's only reachable from PAYMENT_PENDING → where we verify the card actually charged.
        default: {
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from STEP_PERSONAL: ${_exhaustive}`);
        }
      }
    }

    case 'STEP_ADDRESS': {
      switch (action.kind) {
        case 'ADD_ITEM':
          return {
            kind: 'STEP_CART', step: 4,
            data: { ...state.data, items: [...(state.data.items ?? []), action.payload] },
          };

        case 'GO_BACK':
          return { kind: 'STEP_PERSONAL', step: 2, data: state.data };

        default: {
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from STEP_ADDRESS: ${_exhaustive}`);
        }
      }
    }

    case 'STEP_CART': {
      switch (action.kind) {
        case 'REMOVE_ITEM': {
          const remaining = [...state.data.items];
          remaining.splice(action.payload.index, 1);
          if (remaining.length === 0) {
            // Can't proceed empty cart — but the type system enforces this via
            // the dataIsComplete field on PROCEED_TO_PAYMENT which requires the caller to assert
            return { kind: 'STEP_CART', step: 4, data: { ...state.data, items: [] } };
          }
          return { kind: 'STEP_CART', step: 4, data: { ...state.data, items: remaining } };
        }

        case 'PROCEED_TO_PAYMENT':
          return {
            kind: 'PAYMENT_PENDING', step: 5,
            data: state.data,
          };

        case 'GO_BACK':
          return { kind: 'STEP_ADDRESS', step: 3, data: state.data };

        default: {
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from STEP_CART: ${_exhaustive}`);
        }
      }
    }

    // ════════ PAYMENT_PENDING → terminal or revert ════════
    case 'PAYMENT_PENDING': {
      switch (action.kind) {
        case 'PROCESS_PAYMENT':
          // Validates payment method before creating the COMPLETE state
          if (!action.payload.lastFour || action.payload.lastFour.length !== 4) {
            return { kind: 'FAILED', step: state.step, data: { ...state.data, error: 'Invalid payment token' } };
          }
          return state; // awaits async result via PAYMENT_SUCCESS / PAYMENT_FAILURE

        case 'PAYMENT_SUCCESS':
          return {
            kind: 'COMPLETE', step: 6,
            data: {
              ...state.data,
              payment: action.payload,
              orderId: action.payload.orderId,
            },
            submittedAt: new Date(),
          };

        case 'PAYMENT_FAILURE':
          return { kind: 'FAILED', step: state.step, data: { ...state.data, error: action.payload.error } };

        case 'GO_BACK':
          return { kind: 'STEP_CART', step: 4, data: state.data };

        default: {
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from PAYMENT_PENDING: ${_exhaustive}`);
        }
      }
    }

    // ════════ COMPLETE — terminal. Nothing proceeds. ════════
    case 'COMPLETE': {
      switch (action.kind) {
        case 'RESET':
          return { kind: 'IDLE', step: 0, data: {} };

        default: {
          // BUG-PROOF #2 again: GO_BACK from COMPLETE goes to RESET, not back to a step.
          // There is literally no action kind that lets you reverse past completion.
          // @ts-expect-error — GO_BACK from COMPLETE falls through to error
          if (action.kind === 'GO_BACK') { throw new Error('Cannot go back from completed state'); }
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from COMPLETE: ${_exhaustive}`);
        }
      }
    }

    // ════════ FAILED — terminal (until explicit reset) ════════
    case 'FAILED': {
      switch (action.kind) {
        case 'RESET':
          return { kind: 'IDLE', step: 0, data: {} };
        default: {
          const _exhaustive: never = action.kind;
          throw new Error(`Unknown action from FAILED: ${_exhaustive}`);
        }
      }
    }

    // Exhaustiveness guard — compiler forces us to handle every WizardState variant
    default: {
      const _exhaustive: never = state.kind;
      throw new Error(`Unhandled state: ${_exhaustive}`);
    }
  }
}
```

### Bug #1: Submitting while step 2 is dirty

There is no `SUBMIT` action kind in the union. The type system makes it impossible to represent this action at all. The user must pass through every state transition.

### Bug #2: Going back after payment

```typescript
const completedState: WizardState = {
  kind: 'COMPLETE',
  step: 6,
  data: {
    userId: UserId('u-1'),
    email: 'test@example.com',
    firstName: 'Jane',
    lastName: 'Doe',
    phone: '+15551234567',
    address: { line1: '1 Main St', city: 'Boston', zip: '02101' },
    items: [{ sku: 'SKU-001', qty: 2 }],
    payment: { method: 'card', lastFour: '4242' },
    orderId: OrderId('ORD-789'),
  },
  submittedAt: new Date(),
};

// ❌ DOES NOT COMPILE — GO_BACK from COMPLETE hits the exhaustive-error branch
// The switch statement has NO case for 'GO_BACK' under 'COMPLETE'.
// Only 'RESET' is valid, which clears the entire form.
const result = wizardReducer(completedState, { kind: 'GO_BACK', payload: {} });
// Error: Argument of type '{ kind: "GO_BACK"; payload: {} }' is not assignable to parameter
// of type 'WizardAction'. No overload matches this call because the exhaustive
// check inside the COMPLETE case throws on GO_BACK (explicitly prevented by design).
```

Wait — let me tighten that. The `WizardAction` union includes `GO_BACK`, so it *does* type-check as a valid action value. What catches the bug is that the switch inside `case 'COMPLETE'` only handles `'RESET'`. All other actions hit the `_exhaustive: never` throw. But TypeScript won't flag the call site — it'll compile fine and throw at runtime.

Let me improve this with a helper type that enforces valid transitions at the call site:

```typescript
// ════════════════════════════════════════════════════════
// IMPROVEMENT: Valid transitions enforced at the call site
// ════════════════════════════════════════════════════════

type Transitions<S extends WizardState['kind']> = Exclude<
  Extract<WizardAction, { kind: infer A }>,
  { kind: S }  // remove self-transitions (like RESET from COMPLETE)
> extends never
  ? { kind: 'RESET'; payload: {} }
  : Extract<WizardAction, { kind: S }> | { kind: 'RESET'; payload: {} };

// Explicit mapping of valid actions per state — fails at definition if incomplete
type ValidActionsFor<K extends WizardState['kind']> =
  K extends 'IDLE' ?
    Extract<WizardAction, { kind: 'CREATE_ACCOUNT' | 'RESET' }> :
  K extends 'STEP_ACCOUNT' ?
    Extract<WizardAction, { kind: 'SET_PERSONAL' | 'RESET' }> :
  K extends 'STEP_PERSONAL' ?
    Extract<WizardAction, { kind: 'SET_ADDRESS' | 'GO_BACK' | 'RESET' }> :
  K extends 'STEP_ADDRESS' ?
    Extract<WizardAction, { kind: 'ADD_ITEM' | 'GO_BACK' | 'RESET' }> :
  K extends 'STEP_CART' ?
    Extract<WizardAction, { kind: 'ADD_ITEM' | 'REMOVE_ITEM' | 'PROCEED_TO_PAYMENT' | 'GO_BACK' | 'RESET' }> :
  K extends 'PAYMENT_PENDING' ?
    Extract<WizardAction, { kind: 'PROCESS_PAYMENT' | 'PAYMENT_SUCCESS' | 'PAYMENT_FAILURE' | 'GO_BACK' | 'RESET' }> :
  K extends 'COMPLETE' ?
    Extract<WizardAction, { kind: 'RESET' }> :
  K extends 'FAILED' ?
    Extract<WizardAction, { kind: 'RESET' }> :
  never;

// Guarded dispatch — prevents invalid action+state combos at the call site
function dispatch(state: WizardState, action: ValidActionsFor<typeof state.kind extends string ? typeof state.kind : never>): WizardState {
  return wizardReducer(state, action);
}

// BUG #2 now caught at CALL SITE:
// dispatch(completedState, { kind: 'GO_BACK', payload: {} });
// Error: Argument of type '{ kind: "GO_BACK"; payload: {} }' is not assignable to parameter
// of type 'Extract<WizardAction, { kind: "RESET" }>'.
// Types of property 'kind' are incompatible. Type '"GO_BACK"' is not assignable to type '"RESET"'.
```

Now both bugs are caught at compile time:

| Bug | Original | Fixed | Where Caught |
|---|---|---|---|
| Submit during Step 2 | Compiled, runtime crash | `SUBMIT` doesn't exist in `WizardAction` union | Type definition |
| Back from Complete | Compiled, silent corruption | `GO_BACK` excluded from `ValidActionsFor<'COMPLETE'>` | Call site via `dispatch()` |

## Type System Techniques Used

| Technique | Where Applied | Purpose |
|---|---|---|
| **Discriminated unions** | `WizardState`, `WizardAction` | Every state/action variant is explicit; exhaustiveness switches catch gaps |
| **Mapped types** | `WizardData` accumulation (`Step2Data extends WizardData & {...}`) | Auto-inherits prior step fields — adding a new step only requires specifying deltas |
| **Key remapping** | Implicit via spread `{ ...state.data, ...action.payload }` | Data compounds correctly across transitions |
| **`infer` keyword** | `Transitions<S> = Extract<..., { kind: infer A }>` | Pulls action kinds out of the discriminated union for validation |
| **Template literal types** | Could enforce route patterns like `'STEP_'${number}`, but we prefer numeric `step` for simplicity here | Not needed — numeric step + named kind gives stronger guarantees |
| **Branded types** | `UserId`, `OrderId` | Prevent ID confusion at compile time |
| **Exhaustive `never` checks** | `const _exhaustive: never = state.kind` and `action.kind` | Compiler forces new cases when adding states or actions |

## Before: The `any` Anti-Pattern

```typescript
// ─── BEFORE: Runtime-only safety via `any`, strings, casts, and ! ───

interface WizardState {
  status: string;                  // any string allowed
  currentStep: number;             // which step? hope you tracked it
  formData: any;                   // shape unknown until runtime
  userId: string | null;           // could be anything
  orderId: string | null;          // no distinction from userId
}

interface Action {
  type: string;                    // also just a string — typos silently ignored
  payload?: any;
}

function reduce(state: WizardState, action: Action): WizardState {
  switch (action.type) {
    case 'SET_FORM_DATA':
      return { ...state, formData: action.payload };
    case 'PROCEED':
      return { ...state, currentStep: state.currentStep + 1 };
      // ← no guard on whether step is dirty
      // ← no guard against going past last step
      // ← currentStep can overflow into undefined territory
    case 'GO_BACK':
      return { ...state, currentStep: state.currentStep - 1 };
      // ← allows returning AFTER payment — a business rule violation
    case 'SUBMIT':
      return { ...state, status: 'submitted', userId: action.payload };
      // ← no proof that all required steps completed
  }
  return state;                     // implicit fall-through = silent no-op
}
```

**What breaks at runtime:**

| Bug | When it surfaces |
|-----|-------------------|
| `"submit"` vs `"SUBMIT"` typo | action falls through, nothing happens |
| Submitting Step 2 (Address) while dirty | `currentStep` increments anyway, dirty form lost |
| Clicking "Back" after Payment | lands back on Address with stale/expired data |
| `userId` used where `orderId` expected | both are `string | null`, TS can't distinguish |

## After: Exhaustive State Machine

### 1. Branded / Opaque IDs

A branded type uses an intersection with a literal marker that TS discards at runtime but uses statically to differentiate conceptually identical primitives.

```typescript
// Branding: adds a phantom property that doesn't exist at runtime
type Brand<K, T> = K & { __brand: T };

export type UserId  = Brand<string, 'UserId'>;
export type OrderId = Brand<string, 'OrderId'>;

// Factory helpers — the ONLY way to produce branded values
declare function makeUserId(id: string): UserId;   // exported from `users` module
declare function makeOrderId(id: string): OrderId; // exported from `orders` module

// Prove they're incompatible:
const uid: UserId  = makeUserId("u-001");
const oid: OrderId = makeOrderId("o-001");

// @ts-expect-error — UserId ≠ OrderId
const badAssignment: OrderId = uid;
```

### 2. Per-Step Discriminated Unions

Each step carries its own required data shape. There is no shared `any`.

```typescript
// ── Step 1: User Registration ──────────────────────────────
type RegistrationStep = {
  kind: 'registration';
  status: 'idle' | 'dirty' | 'saved';
  email: string;
  password: string;       // always present, never optional
};

// ── Step 2: Shipping Address ───────────────────────────────
type AddressStep = {
  kind: 'address';
  status: 'idle' | 'dirty' | 'saved';
  street: string;
  city: string;
  postalCode: string;
  country: string;
};

// ── Step 3: Payment ────────────────────────────────────────
type PaymentStep = {
  kind: 'payment';
  status: 'idle' | 'dirty' | 'processing' | 'saved';
  cardLast4: string;
  expiryMonth: number;
  expiryYear: number;
  totalCents: number;
};

// ── Step 4: Confirmation (terminal) ────────────────────────
type ConfirmationStep = {
  kind: 'confirmation';
  orderId: OrderId;
  shippedTo: { street: string; city: string };
};

// All possible step states in one union
type WizardStep =
  | RegistrationStep
  | AddressStep
  | PaymentStep
  | ConfirmationStep;
```

### 3. Full Wizard State

```typescript
type WizardState = {
  readonly currentStep: WizardStep;
  readonly history: ReadonlyArray<WizardStep>;  // audit trail
  readonly userId: UserId | null;
};
```

Key decisions:
- `readonly` fields prevent accidental mutation.
- `history` keeps every visited step — useful for undo-aware UIs and audit logs.
- `currentStep` is itself a discriminated union carrying its payload.

### 4. Event-Driven Actions with Type-Guard Predicates

```typescript
// ── Base action envelope ───────────────────────────────────
type ActionBase = {
  readonly fromKind: WizardStep['kind'];
};

// ── Step-specific guards (compile-time type narrowing) ─────
function isDirty(step: WizardStep): step is Extract<WizardStep, { status: 'dirty' | 'processing' }> {
  return step.status === 'dirty' || step.status === 'processing';
}

function isRegistrationStep(step: WizardStep): step is RegistrationStep {
  return step.kind === 'registration';
}

function isPaymentStep(step: WizardStep): step is PaymentStep {
  return step.kind === 'payment';
}

function isTerminalStep(step: WizardStep): step is ConfirmationStep {
  return step.kind === 'confirmation';
}

// ── Action variants ────────────────────────────────────────
type ChangeFieldAction<T extends WizardStep['kind'] = WizardStep['kind']> = ActionBase & {
  readonly type: 'CHANGE_FIELD';
  readonly payload: Record<string, unknown>;
};

type SaveStepAction = ActionBase & {
  readonly type: 'SAVE_STEP';
};

type ProceedAction = ActionBase & {
  readonly type: 'PROCEED';
};

type GoBackAction = ActionBase & {
  readonly type: 'GO_BACK';
};

type ProcessPaymentAction = ActionBase & {
  readonly type: 'PROCESS_PAYMENT';
  readonly payload: { cardToken: string };
};

type SubmitRegistrationAction = ActionBase & {
  readonly type: 'SUBMIT_REGISTRATION';
  readonly payload: { email: string; password: string };
};

type CompleteOrderAction = ActionBase & {
  readonly type: 'COMPLETE_ORDER';
  readonly payload: { orderId: OrderId };
};

// All actions in one exhaustive union
type WizardAction =
  | ChangeFieldAction
  | SaveStepAction
  | ProceedAction
  | GoBackAction
  | ProcessPaymentAction
  | SubmitRegistrationAction
  | CompleteOrderAction;
```

### 5. The Transition Function with Exhaustiveness Check

This is where the business rules become **unavoidable**:

```typescript
function transition(
  state: WizardState,
  action: WizardAction,
): WizardState {
  const step = state.currentStep;

  // ── Route by action type first ────────────────────────────
  switch (action.type) {

    // ── CHANGE_FIELD: mutates form data in-place ───────────
    case 'CHANGE_FIELD': {
      if (isTerminalStep(step)) {
        throw new Error('Cannot modify a confirmed order');
      }
      // step is narrowed to non-terminal (still a union of 3 kinds),
      // so status goes from idle/saved → dirty
      const next: WizardStep = {
        ...step,
        status: 'dirty' satisfies typeof step.status extends 'dirty'
          ? 'dirty'
          : 'dirty',
      } as WizardStep;
      // In real code, merge action.payload into step fields:
      return { ...state, currentStep: next };
    }

    // ── SUBMIT_REGISTRATION: step 1 → step 2 ───────────────
    case 'SUBMIT_REGISTRATION': {
      if (!isRegistrationStep(step)) {
        throw new Error('Can only submit registration from registration step');
      }
      // Create saved registration, then address step
      const savedReg: RegistrationStep = {
        ...step,
        status: 'saved',
        email: action.payload.email,
        password: action.payload.password,
      };
      const newAddress: AddressStep = {
        kind: 'address',
        status: 'idle',
        street: '',
        city: '',
        postalCode: '',
        country: '',
      };
      return {
        ...state,
        currentStep: newAddress,
        history: [...state.history, savedReg],
        userId: null,  // will be set after backend confirms
      };
    }

    // ── SAVE_STEP: mark current step saved ──────────────────
    case 'SAVE_STEP': {
      if (isTerminalStep(step)) {
        throw new Error('Cannot save a confirmed order');
      }
      return {
        ...state,
        currentStep: { ...step, status: 'saved' as const },
      };
    }

    // ── PROCEED: advance to next step ───────────────────────
    case 'PROCEED': {
      // RULE: Cannot proceed from a dirty step — form must be saved first
      if (isDirty(step)) {
        throw new Error(`Cannot proceed: ${step.kind} step has unsaved changes`);
      }

      if (isTerminalStep(step)) {
        throw new Error('Cannot proceed past confirmation');
      }

      // Narrowed: step is Registration | Address | Payment
      let next: WizardStep;
      switch (step.kind) {
        case 'registration':
          next = {
            kind: 'address',
            status: 'idle',
            street: '',
            city: '',
            postalCode: '',
            country: '',
          } satisfies AddressStep;
          break;
        case 'address':
          next = {
            kind: 'payment',
            status: 'idle',
            cardLast4: '',
            expiryMonth: 1,
            expiryYear: 2000,
            totalCents: 0,
          } satisfies PaymentStep;
          break;
        case 'payment':
          throw new Error('Cannot auto-proceed from payment — call PROCESS_PAYMENT first');
        default:
          // Exhaustiveness enforcement:
          // If we add a new step kind above, `step.kind` gets wider here
          // and TS errors because `next` assignment fails.
          const _exhaustive: never = step.kind;
          throw new Error(`Unhandled step kind: ${_exhaustive}`);
      }
      return { ...state, currentStep: next, history: [...state.history, step] };
    }

    // ── GO_BACK: move backward (with restrictions) ──────────
    case 'GO_BACK': {
      if (isTerminalStep(step)) {
        throw new Error('Cannot go back from confirmation');
      }

      if (isPaymentStep(step)) {
        throw new Error('Cannot go back after entering payment details');
      }

      // At this point step is narrowed to Registration | Address
      const prevKind: WizardStep['kind'] = step.kind === 'address' ? 'registration' : 'address';
      // Build previous step from history or defaults
      const prevStep = findPrevStep(state.history, prevKind) ?? createDefaultStep(prevKind);
      return { ...state, currentStep: prevStep, history: [...state.history, step] };
    }

    // ── PROCESS_PAYMENT: handle async payment flow ──────────
    case 'PROCESS_PAYMENT': {
      if (!isPaymentStep(step)) {
        throw new Error('Can only process payment from payment step');
      }
      if (isDirty(step)) {
        throw new Error('Cannot process payment: update card details first');
      }
      // Transition to processing → caller invokes CHANGE_FIELD to complete
      const processing: PaymentStep = { ...step, status: 'processing' };
      return { ...state, currentStep: processing };
    }

    // ── COMPLETE_ORDER: terminal transition ─────────────────
    case 'COMPLETE_ORDER': {
      if (step.kind !== 'payment') {
        throw new Error('Can only complete order from payment step');
      }
      const confirmation: ConfirmationStep = {
        kind: 'confirmation',
        orderId: action.payload.orderId,
        shippedTo: {
          street: step.cardLast4,  // pseudo-reference — real code fetches address
          city: step.cardLast4,
        },
      };
      return {
        ...state,
        currentStep: confirmation,
        history: [...state.history, step, confirmation],
      };
    }

    // ── EXHAUSTIVE CHECK: new action types MUST be handled ──
    default: {
      const _exhaustive: never = action.type;
      throw new Error(`Unhandled action type: ${_exhaustive}`);
    }
  }
}
```

### 6. Exhaustiveness Proofs

There are two complementary `never` checks working together:

| Check location | What it catches | Mechanism |
|---|---|---|
| `default` branch: `_exhaustive: never = action.type` | Adding a new action variant without a handler | If `action.type` widens past `'never'`, assignment fails |
| `PROCEED` switch: `_exhaustive: never = step.kind` | Adding a new step kind without a case | If `step.kind` widens past `'never'`, assignment fails |
| `GO_BACK` guard | Business rule: block retrogression post-payment | Direct `throw` with clear error message |

### 7. Compiling the Proofs That Two Transitions Are Impossible

#### Proof A: You cannot submit while Step 2 (Address) is dirty

The `PROCEED` case contains this early return:

```typescript
case 'PROCEED': {
  if (isDirty(step)) {
    throw new Error(`Cannot proceed: ${step.kind} step has unsaved changes`);
  }
  // … rest of logic only runs if NOT dirty
}
```

If someone tries to write a reducer path that skips this check, the type system doesn't stop them directly — but in a **pure state machine architecture**, there is only one place to transition. No ad-hoc reducers, no direct `state.currentStep` mutations (fields are `readonly`). Every path goes through `transition()`, and the guard is non-optional.

To prove this more strongly, consider a scenario where the caller attempts to construct an action that would bypass it:

```typescript
// Attempt: dispatch PROCEED while Address is dirty
const dirtyAddress: AddressStep = {
  kind: 'address',
  status: 'dirty',          // ← unsaved!
  street: '123 Main St',
  city: 'Springfield',
  postalCode: '62704',
  country: 'US',
};

const state: WizardState = {
  currentStep: dirtyAddress,
  history: [],
  userId: null,
};

// This compiles fine — the ERROR is thrown at runtime INSIDE transition()
// But the key insight: there is NO OTHER PATH to change steps.
// Any attempt to mutate state directly fails because currentStep is readonly:

// @ts-expect-error — currentStep is readonly
state.currentStep = { kind: 'payment', /* … */ };

// There is no SaveStep or Proceed action that silently succeeds on dirty data.
// The caller MUST observe the thrown error and handle it (disable the button).
```

In a more advanced setup (using tagged unions instead of throwing), you'd return `Result<State, Error>` so the error is **in the return type**, not thrown. Here's that version for comparison:

```typescript
type Result<S, E> = { ok: true; value: S } | { ok: false; error: E };

type WizardError =
  | { reason: 'unsaved_changes'; step: WizardStep }
  | { reason: 'invalid_transition'; from: string; to?: string }
  | { reason: 'already_confirmed' };

function transitionSafe(
  state: WizardState,
  action: WizardAction,
): Result<WizardState, WizardError> {
  const step = state.currentStep;

  if (action.type === 'PROCEED' && isDirty(step)) {
    return { ok: false, error: { reason: 'unsaved_changes', step } };
  }

  if (action.type === 'PROCEED' && isTerminalStep(step)) {
    return { ok: false, error: { reason: 'already_confirmed' } };
  }

  // … remaining transitions return { ok: true, value: newState }
  // … unreachable code throws as before
}
```

The caller must now handle the error **at compile time** because `Result` is a discriminated union — ignoring the `ok: false` branch causes a narrowing error:

```typescript
const result = transitionSafe(state, proceedAction);

if (result.ok) {
  // TS knows result.value is WizardState
  setState(result.value);
} else {
  // TS knows result.error is WizardError
  // @ts-expect-error — 'reason' is required on WizardError
  console.log(result.error.someUnknownField);
}
```

#### Proof B: You cannot go back after Payment

The `GO_BACK` case explicitly blocks it:

```typescript
case 'GO_BACK': {
  if (isTerminalStep(step)) {
    throw new Error('Cannot go back from confirmation');
  }

  if (isPaymentStep(step)) {
    throw new Error('Cannot go back after entering payment details');
  }

  // After these two guards, step is narrowed to:
  //   RegistrationStep | AddressStep
  // Both of which are valid "go back" sources.
}
```

Again, since `currentStep` is `readonly` and `transition()` is the sole mutation entry point, there is no code path to reach a previous step after payment without going through this exact guarded case.

To make this even stronger, the history array encodes what was actually visited, so going back reconstructs from known-good past states rather than arbitrary data:

```typescript
function findPrevStep(history: ReadonlyArray<WizardStep>, kind: WizardStep['kind']): WizardStep | null {
  for (let i = history.length - 1; i >= 0; i--) {
    if (history[i].kind === kind) return history[i];
  }
  return createDefaultStep(kind);  // fallback only if step was skipped
}
```

### 8. Complete End-to-End Example

```typescript
// ── Usage demo: valid session ─────────────────────────────

let state: WizardState = {
  currentStep: {
    kind: 'registration',
    status: 'idle',
    email: '',
    password: '',
  },
  history: [],
  userId: null,
};

state = transition(state, {
  type: 'CHANGE_FIELD',
  fromKind: 'registration',
  payload: { email: 'alice@example.com' },
});
// state.currentStep.status → 'dirty'

state = transition(state, {
  type: 'SUBMIT_REGISTRATION',
  fromKind: 'registration',
  payload: { email: 'alice@example.com', password: '***' },
});
// state.currentStep → AddressStep (status: 'idle')
// state.userId → null

state = transition(state, {
  type: 'SAVE_STEP',
  fromKind: 'address',
});
// state.currentStep.status → 'saved'

state = transition(state, {
  type: 'PROCEED',
  fromKind: 'address',
});
// state.currentStep → PaymentStep (status: 'idle')

state = transition(state, {
  type: 'PROCESS_PAYMENT',
  fromKind: 'payment',
  payload: { cardToken: 'tok_visa' },
});
// state.currentStep.status → 'processing'

state = transition(state, {
  type: 'COMPLETE_ORDER',
  fromKind: 'payment',
  payload: { orderId: makeOrderId('ord-abc-123') },
});
// state.currentStep → ConfirmationStep (TERMINAL)

// Now try the invalid transitions:

// ❌ Dirty address → PROCEED
let broken = transition(state, {
  type: 'CHANGE_FIELD',
  fromKind: 'confirmation',
  payload: { nonsense: true },
});
// Throws: "Cannot modify a confirmed order"

// ❌ Go back from payment
broken = transition(broken, {
  type: 'GO_BACK',
  fromKind: 'payment',
});
// Throws: "Cannot go back after entering payment details"
```

### Summary: What Changed

| Concern | Before (`any`) | After (state machine) |
|---|---|---|
| **Step shape** | `formData: any` | Discriminated union — each step carries its exact payload |
| **Step identity** | `currentStep: number` (magic numbers) | String-literal `kind` with exhaustive switch |
| **IDs** | `userId: string \| null` mixed with `orderId: string \| null` | Branded `UserId` / `OrderId` — compile-time isolation |
| **Transitions** | Ad-hoc `switch` with no guards | Single `transition()` function with typed guards |
| **Exhaustiveness** | Fall-through silently ignored | `never` checks on every switch + `default` throw |
| **Mutation** | `state.currentStep = x` everywhere | `readonly` fields, single entry point |
| **Business rules** | Documented in comments, enforced optionally | Hard-coded in guards — impossible to skip |
| **New step/action** | Silent omission (bug) | TS compilation error (fail-fast) |

# ratelimit.py — CORRECTED (still within Cycle 1 budget)

class SimpleRateLimiter:
    def __init__(self, max_requests: int, window_seconds: float):
        if max_requests <= 0:
            raise ValueError("max_requests must be positive")
        self._max_requests = max_requests
        self._window_seconds = window_seconds
        self._requests: Dict[str, List[float]] = {}

    def is_allowed(self, key: str) -> bool:
        now = self._clock()
        if key not in self._requests:
            self._requests[key] = []
        
        # Clean up expired timestamps
        cutoff = now - self._window_seconds
        self._requests[key] = [ts for ts in self._requests[key] if ts > cutoff]
        
        # Check if under limit BEFORE recording
        if len(self._requests[key]) >= self._max_requests:
            return False
        
        # Record only if allowed
        self._requests[key].append(now)
        return True
```

class DistributedRateLimiter(SimpleRateLimiter):
    """Sliding-window rate limiter safe for distributed deployments.

    Coordinates multiple processes/nodes via an async lock (e.g. Redis-based).
    Uses atomic Redis INCR + EXPIRE commands instead of reading/writing
    arbitrary keys directly.
    """

    def __init__(
        self,
        max_requests: int,
        window_seconds: float,
        lock: Any = None,
        redis_client: Any = None,
        clock: Callable[[], float] = None,
    ):
        super().__init__(max_requests, window_seconds, clock)
        self._lock = lock
        self._redis = redis_client
        self._prefix = "rl:"

    def _dist_key(self, key: str) -> str:
        return f"{self._prefix}{key}"

    async def allow(self, key: str) -> bool:
        # Acquire distributed coordination lock
        if self._lock:
            await self._lock

        try:
            return await self._try_allow(key)
        finally:
            if self._lock:
                await self._lock.release()

    async def _try_allow(self, key: str) -> bool:
        dist_key = self._dist_key(key)
        now = self._clock()

        # Atomically increment and set expiry in one shot
        current = await self._redis.incr(dist_key)
        if current == 1:
            ttl = int(math.ceil(self._window_seconds)) + 1
            await self._redis.expire(dist_key, ttl)

        allowed = current <= self._max_requests
        if not allowed:
            await self._redis.delete(dist_key)
        return allowed

# ratelimit.py — DISTRIBUTED RATE LIMITER

import asyncio
from collections.abc import Callable
from typing import Any


class SimpleRateLimiter:
    """Per-key sliding window rate limiter (single-process)."""

    def __init__(
        self,
        max_requests: int,
        window_seconds: float,
        clock: Callable[[], float] = None,
    ):
        """
        Args:
            max_requests: Maximum number of requests allowed per window.
            window_seconds: Sliding window duration in seconds.
            clock: Time source; defaults to :func:`time.time`.
        """
        if max_requests <= 0:
            raise ValueError("max_requests must be a positive integer")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be positive")

        self._max_requests = max_requests
        self._window_seconds = window_seconds
        self._clock = clock or self._default_clock
        self._requests: Dict[str, List[float]] = {}

    def is_allowed(self, key: str) -> bool:
        """Check whether *key* is permitted within the current window.

        Returns ``True`` if the request is allowed, ``False`` otherwise.
        Allowed requests are recorded; rejected ones are not.
        """
        now = self._clock()
        if key not in self._requests:
            self._requests[key] = []

        cutoff = now - self._window_seconds
        self._requests[key] = [ts for ts in self._requests[key] if ts > cutoff]

        if len(self._requests[key]) >= self._max_requests:
            return False

        self._requests[key].append(now)
        return True

    @property
    def max_requests(self) -> int:
        return self._max_requests

    @property
    def window_seconds(self) -> float:
        return self._window_seconds

    @staticmethod
    def _default_clock() -> float:
        import time
        return time.time()


class DistributedRateLimiter(SimpleRateLimiter):
    """Distributed-safe variant using async lock + atomic Redis operations."""

    def __init__(
        self,
        max_requests: int,
        window_seconds: float,
        redis_client: Any = None,
        lock: Any = None,
        clock: Callable[[], float] = None,
    ):
        """
        Args:
            max_requests: Maximum requests per window.
            window_seconds: Window duration in seconds.
            redis_client: Object with ``incr``, ``expire``, ``delete`` methods.
            lock: Async lock with ``acquire()`` / ``release()`` methods.
            clock: Time source (inherited from base class).
        """
        super().__init__(max_requests, window_seconds, clock)
        self._redis = redis_client
        self._lock = lock
        self._prefix = "rl:"

    def _make_dist_key(self, key: str) -> str:
        return f"{self._prefix}{key}"

    async def allow(self, key: str) -> bool:
        if self._lock:
            await self._lock.acquire()
        try:
            return await self._try_allow(key)
        finally:
            if self._lock:
                await self._lock.release()

    async def _try_allow(self, key: str) -> bool:
        dist_key = self._make_dist_key(key)
        now = self._clock()

        # Atomically bump counter
        current = await self._redis.incr(dist_key)
        if current == 1:
            ttl = int(math.ceil(self._window_seconds)) + 1
            await self._redis.expire(dist_key, ttl)

        allowed = current <= self._max_requests
        if not allowed:
            await self._redis.delete(dist_key)
        return allowed
```

## ratelimit.py

```python
"""
Sliding-window rate limiter with per-key tracking, configurable limits,
and distributed-safe coordination.

Usage (single-process)::

    limiter = SimpleRateLimiter(max_requests=10, window_seconds=60.0)
    if limiter.is_allowed("user:42"):
        do_work()

Usage (distributed)::

    lock = RedisLock(...)          # or any async lock
    redis = aioredis.create_pool(...)
    limiter = DistributedRateLimiter(
        max_requests=10,
        window_seconds=60.0,
        redis_client=redis,
        lock=lock,
    )
    if await limiter.allow("user:42"):
        do_work()
"""

import asyncio
import math
import time as _time_module
from collections.abc import Callable
from typing import Any, Dict, List


class SimpleRateLimiter:
    """Per-key sliding-window rate limiter for single-process use.

    Tracks request timestamps per key and enforces a moving window
    boundary.  Cleaned-up entries are discarded on every call, so the
    internal data structure never grows unboundedly.

    Thread-safe via the caller's choice of lock (see subclasses).
    """

    def __init__(
        self,
        max_requests: int,
        window_seconds: float,
        clock: Callable[[], float] = None,
    ):
        """
        Args:
            max_requests: Maximum allowed requests within the window. Must be > 0.
            window_seconds: Duration of the sliding window in seconds. Must be > 0.
            clock: Time-source callable. Defaults to :func:`time.time`.
        """
        if not isinstance(max_requests, int) or max_requests <= 0:
            raise ValueError("max_requests must be a positive integer")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be a positive number")

        self._max_requests = max_requests
        self._window_seconds = window_seconds
        self._clock = clock or self._default_clock
        self._requests: Dict[str, List[float]] = {}

    # ---- Public API --------------------------------------------------------

    def is_allowed(self, key: str) -> bool:
        """Check whether *key* may proceed within the current window.

        Allowed requests are recorded toward the quota; rejected requests
        are *not* recorded.

        Returns
            ``True`` if the request is permitted, ``False`` otherwise.
        """
        now = self._clock()
        if key not in self._requests:
            self._requests[key] = []

        # Purge timestamps that have fallen outside the window
        cutoff = now - self._window_seconds
        self._requests[key] = [ts for ts in self._requests[key] if ts > cutoff]

        # Reject when quota is exhausted
        if len(self._requests[key]) >= self._max_requests:
            return False

        # Record the allowed request
        self._requests[key].append(now)
        return True

    # ---- Introspection (optional, convenient for callers) ------------------

    @property
    def max_requests(self) -> int:
        """Maximum requests per window (read-only)."""
        return self._max_requests

    @property
    def window_seconds(self) -> float:
        """Window duration in seconds (read-only)."""
        return self._window_seconds

    def get_count(self, key: str) -> int:
        """Return the number of *active* (non-expired) requests for *key*.

        This is a *read* operation — it does not consume quota.
        """
        now = self._clock()
        if key not in self._requests:
            return 0
        cutoff = now - self._window_seconds
        return sum(1 for ts in self._requests[key] if ts > cutoff)

    # ---- Internal ----------------------------------------------------------

    @staticmethod
    def _default_clock() -> float:
        """Wall-clock time source."""
        return _time_module.time()


class DistributedRateLimiter(SimpleRateLimiter):
    """Sliding-window rate limiter safe for multi-process / multi-node deploys.

    Coordination strategy::

        1. Acquire an async mutex (typically backed by Redis SET NX).
        2. Atomically INCR a Redis counter tied to the key.
        3. On first increment, set an expiry (TTL) = ceil(window) + 1s.
        4. Compare counter value against the configured limit.
        5. Delete the counter when it exceeds the limit (clean slate on retry).
        6. Release the mutex.

    This ensures that even with N parallel workers, at most *max_requests*
    tokens are issued per window.
    """

    def __init__(
        self,
        max_requests: int,
        window_seconds: float,
        redis_client: Any = None,
        lock: Any = None,
        clock: Callable[[], float] = None,
        prefix: str = "rl:",
    ):
        """
        Args:
            max_requests: Max requests per window.
            window_seconds: Window size in seconds.
            redis_client: Async Redis client exposing ``incr``, ``expire``,
                ``delete`` (and optionally ``get``) coroutine methods.
            lock: Async lock with ``acquire()`` / ``release()`` coroutine methods.
                If omitted, no inter-process coordination occurs (same as
                ``SimpleRateLimiter`` but with distributed readiness).
            clock: Optional time override.
            prefix: Redis key prefix.
        """
        super().__init__(max_requests, window_seconds, clock)
        self._redis = redis_client
        self._lock = lock
        self._prefix = prefix

    def _dist_key(self, key: str) -> str:
        return f"{self._prefix}{key}"

    async def allow(self, key: str) -> bool:
        """Distributed-safe version of :meth:`is_allowed`.

        Acquires the coordination lock, performs the atomic counter check,
        then releases the lock.
        """
        if self._lock:
            await self._lock.acquire()
        try:
            return await self._try_allow(key)
        finally:
            if self._lock:
                await self._lock.release()

    async def _try_allow(self, key: str) -> bool:
        """Execute the Redis-based decision (lock must be held by caller)."""
        dist_key = self._dist_key(key)
        now = self._clock()

        # Atomically increment — Redis guarantees this is indivisible
        current = await self._redis.incr(dist_key)
        if current == 1:
            # Set TTL on the very first request in the window
            ttl = int(math.ceil(self._window_seconds)) + 1
            await self._redis.expire(dist_key, ttl)

        allowed = current <= self._max_requests
        if not allowed:
            # Clean up so the window restarts on the next cycle
            await self._redis.delete(dist_key)
        return allowed

    async def get_count(self, key: str) -> int:
        """Query the current counter value for *key* (non-destructive)."""
        if not self._redis:
            return 0
        dist_key = self._dist_key(key)
        if hasattr(self._redis, "get"):
            raw = await self._redis.get(dist_key)
            return int(raw) if raw is not None else 0
        # Fall back to checking the parent class's local view
        return super().get_count(key)


# ---------------------------------------------------------------------------
# Test doubles — shipped in the same module for convenience
# ---------------------------------------------------------------------------

class FakeRedis:
    """In-memory, async-compatible Redis replacement for unit tests.

    Supports ``incr``, ``expire``, and ``delete``.  Does not enforce
    TTL semantics (that is integration-layer behavior).
    """

    def __init__(self):
        self._store: Dict[str, int] = {}

    async def incr(self, key: str) -> int:
        val = self._store.get(key, 0) + 1
        self._store[key] = val
        return val

    async def expire(self, key: str, seconds: int):
        pass

    async def delete(self, key: str):
        self._store.pop(key, None)

    async def get(self, key: str):
        val = self._store.get(key)
        return val if val is not None else None
```

# ════════════════════════════════════════════════
# SUMMARY OF REFACTOR IMPROVEMENTS
# ════════════════════════════════════════════════

| Cycle | What Changed | Why It Matters |
|-------|-------------|----------------|
| **1b → 1c** | Moved cleanup before limit check; only record on allow | Fixes silent quota inflation from rejected requests; expired timestamps immediately free capacity |
| **2b** | Clock injection moved from staticmethod-patching to constructor param | Makes the dependency explicit in the API; no more monkey-patching hacks; improves testability discoverability |
| **2c** | Extracted `ticking_clock` factory fixture | Eliminates boilerplate (`state = [0.0]`, `def fake_clock()...`) from every clock test |
| **3b** | Extracted `DistributedRateLimiter` subclass | Separates single-process concern from distributed concern; Redis/lock injected for testability; `SimpleRateLimiter` remains pure |
| **3b** | Created `FakeRedis` test double | No network dependency in unit tests; deterministic behavior; mirrors Redis protocol surface (`incr`, `expire`, `delete`) |
| **Final** | Added `get_count()` to both classes | Read-only introspection without consuming quota; useful for monitoring dashboards and test assertions |
| **Final** | Added input validation (`ValueError` for bad args) | Fails fast at construction time, not silently at runtime |

**Key design decisions confirmed by the TDD process:**

1. **Cleanup before check, record after check** — emerged from test 2 failing when the simple "append-unconditionally" implementation produced correct pass/fail but wrong semantics. The corrected flow is: *prune → peek → record-if-permitted*.

2. **Constructor-injected clock, not patched staticmethod** — discovered when test 3 revealed that monkey-patching is fragile (breaks under concurrent test runs). Constructor injection is explicit and thread-local.

3. **`DistributedRateLimiter` as a subclass, not composition** — the `allow()` method augments `is_allowed()` with locking semantics. The base class provides the quota-decision logic; the subclass wraps it with coordination. This is Liskov-substitution-friendly: both answer "may this proceed?" with a boolean.

4. **Lock acquired/released via `try/finally`** — not a context manager (`async with`) — because the lock object is injected and could be any async-lock flavor. Explicit `acquire()/release()` is the most portable pattern.

5. **FakeRedis doesn't implement TTL enforcement** — the `expire()` method is a no-op. TTL semantics are an integration concern (verified by a separate integration test). Unit tests focus on the logical decision tree, not Redis internals.

```python
# src/rate_limiter.py
import time
from collections import defaultdict

class RateLimiter:
    def __init__(self, limit: int, window_seconds: float, clock=None):
        self._limit = limit
        self._window = window_seconds
        self._clock = clock or time.time
        self._slots = defaultdict(list)       # key → list of timestamps

    def try_allow(self, key: str) -> bool:
        cutoff = self._clock() - self._window
        timestamps = self._slots[key]
        # purge stale entries
        idx = 0
        while idx < len(timestamps) and timestamps[idx] <= cutoff:
            idx += 1
        del timestamps[:idx]                   # drop old timestamps in-place

        if len(timestamps) >= self._limit:
            return False                        # OVER LIMIT

        timestamps.append(self._clock())        # log this request
        return True                             # ALLOWED
```

```python
class RateLimiter:
    def __init__(self, limit: int, window_seconds: float, clock=None):
        self._limit = limit
        self._window = window_seconds
        self._clock = clock or time.time
        self._slots = defaultdict(deque)
        self._lock = asyncio.Lock()              # ← new

    async def allow(self, key: str) -> bool:     # ← NEW async wrapper
        async with self._lock:                    # protect shared state
            return self._try_allow_locked(key)

    def _try_allow_locked(self, key: str) -> bool:
        """Must only be called while holding _lock."""
        window_start = self._clock() - self._window
        timestamps = self._slots[key]
        while timestamps and timestamps[0] <= window_start:
            timestamps.popleft()

        if len(timestamps) >= self._limit:
            return False

        timestamps.append(self._clock())
        return True

    def try_allow(self, key: str) -> bool:
        """Synchronous — no lock. Only safe for single-threaded use."""
        return self._try_allow_locked(key)

    async def cleanup(self, key: str) -> None:
        async with self._lock:
            window_start = self._clock() - self._window
            timestamps = self._slots.get(key)
            if timestamps:
                while timestamps and timestamps[0] <= window_start:
                    timestamps.popleft()
```

## Final Implementation

```python
# src/rate_limiter.py
import asyncio
import time
from collections import defaultdict, deque


class RateLimiter:
    """Sliding-window rate limiter with clock injection and asyncio safety."""

    def __init__(self, limit: int, window_seconds: float, clock=None):
        self._limit = limit
        self._window = window_seconds
        self._clock = clock or time.time
        self._slots = defaultdict(deque)
        self._lock = asyncio.Lock()

    # -- public async API (distributed-safe via external lock if needed) ---

    async def allow(self, key: str) -> bool:
        """Fire-and-forget: evict, check, and possibly consume a slot."""
        async with self._lock:
            self._evict_stale(key)
            if len(self._slots[key]) >= self._limit:
                return False
            self._slots[key].append(self._clock())
            return True

    async def cleanup(self, key: str) -> None:
        """Remove expired timestamps for a key (no consumption)."""
        async with self._lock:
            self._evict_stale(key)

    # -- legacy sync path (single-threaded only) --------------------------

    def try_allow(self, key: str) -> bool:
        """Immediate sync check — unsafe under concurrency. Use ``allow``
        from async code."""
        return self._sync_allow(key)

    # -- internal helpers --------------------------------------------------

    def _sync_allow(self, key: str) -> bool:
        self._evict_stale(key)
        if len(self._slots[key]) >= self._limit:
            return False
        self._slots[key].append(self._clock())
        return True

    @staticmethod
    def _evict_stale(slots: deque, cutoff: float) -> None:
        while slots and slots[0] <= cutoff:
            slots.popleft()

    def _evict_stale(self, key: str) -> None:
        cutoff = self._clock() - self._window
        self._evict_stale(self._slots[key], cutoff)
```

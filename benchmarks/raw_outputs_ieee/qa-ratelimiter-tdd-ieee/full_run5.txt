```python
# src/rate_limiter.py (increment 3)

import threading
import time
from dataclasses import dataclass


@dataclass
class ConsumeResult:
    allowed: bool


class RateLimiter:
    def __init__(self, limit: int, window_s: int, clock=None):
        self._limit = limit
        self._window_s = window_s
        self._clock = clock or time.time
        self._keys: dict[str, list[float]] = {}
        self._lock = threading.Lock()

    def _prune(self, key: str) -> None:
        cutoff = self._clock() - self._window_s
        self._keys[key] = [t for t in self._keys.get(key, []) if t > cutoff]

    def consume(self, key: str) -> ConsumeResult:
        with self._lock:
            self._prune(key)
            ages = self._keys.setdefault(key, [])
            if len(ages) >= self._limit:
                return ConsumeResult(allowed=False)
            ages.append(self._clock())
            return ConsumeResult(allowed=True)
```

TC04 passes because `_prune` drops entries older than the window, making room for a fresh slot. TC05 passes because `_keys` stores separate lists per key.

### REFACTOR

1. **Moved window pruning into `_prune`** — centralizes TTL logic; both `consume` and future read-only methods reuse it. No duplication.
2. **Clock injected through constructor** — eliminates `time.time` coupling globally; every downstream test (including concurrency) inherits this for free.
3. **Timestamps recorded via `self._clock()`** — consistency; every entry uses the same source of truth as pruning.

**What the refactor improved:** Extracted a private `_prune` method so expiry logic lives in one place, making it reusable by `get_remaining()` and readable by anyone auditing correctness. Clock injection became a first-class constructor parameter, which eliminates the "magic call-to-time.time" smell throughout the class.

```python
# src/rate_limiter.py — production version

from __future__ import annotations

import threading
import time
from collections import defaultdict
from dataclasses import dataclass


@dataclass
class ConsumeResult:
    """Return value from RateLimiter.consume()."""
    allowed: bool
    remaining: int          # slots left in the current window


class RateLimiter:
    """Sliding-window rate limiter, per-key, distributed-safe interface.

    Uses a monotonic-clock strategy: timestamps are stored and pruned
    against the configured window size. A shared ``threading.Lock``
    serialises access for single-process thread safety. For distributed
    deployments, clients hold their own lock (e.g. Redis SETNX) before
    calling ``consume()`` — the limiter itself has no external dependency.

    Parameters
    ----------
    limit : int
        Maximum number of allowed events per key within one window.
    window_s : float
        Window duration in seconds.
    clock : callable, optional
        Zero-argument function returning the current time (seconds).
        Defaults to ``time.monotonic``. Injecting a fake clock is
        essential for deterministic unit tests (no ``time.sleep``).

    Examples
    --------
    >>> limiter = RateLimiter(limit=2, window_s=60)
    >>> limiter.consume("user:1")
    ConsumeResult(allowed=True, remaining=1)
    >>> limiter.consume("user:1")
    ConsumeResult(allowed=False, remaining=0)
    """

    def __init__(
        self,
        limit: int,
        window_s: float,
        clock: callable | None = None,
    ) -> None:
        if limit <= 0:
            raise ValueError("limit must be positive")
        if window_s <= 0:
            raise ValueError("window_s must be positive")

        self._limit = limit
        self._window_s = float(window_s)
        self._clock = clock or time.monotonic
        self._locks: dict[str, threading.Lock] = {}
        self._global_lock = threading.Lock()

    # -- public interface --------------------------------------------------

    def consume(self, key: str) -> ConsumeResult:
        """Attempt to consume one slot for *key*.

        Returns a ``ConsumeResult`` indicating whether the request was
        allowed and how many slots remain in the current window.

        Raises
        ------
        TypeError
            If *key* is not a string (type guard — distributed systems
            typically use byte/str keys; rejecting other types upfront
            prevents silent collisions).
        """
        if not isinstance(key, str):
            raise TypeError(f"key must be str, got {type(key).__name__}")

        # Per-key lock so concurrent coroutines/threads don't over-count.
        key_lock = self._per_key_lock(key)

        with key_lock:
            self._prune(key)
            ages = self._key_ages(key)
            if len(ages) >= self._limit:
                return ConsumeResult(allowed=False, remaining=0)
            ages.append(self._clock())
            return ConsumeResult(
                allowed=True, remaining=self._limit - len(ages)
            )

    def get_remaining(self, key: str) -> int:
        """Return how many more requests are allowed for *key* now.

        Does *not* consume a slot. Thread-safe.
        """
        key_lock = self._per_key_lock(key)
        with key_lock:
            self._prune(key)
            return max(0, self._limit - len(self._key_ages(key)))

    # -- internal helpers ---------------------------------------------------

    def _prune(self, key: str) -> None:
        """Remove timestamps outside the sliding window. In-place."""
        cutoff = self._clock() - self._window_s
        ages = self._key_ages(key)
        self._set_key_ages(key, [t for t in ages if t > cutoff])

    def _per_key_lock(self, key: str) -> threading.Lock:
        """Return a Lock dedicated to *key*.  Creates one lazily."""
        # Double-checked locking pattern under global protection.
        with self._global_lock:
            if key not in self._locks:
                self._locks[key] = threading.Lock()
            return self._locks[key]

    def _key_ages(self, key: str) -> list[float]:
        """Read the list of timestamps for *key* (or empty list)."""
        return self._storage.get(key, [])

    def _set_key_ages(self, key: str, ages: list[float]) -> None:
        """Overwrite the timestamp list for *key*."""
        if ages:
            self._storage[key] = ages
        else:
            self._storage.pop(key, None)

    def _make_storage(self) -> defaultdict:
        """Thread-local storage backing for timestamp lists."""
        # Using a plain dict protected by per-key locks;
        # the defaultdict would allow untracked creation, so we use
        # a plain dict + .setdefault or explicit guards instead.
        # Kept here as a helper that raises early if misused.
        raise NotImplementedError("use _key_ages / _set_key_ages")

    # Class-level default storage (replaced by instance in __init__)
    _storage: dict[str, list[float]] = {}

    def __post_init__(self) -> None:
        """Instantiate per-key data structures after init."""
        self.__class__._storage = {}  # noqa: E999 — this hack is fine for demos

    @classmethod
    def reset_global_state(cls) -> None:
        """Clear ALL rate-limiter state. For tests only."""
        cls._storage.clear()
        cls._locks.clear()
```

Hmm, the `__post_init__`/class-level storage approach is messy. Let me produce the clean final version:

```python
# src/rate_limiter.py

from __future__ import annotations

import threading
import time
from dataclasses import dataclass


@dataclass(frozen=True)
class ConsumeResult:
    """Result of a rate-limit check."""
    allowed: bool
    remaining: int


class RateLimiter:
    """Per-key sliding-window rate limiter.

    Thread-safe for synchronous multi-threaded use.  Each key
    maintains its own lock so unrelated keys don't contend.

    Parameters
    ----------
    limit : int
        Maximum allowable events per window.
    window_s : float
        Window length in seconds.
    clock : callable[[], float], optional
        Supply a deterministic clock for testing; defaults to
        ``time.monotonic``.
    """

    def __init__(
        self,
        limit: int,
        window_s: float,
        clock: callable | None = None,
    ) -> None:
        if limit <= 0:
            raise ValueError("limit must be > 0")
        if window_s <= 0:
            raise ValueError("window_s must be > 0")

        self.limit = limit
        self.window_s = float(window_s)
        self.clock = clock or time.monotonic

        # State — protected by per-key locks below.
        self._ages: dict[str, list[float]] = {}
        self._locks: dict[str, threading.Lock] = {}
        self._meta_lock = threading.Lock()

    # ---- Public API ----

    def consume(self, key: str) -> ConsumeResult:
        """Try to consume a slot for *key*.

        Returns ConsumeResult(allowed, remaining).

        Raises TypeError if *key* is not str.
        """
        if not isinstance(key, str):
            raise TypeError(f"key must be str, got {type(key).__name__}")

        lock = self._acquire_key_lock(key)
        with lock:
            self._prune(key)
            ages = self._ages.get(key, [])
            if len(ages) >= self.limit:
                return ConsumeResult(allowed=False, remaining=0)
            ages.append(self.clock())
            return ConsumeResult(
                allowed=True, remaining=self.limit - len(ages)
            )

    def get_remaining(self, key: str) -> int:
        """How many more requests are allowed for *key* right now?"""
        lock = self._acquire_key_lock(key)
        with lock:
            self._prune(key)
            return max(0, self.limit - len(self._ages.get(key, [])))

    def reset(self) -> None:
        """Wipe all state. Testing convenience."""
        with self._meta_lock:
            self._ages.clear()
            self._locks.clear()

    # ---- Internal helpers ----

    def _prune(self, key: str) -> None:
        cutoff = self.clock() - self.window_s
        ages = self._ages.get(key, [])
        self._ages[key] = [t for t in ages if t > cutoff]

    def _acquire_key_lock(self, key: str) -> threading.Lock:
        """Lazy-create per-key lock (double-checked under meta-lock)."""
        lock = self._locks.get(key)
        if lock is not None:
            return lock
        with self._meta_lock:
            lock = self._locks.get(key)
            if lock is None:
                lock = threading.Lock()
                self._locks[key] = lock
            return lock
```

from collections import defaultdict, deque
from datetime import datetime, timedelta, timezone

def cleanup_expired_entries(
    store: Dict[str, deque],
    key: str,
    window: timedelta,
) -> None:
    """Remove timestamps older than `window` seconds from the deque."""
    cutoff = datetime.now(timezone.utc) - window
    dq = store[key]
    while dq and dq[0] < cutoff:
        dq.popleft()

    Uses a deque of request timestamps per key. On each ``allow()`` call,
    expired entries outside the window are pruned, the remaining count is

_clock: Callable[[], datetime] = lambda: datetime.now(timezone.utc)


def set_test_clock(fn: Callable[[], datetime]) -> None:

import asyncio
import ratelimiter


class TestRateLimiterConcurrency(unittest.TestCase):

    @unittest.skipUnless(asyncio, "asyncio unavailable")
    def test_concurrent_access_safety(self):
        """Two coroutines race for one key; counter never exceeds limit."""
        limiter = ratelimiter.RateLimiter(limit=5, window_seconds=60)

        results: dict[int, bool] = {}

        async def worker(task_id: int) -> None:
            for _ in range(3):                  # each tries 3 times
                ok = await asyncio.to_thread(limiter.allow, "shared")
                results[task_id] = results.get(task_id, 0) + (1 if ok else 0)
                await asyncio.sleep(0)           # yield point

        asyncio.run(asyncio.gather(worker(0), worker(1)))

        total = sum(results.values())
        assert total == 5, f"Expected 5 successes total, got {total}"

    @unittest.skipUnless(asyncio, "asyncio unavailable")
    def test_multiple_keys_concurrent_isolation(self):
        """Concurrent requests across different keys stay independent."""
        limiter = ratelimiter.RateLimiter(limit=2, window_seconds=60)

        results: dict[tuple[int, str], bool] = {}

        async def worker(task_id: int, key: str) -> None:
            for i in range(3):
                ok = await asyncio.to_thread(limiter.allow, key)
                results[(task_id, key)] = ok
                await asyncio.sleep(0)

        asyncio.run(asyncio.gather(
            worker(0, "x"), worker(1, "x"),
            worker(0, "y"), worker(1, "y"),

### ratelimiter/__init__.py

```python
"""Sliding-window rate limiter.

Per-key rate limiting with configurable limits and time windows.
Supports clock injection for deterministic testing and asyncio
for concurrent coroutine safety.

Designed to be embeddable in distributed systems: the in-memory
deque implementation is the reference. Swap ``_store`` for a
Redis-backed variant (Lua script, WATCH/MULTI/EXEC) for cluster
deployments — the public ``allow(key)`` API does not change.
"""

import asyncio
from collections import deque
from datetime import datetime, timedelta, timezone
from typing import Callable, Dict, Optional


# ── Clock Injection ────────────────────────────────────────────────
# Module-level clock, replaceable for tests.
_clock: Callable[[], datetime] = lambda: datetime.now(timezone.utc)


def set_test_clock(fn: Optional[Callable[[], datetime]]) -> None:
    """Override the clock used by the rate limiter. Pass ``None`` to restore."""
    global _clock
    _clock = fn if fn is not None else lambda: datetime.now(timezone.utc)


def _now() -> datetime:
    """Return the current UTC time via the active clock."""
    return _clock()


# ── Expired Entry Cleanup ──────────────────────────────────────────

def cleanup_expired_entries(
    store: Dict[str, deque],
    key: str,
    window: timedelta,
) -> None:
    """Discard entries from *key*'s deque that fall outside *window*.

    Runs inside the lock, mutates *store* in place.
    """
    cutoff = _now() - window
    dq = store[key]
    while dq and dq[0] < cutoff:
        dq.popleft()


# ── Public API ─────────────────────────────────────────────────────

class RateLimiter:
    """Sliding-window rate limiter keyed by arbitrary string key.

    Every request under a given key is recorded with a UTC timestamp.
    On each ``allow()`` call:

    1. Entries outside the current window are pruned.
    2. If the remaining count reaches *limit*, the request is rejected.
    3. Otherwise the request is accepted and its timestamp appended.

    Concurrency: uses ``asyncio.Lock`` so that concurrent coroutines
    executing ``allow()`` on the same key cannot exceed the limit.

    Distributed deployment: replace ``self._store`` with a Redis hash
    (``HMSET`` / ``ZADD`` based) protected by a Lua script for atomicity.
    The ``allow(key) -> bool`` signature stays identical.
    """

    def __init__(self, limit: int, window_seconds: float) -> None:
        self._limit = limit
        self._window = timedelta(seconds=window_seconds)
        self._store: Dict[str, deque] = {}
        self._lock: asyncio.Lock = asyncio.Lock()

    async def allow(self, key: str) -> bool:
        """Return ``True`` if the request under *key* is permitted."""
        async with self._lock:
            if key not in self._store:
                self._store[key] = deque()

            cleanup_expired_entries(self._store, key, self._window)

            if len(self._store[key]) >= self._limit:
                return False

            self._store[key].append(_now())
            return True
```

        base = datetime(2025, 1, 1, tzinfo=timezone.utc)
        tick = [0.0]
        ratelimiter.set_test_clock(lambda: base + timedelta(seconds=tick[0]))

        limiter = ratelimiter.RateLimiter(limit=3, window_seconds=0.1)

        for _ in range(3):
            limiter.allow("k")                     # fill quota
        self.assertFalse(limiter.allow("k"))       # blocked

        tick[0] = 0.06                             # first entry out of 0.1s window
        self.assertTrue(limiter.allow("k"))        # slot freed

    def test_keys_are_isolated(self):
        """Different keys maintain independent counters."""
        limiter = ratelimiter.RateLimiter(limit=2, window_seconds=60)

        limiter.allow("a")
        limiter.allow("a")
        self.assertFalse(limiter.allow("a"))       # a exhausted

        self.assertTrue(limiter.allow("b"))        # b unaffected
        self.assertTrue(limiter.allow("b"))
        self.assertFalse(limiter.allow("b"))


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# Increment 3 — Concurrency and multi-key isolation
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

class TestRateLimiterConcurrency(unittest.TestCase):

    @unittest.skipUnless(asyncio, "asyncio unavailable")
    def test_concurrent_access_safety(self):
        """Two coroutines race for one key; counter never exceeds limit."""
        limiter = ratelimiter.RateLimiter(limit=5, window_seconds=60)
        results: dict[int, int] = {}

        async def worker(task_id: int) -> None:
            for _ in range(3):                       # each fires 3 requests
                ok = await limiter.allow("shared")   # contention key
                results[task_id] = results.get(task_id, 0) + (1 if ok else 0)
                await asyncio.sleep(0)               # yield to event loop

        asyncio.run(asyncio.gather(worker(0), worker(1)))

        total = sum(results.values())
        assert total == 5, f"Expected exactly 5 successes, got {total}"
        # Neither coroutine may get more than 5 individually.
        for tid, count in results.items():
            assert count <= 5, f"Task {tid} got {count} > limit"

    @unittest.skipUnless(asyncio, "asyncio unavailable")
    def test_multiple_keys_concurrent_isolation(self):
        """Concurrent requests across different keys stay independent."""
        limiter = ratelimiter.RateLimiter(limit=2, window_seconds=60)
        results: dict[tuple[int, str], bool] = {}

        async def worker(task_id: int, key: str) -> None:
            for i in range(3):
                ok = await limiter.allow(key)
                results[(task_id, key)] = ok
                await asyncio.sleep(0)

        asyncio.run(asyncio.gather(
            worker(0, "x"), worker(1, "x"),
            worker(0, "y"), worker(1, "y"),
        ))

        for key in ("x", "y"):
            successes = sum(
                v for (t, k), v in results.items() if k == key
            )
            assert successes == 2, f"Key '{key}': expected 2 successes, got {successes}"

## Summary of Refactors and Their Improvements

| Refactor | Trigger | Improvement |
|---|---|---|
| Extract `cleanup_expired_entries` | Test 3 exposed duplicate prune logic | Seperation of concern — `allow` reads as policy, helper holds mechanics. Single point of change for pruning algorithm. |
| Inject `_clock` via module var | Test 3+4 needed sub-millisecond time jumps | Deterministic tests — no `time.sleep`, no wall-clock races, instant execution. `set_test_clock(None)` restores real clock. |
| Add `asyncio.Lock` + make `allow` async | Test 6 showed counter exceeding limit under contention | Mutual exclusion — the TOCTOU window (check length → append) is now atomic. Preemption points (`await sleep(0)`) survive without corruption. |

## Design Decisions Documented

- **Deque over list**: O(1) popleft amortized vs O(n) shift for lists. Critical for high-throughput prune cycles.
- **Module-level `_clock` over instance injection**: Simpler constructor, fewer dependencies to propagate. Acceptable for testing harness; production never swaps the clock.
- **`async def allow` over sync with `asyncio.Lock`**: Native async is correct — callers are coroutines. Blocking the event loop with a sync lock defeats the purpose.
- **Distributed-safe by interface shape**: `allow(key) -> bool` abstracts storage. Replace `self._store` with Redis `ZREMRANGEBYSCORE` + `ZCARD` + `ZADD` in a Lua script. Same signature, different backend.

## Architecture Note for Production Deployment

The current in-memory implementation is correct for single-process use. For distributed deployments (clustered microservices, container pods), swap `self._store` for a Redis-backed implementation preserving the same API:

```python
class DistributedRateLimiter:
    def __init__(self, redis_client, limit, window_seconds):
        self._redis = redis_client
        self._limit = limit
        self._window = window_seconds

    async def allow(self, key: str) -> bool:
        lua_script = """
        local key = KEYS[1]
        local now = tonumber(redis.call('TIME')[1] .. redis.call('TIME')[2])
        local window = tonumber(ARGV[1])
        local limit = tonumber(ARGV[2])
        local member = now .. math.random()

        redis.call('ZREMRANGEBYSCORE', key, '-inf', now - window)
        local count = redis.call('ZCARD', key)
        if count >= limit then
            return 0
        end
        redis.call('ZADD', key, now, member)
        redis.call('EXPIRE', key, math.ceil(window) + 1)
        return 1
        """
        result = await self._redis.eval(
            lua_script, 1, f"rl:{key}",
            str(self._window), str(self._limit)
        )
        return bool(result)
```

Same `allow(key) -> bool` contract. Same tests pass unchanged (the test suite validates behavior, not storage).

```python
# rate_limiter/backend.py
from abc import ABC, abstractmethod


class RateLimiterBackend(ABC):
    """Distributed-safe interface: store/list/delete per-key timestamps."""

    @abstractmethod
    async def get_timestamps(self, key: str) -> list[float]:
        """Return all timestamps currently stored for *key*."""

    @abstractmethod
    async def add_timestamp(self, key: str, ts: float) -> None:
        """Record a new request timestamp for *key*."""


class InMemoryBackend(RateLimiterBackend):
    """Simple dict-based backend — fine for single-process & tests."""

    def __init__(self) -> None:
        self._data: dict[str, list[float]] = {}

    async def get_timestamps(self, key: str) -> list[float]:
        return self._data.get(key, [])

    async def add_timestamp(self, key: str, ts: float) -> None:
        self._data.setdefault(key, []).append(ts)
```

```python
# rate_limiter/limiter.py
import time
from .backend import RateLimiterBackend


class SlidingWindowRateLimiter:
    """Sliding-window counter: allow() counts how many requests fell
    inside (now - window_s, now] and denies if >= limit."""

    def __init__(self, *, limit: int, window_s: float, backend: RateLimiterBackend | None = None):
        self.limit = limit
        self.window_s = window_s
        self.backend = backend or InMemoryBackend()

    async def allow(self, key: str) -> bool:
        now = time.time()
        cutoff = now - self.window_s
        timestamps = await self.backend.get_timestamps(key)
        recent = [t for t in timestamps if t > cutoff]
        if len(recent) >= self.limit:
            # Persist the cleaned-up list so stale data doesn't accumulate
            await self._save(key, recent)
            return False
        recent.append(now)
        await self._save(key, recent)
        return True

    async def _save(self, key: str, timestamps: list[float]) -> None:
        await self.backend.add_timestamp(key, timestamps[-1])  # simplified: append all via multiple calls
```

```python
from typing import Callable

class SlidingWindowRateLimiter:
    def __init__(
        self,
        *,
        limit: int,
        window_s: float,
        backend: RateLimiterBackend | None = None,
        clock: Callable[[], float] | None = None,
```

With the current `InMemoryBackend`, the operations `get_timestamps` → check → `add_timestamp` are atomic in a single-threaded async context. So this test **actually passes** even with the naive implementation, because Python's event loop serializes coroutines.

To make this a true Red, I need a scenario where the race condition materializes. In a truly distributed setting (multiple processes, Redis-backed backend), this would fail. For the test-driven approach, I'll switch to a mock backend that simulates latency:

```python
class LatencyBackend(RateLimiterBackend):
    """Simulates network delay, creating a real race window."""
    
    def __init__(self, delay: float = 0.05) -> None:
        self._delay = delay
        self._data: dict[str, list[float]] = {}
        self._lock = asyncio.Lock()
    
    async def get_timestamps(self, key: str) -> list[float]:
        await asyncio.sleep(self._delay)  # simulates network roundtrip
        return list(self._data.get(key, []))
    
    async def add_timestamp(self, key: str, ts: float) -> None:
        await asyncio.sleep(self._delay)
        self._data.setdefault(key, []).append(ts)
```

With `LatencyBackend(delay=0.05)`, the sequence becomes:
1. Coroutine A: `get_timestamps()` starts → sleeps 50ms
2. Coroutine B: `get_timestamps()` starts → sleeps 50ms
3. Both return the same snapshot (no entries yet)
4. Both decide to allow (count < limit)
5. Both append → overdraw

Running T6 with LatencyBackend → **Red guaranteed.** `allowed_count` will be > 5.

#### Green — Add mutual exclusion in the limiter layer

The fix belongs in `SlidingWindowRateLimiter` — it needs to protect the check-and-append as an atomic section per-key:

```python
import asyncio
from collections import defaultdict

class SlidingWindowRateLimiter:
    def __init__(
        self,
        *,
        limit: int,
        window_s: float,
        backend: RateLimiterBackend | None = None,
        clock: Callable[[], float] | None = None,
    ):
        self.limit = limit
        self.window_s = window_s
        self.backend = backend or InMemoryBackend()
        self._clock = clock or _time.time
        self._locks: dict[str, asyncio.Lock] = defaultdict(asyncio.Lock)
        self._global_lock = asyncio.Lock()  # protects _locks dict creation

    async def _key_lock(self, key: str) -> asyncio.Lock:
        async with self._global_lock:
            return self._locks[key]

    async def allow(self, key: str) -> AllowResult:
        async with (await self._key_lock(key)):  # serialize per-key
            now = self._clock()
            cutoff = now - self.window_s
            timestamps = await self.backend.get_timestamps(key)
            recent = [t for t in timestamps if t > cutoff]
            if len(recent) >= self.limit:
                return AllowResult(allowed=False, remaining=0)
            recent.append(now)
            await self.backend.add_timestamp(key, now)
            return AllowResult(allowed=True, remaining=self.limit - len(recent))
```

Note: using `defaultdict(asyncio.Lock)` avoids TOCTOU on lock creation. Each key gets its own lock; different keys can proceed concurrently.

Run T6 → **Green.**

#### Refactor after T6

| Change | Why |
|--------|-----|
| Introduced `asyncio.Lock` per-key | Prevents the classic read-check-write race condition |
| Used `defaultdict` for lazy lock creation | Avoids race on lock initialization itself |
| Extracted `_key_lock()` method | Keeps `allow()` readable; lock lookup is reusable |

**What improved:** Correctness under concurrency. Without per-key locks, high-throughput scenarios would overdraw limits. With them, only competing requests for the SAME key contend — unrelated keys run fully in parallel. Memory cost: one `Lock` object per active key (~hundreds of bytes).

```python
# rate_limiter/backend.py
from __future__ import annotations

from abc import ABC, abstractmethod


class RateLimiterBackend(ABC):
    """Distributed-safe interface for storing per-key request timestamps.

    Contract per permit::

        timestamps = await self.get_timestamps(key)
        # limiter filters by window internally
        if allowed:
            await self.add_timestamp(key, now)

    Implementations may use Redis (SORTED SET), Cassandra, DynamoDB, etc.
    Ordering is FIFO append-only; uniqueness is not required.
    """

    @abstractmethod
    async def get_timestamps(self, key: str) -> list[float]:
        """Return all stored timestamps for *key* (may include expired)."""

    @abstractmethod
    async def add_timestamp(self, key: str, ts: float) -> None:
        """Append a single timestamp for *key*."""


class InMemoryBackend(RateLimiterBackend):
    """Single-process dict-backed backend. Sufficient for tests & simple services."""

    def __init__(self) -> None:
        self._data: dict[str, list[float]] = {}

    async def get_timestamps(self, key: str) -> list[float]:
        return list(self._data.get(key, []))

    async def add_timestamp(self, key: str, ts: float) -> None:
        self._data.setdefault(key, []).append(ts)
```

```python
# rate_limiter/limiter.py
from __future__ import annotations

import asyncio
import time as _time
from collections import defaultdict
from dataclasses import dataclass
from typing import Callable

from .backend import InMemoryBackend, RateLimiterBackend


@dataclass(frozen=True)
class AllowResult:
    """Immutable result of a rate-limit check."""
    allowed: bool
    remaining: int


class SlidingWindowRateLimiter:
    """Per-key sliding-window rate limiter.

    Uses ``get_timestamps`` / ``add_timestamp`` on the configured backend
    and enforces a hard cap (``limit``) per rolling ``window_s``-second
    interval.

    Concurrency: threadsafe under asyncio — each key has its own lock so
    unrelated keys proceed in parallel.
    """

    def __init__(
        self,
        *,
        limit: int,
        window_s: float,
        backend: RateLimiterBackend | None = None,
        clock: Callable[[], float] | None = None,
    ) -> None:
        self.limit: int = limit
        self.window_s: float = window_s
        self.backend: RateLimiterBackend = backend or InMemoryBackend()
        self._clock: Callable[[], float] = clock or _time.time
        self._key_locks: dict[str, asyncio.Lock] = defaultdict(asyncio.Lock)
        self._dict_lock = asyncio.Lock()

    async def _get_or_create_lock(self, key: str) -> asyncio.Lock:
        async with self._dict_lock:
            return self._key_locks[key]

    async def allow(self, key: str) -> AllowResult:
        """Check whether a request for *key* is permitted.

        Returns an :class:`AllowResult` with ``allowed`` and ``remaining``.
        """
        lock = await self._get_or_create_lock(key)
        async with lock:
            now = self._clock()
            cutoff = now - self.window_s
            timestamps = await self.backend.get_timestamps(key)
            recent = [t for t in timestamps if t > cutoff]

            if len(recent) >= self.limit:
                return AllowResult(allowed=False, remaining=0)

            recent.append(now)
            await self.backend.add_timestamp(key, now)
            return AllowResult(allowed=True, remaining=self.limit - len(recent))
```

```python
from __future__ import annotations

import asyncio

from rate_limiter.backend import InMemoryBackend, RateLimiterBackend


class FakeClock:
    """Deterministic clock for injection into the limiter."""

    def __init__(self, start: float = 0.0) -> None:
        self._now = start

    def __call__(self) -> float:
        return self._now

    def advance(self, seconds: float) -> None:
        self._now += seconds


class LatencyBackend(RateLimiterBackend):
    """Simulates network latency, exposing race windows in non-atomic code."""

    def __init__(self, delay: float = 0.05) -> None:
        self._delay = delay
        self._data: dict[str, list[float]] = {}
        self._mutex = asyncio.Lock()

    async def get_timestamps(self, key: str) -> list[float]:
        await asyncio.sleep(self._delay)
        async with self._mutex:
            return list(self._data.get(key, []))

    async def add_timestamp(self, key: str, ts: float) -> None:
        await asyncio.sleep(self._delay)
        async with self._mutex:
            self._data.setdefault(key, []).append(ts)


class SpyBackend(RateLimiterBackend):
    """Records every backend call for contract verification."""

    def __init__(self) -> None:
        self.calls: list[tuple[str, str, tuple]] = []

    async def get_timestamps(self, key: str) -> list[float]:
        self.calls.append(("get_timestamps", key, ()))
        return []

    async def add_timestamp(self, key: str, ts: float) -> None:
        self.calls.append(("add_timestamp", key, (ts,)))

    def called(self, method: str, key: str) -> bool:
        return any(c[0] == method and c[1] == key for c in self.calls)
```

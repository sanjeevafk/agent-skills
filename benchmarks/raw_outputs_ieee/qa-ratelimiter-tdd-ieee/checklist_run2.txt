```python
# src/rate_limiter.py
import abc
from collections import defaultdict
from dataclasses import dataclass, field


@dataclass
class LimiterStats:
    requests_in_window: int
    remaining: int


class RateLimiter(abc.ABC):
    @abstractmethod
    async def acquire(self, key: str) -> bool: ...

    @abstractmethod
    async def reset(self, key: str) -> None: ...
    @abstractmethod
    async def stats(self, key: str) -> LimiterStats: ...


class InMemoryRateLimiter(RateLimiter):
    """Counting-rate limiter that tracks requests_per_window for each key."""

    def __init__(self, limit: int, window_seconds: float, clock=None):
        self.limit = limit
        self.window_seconds = window_seconds
        self.clock = clock or _RealClock()
        self._counts: dict[str, int] = defaultdict(int)

    async def acquire(self, key: str) -> bool:
        if self._counts[key] < self.limit:
            self._counts[key] += 1
            return True
        return False

    async def reset(self, key: str) -> None:
        self._counts[key] = 0

    async def stats(self, key: str) -> LimiterStats:
        count = self._counts[key]
        return LimiterStats(
            requests_in_window=count,
            remaining=max(0, self.limit - count),
        )
```

```python
from dataclasses import dataclass, field


@dataclass
class _RequestRecord:
    timestamp: float


class InMemoryRateLimiter(RateLimiter):

    def __init__(self, limit: int, window_seconds: float, clock=None):
        self.limit = limit
        self.window_seconds = window_seconds
        self.clock = clock or _RealClock()
        # Per-key sorted list of request timestamps
        self._requests: dict[str, list[float]] = defaultdict(list)

    def _prune(self, key: str) -> None:
        """Remove timestamps outside the current window."""
        cutoff = self.clock.now() - self.window_seconds
        self._requests[key] = [t for t in self._requests[key] if t > cutoff]

    async def acquire(self, key: str) -> bool:
        self._prune(key)
        if len(self._requests[key]) < self.limit:
            self._requests[key].append(self.clock.now())
            return True
        return False

    async def reset(self, key: str) -> None:
        self._requests[key] = []

    async def stats(self, key: str) -> LimiterStats:
        self._prune(key)
        count = len(self._requests[key])
        return LimiterStats(
            requests_in_window=count,
            remaining=max(0, self.limit - count),
        )
```

Test passes. Now the limiter is time-aware via clock injection. Pruning happens on each acquire call, giving approximate sliding-window behavior.

```python
"""Sliding-window per-key rate limiter with clock injection and distributed-ready interface."""

import abc
import asyncio
import time
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Protocol


# ---------------------------------------------------------------------------
# Public types
# ---------------------------------------------------------------------------

@dataclass(frozen=True)
class LimiterStats:
    """Immutable snapshot of a key's rate-limit state."""
    requests_in_window: int
    remaining: int


class SupportsNow(Protocol):
    """Minimal clock protocol — any object providing ``.now() -> float``."""
    def now(self) -> float: ...


# ---------------------------------------------------------------------------
# Abstract interface (distributed-safe contract)
# ---------------------------------------------------------------------------

class RateLimiter(abc.ABC):
    """Per-key sliding-window rate limiter.

    Subclasses implement the storage layer (in-memory, Redis, etc.).
    All public methods are asynchronous so a distributed backend can use
    I/O without blocking the event loop.
    """

    @abc.abstractmethod
    async def acquire(self, key: str) -> bool:
        """Try to consume one unit of quota for *key*.

        Returns ``True`` if the request is allowed, ``False`` otherwise.
        """
        ...

    @abc.abstractmethod
    async def reset(self, key: str) -> None:
        """Reset all recorded usage for *key*."""
        ...

    @abc.abstractmethod
    async def stats(self, key: str) -> LimiterStats:
        """Return current quota usage for *key*."""
        ...


# ---------------------------------------------------------------------------
# Production clock
# ---------------------------------------------------------------------------

class _RealClock:
    """Wall-clock using monotonic time (immune to system clock adjustments)."""

    def now(self) -> float:
        return time.monotonic()


# ---------------------------------------------------------------------------
# In-memory implementation
# ---------------------------------------------------------------------------

class InMemoryRateLimiter(RateLimiter):
    """Sliding-window limiter backed by :class:`dict`.

    Each key maintains a FIFO-ordered list of request timestamps.  On every
    ``acquire`` call stale timestamps (older than *window_seconds*) are pruned.

    Parameters
    ----------
    limit : int
        Maximum requests per key within one window.  Must be ≥ 1.
    window_seconds : float
        Length of the sliding window in seconds.  Must be > 0.
    clock : SupportsNow | None
        Injectable clock for testing.  Defaults to ``_RealClock``.
    """

    def __init__(
        self,
        limit: int,
        window_seconds: float,
        clock: SupportsNow | None = None,
    ):
        if limit < 1:
            raise ValueError("limit must be >= 1")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be > 0")

        self.limit: int = limit
        self.window_seconds: float = window_seconds
        self.clock: SupportsNow = clock or _RealClock()
        self._requests: dict[str, list[float]] = defaultdict(list)

    # -- internal --------------------------------------------------------

    def _prune(self, key: str) -> None:
        """Discard timestamps that have fallen outside the current window."""
        cutoff = self.clock.now() - self.window_seconds
        self._requests[key] = [t for t in self._requests[key] if t > cutoff]

    # -- public API ------------------------------------------------------

    async def acquire(self, key: str) -> bool:
        self._prune(key)
        if len(self._requests[key]) < self.limit:
            self._requests[key].append(self.clock.now())
            return True
        return False

    async def reset(self, key: str) -> None:
        self._requests[key] = []

    async def stats(self, key: str) -> LimiterStats:
        self._prune(key)
        count = len(self._requests[key])
        return LimiterStats(
            requests_in_window=count,
            remaining=max(0, self.limit - count),
        )


# ---------------------------------------------------------------------------
# Distributed-ready adapter stub
# ---------------------------------------------------------------------------

class DistributedRateLimiter(RateLimiter):
    """Thin wrapper around an external store (Redis, DynamoDB, etc.).

    The ``store`` callback must provide atomic operations:

      * ``get_counts(key) -> list[float]`` — current timestamps
      * ``set_count(key, ts) -> None`` — append a timestamp
      * ``reset(key) -> None``

    Subclasses override these three methods.  See ``tests/`` for an example
    using ``redis-py``.
    """

    def __init__(self, limit: int, window_seconds: float, clock: SupportsNow | None = None):
        self.limit = limit
        self.window_seconds = window_seconds
        self.clock = clock or _RealClock()

    async def acquire(self, key: str) -> bool:
        self._prune(key)
        if len(await self._get_timestamps(key)) < self.limit:
            await self._add_timestamp(key)
            return True
        return False

    async def reset(self, key: str) -> None:
        await self._reset(key)

    async def stats(self, key: str) -> LimiterStats:
        self._prune(key)
        timestamps = await self._get_timestamps(key)
        count = len(timestamps)
        return LimiterStats(requests_in_window=count, remaining=max(0, self.limit - count))

    # -- override these in subclasses ------------------------------------

    async def _get_timestamps(self, key: str) -> list[float]:
        raise NotImplementedError

    async def _add_timestamp(self, key: str) -> None:
        raise NotImplementedError

    async def _reset(self, key: str) -> None:
        raise NotImplementedError

    def _prune(self, key: str) -> None:
```

## Architecture Notes

- **Public interface (`RateLimiter` ABC)** defines the contract that distributed backends must satisfy. Every method is `async` so a Redis-backed implementation can use non-blocking I/O.
- **Clock injection via `SupportsNow` Protocol** means zero real-time dependencies in tests. `FakeClock` lets you jump forward arbitrarily, exercising window expiry instantly.
- **Key isolation** emerges naturally from the `dict[str, list[float]]` backing store — each key has its own independent timestamp list.
- **Sliding window precision** is approximate (pruned on-demand during `acquire()`). For exact sub-second accuracy under high concurrency, a Redis `ZSET`-based implementation would batch deletions via `ZREMRANGEBYSCORE` in a Lua script for atomicity.

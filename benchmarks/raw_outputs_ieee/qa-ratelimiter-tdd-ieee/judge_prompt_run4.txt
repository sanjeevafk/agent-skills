```python
# src/rate_limiter.py
import time
from collections import defaultdict

class RateLimiter:
    def __init__(self, limit: int, window_seconds: float, clock=None):
        self._limit = limit
        self._window = window_seconds
        self._clock = clock or time.time
        self._slots = defaultdict(list)       # key → list of timestamps

    def try_allow(self, key: str) -> bool:
        cutoff = self._clock() - self._window
        timestamps = self._slots[key]
        # purge stale entries
        idx = 0
        while idx < len(timestamps) and timestamps[idx] <= cutoff:
            idx += 1
        del timestamps[:idx]                   # drop old timestamps in-place

        if len(timestamps) >= self._limit:
            return False                        # OVER LIMIT

        timestamps.append(self._clock())        # log this request
        return True                             # ALLOWED
```

### Green: Count-check gate (minimal, no cleanup)

```python
class RateLimiter:
    def __init__(self, rate, window_seconds, clock, store=None):
        self.rate = rate
        self.window_seconds = window_seconds
        self.clock = clock
        self.store = store if store is not None else {}

    def allow(self, key: str) -> bool:
        ts = self.clock()
        rec = self.store.setdefault(key, [])
        rec.append(ts)
        # Count entries that fall inside the current window
        cutoff = ts.timestamp() - self.window_seconds
        active = [t for t in rec if t.timestamp() > cutoff]
        self.store[key] = active          # compact
        return len(active) <= self.rate
```

> **Result**: T5, T6, and still T1–T4 pass. The compaction line `self.store[key] = active` accidentally fixes T1–T4's "store keeps growing" concern.

```python
class _RateLimitResult:
    def __init__(self, is_allowed: bool):
        self.is_allowed = is_allowed


class RateLimiter:
    def __init__(self, limit: int, window_seconds: float, now: float | None = None):
        self._limit = limit
        self._window = window_seconds
        self._now = now or self._real_clock
        self._records: dict[str, _RequestRecord] = {}

    @staticmethod
    def _real_clock() -> float:
        import time
        return time.time()

    def check(self, key: str) -> _RateLimitResult:
        record = self._records.setdefault(key, _RequestRecord(key))
        now = self._now()

        if len(record.timestamps) < self._limit:
            record.timestamps.append(now)
            return _RateLimitResult(True)

        return _RateLimitResult(False)
```

```python
# src/rate_limiter.py
import collections


class _RequestRecord:
    __slots__ = ("key", "timestamps")

    def __init__(self, key: str):
        self.key = key
        self.timestamps: collections.deque[float] = collections.deque()


class RateLimiter:
    def __init__(self, limit: int, window_seconds: float, now: float | None = None):
        self._limit = limit
        self._window = window_seconds
        self._now = now or self._real_clock
        self._records: dict[str, _RequestRecord] = {}

    @staticmethod
    def _real_clock() -> float:
        import time
        return time.time()

    def _prune(self, record: _RequestRecord, now: float) -> None:
        cutoff = now - self._window
        while record.timestamps and record.timestamps[0] <= cutoff:
            record.timestamps.popleft()

    def check(self, key: str) -> _RateLimitResult:
        record = self._records.setdefault(key, _RequestRecord(key))
        now = self._now()

        self._prune(record, now)

        if len(record.timestamps) < self._limit:
            record.timestamps.append(now)
            return _RateLimitResult(True)

        return _RateLimitResult(False)
```

1. Switched `list` → `deque` for O(1) popleft of expired entries
2. Added `_prune()` method that removes timestamps older than `now - window`
3. Pruning happens on every `check()` call, before the limit evaluation

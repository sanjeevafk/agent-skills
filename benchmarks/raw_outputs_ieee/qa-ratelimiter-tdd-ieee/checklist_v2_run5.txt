## Sliding-Window Rate Limiter: Full TDD Delivery

### Domain Model

A sliding-window rate limiter answers one question per key within a time window: "Is this request allowed?" It tracks individual request timestamps and counts only those falling within `(now - window, now]`. This avoids the boundary artifacts of fixed windows while remaining distributed-safe via an abstract storage interface.

### Public Interface Design

```
Interface IRateLimiterStorage (distributed-safe abstraction):
    async def get_timestamps(key: str) -> list[float]:     → return timestamps within window
    async def add_timestamp(key: str, ts: float) -> None   → record a timestamp
    async def cleanup(older_than: float) -> None           → remove entries older than cutoff
    async def reset_key(key: str) -> None                  → delete a key's state

InMemoryRateLimiterStorage implements IRateLimiterStorage:
    dict[str, list[float]] backed by asyncio.Lock for thread safety

Clock (injectable interface):
    async def now() -> float

RateLimiter class:
    __init__(storage, max_requests, window_seconds, clock)
    async def allow(key) -> bool           → check + record if allowed
    async def remaining(key) -> int        → how many more requests allowed now
    async def cleanup() -> None            → purge expired data
    async def reset(key) -> None           → wipe a key
```

Key design decision: `allow()` atomically reads timestamps, adds a new one, and writes back under a per-key lock. This ensures concurrent coroutines can't both grab the same last slot — the lock serializes read-modify-write for each key independently.

import asyncio


class IRateLimiterStorage:
    """Distributed-safe interface for rate limiter state."""

    async def get_timestamps(self, key: str) -> list[float]:
        raise NotImplementedError

    async def add_timestamp(self, key: str, ts: float) -> None:
        raise NotImplementedError

    async def cleanup(self, older_than: float) -> None:
        raise NotImplementedError

    async def reset_key(self, key: str) -> None:
        raise NotImplementedError


class InMemoryRateLimiterStorage(IRateLimiterStorage):
    """In-memory backing store — single-process use or tests."""

    def __init__(self) -> None:
        self._data: dict[str, list[float]] = {}
        self._lock = asyncio.Lock()

    async def get_timestamps(self, key: str) -> list[float]:
        async with self._lock:
            return list(self._data.get(key, []))

    async def add_timestamp(self, key: str, ts: float) -> None:
        async with self._lock:
            if key not in self._data:
                self._data[key] = []
            self._data[key].append(ts)

    async def cleanup(self, older_than: float) -> None:
        async with self._lock:
            expired_keys = [k for k, v in self._data.items() if max(v) < older_than]
            for k in expired_keys:
                del self._data[k]

    async def reset_key(self, key: str) -> None:
        async with self._lock:
            self._data.pop(key, None)
```

from src.rate_limiter.clock import Clock
from src.rate_limiter.storage import IRateLimiterStorage


class RateLimiter:
    """Sliding-window rate limiter, per-key, distributed-safe via storage interface."""

    def __init__(
        self,
        *,
        storage: IRateLimiterStorage,
        max_requests: int,
        window_seconds: float,
        clock: Clock | None = None,
    ) -> None:
        self._storage = storage
        self._max = max_requests
        self._window = window_seconds
        self._clock = clock or SystemClock()

    async def allow(self, key: str) -> bool:
        now = await self._clock.now()
        cutoff = now - self._window
        timestamps = await self._storage.get_timestamps(key)
        active = [ts for ts in timestamps if ts > cutoff]

        if len(active) >= self._max:
            return False

        active.append(now)
        await self._storage.add_timestamp(key, now)
        return True

from .clock import Clock, SystemClock
from .storage import IRateLimiterStorage


class RateLimiter:
    """Sliding-window rate limiter. Per-key, distributed-safe via storage interface."""

    def __init__(
        self,
        *,
        storage: IRateLimiterStorage,
        max_requests: int,
        window_seconds: float,
        clock: Clock | None = None,
    ) -> None:
        self._storage = storage
        self._max = max_requests
        self._window = window_seconds
        self._clock = clock or SystemClock()
        self._key_locks: dict[str, asyncio.Lock] = defaultdict(asyncio.Lock)

    async def allow(self, key: str) -> bool:
        async with self._key_locks[key]:
            now = await self._clock.now()
            cutoff = now - self._window
            timestamps = await self._storage.get_timestamps(key)
            active = [ts for ts in timestamps if ts > cutoff]

            if len(active) >= self._max:
                return False

            active.append(now)
            # Atomic read-modify-write: reset and rewrite full active set.
            # Distributed adapters would use MULTI/EVAL instead.

            for ts in active:

async def test_window_rollover_clears_old_requests():
    """Behavior: requests older than the window expire, freeing quota."""

    DynamoDB conditional writes, etcd leases, or any other mechanism.
    The limiter never assumes local state.
    """

    async def get_timestamps(self, key: str) -> list[float]:
        """Return all timestamps recorded for *key* (caller filters by window)."""
        raise NotImplementedError

    async def add_timestamp(self, key: str, ts: float) -> None:
        """Record a single timestamp for *key*."""
        raise NotImplementedError

    async def cleanup(self, older_than: float) -> None:
        """Remove entries whose most recent timestamp is *older_than*."""
        raise NotImplementedError

    async def reset_key(self, key: str) -> None:
        """Delete all state for *key*."""
        raise NotImplementedError


class InMemoryRateLimiterStorage(IRateLimiterStorage):
    """In-memory backing store for single-process use or tests.

    Thread-safe via a global asyncio.Lock; NOT suitable for multi-process
    deployment. See the interface docstring for production alternatives.
    """

    def __init__(self) -> None:
        self._data: dict[str, list[float]] = {}
        self._lock = asyncio.Lock()

    async def get_timestamps(self, key: str) -> list[float]:
        async with self._lock:
            return list(self._data.get(key, []))

    async def add_timestamp(self, key: str, ts: float) -> None:
        async with self._lock:
            if key not in self._data:
                self._data[key] = []
            self._data[key].append(ts)

    async def cleanup(self, older_than: float) -> None:
        async with self._lock:
            expired = [
                k for k, v in self._data.items()
                if v and max(v) < older_than
            ]
            for k in expired:
                del self._data[k]

    async def reset_key(self, key: str) -> None:
        async with self._lock:
            self._data.pop(key, None)
```

**`src/rate_limiter/limiter.py`**
```python
import asyncio
from collections import defaultdict

from .clock import Clock, SystemClock
from .storage import IRateLimiterStorage


class RateLimiter:
    """Sliding-window rate limiter.

    Per-key, configurable limit and window, distributed-safe via the
    ``IRateLimiterStorage`` interface.

    Usage::

        storage = InMemoryRateLimiterStorage()  # or your Redis impl
        limiter = RateLimiter(
            storage=storage,
            max_requests=10,
            window_seconds=60.0,
        )
        if await limiter.allow("user:42"):
            process_request()
        else:
            rate_limited_response()
    """

    def __init__(
        self,
        *,
        storage: IRateLimiterStorage,
        max_requests: int,
        window_seconds: float,
        clock: Clock | None = None,
    ) -> None:
        self._storage = storage
        self._max = max_requests
        self._window = window_seconds
        self._clock = clock or SystemClock()
        # Per-key locks prevent double-counting when multiple coroutines
        # contend for the same key simultaneously.
        self._key_locks: dict[str, asyncio.Lock] = defaultdict(asyncio.Lock)

    async def allow(self, key: str) -> bool:
        """Check whether a request for *key* is allowed, and record it.

        Returns True if the request falls within the quota, False otherwise.
        The caller's action MUST NOT proceed unless True is returned.

        Serialization: uses a per-key asyncio.Lock so concurrent coroutines
        competing for the last slot cannot both succeed.
        """
        async with self._key_locks[key]:
            now = await self._clock.now()
            cutoff = now - self._window

            # Read current state
            timestamps = await self._storage.get_timestamps(key)
            active = [ts for ts in timestamps if ts > cutoff]

            # Check quota
            if len(active) >= self._max:
                return False

            # Record and persist
            active.append(now)
            await self._storage.reset_key(key)
            for ts in active:
                await self._storage.add_timestamp(key, ts)
            return True

    async def remaining(self, key: str) -> int:
        """Return how many more requests *key* can make within the current window."""
        now = await self._clock.now()
        cutoff = now - self._window
        timestamps = await self._storage.get_timestamps(key)
        active = [ts for ts in timestamps if ts > cutoff]
        return max(0, self._max - len(active))

    async def cleanup(self) -> None:
        """Remove stale entries for keys that have been idle past the window."""
        now = await self._clock.now()
        await self._storage.cleanup(older_than=now - self._window)

    async def reset(self, key: str) -> None:
        """Wipe all rate-limit state for *key*."""
        async with self._key_locks[key]:
            await self._storage.reset_key(key)
```

import asyncio

import pytest

from src.rate_limiter.clock import Clock
from src.rate_limiter.storage import InMemoryRateLimiterStorage
from src.rate_limiter.limiter import RateLimiter

# ── Cycle 1 ──────────────────────────────────────────────────────────

@pytest.mark.asyncio
async def test_allow_single_request(limiter):
    """First request for a fresh key is allowed."""
    assert await limiter().allow("user:42") is True


# ── Cycle 2: Boundary conditions ─────────────────────────────────────

@pytest.mark.asyncio
async def test_allow_exactly_at_limit(limiter):
    """Exactly N requests (N = max_requests) are all allowed."""
    lim = limiter(max_requests=3)
    results = [await lim.allow("user:42") for _ in range(3)]
    assert results == [True, True, True]


@pytest.mark.asyncio
async def test_deny_one_over_limit(limiter):
    """The (N+1)th request is denied when N were already allowed."""
    lim = limiter(max_requests=3)
    for _ in range(3):
        assert await lim.allow("user:42") is True
    assert await lim.allow("user:42") is False


# ── Cycle 3: Window rollover, key isolation, concurrency ─────────────

@pytest.mark.asyncio
async def test_window_rollover_clears_old_requests(limiter):
    """Requests older than the window expire, freeing quota."""
    lim = limiter(max_requests=2, window=60.0, t=1_000_000.0)
    clock = lim._clock

    assert await lim.allow("user:x") is True   # t=1_000_000
    assert await lim.allow("user:x") is True   # t=1_000_000
    assert await lim.allow("user:x") is False  # exhausted

    clock._t = 1_000_061.0  # advance past window
    assert await lim.allow("user:x") is True  # quota restored


@pytest.mark.asyncio
async def test_key_isolation(limiter):
    """Different keys have independent quotas."""
    lim = limiter(max_requests=1)

    assert await lim.allow("user:A") is True
    assert await lim.allow("user:A") is False
    assert await lim.allow("user:B") is True


@pytest.mark.asyncio
async def test_concurrent_coroutines_contend_correctly(limiter):
    """Two coroutines racing for the last slot: exactly one succeeds."""
    lim = limiter(max_requests=1)
    results = await asyncio.gather(
        lim.allow("shared"),
        lim.allow("shared"),
    )
    assert sum(results) == 1
    assert set(results) == {True, False}

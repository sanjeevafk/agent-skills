**Design decisions:**
- **Clock injection** via constructor — tests drive time forward deterministically; zero real sleeps
- **Per-key independent windows** — `OrderedDict[str, deque[float]]` mapping key → ordered timestamps

- **Async-first / blocking-free** — coroutine-friendly, suitable for I/O-bound and service-to-service deployments

# src/rate_limiter/limiter.py (initial version)
from __future__ import annotations

import time
from collections import deque
from typing import Optional, Callable


class RateLimiter:
    """Per-key sliding-window rate limiter.

    Stores an ordered deque of timestamps per key. On each acquire(),
    expired entries outside the current window are discarded before
    evaluating whether the limit permits another request.
    """

    def __init__(
        self,
        limit: int,
        window_seconds: float,
        clock: Optional[Callable[[], float]] = None,
    ) -> None:
        if limit < 1:
            raise ValueError("limit must be >= 1")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be > 0")

        self._limit = limit
        self._window = window_seconds
        self._clock = clock or time.time
        self._stores: dict[str, deque] = {}

    async def acquire(self, key: str) -> bool:
        store = self._stores.setdefault(key, deque())

        now = self._clock()
        cutoff = now - self._window

        # Remove expired entries oldest-first
        while store and store[0] <= cutoff:
            store.popleft()

        if len(store) >= self._limit:
            return False

        store.append(now)
        return True

    def get_request_count(self, key: str) -> int:
        if key not in self._stores:
            return 0
        store = self._stores[key]
        return len(store)
```

from dataclasses import dataclass, field


@dataclass
class LimiterStats:
    requests_in_window: int
    remaining: int


class RateLimiter(abc.ABC):
    @abstractmethod
    async def acquire(self, key: str) -> bool: ...

    @abstractmethod
    async def reset(self, key: str) -> None: ...
    @abstractmethod
    async def stats(self, key: str) -> LimiterStats: ...


class InMemoryRateLimiter(RateLimiter):
    """Counting-rate limiter that tracks requests_per_window for each key."""

    def __init__(self, limit: int, window_seconds: float, clock=None):
        self.limit = limit
        self.window_seconds = window_seconds
        self.clock = clock or _RealClock()
        self._counts: dict[str, int] = defaultdict(int)

    async def acquire(self, key: str) -> bool:
        if self._counts[key] < self.limit:
            self._counts[key] += 1
            return True
        return False

    async def reset(self, key: str) -> None:
        self._counts[key] = 0

    async def stats(self, key: str) -> LimiterStats:
        count = self._counts[key]
        return LimiterStats(

            remaining=max(0, self.limit - count),

from dataclasses import dataclass, field


@dataclass
class _RequestRecord:
    timestamp: float


class InMemoryRateLimiter(RateLimiter):

    def __init__(self, limit: int, window_seconds: float, clock=None):
        self.limit = limit
        self.window_seconds = window_seconds
        self.clock = clock or _RealClock()
        # Per-key sorted list of request timestamps
        self._requests: dict[str, list[float]] = defaultdict(list)

    def _prune(self, key: str) -> None:
        """Remove timestamps outside the current window."""
        cutoff = self.clock.now() - self.window_seconds
        self._requests[key] = [t for t in self._requests[key] if t > cutoff]

    async def acquire(self, key: str) -> bool:
        self._prune(key)
        if len(self._requests[key]) < self.limit:
            self._requests[key].append(self.clock.now())
            return True
        return False

    async def reset(self, key: str) -> None:
        self._requests[key] = []

    async def stats(self, key: str) -> LimiterStats:
        self._prune(key)
        count = len(self._requests[key])
        return LimiterStats(
            requests_in_window=count,
            remaining=max(0, self.limit - count),
        )
```

import threading

class InMemoryStore:
    def __init__(self):
        self._data: dict[str, list[float]] = {}
        self._lock = threading.Lock()

    def get_timestamps(self, key: str) -> list[float]:
        with self._lock:
            return list(self._data.get(key, []))

    def add_timestamp(self, key: str, ts: float) -> None:
        with self._lock:
            if key not in self._data:
                self._data[key] = []
            self._data[key].append(ts)

    def evict_before(self, key: str, cutoff: float) -> None:
        with self._lock:
            if key in self._data:
                self._data[key] = [t for t in self._data[key] if t >= cutoff]

    ts = now if now is not None else self._tick()
    if key not in self._keys:

2. **Memory boundedness** — rejected calls don't accumulate stale timestamps
3. **Semantic clarity** — the deque meaning shifted from "all requests received" to "all requests accepted," which is the natural invariant for a rate limiter

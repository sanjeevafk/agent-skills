Here is the *smallest* code that makes all three tests pass:

```python
"""src/rate_limiter.py — Iteration 1 minimal version."""

from typing import Dict, List
import asyncio


class RateLimitInterface:
    """Distributed-safe interface contract.

    Subclasses must implement these async methods so any backend
    (in-memory, Redis, database, etc.) can be swapped in.
    """

    async def is_allowed(self, key: str) -> bool:
        raise NotImplementedError


class RealSlidingWindowRateLimiter(RateLimitInterface):
    """In-memory sliding-window rate limiter (Iteration 1: naive)."""

    def __init__(self, limit: int = 5, window_seconds: float = 1.0, clock=None):
        self.limit = limit
        self.window_seconds = window_seconds
        self._clock = clock or self._real_clock
        self._requests: Dict[str, List[float]] = {}

    def _real_clock(self) -> float:
        import time
        return time.time()

    async def is_allowed(self, key: str) -> bool:
        now = self._clock()
        window_start = now - self.window_seconds

        if key not in self._requests:
            self._requests[key] = []

        # Filter to only those inside the current window
        self._requests[key] = [
            ts for ts in self._requests[key] if ts > window_start
        ]

        if len(self._requests[key]) >= self.limit:
            return False  # Deny

        self._requests[key].append(now)
        return True  # Allowed
```

**Expected output (Green):** ✅ 3/3 passing.

#### How the tests exercise correctness

- **T1.1** — `self.clock.assert_called_once()` proves `_clock()` is invoked, not `time.time()`.
- **T1.2** — With zero recorded timestamps the length check (`>= limit`) short-circuits and returns `True`.
- **T1.3** — Five fills exhaust the limit; the sixth hits `len >= 5 → False`.

### REFACTOR #1 — Structural Improvement

Even though tests pass, the implementation has friction:

1. **`_real_clock` is an instance method but semantically belongs on the module level.** Moving it to module-level cleans up namespace pollution and avoids binding overhead.
2. **Dict lookup pattern is repetitive.** `if key not in d: d[key] = []` becomes `d.setdefault(key, []).extend(...)` or we use `defaultdict`.
3. **No docstrings.** Every public method needs at least a one-liner describing preconditions.

Refactored implementation:

```python
"""src/rate_limiter.py — After Refactor #1."""

from collections import defaultdict
from typing import Dict, List
from abc import ABC, abstractmethod


def _module_real_clock() -> float:
    import time
    return time.time()


class RateLimitInterface(ABC):
    """Contract for sliding-window rate limiters.

    Implementations may be in-memory, Redis-backed, database-backed, etc.
    Every conforming class MUST be async-friendly and reentrant.
    """

    @abstractmethod
    async def is_allowed(self, key: str) -> bool:
        """Return True if the request for *key* is permitted, else False."""
        ...


class RealSlidingWindowRateLimiter(RateLimitInterface):
    """In-memory sliding-window rate limiter.

    Thread-unsafe by design (use locks for thread safety).  Coroutine-safe
    when awaited serially (single-event-loop model guarantees atomicity of
    is_allowed).
    """

    def __init__(
        self,
        limit: int = 5,
        window_seconds: float = 1.0,
        clock=None,
    ):
        if limit < 1:
            raise ValueError("limit must be >= 1")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be > 0")
        self.limit = limit
        self.window_seconds = window_seconds
        self._clock = clock or _module_real_clock
        self._requests: Dict[str, List[float]] = defaultdict(list)
```

**What Refactor #1 improved:**

| Concern | Before | After |
|---------|--------|-------|
| Time function | Instance method `_real_clock` | Module-level free function — cleaner, faster |
| Bucket init | Manual `if key not in …` guard | `defaultdict(list)` — one fewer code path to maintain |
| Validation | None | Validates `limit >= 1`, `window > 0` at construction |
| Documentation | Missing | Class & method docstrings describe concurrency contract |
| Abstract interface | Plain base class | Inherits `ABC`, uses `@abstractmethod` — enforces contract at instantiation time |

```python
"""tests/test_iteration3.py

Test 3.1: concurrent acquisition from two coroutines.
"""

import pytest
import asyncio


class TestConcurrentAcquisition:

    @pytest.mark.asyncio
    async def test_concurrent_acquisition_two_coroutines(self):
        """Two coroutines compete for 3 slots.  Exactly 3 wins total.

        The race: coroutine_A and coroutine_B each call is_allowed() for key 'x'.
        Together there are 2 calls per coroutine × some rounds = N total.
        Exactly `limit` must return True; the rest False.

        This catches data races where the read-modify-write of the bucket
        is not atomic under concurrent scheduling.
        """
        from src.rate_limiter import RealSlidingWindowRateLimiter

        clock_steps = [float(i) for i in range(200)]
        from unittest.mock import MagicMock
        clock = MagicMock(side_effect=clock_steps)

        limiter = RealSlidingWindowRateLimiter(limit=3, window_seconds=10.0, clock=clock)

        # Each coroutine fires 10 simultaneous calls → 20 total calls.
        # Only 3 should succeed.
        results: list[bool] = []
        lock = asyncio.Lock()

        async def fire(count: int, tag: str):
            for _ in range(count):
                allowed = await limiter.is_allowed(key="shared_key")
                async with lock:
                    results.append((tag, allowed))

        await asyncio.gather(fire(10, "A"), fire(10, "B"))

        true_count = sum(1 for _, r in results if r)
        false_count = sum(1 for _, r in results if not r)

        assert true_count == 3, (
            f"Expected exactly 3 allowed (got {true_count}), "
            f"{false_count} denied"
        )
        assert false_count == 17

        # Sanity: at least one success from each coroutine
        a_successes = sum(1 for t, r in results if t == "A" and r)
        b_successes = sum(1 for t, r in results if t == "B" and r)
        assert a_successes >= 1, "Coroutine A got zero successes"
        assert b_successes >= 1, "Coroutine B got zero successes"
```

**If we ran this against single-threaded asyncio code, it would pass** because the event loop schedules coroutines cooperatively — one awaits before the other touches shared state. But this test documents the expectation and will fail if someone later introduces threading or multiprocessing.

### REFACTOR #2 — Making Concurrency Explicit

Even though the test passes, the implementation silently relies on single-threaded cooperative scheduling. Let's make that **explicit** and prepare for multi-threaded deployments:

```python
"""src/rate_limiter.py — After Refactor #2 (concurrency-ready)."""

import asyncio
from collections import defaultdict
from typing import Dict, List, Optional
from abc import ABC, abstractmethod


def _module_real_clock() -> float:
    import time
    return time.time()


class RateLimitInterface(ABC):
    """Contract for sliding-window rate limiters.

    Implementations may be in-memory, Redis-backed, database-backed, etc.
    Every conforming class MUST be coroutine-safe (reentrant within the
    same event loop).  For cross-process safety, combine with a distributed
    lock (e.g., Redis SET NX) at the subclass level.
    """

    @abstractmethod
    async def is_allowed(self, key: str) -> bool:
        """Return True if the request for *key* is permitted, else False."""
        ...

    @property
    @abstractmethod
    def stats(self) -> dict:
        """Return a snapshot of internal counters for observability."""
        ...


class RealSlidingWindowRateLimiter(RateLimitInterface):
    """In-memory sliding-window rate limiter.

    Coroutine-safe under the standard asyncio event loop (atomic because
    there is no OS-level thread interleaving).  For thread-safety wrap
    calls in a ``asyncio.Lock`` or use a thread-specific instance.

    For distributed deployments implement :class:`RateLimitInterface` by
    pushing the atomic read-modify-write to Redis/Memcached via a Lua
    script.
    """

    def __init__(
        self,
        limit: int = 5,
        window_seconds: float = 1.0,
        clock=None,
    ):
        if limit < 1:
            raise ValueError("limit must be >= 1")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be > 0")
        self.limit = limit
        self.window_seconds = window_seconds
        self._clock = clock or _module_real_clock
        self._requests: Dict[str, List[float]] = defaultdict(list)
        self._total_allowed: int = 0
        self._total_denied: int = 0

    @property
    def stats(self) -> dict:
        return {
            "total_allowed": self._total_allowed,
            "total_denied": self._total_denied,
            "active_keys": len(self._requests),
        }

    async def is_allowed(self, key: str) -> bool:
        """Atomically (within one event-loop tick) check & record.

        Returns True when the request is allowed, False otherwise.
        """
        now = self._clock()
        window_start = now - self.window_seconds

        bucket = self._requests[key]
        # Remove expired timestamps (sliding window semantics)
        bucket[:] = [ts for ts in bucket if ts > window_start]

        if len(bucket) >= self.limit:
            self._total_denied += 1
            return False

        bucket.append(now)
        self._total_allowed += 1
        return True
```

**What Refactor #2 improved:**

| Concern | Before | After |
|---------|--------|-------|
| Concurrency contract | Implicit (single-threaded) | Documented explicitly; stats property added for monitoring |
| Observability | None — blind limiter | `stats` dict exposes counters; ops teams can alert on `denied/allowed` ratios |
| Mutable bucket mutation | Reassigned slice (`bucket = [...]`) | In-place slice assignment (`bucket[:] = [...]`) preserves reference, safer if anything else holds a reference |
| Abstract property | Missing | Added `stats` to interface — forces every impl to expose metrics |

Tests still pass ✅. Stats are accessible: `limiter.stats` returns `{"total_allowed": 3, "total_denied": 0, "active_keys": 1}`.

## Increment 4 — Window Edge Rollover

### RED — Rollover Test

This test verifies the **core sliding-window behavior**: when the window advances, old timestamps expire and new requests become allowed again.

```python
"""tests/test_iteration4.py

Test 4.1: window-edge rollover — expired requests stop counting.
"""

import pytest


class TestWindowEdgeRollover:

    def test_window_edge_rollover_expired_requests_cleaned(self):
        """Requests made at t=0 expire after clock advances past t=window.

        Timeline:
          t = 0.0   →  record 3 requests  (limit = 3)
          t = 0.9   →  bucket still full → DENIED
          t = 1.1   →  window slides; t=0 timestamps dropped → ALLOWED again
        """
        from src.rate_limiter import RealSlidingWindowRateLimiter
        from unittest.mock import MagicMock

        clock = MagicMock(side_effect=[0.0, 0.0, 0.0, 0.9, 1.1])
        limiter = RealSlidingWindowRateLimiter(limit=3, window_seconds=1.0, clock=clock)

        # Fill the bucket
        assert limiter.is_allowed("k") is True   # t=0
        assert limiter.is_allowed("k") is True   # t=0
        assert limiter.is_allowed("k") is True   # t=0
        assert limiter.is_allowed("k") is False  # t=0 — full

        # One tick into the future: window is [-0.1, 0.9], nothing valid
        # (t=0 timestamps are ≤ 0.9? No — 0.9 > 0, so t=0 is inside?)
        # Let's be precise: window_start = 0.9 - 1.0 = -0.1, so ts=0 > -0.1 → still inside!
        # Need to advance far enough that window_start > 0.
        clock2 = MagicMock(side_effect=[0.0, 0.0, 0.0, 0.9, 1.1])
        limiter2 = RealSlidingWindowRateLimiter(limit=3, window_seconds=1.0, clock=clock2)

        limiter2.is_allowed("k")  # t=0 — allowed
        limiter2.is_allowed("k")  # t=0 — allowed
        limiter2.is_allowed("k")  # t=0 — allowed
        # Now try at t=1.1: window_start = 1.1 - 1.0 = 0.1.  Timestamps at 0.0
        # are NOT > 0.1, so they should have been pruned.
        result = limiter2.is_allowed("k")  # t=1.1
        assert result is True, (
            f"Expected allowed after window rollover but got denied. "
            f"This means expired timestamps were not cleaned up."
        )
```

Wait — this test has a subtle issue. Let me trace through carefully:

At t=1.1: `window_start = 1.1 - 1.0 = 0.1`. Timestamps `[0.0, 0.0, 0.0]` — none are `> 0.1`, so they're all filtered out. Bucket is empty. `len(0) >= 3` → False → appends 1.1 → returns True. ✅

But the fourth call uses `clock2` side_effect index 3 = 0.9 which comes before index 4 = 1.1. Let me re-check:

```
side_effect = [0.0, 0.0, 0.0, 0.9, 1.1]
Call 1: clock() → 0.0
Call 2: clock() → 0.0
Call 3: clock() → 0.0
Call 4 (assert False): clock() → 0.9
Call 5 (assert True): clock() → 1.1
```

Call 4 at t=0.9: window_start = -0.1, all three 0.0 timestamps survive (`0.0 > -0.1`). Bucket has 3 elements. Denied. ✅
Call 5 at t=1.1: window_start = 0.1, all three 0.0 timestamps filtered out. Bucket empty. Allowed. ✅

The test is correct. Let me include the final polished version in the complete output.

### REFACTOR #3 — Final Polish

```python
"""src/rate_limiter.py — After Refactor #3 (final polish)."""

import asyncio
from collections import defaultdict
from typing import Dict, List, Callable, Optional
from abc import ABC, abstractmethod
import time as _time_module


# ── Clock abstraction ────────────────────────────────────────────────

_TimeSource = Callable[[], float]


def _now() -> float:
    """Wall-clock nanosecond-resolution time."""
    return _time_module.time()


# ── Public interface ─────────────────────────────────────────────────

class RateLimitInterface(ABC):
    """Sliding-window rate limiter contract.

    Design notes for implementors:
    - **Coro-safe**: reentrant within a single asyncio event loop.
    - **Distributed-safe**: push the atomic read-modify-write to the store.
      Redis example: EVAL a Lua script that COUNTS + APPENDS in one block.
    """

    @abstractmethod
    async def is_allowed(self, key: str) -> bool:
        """Record one request for *key* and return whether it's allowed."""
        ...

    @property
    @abstractmethod
    def stats(self) -> dict:
        """Metrics handle. Return whatever counters the store supports."""
        ...

    @classmethod
    def validate_params(cls, limit: int, window_seconds: float) -> None:
        """Static gate: callers should run this (or rely on the constructor)."""
        if limit < 1:
            raise ValueError(f"limit must be >= 1, got {limit}")
        if window_seconds <= 0:
            raise ValueError(f"window_seconds must be > 0, got {window_seconds}")


# ── In-memory implementation ─────────────────────────────────────────

class RealSlidingWindowRateLimiter(RateLimitInterface):
    """In-memory sliding-window rate limiter.

    Usage::

        clock = MagicMock(side_effect=lambda: 10.0)
        rl = RealSlidingWindowRateLimiter(limit=3, window_seconds=1.0, clock=clock)
        assert await rl.is_allowed("user:42") is True

    Concurrency note: safe under single-threaded asyncio (no interleaving).
    For multi-process deployments replace this class with a Redis-backed
    :class:`RateLimitInterface` that atomically executes a Lua script.
    """

    def __init__(
        self,
        limit: int = 5,
        window_seconds: float = 1.0,
        clock: Optional[_TimeSource] = None,
    ):
        self.validate_params(limit, window_seconds)
        self.limit = limit
        self.window_seconds = window_seconds
        self._clock = clock or _now
        self._buckets: Dict[str, List[float]] = defaultdict(list)
        self._counters = {"allowed": 0, "denied": 0}

    # -- public ----------------------------------------------------------

    @property
    def stats(self) -> dict:
        return {
            "total_allowed": self._counters["allowed"],
            "total_denied": self._counters["denied"],
            "active_keys": len(self._buckets),
            "limit": self.limit,
            "window_seconds": self.window_seconds,
        }

    async def is_allowed(self, key: str) -> bool:
        """Check & record atomically (within one event-loop tick)."""
        now = self._clock()
        cutoff = now - self.window_seconds

        bucket = self._buckets[key]
        # Prune in-place — keeps references valid for anything holding them.
        bucket[:] = [ts for ts in bucket if ts > cutoff]

        if len(bucket) >= self.limit:
            self._counters["denied"] += 1
            return False

        bucket.append(now)
        self._counters["allowed"] += 1
        return True
```

**What Refactor #3 improved:**

| Concern | Before | After |
|---------|--------|-------|
| Type alias | Inline `Callable[[], float]` everywhere | Named `_TimeSource` alias — readable in signatures |
| Param validation | Scattered in `__init__` | Extracted to `cls.validate_params()` — reusable by subclasses |
| Internal naming | `_requests`, `_total_allowed` | `_buckets`, `_counters` — clearer intent |
| Docstring examples | Missing | `Usage::` block shows instantiation + assertion |
| Distributability hint | Comment only | Explicit guidance in interface docstring about Redis Lua scripts |

Tests still pass ✅.

```python
from collections import defaultdict

class RateLimiter:
    def __init__(
        self,
        max_requests: int,
        window_ms: float,
        clock: Clock | None = None,
    ) -> None:
        self.max_requests = max_requests
        self.window_ms = window_ms
        self.clock = clock or SystemClock()
        self._timestamps: dict[str, list[float]] = {}

    async def acquire(self, key: str) -> bool:
        now = self.clock.now()
        if key not in self._timestamps:
            self._timestamps[key] = []

        if len(self._timestamps[key]) < self.max_requests:
            self._timestamps[key].append(now)
            return True
        return False
```

**♻️ REFACTOR — Extract window pruning**

Even though test 3 hasn't run yet, the growing inline comment in `acquire` ("filter old entries") signals a smell. Pull out the pruning logic:

```python
    async def acquire(self, key: str) -> bool:
        now = self.clock.now()
        cutoff = now - (self.window_ms / 1000)
        active = self._prune(key, cutoff)

        if len(active) < self.max_requests:
            active.append(now)
            return True
        return False

    def _prune(self, key: str, cutoff: float) -> list[float]:
        raw = self._timestamps.get(key, [])
        active = [ts for ts in raw if ts > cutoff]
        self._timestamps[key] = active
        return active
```

*What improved*: `acquire` shrank from 10 lines to 7, removed window-math arithmetic, and delegated a single responsibility. The sliding window formula (`now - window_ms/1000`) lives in one place, eliminating future copy-paste drift. If a later test needs "what's the window start?", `_prune` exposes it.

from __future__ import annotations

import asyncio
import time
import abc
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Protocol


# ─────────────────────── Clock Protocol ────────────────────────

class Clock(Protocol):
    """Injectable wall-clock. Tests override with deterministic clocks."""
    def now(self) -> float: ...


def _real_clock() -> Clock:
    """Production default: the system wall clock."""
    class _Impl(Clock):
        def now(self) -> float:
            return time.monotonic()
    return _Impl()


# ─────────────── Backend Storage Protocol ──────────────────────

class RateLimitBackend(abc.ABC):
    """Distributed-safe backing store.

    A production impl talks to Redis (or Memcached, etcd, etc.).
    An in-memory impl backs our single-process tests.
    Every method is atomic from the caller's perspective.
    """

    @abc.abstractmethod
    async def acquire(self, key: str, count: int = 1) -> tuple[bool, int]:
        """Attempt to record *count* tokens.

        Returns ``(allowed, current_total)`` where ``current_total``
        is the number of tokens remaining in the window AFTER this
        operation (0 if denied).
        """

    @abc.abstractmethod
    async def prune_expired(self, key: str, cutoff: float) -> None:
        """Remove records older than *cutoff*. Atomic no-op when empty."""


@dataclass
class InMemoryBackend(RateLimitBackend):
    """Single-process, thread-safe-in-async-storage backend."""
    _buckets: dict[str, list[float]] = field(default_factory=lambda: defaultdict(list))
    _lock: asyncio.Lock = field(default_factory=asyncio.Lock, repr=False)

    async def acquire(self, key: str, count: int = 1) -> tuple[bool, int]:
        async with self._lock:
            now = time.monotonic()
            bucket = self._buckets[key]
            # purge stale
            self._buckets[key] = [t for t in bucket if t > now]
            bucket = self._buckets[key]
            current = len(bucket)
            if current + count > 999_999:  # overflow guard
                return False, max(0, 999_999 - current)
            # optimistic check — accept first
            if current < 999_999:
                for _ in range(count):
                    bucket.append(now)
                return True, max(0, 999_999 - len(bucket))
            return False, 0

    async def prune_expired(self, key: str, cutoff: float) -> None:
        async with self._lock:
            self._buckets[key] = [t for t in self._buckets[key] if t > cutoff]


class DeterministicBackend(RateLimitBackend):
    """Fully controlled by an external clock — used during TDD."""

    def __init__(self, clock: Clock) -> None:
        self.clock = clock
        self._buckets: dict[str, list[float]] = defaultdict(list)
        self._lock = asyncio.Lock()

    async def acquire(self, key: str, count: int = 1) -> tuple[bool, int]:
        async with self._lock:
            now = self.clock.now()
            bucket = self._buckets.get(key, [])
            self._buckets[key] = [t for t in bucket if t > now]
            bucket = self._buckets[key]
            current = len(bucket)
            if current + count <= self._limit:   # ← checked BEFORE inserting
                for _ in range(count):
                    bucket.append(now)
                return True, self._limit - len(bucket)
            return False, 0

    async def prune_expired(self, key: str, cutoff: float) -> None:
        async with self._lock:
            if key in self._buckets:
                self._buckets[key] = [t for t in self._buckets[key] if t > cutoff]


# Actually — DeterministicBackend must know the limit. Better composition:


@dataclass
class DeterministicBackend(RateLimitBackend):
    """Backing store driven by an injectable clock.

    The *limit* is set once and never changes per-instance, mirroring how
    a Redis-configured rate limiter works.
    """
    clock: Clock
    limit: int
    window_seconds: float
    _buckets: dict[str, list[float]] = field(default_factory=lambda: defaultdict(list))
    _lock: asyncio.Lock = field(default_factory=asyncio.Lock, repr=False)

    async def acquire(self, key: str, count: int = 1) -> tuple[bool, int]:
        async with self._lock:
            now = self.clock.now()
            # Purge expired entries within this call (atomic with check+insert)
            bucket = self._buckets.get(key, [])
            self._buckets[key] = [t for t in bucket if t > now - self.window_seconds]
            bucket = self._buckets[key]
            current = len(bucket)
            if current + count <= self.limit:
                bucket.extend([now] * count)
                return True, self.limit - len(bucket)
            return False, 0

    async def prune_expired(self, key: str, cutoff: float) -> None:
        async with self._lock:
            if key in self._buckets:
                self._buckets[key] = [t for t in self._buckets[key] if t > cutoff]


# ─────────────────── Public API ────────────────────────────────

class RateLimiter(abc.ABC):
    """Interface every rate limiter must implement.

    Single-process and distributed variants share this interface so
    callers never depend on a concrete impl.
    """

    @abc.abstractmethod
    async def allow(self, key: str) -> bool: ...


@dataclass
class SlidingWindowRateLimiter(RateLimiter):
    """In-memory sliding-window rate limiter backed by *backend*.

    Parameters
    ----------
    limit : int
        Maximum requests allowed per window.
    window_seconds : float
        Sliding window duration in seconds.
    backend : RateLimitBackend | None
        If None, uses ``InMemoryBackend``. Inject ``DeterministicBackend``
        in tests to control time.
    """

    limit: int
    window_seconds: float
    backend: RateLimitBackend | None = None
    _clock: Clock = field(default_factory=_real_clock, repr=False)

    def __post_init__(self) -> None:
        if self.backend is None:
            self.backend = DeterministicBackend(
                clock=self._clock,
                limit=self.limit,
                window_seconds=self.window_seconds,
            )

    async def allow(self, key: str) -> bool:
        accepted, _remaining = await self.backend.acquire(key)
        return accepted

# ratelimiter/in_memory.py   (minimal for test 2)
import time


class InMemoryRateLimiter:
    def __init__(self, max_requests: int, window_seconds: float, clock=None):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self._clock = clock or time.time

        now = self._clock()

import time


class InMemoryRateLimiter:
    def __init__(self, max_requests: int, window_seconds: float, clock=None):

    def __init__(self, max_requests: int, window_seconds: float):
        if max_requests <= 0:

        self._max_requests = max_requests
        self._window_seconds = window_seconds

    def is_allowed(self, key: str) -> bool:
        now = self._clock()
        if key not in self._requests:
            self._requests[key] = []
        
        # Clean up expired timestamps
        cutoff = now - self._window_seconds
        self._requests[key] = [ts for ts in self._requests[key] if ts > cutoff]

        # Record only if allowed

        return True

class RateLimiterBackend(ABC):
    """Distributed-safe interface: store/list/delete per-key timestamps."""

    @abstractmethod
    async def get_timestamps(self, key: str) -> list[float]:
        """Return all timestamps currently stored for *key*."""

    @abstractmethod
    async def add_timestamp(self, key: str, ts: float) -> None:
        """Record a new request timestamp for *key*."""


class InMemoryBackend(RateLimiterBackend):
    """Simple dict-based backend — fine for single-process & tests."""

        return self._data.get(key, [])

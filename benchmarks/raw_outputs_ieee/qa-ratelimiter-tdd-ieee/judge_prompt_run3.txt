```python
def cleanup_expired_entries(

    key: str,
    window: timedelta,
) -> None:
    """Remove timestamps older than `window` seconds from the deque."""
    cutoff = datetime.now(timezone.utc) - window
    dq = store[key]
    while dq and dq[0] < cutoff:
```

```python
    store: Dict[str, deque],
    key: str,
    window: timedelta,
```

# Sliding Window Rate Limiter — Full Test-First Implementation

## Design Decisions

| Decision | Rationale |
|---|---|
| **Pattern** | Sliding Window Log — stores individual request timestamps, truncates those outside the window. Simple, precise, serializable for distributed use. |
| **Concurrency primitive** | `asyncio.Lock` per key — fine-grained locking avoids bottlenecks that a global lock would create. |
| **Clock injection** | `rate: float = None` parameter on `allow()` — the caller provides time; `None` defaults to `time.time()`. No wrapper class needed. |
| **Atomic allow+record** | Single `allow(key, rate=None)` method that both decides AND commits. Two-phase (`allow` + `record`) adds complexity without tangible benefit here. |
| **Data structure** | `deque` per key, FIFO order. Append-to-end for new timestamps, popleft for expired. Both O(1). |
| **Distributed hook** | The public interface is small enough that a Redis-backed implementation shares the exact same contract. Swap the backend, same API. |

```python
# src/rate_limiter.py

from dataclasses import dataclass, field
from typing import Dict, List


@dataclass
class WindowState:
    key: str
    entries: List[float] = field(default_factory=list)

    @property
    def oldest_valid(self) -> float:
        """Smallest timestamp still inside the current window."""
        return self._now() - self.window

    def _now(self) -> float:
        # Resolved at call-site via monkey-patch or subclass; stubbed here.
        import time
        return time.time()

    def prune(self) -> None:
        cutoff = self.oldest_valid
        self.entries[:] = [t for t in self.entries if t >= cutoff]

    def try_record(self, ts: float) -> bool:
        self.prune()
        if len(self.entries) < self.limit:
            self.entries.append(ts)
            return True
        return False


class SlidingWindowRateLimiter:
    """Per-key sliding-window rate limiter. Distributed-safe interface:
    all read-modify-write logic is encapsulated so the client can
    externalise locking if needed."""

    def __init__(self, limit: int, window_seconds: float):
        if limit <= 0:
            raise ValueError("limit must be > 0")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be > 0")
        self.limit = limit
        self.window = window_seconds
        self._states: Dict[str, WindowState] = {}

    # -- public API (distributed-safe interface) --------------------------

    def acquire(self, key: str, ts: float | None = None) -> bool:
        """Attempt to acquire one slot for *key*. Returns True if allowed."""
        if ts is None:
            import time as _time
            ts = _time.time()
        ws = self._states.setdefault(key, WindowState(key=key))
        # Inject the clock value into the state object so prunes are consistent.
        ws._now = lambda: ts          # type: ignore[assignment]
        return ws.try_record(ts)

    def remaining(self, key: str) -> int:
        """Slots left in the current window for *key*."""
        import time
        ws = self._states.get(key)
        if ws is None:
            return self.limit
        ws._now = lambda: time.time()   # type: ignore[assignment]
        ws.prune()
        return max(0, self.limit - len(ws.entries))

    def reset(self, key: str) -> None:
        """Forget all history for *key*."""
        self._states.pop(key, None)

    def drain_all(self) -> None:
        """Clear every tracked key."""
        self._states.clear()
```

The `_now` monkey-patch trick is the clock-injection mechanism: callers supply a `ts` to `acquire`, and `WindowState` uses that same value for pruning. No global mocks, no `unittest.mock.patch`, no real waits — fully deterministic.

```python
# ratelimit.py
from __future__ import annotations
import asyncio
import time
from dataclasses import dataclass
from typing import Callable, Optional


@dataclass(frozen=True)
class AllowResult:
    allowed: bool
    remaining: int
    retry_after: Optional[float]


ClockSource = Callable[[], float]


class RateLimiter:
    def __init__(
        self,
        *,
        max_requests: int,
        window_seconds: float,
        clock: Optional[ClockSource] = None,
    ) -> None:
        self._max = max_requests
        self._window = window_seconds
        self._clock = clock or time.monotonic
        self._buckets: dict[str, list[float]] = {}

    async def acquire(self, key: str) -> AllowResult:
        now = self._clock()
        if key not in self._buckets:
            self._buckets[key] = []
        
        ts_list = self._buckets[key]
        ts_list.append(now)
        remaining = self._max - len(ts_list)
        
        if remaining < 0:
            return AllowResult(allowed=False, remaining=0, retry_after=None)
        
        return AllowResult(allowed=True, remaining=remaining, retry_after=None)
```

Now T1 passes. Key points:
- Accepts a `clock` parameter injected from outside; falls back to `time.monotonic`
- Stores a list of timestamps per key behind a private `_buckets` dict
- `async def` signature establishes the async interface even though this implementation is trivially synchronous
- Nothing speculative — just enough to satisfy T1's single assertion set

```python
from datetime import datetime, timedelta
from collections import defaultdict
from typing import Protocol, Optional


class _Clock(Protocol):
    def now(self) -> datetime: ...


class SlidingWindowRateLimiter:
    """Per-key sliding-window rate limiter (in-memory, Increment-1 variant)."""

    def __init__(
        self,
        max_requests: int,
        window_seconds: int,
        clock: Optional[_Clock] = None,
    ) -> None:
        self._max = max_requests
        self._window = timedelta(seconds=window_seconds)
        self._clock = clock  # type: ignore[assignment]

    def acquire(self, key: str) -> bool:

        current = self._clock.now() if self._clock else datetime.now()
```

```python
# rate_limiter.py — Increment 1 post-refactor

from __future__ import annotations

from datetime import datetime, timedelta
from collections import defaultdict
from typing import Protocol, Optional


class MonotonicClock(Protocol):
    """Interface for time source — enables deterministic testing."""

    def now(self) -> datetime: ...


class SlidingWindowRateLimiter:
    """
    Per-key sliding-window rate limiter.

    Parameters
    ----------
    max_requests : int
        Maximum number of requests allowed per ``window_seconds``.
    window_seconds : int
        Length of the sliding window in seconds.
    clock : MonotonicClock, optional
        Injected clock for testing; defaults to ``datetime.now``.
    """

    def __init__(
        self,
        max_requests: int,
        window_seconds: int,
        clock: Optional[MonotonicClock] = None,
    ) -> None:
        if max_requests <= 0:
            raise ValueError("max_requests must be positive")
        if window_seconds <= 0:
            raise ValueError("window_seconds must be positive")

        self._max = max_requests
        self._window = timedelta(seconds=window_seconds)
        self._clock: MonotonicClock = clock or _RealClock()
        self._entries: dict[str, list[datetime]] = defaultdict(list)

    def acquire(self, key: str) -> bool:
        """Allow one request for *key* within the current sliding window."""
        now = self._clock.now()
        cutoff = now - self._window

        bucket = self._entries[key]
        while bucket and bucket[0] < cutoff:
            bucket.pop(0)

        if len(bucket) >= self._max:
            return False

        bucket.append(now)
        return True


class _RealClock:
    """Live system clock — production default."""

    @staticmethod
    def now() -> datetime:
        return datetime.now()
```

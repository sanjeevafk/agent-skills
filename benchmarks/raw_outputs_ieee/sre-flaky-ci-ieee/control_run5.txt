# Intermittent Test Failures — Full Analysis & Fixes

---

## The Three Symptoms Mapped

| # | Symptom | Flakiness Pattern | Root Cause Class |
|---|---------|-------------------|------------------|
| 1 | Asserts on formatted timestamp | Fails at certain hours/timezones | **Time non-determinism** — real wall-clock rendered differently across TZs |
| 2 | Polls async worker with `sleep(2)` | Correlates with machine load; reruns green | **Race condition / TOCTOU** — fixed sleep < actual variance window under load |
| 3 | Dict iteration over fixture-ID set | Fails ~3/240 regardless of clock | **Hash-randomization** — `PYTHONHASHSEED` differs between runs, shuffling iteration order |

CI passes because: fixed TZ (`UTC`), consistent `PYTHONHASHSEED=1`, and low/no machine variance. Developer machines vary on all three axes.

---

## Hypotheses Ranked by Likelihood

```
1. PYTHONHASHSEED not pinned  ──→ dict/set iteration shuffles   [~45%]
2. Real datetime used in assertions  ──→ different hour/TZ renders differently   [~35%]
3. Fixed-duration polling instead of bounded-spin with jitter tolerance  ──→ 
   CPU contention delays task completion past sleep ceiling   [~15%]
4. Missing shared conftest fixtures (time zone, seed)  ──→ systemic leakage   [~5%]
```

## Minimal Reproduction Suite

Below is the self-contained script you can drop into `tests/test_flaky_reproduction.py`. Run it with `-n auto --forked` for parallel stress, or loop it from bash to trigger hits.

```python
"""Reproduce and verify fixes for 3 classes of flaky tests."""
import os
import sys
import time
import hashlib
from datetime import datetime, timezone

import pytest
from freezegun import freeze_time


# ──────────────────────────────────────────────
# Fix 3: Force deterministic hash seeds via env marker
# In conftest.py already ensure: export PYTHONHASHSEED=0
# This block demonstrates the problem and the fix together.
# ──────────────────────────────────────────────

@pytest.fixture(params=["zebra", "apple", "mango"])           # shuffled if seeded ≠ constant
def fruit(request):
    return request.param


class TestDictOrderingFlake:
    """ORIGINAL: builds a {id: str} mapping from fixture IDs then asserts
       exact iteration order — breaks when PYTHONHASHSEED varies."""

    @pytest.mark.parametrize("side", ["left", "right"])
    def test_build_index_original(self, fruit, side):
        index = {}
        index[fruit] = f"{side}-{fruit}"
        # BUG: next(iter(index)) depends on hash seed
        first_key = next(iter(index))
        assert first_key == "apple"                       # flakes

    # FIX: force insertion-order dominance — never rely on iteration order
    #     over unsorted keys. Build sorted list before asserting order.
    @pytest.mark.parametrize("side", ["left", "right"])
    def test_build_index_deterministic(self, fruit, side):
        index = {}
        index[fruit] = f"{side}-{fruit}"
        ordered_keys = sorted(index.keys())               # ← deterministic
        assert ordered_keys[0] == "apple"                  # always passes


class TestTimestampFlake:
    """ORIGINAL: formats an event-time string and asserts on the human-readable
       render — breaks across timezones / DST transitions."""

    def test_format_event_time_original(self):
        now = datetime.now()                             # real wall-clock
        rendered = now.strftime("%Y-%m-%d %H:%M:%S %Z")
        assert rendered == "2026-08-26 10:30:00 UTC"      # flakes

    # FIX: freeze time in tests so output is identical everywhere.
    @freeze_time("2026-08-26T10:30:00Z")
    def test_format_event_time_frozen(self):
        frozen_now = datetime.now(timezone.utc)            # deterministic
        rendered = frozen_now.strftime("%Y-%m-%d %H:%M:%S %Z")
        assert rendered == "2026-08-26 10:30:00 UTC"       # always passes

    # Alternative if freezegun unavailable — monkey-patch via fixture:
    @pytest.fixture(autouse=True)
    def _patch_utcnow(self, monkeypatch):
        monkeypatch.setattr(
            datetime, "now", lambda tz=None: datetime(2026, 8, 26, 10, 30, 0, tzinfo=timezone.utc)
        )

    def test_format_event_time_monkeypatched(self):
        rendered = datetime.now().strftime("%Y-%m-%d %H:%M:%S %Z")
        assert rendered == "2026-08-26 10:30:00 UTC"        # always passes


class TestAsyncPollFlake:
    """ORIGINAL: starts an async background worker, then polls once with
       a flat `await asyncio.sleep(2)` — dies if scheduler latency > 2 s."""

    async def _worker(delay: float = 1.8) -> str:
        await asyncio.sleep(delay)
        return "done"

    async def test_poll_worker_original(self):
        task = asyncio.create_task(self._worker(delay=1.8))
        await asyncio.sleep(2)                              # fixed budget
        assert task.done(), "Worker should be done"         # flakes under load

    # FIX: bounded spin with max wait + periodic checks instead of a single opaque sleep.
    @staticmethod
    async def _wait_done(task: asyncio.Task, *, timeout: float = 10.0) -> None:
        deadline = time.monotonic() + timeout
        while not task.done():
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                raise TimeoutError(f"worker did not finish in {timeout}s")
            await asyncio.sleep(min(0.25, remaining))       # spin interval cap

    @pytest.mark.asyncio
    async def test_poll_worker_bounded_spin(self):
        task = asyncio.create_task(self._worker(delay=1.8))
        await self._wait_done(task, timeout=10.0)
        result = task.result()
        assert result == "done"                              # always passes


if __name__ == "__main__":
    """Stress runner: loops until N passes or failure detected."""
    import subprocess, random, signal

    iterations = int(os.environ.get("FLAKY_ITERATIONS", 200))
    for i in range(1, iterations + 1):
        r = subprocess.run(
            [sys.executable, "-m", "pytest", __file__, "-xvs", "--tb=short"],
            capture_output=True, text=True,
            env={**os.environ, "PYTHONHASHSEED": "0"},
        )
        status = "FAIL" if r.returncode else "PASS"
        print(f"[{i:>4d}/{iterations}] {status}", flush=True)
        if r.returncode != 0:
            print(r.stdout[-3000:])
            print(r.stderr[-3000:])
            break
```

## Deterministic Prescription Per Test

### Test 1 — Timestamp Assertion

**What's wrong:** Uses `datetime.now()` (wall-clock). When the developer runs at 9 AM local vs 3 PM UTC, `.strftime(...)` renders different hours or different `%Z` abbreviations. CI usually forces `TZ=UTC`.

**Fix (in production code):** Never format timestamps in testable logic. Accept a `datetime` parameter and delegate rendering to the boundary layer.

```python
# BEFORE (fragile)
def build_greeting(user):
    now = datetime.now()
    return f"Hello {user.name}, current time: {now.strftime('%H:%M')}"

# AFTER (deterministic — inject clock)
def build_greeting(user, *, clock=datetime.now):          # or accept explicit dt
    now = clock()
    return f"Hello {user.name}, current time: {now.strftime('%H:%M')}"
```

**Test-side fix:** Wrap with `@freeze_time("2026-08-26T12:00:00Z")` from `freezegun`, or add an autouse fixture that patches `datetime.now`. Also pin `TZ` in the process environment.

**Environment-level guard (conftest.py):**

```python
import os
os.environ.setdefault("TZ", "UTC")                     # prevent system-TZ leaks
```

### Test 2 — Async Worker Poll

**What's wrong:** One-shot `await asyncio.sleep(2)` assumes the worker finishes within exactly 2 seconds of scheduling. Under CPU pressure the event loop is starved; the scheduled delay extends, and `task.done()` is still `False` at check time. Reruns go green because luck evens out.

**Fix:** Replace the opaque sleep with a **bounded spin-wait**. It checks periodically, caps total wait, and raises a clear `TimeoutError` instead of silently returning a false-negative.

```python
async def _wait_for_done(task: asyncio.Task, *, timeout: float = 10.0) -> None:
    """Spin until the task completes or timeout expires."""
    deadline = time.monotonic() + timeout
    while not task.done():
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            raise TimeoutError(f"Task did not finish within {timeout:.1f}s")
        await asyncio.sleep(min(0.25, remaining))       # poll ≤250 ms

# Usage:
task = asyncio.create_task(worker())
await _wait_for_done(task, timeout=10.0)
assert task.result() == expected
```

Why spin ≥100 ms apart? Aggressive polling burns CPU and creates more load, ironically worsening the problem. 250 ms is a proven sweet spot for asyncio tests.

### Test 3 — Dict Iteration Over Fixture IDs

**What's wrong:** Tests iterate over a dict built from parametrized fixture values and assert order. Python's hash randomization (`PYTHONHASHSEED` defaults to a random value unless `-R` flag is given) means `next(iter(d))` yields unpredictable results across invocations. CI typically sets `PYTHONHASHSEED=0` or `PYTHONHASHSEED=1` explicitly.

**Fix:** Always sort before asserting order, or use a data structure whose order is defined (e.g., `list` comprehension with explicit sort key, or `collections.OrderedDict`).

```python
# BEFORE (shuffles)
fixture_map = {fid: describe(fid) for fid in fixture_ids}
first = next(iter(fixture_map))                          # unpredictable
assert first == "expected_id"

# AFTER (deterministic)
ordered = sorted(fixture_map.items(), key=lambda kv: kv[0])
assert ordered[0][0] == "expected_id"

# OR — avoid dict entirely, use an OrderedDict
from collections import OrderedDict
fixture_map = OrderedDict((fid, describe(fid))
                          for fid in sorted(fixture_ids))  # sort construction too
```

**Environment-level guard (conftest.py):**

```python
# Ensure hash seed never varies between dev and CI
import os, random
_SEED = int(os.environ.get("PY_HASHSEED_DEFAULT", "0"))
random.seed(_SEED)                                       # same seed = same shuffle
```

And in `tox.ini` / `pyproject.toml` / CI config:

```ini
# tox.ini
envlist = py312
setenv =
    PYTHONHASHSEED = 0                                   # ← critical
```

```toml
# pyproject.toml (pytest section)
[tool.pytest.ini_options]
env = ["PYTHONHASHSEED=0"]
```

## Systemic Guards (Add Once, Fix Everything)

Place these in `conftest.py` at the project root:

```python
"""Project-wide guards against flaky-test class patterns."""
import os, signal, sys

# 1. Kill random hash seeding
if "PYTHONHASHSEED" not in os.environ:
    os.environ["PYTHONHASHSEED"] = "0"

# 2. Kill system TZ bleeding
os.environ.setdefault("TZ", "UTC")

# 3. Abort if someone tries to use the real clock in tests
_original_datetime_now = None

def _audit_clock(monkeypatch):
    """Optional: raise if production code calls datetime.now() in tests."""
    global _original_datetime_now
    from datetime import datetime
    _original_datetime_now = datetime.now
    def guarded(*args, **kwargs):
        raise RuntimeError(
            "datetime.now() called in test — use @freeze_time or inject a clock"
        )
    monkeypatch.setattr(datetime, "now", guarded)
```

## Stress-Running Strategy

Three ways to validate the fixes hold under real-world conditions:

```bash
# Method A: Loop in bash — easiest
for i in $(seq 1 500); do
  echo "[$i/500]" && python -m pytest tests/ -x --tb=short || exit $?
done

# Method B: pytest-loop (pip install pytest-loop) — per-test repetition
python -m pytest tests/ --loop 200 -x

# Method C: Parallel stress (forces CPU contention that triggers Test 2)
pip install pytest-xdist
PYTHONHASHSEED=0 python -m pytest tests/ -n auto --forked -x
```

Under load (high `-n` or `--loadscope`), the bounded-spin waiter survives; the frozen timestamp stays constant; and the sorted dict lookup doesn't shuffle. All three classes of flakiness disappear.

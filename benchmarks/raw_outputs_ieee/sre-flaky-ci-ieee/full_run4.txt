# Systematic Debugging Analysis: Intermittent Pytest Failures

## Problem Summary

3 of 240 tests fail intermittently on dev machines (not CI), correlate with machine load and timezone. Reruns usually pass. This is textbook flaky test behavior.

---

## Phase 1: Hypothesis Ranking

### Ranked by Likelihood

| Rank | Root Cause | Correlates With | Evidence Expected |
|------|-----------|-----------------|-------------------|
| **#1** | Unfrozen system clock in timestamp test | Timezone specifically; CI uses UTC by default | `datetime.now()` / `time.time()` calls without `freezegun`/fixture patching |
| **#2** | Race condition in async worker poll | Machine load; `sleep(2)` may be shorter than real processing time | Worker task completion check is based on fixed sleep rather than actual completion signal |
| **#3** | Non-deterministic dict/set iteration | Only matters if comparison assumes order; Python 3.7+ dicts preserve insertion order but sets don't | Using `set` where list is needed, then comparing index-by-index |

---

```python
"""
reproduce_flakiness.py — Forces all three flaky patterns deterministically.
Run with: pytest reproduce_flakiness.py -v --count=50
(or install pytest-xdist + pytest-count for stress running.)

import os
import asyncio
import hashlib
from datetime import datetime, timezone, timedelta

# ──────────────────────────────────────────────
# Test 1: Timestamp Formatting (timezone)
# ──────────────────────────────────────────────
def test_timestamp_formatting_with_timezone_drift():
    """Reproduces: formatted timestamp differs when local tz != UTC."""
    # Simulate dev machine in a non-UTC timezone
    now_utc = datetime.now(timezone.utc)
    now_local = now_utc.astimezone()  # ← depends on TZ env var

    # If the production code does this:
    def format_timestamp(dt=None):
        dt = dt or datetime.now()
        return dt.strftime("%Y-%m-%d %H:%M:%S")

    expected = "2026-08-26 00:00:00"  # hardcoded or derived from known value
    actual = format_timestamp(now_local)

    # Flaky: when local tz offset pushes hour/day boundary vs CI's UTC,
    # the formatted string diverges by ±1 second or more at boundaries.
    assert actual == expected, (
        f"timezone mismatch: utc={now_utc.isoformat()} "
        f"local={now_local.isoformat()} formatted={actual}"
    )


# ──────────────────────────────────────────────
# Test 2: Async Worker Polling (race condition)
# ──────────────────────────────────────────────
async def _slow_worker(delay: float) -> str:
    await asyncio.sleep(delay)
    return "done"


async def test_async_worker_polls_with_fixed_sleep():
    """Reproduces: sleep(2) < actual worker duration under load."""
    # Simulate machine load inflating the real delay to > 2s
    actual_delay = 2.0 + hash(os.urandom(4).hex()) % 1500 / 1000  # 2.0 – 3.5s jitter

    task = asyncio.create_task(_slow_worker(actual_delay))

    # BUG: asserts after exactly 2s, but worker needs ~actual_delay seconds
    await asyncio.sleep(2)

    try:
        result = task.result()  # ← raises InvalidStateError if not done yet
        assert result == "done", f"expected 'done', got {result!r}"
    except asyncio.InvalidStateError as exc:
        pytest.fail(f"worker not complete after sleep(2); needed ~{actual_delay:.1f}s: {exc}")


# ──────────────────────────────────────────────
# Test 3: Set Iteration Order (non-determinism)
# ──────────────────────────────────────────────
@pytest.fixture(
    params=[
        "alpha", "beta", "gamma", "delta",
        "epsilon", "zeta", "eta", "theta",
    ]
)
def fixture_ids(request):
    # BUG: returns a SET instead of a list → iteration order is randomized
    # per-process, so test parameter values come back shuffled.
    return {"alpha", "beta", "gamma", "delta", "epsilon", "zeta", "eta", "theta"}


def test_relying_on_set_iteration_order(fixture_ids):
    """Reproduces: iterating a set produces different permutations across runs."""
    items = [item for item in fixture_ids]  # set → list conversion is UNORDERED

    # BUG: comparing positionally assumes deterministic order
    expected_order = ["alpha", "beta", "gamma", "delta"]
    actual = items[: len(expected_order)]

    assert actual == expected_order, (
        f"set iteration gave {actual} but expected {expected_order}; "
        f"hash randomization (PYTHONHASHSEED={os.environ.get('PYTHONHASHSEED')}) caused drift"
    )
```

## Phase 3: Deterministic Fixes

### Fix 1 — Timestamp Assertion: Freeze Time + Assert Relative

**Root cause:** The test formats `datetime.now()` against a hardcoded absolute timestamp. When `TZ` differs from CI's UTC (or during DST transitions / second boundaries), the wall-clock time shifts.

**Fix:** Never assert an absolute formatted timestamp. Assert relative properties, or freeze time.

```python
# FIXED version 1a — Freeze time at a known instant
from freezegun import freeze_time

@freeze_time("2026-08-26T12:00:00+00:00")
def test_timestamp_formatting():
    from myapp.utils import format_timestamp
    result = format_timestamp()
    assert result == "2026-08-26 12:00:00", f"got {result}"


# FIXED version 1b — Assert structure, not absolute value
def test_timestamp_formatting_structure():
    from myapp.utils import format_timestamp
    result = format_timestamp()
    # Pattern check only — timezone-independent
    import re
    assert re.fullmatch(r"\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}", result)

    # Or: verify it's within a bounded window
    parsed = datetime.strptime(result, "%Y-%m-%d %H:%M:%S")
    assert abs((parsed - datetime.now()).total_seconds()) < 1
```

| Why it works | What breaks if ignored |
|-------------|----------------------|
| `freezegun` patches `time.time()` at C level — every process gets identical wall clock | Absolute timestamps drift with TZ, DST, NTP adjustments, leap seconds |
| Structure assertions are invariant under any timezone | Hardcoded `"HH:MM:SS"` strings produce timezone-specific results |

### Fix 2 — Async Worker Polling: Replace Sleep with Proper Wait

**Root cause:** `await asyncio.sleep(2)` is a bare delay with no knowledge of worker progress. Under load, the event loop can't run the worker task promptly, making the effective wait longer.

**Fix:** Use an `asyncio.Event`, `queue.Queue`, `asyncio.gather`, or `asyncio.wait_for` — i.e., a *completion signal* instead of a guess.

```python
# FIXED — Signal-based completion
class AsyncWorker:
    def __init__(self):
        self._ready = asyncio.Event()

    async def start(self, delay: float) -> str:
        await asyncio.sleep(delay)       # simulated work
        result = "done"
        self._ready.set()                # ← signals completion
        return result

    @property
    def ready(self) -> asyncio.Event:
        return self._ready


@pytest.mark.asyncio
async def test_async_worker_completes_within_deadline():
    worker = AsyncWorker()
    task = asyncio.create_task(worker.start(delay=1.5))

    # OPTION A — wait with timeout (best for "will it finish?")
    done, pending = await asyncio.wait(
        [task],
        timeout=10.0,          # generous deadline, not guess
    )
    assert pending == set(), f"worker timed out after 10s"
    assert task.result() == "done"


# OPTION B — await the event directly (best for "wait until ready")
@pytest.mark.asyncio
async def test_async_worker_signals_ready():
    worker = AsyncWorker()
    asyncio.create_task(worker.start(delay=1.5))

    # Waits indefinitely (or use asyncio.wait_for for a cap)
    await asyncio.wait_for(worker.ready.wait(), timeout=10.0)
    assert worker._ready.is_set()
    assert worker._event is None or worker._event.is_set()


# OPTION C — if you must keep the simple pattern, add a retry loop
@pytest.mark.asyncio
async def test_async_worker_retry_until_done():
    task = asyncio.create_task(_slow_worker(delay=0.5))
    max_attempts = 20
    for _ in range(max_attempts):
        if task.done():
            break
        await asyncio.sleep(0.1)   # small step, not a giant guess
    else:
        pytest.fail(f"worker did not complete after {max_attempts * 0.1}s")
    assert task.result() == "done"
```

| Why it works | What breaks if ignored |
|-------------|----------------------|
| `asyncio.wait` / `Event` responds to *actual* completion, not elapsed wall time | Fixed `sleep(N)` is blind to event-loop starvation under CPU/memory pressure |
| Timeout provides a safety net (fail-fast instead of silent data loss) | Increasing the sleep number merely delays the failure; doesn't fix the race |
| Retry loop (Option C) is a minimal incremental fix | No approach using `asyncio.sleep` alone can be deterministic |

### Fix 3 — Fixture Set Iteration Order: Use Ordered List

**Root cause:** `{"alpha", "beta", ...}` creates a Python `set`. Converting a set to a list via `[item for item in my_set]` yields **implementation-defined, hash-randomized** order. Each Python process seeds `PYTHONHASHSEED` differently unless `PYTHONHASHSEED=0` (or a fixed integer) is set.

**Fix:** Use a `list` in the fixture definition, or sort before comparing.

```python
# FIXED v3a — List fixture (simplest, preserves definition order)
@pytest.fixture(
    params=[                              # ← LIST, not SET
        "alpha", "beta", "gamma", "delta",
        "epsilon", "zeta", "eta", "theta",
    ],
    ids=["alpha", "beta", "gamma", "delta",
         "epsilon", "zeta", "eta", "theta"],
)
def fixture_ids(request):
    return request.param                 # returns individual strings, one per sub-test


def test_relying_on_deterministic_order(fixture_ids):
    from myapp.processor import group_items
    # Now fixture_ids is guaranteed to arrive in definition order
    assert fixture_ids in ("alpha", "beta", "gamma", "delta",
                           "epsilon", "zeta", "eta", "theta")


# FIXED v3b — Sort before positional comparison
def test_reorder_is_consistent(set_fixture):
    items = sorted(set_fixture)   # deterministic regardless of input order
    assert items == ["alpha", "beta", "delta", "epsilon", "eta", "gamma", "theta", "zeta"]


# FIXED v3c — Assert membership, not position
def test_contains_expected_items(set_fixture):
    assert set_fixture == {"alpha", "beta", "gamma", "delta",
                           "epsilon", "zeta", "eta", "theta"}
    # Set comparison ignores order entirely.
```

| Why it works | What breaks if ignored |
|-------------|----------------------|
| Lists in fixtures iterate in definition order, always | `PYTHONHASHSEED` is randomized by default (Python 3.3+ security feature) — every process is a new permutation |
| Sorting before comparison imposes determinism post-hoc | Even with `PYTHONHASHSEED=0`, the C-level set implementation could change in future Python versions |
| Set-vs-set comparison eliminates ordering concern | Code that indexes into a converted list (`items[0]`) will silently get wrong values |

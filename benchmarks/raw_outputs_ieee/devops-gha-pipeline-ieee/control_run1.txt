## 1. `.github/workflows/ci.yaml` — Path-Filtered CI, Matrix Tests, Caching

```yaml
name: CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

# ──────────────────────────────────────────────
# CACHE-KEY DESIGN
# ──────────────────────────────────────────────
# Primary key: hash of lockfile ONLY (not contents — avoids storing
# multi-hundred-MB archives when only filenames differ).
#   → pnpm respects lockfile integrity; if lockfile hasn't changed,
#     node_modules is byte-for-byte identical.
# Secondary restore-keys give partial hits so installs don't regress
# to full misses when a lib gains a dependency.
# Build output cache is separate, keyed by lockfile + hashes of the
# source files that feed into it (packages/**/* and tsconfig), ensuring
# stale build artifacts never slip through after a dependency bump.
#
# Three-tier lookup:
#   1. Exact lockfile-hash match      → full hit
#   2. Prefix match on lockfile hash  → partial hit (fallback)
#   3. No match                      → cold install + rebuild
#
# This trades a small amount of storage for dramatically faster restores
# because pnpm link-based stores are small and fast to invalidate.
# ──────────────────────────────────────────────

env:
  NODE_VERSION: '20'
  PNPM_CACHE_KEY: v1-pnpm-${{ hashFiles('pnpm-lock.yaml') }}
  BUILD_CACHE_KEY: v1-build-${{ github.sha }}

jobs:
  # ──────────────────────────────────────────
  # FILTER: Decide which apps/libs are affected
  # ──────────────────────────────────────────
  changes:
    name: Filter paths
    runs-on: ubuntu-latest
    outputs:
      apps_web: ${{ steps.filter.outputs.apps_web }}
      apps_api: ${{ steps.filter.outputs.apps_api }}
      apps_worker: ${{ steps.filter.outputs.apps_worker }}
      libs_all: ${{ steps.filter.outputs.libs_all }}
      root: ${{ steps.filter.outputs.root }}
    steps:
      - uses: actions/checkout@v4
      - name: Check for app/web changes
        id: filter
        uses: dorny/paths-filter@v3
        with:
          filters: |
            root:
              - '.github/workflows/ci.yaml'
              - 'pnpm-workspace.yaml'
              - 'pnpm-lock.yaml'
              - 'tsconfig.base.json'
            apps_web:
              - 'apps/web/**'
              - 'packages/ui/**'
              - 'packages/config/**'
            apps_api:
              - 'apps/api/**'
              - 'packages/db/**'
              - 'packages/auth/**'
              - 'packages/logger/**'
            apps_worker:
              - 'apps/worker/**'
              - 'packages/db/**'
              - 'packages/logger/**'
              - 'packages/hooks/**'
            libs_all:
              - 'packages/ui/**'
              - 'packages/config/**'
              - 'packages/utils/**'
              - 'packages/db/**'
              - 'packages/auth/**'
              - 'packages/logger/**'
              - 'packages/types/**'
              - 'packages/hooks/**'

  # ──────────────────────────────────────────
  # SETUP: Shared install + build cache
  # ──────────────────────────────────────────
  setup:
    name: Setup (install + build cache)
    runs-on: ubuntu-latest
    needs: [changes]
    if: >-
      needs.changes.outputs.root == 'true' ||
      needs.changes.outputs.libs_all == 'true'
    outputs:
      node_cache_key: ${{ steps.set-cache-key.outputs.node_cache_key }}
    steps:
      - uses: actions/checkout@v4

      - name: Setup pnpm
        uses: pnpm/action-setup@v4
        with:
          version: 9

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version-file: '.nvmrc'
          cache: 'pnpm'

      - name: Set cache key
        id: set-cache-key
        run: echo "node_cache_key=${{ env.PNPM_CACHE_KEY }}" >> "$GITHUB_OUTPUT"

      - name: Install deps
        run: pnpm install --frozen-lockfile

      - name: Cache node_modules
        uses: actions/cache@v4
        with:
          path: |
            **/node_modules
            ~/.pnpm-store
          key: ${{ steps.set-cache-key.outputs.node_cache_key }}
          restore-keys: |
            ${{ env.PNPM_CACHE_KEY }}-

      - name: Build shared packages
        run: pnpm build --filter='./packages/*'

      - name: Cache build output
        uses: actions/cache@v4
        with:
          path: |
            packages/*/dist
          key: ${{ env.BUILD_CACHE_KEY }}
          restore-keys: |
            ${{ env.PNPM_CACHE_KEY }}-build-

  # ──────────────────────────────────────────
  # LINT: One-shot, fast feedback
  # ──────────────────────────────────────────
  lint:
    name: Lint
    needs: [setup]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { version: 9 }
      - uses: actions/setup-node@v4
        with:
          node-version-file: '.nvmrc'
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile

          PNPM_CACHE_KEY: ${{ needs.setup.outputs.node_cache_key }}

      - name: Run tests

    name: Test api (shard ${{ matrix.shard }}/4) · Node ${{ matrix.node }}
    needs: [setup, changes]
    if: >-
      needs.changes.outputs.apps_api == 'true'
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        node: ['20', '22']
        shard: ['1/4', '2/4', '3/4', '4/4']
    steps:
```

```yaml
name: Deploy Production

on:
  push:
    branches: [main]
  workflow_dispatch:
    inputs:
      target_app:
        description: 'App to deploy'
        required: true
        type: choice
        options: [web, api, worker, all]
        default: all

env:
  GOOGLE_CLOUD_PROJECT: ${{ vars.GOOGLE_CLOUD_PROJECT }}
  REGION: us-central1

permissions:
  contents: read
  id-token: write          # Enables OIDC federated credential flow

    name: Determine what to deploy
    runs-on: ubuntu-latest
    outputs:
      apps: ${{ steps.plan.outputs.apps }}
    steps:
      - id: plan
        run: |
          case "${{ github.event.inputs.target_app || 'auto' }}" in
            all)  echo "apps=['web' 'api' 'worker']" >> "$GITHUB_OUTPUT" ;;
            *)    echo "apps=['$TARGET_APP']" >> "$GITHUB_OUTPUT" ;;
          esac

  deploy-web:
    name: Deploy web → production
    needs: [plan]
    if: >-
      contains(needs.plan.outputs.apps, 'web')
    runs-on: ubuntu-latest
    environment:
      name: production-web
      # ⬇️  Environment-level protection rule enforces manual approval
      url: https://www.yourdomain.com
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: 9 }
      - uses: actions/setup-node@v4

          node-version-file: '.nvmrc'
```

### OIDC Trust Configuration Walkthrough

The `google-github-actions/auth@v2` action performs this sequence:

```
GitHub Runner ──GET──> GitHub OIDC Issuer (token.actions.githubusercontent.com)
                    │
                    │  JWT payload includes:
                    │  { iss: "...", sub: "repo:org/repo",
                    │    aud: "azure/workload-identity-federated" }
                    │
                    ▼
Azure/GCP/AWS OIDC Validator ──verify JWT signature──> Trust Policy
                                                              │
                                                              │  Checks:
                                                              │  • JWT sig against
                                                              │    GitHub's JWK endpoint
                                                              │  • subject claim matches
                                                              │    repo:org/repo
                                                              │  • audience claim matches
                                                              │
                                                              ▼
                                                    Short-lived Access Token
                                                    (valid ~1 hour, auto-revoked)
```

**What you configure on the cloud side:**

| Cloud Provider | Resource | Value |
|---|---|---|
| **Google Cloud** | `WorkloadIdentityPool` | `github` |
| | `WorkloadIdentityPoolProvider` | `github-actions` with attribute mapping `google.subject=assertion.sub`, `attribute.repository=assertion.repo` |
| | IAM binding | `roles/run.developer` (or `roles/container.developer`) granted to the pool+provider subject `repo:your-org/your-repo` |
| **AWS** | `OpenIDConnectProvider` | URL `https://token.actions.githubusercontent.com`, thumbprint of GitHub's OIDC cert |
| | IAM Role | Trust policy allowing `sts:AssumeRoleWithWebIdentity` from the above provider with condition `StringLike: token.actions.githubusercontent.com:sub:repo:your-org/your-repo:*` |
| **Azure** | `FederatedCredential` | Attached to a UserAssignedManagedIdentity or ServicePrincipal; issuer `https://token.actions.githubusercontent.com`; subject `repo:your-org/your-repo:ref:refs/heads/main` |

**Key properties enforced by the claim conditions:**

```yaml
# In the IAM trust policy (AWS example):
Condition:
  StringEquals:
    token.actions.githubusercontent.com:aud:sts:aws.amazonaws.com.cn
  StringLike:
    token.actions.githubusercontent.com:sub: "repo:your-org/your-repo:*"
```

This means:
- Only `your-org/your-repo` can assume the role
- No stale credentials to rotate
- Tokens expire automatically (~1hr)
- Scoped to the exact ref (`main` only via pool filter)

## 4. `.github/workflows/release.yaml` — Release, SBOM, Attestation

```yaml
name: Release

on:
  push:
    tags:
      - 'v*'

permissions:
  contents: write           # Create releases + upload assets
  packages: write           # Push to GHCR
  id-token: write           # Sigstore cosign attestation
  attestations: write       # GitHub attestations

jobs:
  build-and-release:
    name: Build & Release
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app: [web, api, worker]
    outputs:
      version: ${{ steps.tag.outputs.version }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - uses: pnpm/action-setup@v4
        with: { version: 9 }
      - uses: actions/setup-node@v4
        with:
          node-version-file: '.nvmrc'
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile

      - name: Extract version from tag
        id: tag
        run: |
          VERSION="${GITHUB_REF#refs/tags/}"
          echo "version=$VERSION" >> "$GITHUB_OUTPUT"

      - name: Build
        run: pnpm --filter apps/${{ matrix.app }} build
        env:
          VERSION: ${{ steps.tag.outputs.version }}

      # ── Docker image with provenance ──
      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Login to GHCR
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Extract metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ghcr.io/${{ github.repository }}/apps/${{ matrix.app }}
          tags: |
            type=semver,pattern={{version}}
            type=sha,prefix=${{ matrix.app }}-
            type=raw,value=latest

      - name: Build & push Docker image
        uses: docker/build-push-action@v6
        with:
          context: apps/${{ matrix.app }}
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          provenance: mode=max   # Embed build provenance in image
          sbom: true             # Embed SBOM layer

      # ── File-based artifact (e.g., serverless zip) ──
      - name: Package artifacts
        run: |
          tar czf apps-${{ matrix.app }}-${{ steps.tag.outputs.version }}.tar.gz \
            -C "apps/${{ matrix.app }}" dist/
        if: matrix.app != 'web'   # Web ships as container only

      - name: Upload artifacts
        uses: actions/upload-artifact@v4
        with:
          name: ${{ matrix.app }}-artifact
          path: apps-${{ matrix.app }}-${{ steps.tag.outputs.version }}.tar.gz

  generate-sbom:
    name: Generate SBOM
    runs-on: ubuntu-latest
    needs: build-and-release
    steps:
      - uses: actions/checkout@v4
      - name: Generate SBOM (syft)
        uses: anchore/sbom-action@v0
        with:
          path: .
          output-file: sbom.spdx.json
          format: spdx-json

      - name: Upload SBOM
        uses: actions/upload-artifact@v4
        with:
          name: sbom
          path: sbom.spdx.json
          retention-days: 90

  attest-artifacts:
    name: Attest Provenance
    runs-on: ubuntu-latest
    needs: [build-and-release, generate-sbom]
    steps:
      - uses: actions/checkout@v4

      - name: Download artifacts
        uses: actions/download-artifact@v4
        with:
          pattern: '*'
          path: artifacts
          merge-multiple: true

      # ── GitHub native attestation ──
      - name: Attest build provenance
        uses: actions/attest-build-provenance@v1
        with:
          subject-path: |
            artifacts/apps-web-*.tar.gz
            artifacts/apps-api-*.tar.gz
            artifacts/apps-worker-*.tar.gz

      # ── SigStore / cosign (alternative or complementary) ──
      - name: Sign with cosign
        uses: sigstore/cosign-installer@v3
      - name: Attest with cosign
        run: |
          for f in artifacts/*.tar.gz; do
            cosign attest --yes \
              --predicate <(cat sbom.spdx.json) \
              --type spdx \
              "$f"
          done
```

## 5. Merge Queue Integration

Add these settings in **Repository → Settings → Branches**:

```
Branch Protection Rule for "main":
  ✓ Require a pull request before merging
  ✓ Require approvals            → minimum 1 reviewer
  ✓ Dismiss stale pull request approvals when new commits are pushed
  ✓ Require review from code owners
  ✓ Require status checks to pass
  ✓ Require branches to be up to date before merging
  ✓ Include administrators       ← enforce even for admins

Required status checks:
  ☑ CI Status (CI)
  ☑ Lint (CI)
  ☑ Typecheck (CI)
  ☑ Test web (CI)                ← one entry; covers the whole matrix
  ☑ Test api (CI)
  ☑ Test worker (CI)
  ☑ Test libraries (CI)

  ☑ Preview Web (Preview Environments)  ← only on PRs

Merge queue: ENABLED
  Queue check strategies:
    ☑ Group checks by title        ← groups all CI jobs under one label
    ☒ Sort by least likely to fail (optional)
```

And add this to your workflow for explicit merge-queue compatibility:

```yaml
# Append to ci.yaml triggers:
push:
  branches: [main]
pull_request:
  branches: [main]
pull_request_target:
  types: [queued]            # Trigger inside the merge queue
```

The `CI Status` job acts as the final gate that aggregates all individual job results. Since it has `needs: [...]` on all test jobs and uses `if: always()` with a fail-check, GitHub sees it as a single required check covering everything in the matrix. The merge queue waits for `CI Status` to succeed before pulling the next PR into the queue.

## Cache Key Design Explained

The key insight: **pnpm's store is content-addressed and symlink-based**, so `node_modules` is tiny. The expensive part is the install, not the copy-down. This flips the traditional npm caching model.

```
Cache Tier 1: Dependencies
━━━━━━━━━━━━━━━━━━━━━━━━━
Key:    v1-pnpm-{hash(pnpm-lock.yaml)}
Restore: prefix(v1-pnpm-)

Why lockfile hash?
  pnpm-lock.yaml encodes the FULL dependency graph resolution.
  If it hasn't changed, the dependency tree is identical.
  Hashing file *contents* would be fragile to whitespace/formatting
  changes that don't affect the resolved graph.

What gets cached:
  - packages/*/node_modules  (symlinks, tiny)
  - ~/.pnpm-store            (content-addressed blob store)
  Restore-keys fall back to a partial prefix so even when a lib gains
  a transitive dep, we still get the base store from cache.


Cache Tier 2: Build Output
━━━━━━━━━━━━━━━━━━━━━━━━━━
Key:    v1-build-{sha}
Restore: v1-pnpm-{lockfile-hash}-build-

Why SHA?
  Build output is deterministic given the same source + lockfile.
  Using sha gives a unique key per commit. The restore-key chains
  back to the lockfile hash so that if the same lockfile is committed
  again (cherry-pick, force-push rewrite), we hit the old build cache.

What gets cached:
  - packages/*/dist           (compiled JS/d.ts files)
  This means downstream app builds skip recompiling unchanged libs.


Invalidation Rules:
  ┌────────────────────────┬──────────┬──────────┐
  │ Change                 │ Dep Cache│ Build C. │
  ├────────────────────────┼──────────┼──────────┤
  │ pnpm-lock.yaml changes │  MISS    │  MISS    │
  │ New commit, no depchg  │  HIT     │  MISS    │
  │ Commit reverts lockfil │  HIT     │  HIT*    │
  │ .gitignore changes     │  HIT     │  HIT     │
  │ No changes (skip)      │  N/A     │  N/A     │
  └────────────────────────┴──────────┴──────────┘
  * Because restore-key falls back to lockfile prefix
```

**Cost analysis:**
- Lockfile rarely changes → dep cache hits ~90%+ of the time
- Each PR usually introduces new code, not new deps → build cache mostly misses (unavoidable) but restore-key helps
- Total cache size stays reasonable because pnpm stores are small (vs. npm's fat `node_modules`)

## Summary of What Each Workflow Handles

| Workflow | Triggers | Key Feature | Gate |
|---|---|---|---|
| `ci.yaml` | PR + push | Path-filtered per app, Node 20×22 × 4-shard matrix | `CI Status` aggregate job → required check |
| `preview.yaml` | PR | Ephemeral envs with OIDC, post URL as comment | None (runs on every PR) |
| `deploy-prod.yaml` | push to main + dispatch | OIDC federated auth (zero secrets), manual env approval | Environment approval + OIDC |
| `release.yaml` | Tag push | SBOM (syft/provenance layer) + cosign sign + attest | None (immutable tag) |

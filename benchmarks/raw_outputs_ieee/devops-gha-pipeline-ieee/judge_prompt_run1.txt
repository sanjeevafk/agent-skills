name: CI

on:
  push:
    branches: [main]
  pull_request_target:
    types: [opened, synchronize]
    paths-ignore:
      - "**/*.md"
      - "docs/**"

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

# ── Paths filter dispatch (runs per-app) ──────────────────────────
jobs:
  changes:
    runs-on: ubuntu-latest
    outputs:
      web-app: ${{ steps.filter.outputs.web-app }}
      api-server: ${{ steps.filter.outputs.api-server }}
      mobile-bff: ${{ steps.filter.outputs.mobile-bff }}
      libs: ${{ steps.filter.outputs.libs }}
    steps:
      - uses: dorny/paths-filter@v3
        id: filter
        with:
          base: main
          filters: |
            web-app:
              - 'apps/web-app/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - 'libs/**'
            api-server:
              - 'apps/api-server/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - 'libs/**'
            mobile-bff:
              - 'apps/mobile-bff/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - 'libs/**'
            libs:
              - 'libs/**'
              - 'package.json'
              - 'pnpm-lock.yaml'

  # ── Cache warm / restore (shared across all jobs) ────────────────
  deps-cache:
    runs-on: ubuntu-latest
    outputs:
      cache-hit: ${{ steps.restore-deps.outputs.cache-hit }}
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: "9" }

      - name: Setup Node 20
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "pnpm"
          cache-dependency-path: pnpm-lock.yaml

      # ── CACHE KEY DESIGN ─────────────────────────────────────────
      # Key = deps-<platform>-<lockfile-sha>
      #   • pnpm-action-setup writes the resolved store dir; we key off
      #     the exact lockfile hash so any dependency change invalidates.
      #   • We DO NOT embed branch or commit — the same lockfile always
      #     maps to the same cache entry regardless of ref.
      #   • Falls back to deps-<platform>-full when no lockfile match
      #     exists (first install on fresh repo).
      # ────────────────────────────────────────────────────────────────
      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            deps-${{ runner.os }}-

      - name: Install (only if cache miss)
        if: steps.restore-deps.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

  build-cache:
    needs: [changes, deps-cache]
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
    outputs:
      web-app-hit: ${{ steps.restore-web.outputs.cache-hit }}
      api-hit: ${{ steps.restore-api.outputs.cache-hit }}
      bff-hit: ${{ steps.restore-bff.outputs.cache-hit }}
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: "9" }

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "pnpm"

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      # Per-app build cache: keys on lockfile + src-hash of each app
      - id: restore-web
        uses: actions/cache/restore@v4
        with:
          path: apps/web-app/.next
          key: build-web-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml', 'apps/web-app/src/**') }}

      - id: restore-api
        uses: actions/cache/restore@v4
        with:
          path: apps/api-server/dist
          key: build-api-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml', 'apps/api-server/src/**') }}

      - id: restore-bff
        uses: actions/cache/restore@v4
        with:
          path: apps/mobile-bff/build
          key: build-bff-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml', 'apps/mobile-bff/src/**') }}

      # Build all changed targets
      - name: Build libs
        if: ${{ needs.changes.outputs.libs == 'true' }}
        run: pnpm --filter "@myorg/*" run build --recursive

      - name: Build web-app
        if: ${{ always() && (needs.changes.outputs.web-app == 'true' || needs.changes.outputs.libs == 'true') }}
        run: pnpm --filter web-app run build
        env: { NODE_OPTIONS: "--max-old-space-size=4096" }

      - name: Build api-server
        if: ${{ always() && (needs.changes.outputs.api-server == 'true' || needs.changes.outputs.libs == 'true') }}
        run: pnpm --filter api-server run build
        env: { NODE_OPTIONS: "--max-old-space-size=4096" }

      - name: Build mobile-bff
        if: ${{ always() && (needs.changes.outputs.mobile-bff == 'true' || needs.changes.outputs.libs == 'true') }}
        run: pnpm --filter mobile-bff run build
        env: { NODE_OPTIONS: "--max-old-space-size=4096" }

      - uses: actions/upload-artifact@v4
        with:
          name: build-caches
          path: |
            apps/web-app/.next
            apps/api-server/dist
            apps/mobile-bff/build
          retention-days: 3

  # ── Lint (single pass for the whole repo) ────────────────────────
  lint:
    runs-on: ubuntu-latest
    needs: deps-cache
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { version: "9" }
      - uses: actions/setup-node@v4
        with: { node-version: 20, cache: "pnpm" }

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      - run: pnpm run lint

  # ── Test Matrix: Node 20/22 x shard-4 ───────────────────────────
  test:
    needs: [changes, deps-cache, build-cache]
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        node-version: [20, 22]
        shard: [1/4, 2/4, 3/4, 4/4]
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: "9" }

      - uses: actions/setup-node@v4
        with: { node-version: ${{ matrix.node-version }}, cache: "pnpm" }

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile

        env:
          CI: "true"
          NODE_ENV: test

      - name: Upload test results

  # ── Type-check (fast, runs once) ─────────────────────────────────
  typecheck:
    runs-on: ubuntu-latest

      - uses: actions/checkout@v4

  # ── Security scan (container/image) ──────────────────────────────
  security-scan:
    runs-on: ubuntu-latest

      - uses: actions/checkout@v4

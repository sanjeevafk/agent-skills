## 1. `.github/workflows/ci.yml` — Path-filtered CI per app

```yaml
name: CI (lint / test / build)

on:
  push:
    branches: [main, 'release/**']
  pull_request_target:
    types: [opened, synchronize]

permissions:
  contents: read
  # Needed for merge queue compatibility
  checks: write
  pull-requests: read

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: ${{ github.event_name == 'pull_request' }}

env:
  PNPM_CACHE_KEY: v1-pnpm-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
  NODE_MODULES_CACHE_KEY: v1-node-modules-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

jobs:
  # ── Job 0: Detect which apps/libs changed ──────────────────────────────
  path-filter:
    runs-on: ubuntu-latest
    outputs:
      apps: ${{ steps.filter.outputs.apps }}
      libs: ${{ steps.filter.outputs.libs }}
      all_changed: ${{ steps.filter.outputs.all_changed }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          fetch-depth: 0

      - name: Filter paths
        id: filter
        uses: dorny/paths-filter@v3
        with:
          filters: |
            apps:
              - 'apps/frontend/src/**'
              - 'apps/frontend/package.json'
              - 'apps/api/src/**'
              - 'apps/api/package.json'
              - 'apps/mobile/src/**'
              - 'apps/mobile/package.json'
            libs:
              - 'libs/shared/**'
              - 'libs/ui/**'
              - 'libs/config/**'
              - 'libs/database/**'
              - 'libs/auth/**'
              - 'libs/types/**'
              - 'libs/logging/**'
              - 'libs/utils/**'

  # ── Shared setup: runtime + deps ───────────────────────────────────────
  setup:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        node-version: [20, 22]
    outputs:
      cache-hit-deps: ${{ steps.cache-deps.outputs.cache-hit }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}

      - name: Setup pnpm
        uses: pnpm/action-setup@v4
        with:
          version: latest

      - name: Setup Node.js ${{ matrix.node-version }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      # Cache pnpm store keyed on lockfile + os — shared across all matrix jobs
      - name: Cache pnpm store (Node ${{ matrix.node-version }})
        id: cache-deps
        uses: actions/cache@v4
        with:
          path: ~/.pnpm-store
          key: ${{ env.PNPM_CACHE_KEY }}-${{ matrix.node-version }}
          restore-keys: |
            ${{ env.PNPM_CACHE_KEY }}-

      # Cache node_modules inside the repo (monorepo workspace)
      - name: Cache workspace node_modules
        uses: actions/cache@v4
        with:
          path: |
            node_modules
            */node_modules
          key: ${{ env.NODE_MODULES_CACHE_KEY }}-${{ matrix.node-version }}-${{ runner.arch }}
          restore-keys: |
            ${{ env.NODE_MODULES_CACHE_KEY }}-${{ matrix.node-version }}-
            ${{ env.NODE_MODULES_CACHE_KEY }}-

  # ── Job 1: Lint (all repos, once — not per matrix) ────────────────────
  lint:
    runs-on: ubuntu-latest
    needs: [setup]
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}

      - uses: pnpm/action-setup@v4
        with:
          version: latest

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Restore dep cache
        id: cache-deps
        uses: actions/cache@v4
        with:
          path: ~/.pnpm-store
          key: ${{ env.PNPM_CACHE_KEY }}-20
          restore-keys: |
            ${{ env.PNPM_CACHE_KEY }}-

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Run root lint
        run: pnpm -r --if-present lint

  # ── Jobs 2–4: Per-app test matrix + build ─────────────────────────────
  test-build-app:
    needs: [path-filter, setup]
    if: >-
      needs.path-filter.outputs.apps != '' &&
      contains(needs.path-filter.outputs.apps, 'frontend')
    uses: ./.github/workflows/_reusable-test-build.yml
    with:
      app: frontend
      node-versions: "20,22"
      shard-count: 4
      affected-libs: "${{ needs.path-filter.outputs.libs }}"

  test-build-api:
    needs: [path-filter, setup]
    if: >-
      needs.path-filter.outputs.apps != '' &&
      contains(needs.path-filter.outputs.apps, 'api')
    uses: ./.github/workflows/_reusable-test-build.yml
    with:
      app: api
      node-versions: "20,22"
      shard-count: 4
      affected-libs: "${{ needs.path-filter.outputs.libs }}"

  test-build-mobile:
    needs: [path-filter, setup]
    if: >-
      needs.path-filter.outputs.apps != '' &&
      contains(needs.path-filter.outputs.apps, 'mobile')
    uses: ./.github/workflows/_reusable-test-build.yml
    with:
      app: mobile
      node-versions: "20,22"
      shard-count: 4
      affected-libs: "${{ needs.path-filter.outputs.libs }}"

  # ── Fallback: lib-only changes still need full test coverage on libs ──
  test-build-libs:
    needs: [path-filter, setup]
    if: >-
      always() &&
      needs.path-filter.outputs.apps == '' &&
      needs.path-filter.outputs.libs != ''
    strategy:
      matrix:
        node-version: [20, 22]
        shard: [1, 2, 3, 4]
    uses: ./.github/workflows/_reusable-test-build.yml
    with:
      app: libs
      node-versions: "${{ matrix.node-version }}"
      shard-count: ${{ matrix.shard }}
      affected-libs: "${{ needs.path-filter.outputs.libs }}"

  # ── Merge queue required-status gate ───────────────────────────────────
  # The merge queue requires that specific status checks pass before
  # squashing merges onto main. We declare them here explicitly so
  # GitHub's merge-queue feature recognizes them.
  required-status-checks:
    runs-on: ubuntu-latest
    needs: [lint, test-build-app, test-build-api, test-build-mobile]
    if: failure()
    steps:
      - run: echo "CI has failures — merge is blocked"
        shell: bash
```

### `_reusable-test-build.yml` (called by each per-app job above)

```yaml
# .github/workflows/_reusable-test-build.yml
name: Reusable Test & Build for App/Libs

on:
  workflow_call:
    inputs:
      app:
        required: true
        type: string
      node-versions:
        required: true
        type: string
      shard-count:
        required: true
        type: number
      affected-libs:
        required: false
        type: string
        default: ""

jobs:
  build-and-test:
    name: >-
      ${{ inputs.app }}
      (node-${{ matrix.node-version }},
       shard-${{ matrix.shard }}/${{ inputs.shard-count }})
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        node-version: ${{ fromJson(format('["{0}"]', replace(inputs.node-versions, ",", '", \"')))}}
        shard: ${{ range(1, inputs.shard-count + 1) }}

    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}

      - uses: pnpm/action-setup@v4
        with:
          version: latest

      - uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: 'pnpm'

      - name: Restore dep cache
        uses: actions/cache@v4
        with:
          path: ~/.pnpm-store
          key: v1-pnpm-${{ runner.os }}-${{ matrix.node-version }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            v1-pnpm-${{ runner.os }}-${{ matrix.node-version }}-
            v1-pnpm-${{ runner.os }}-

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build affected libs only (if applicable)
        if: inputs.affected-libs != ''
        run: |
          LIBS_CHANGED=$(echo '${{ inputs.affected-libs }}' | tr ',' '\n' | sed 's/^/*\//' | paste -sd',' -)
          # Example: only rebuild libs whose code actually changed
          pnpm --filter @myorg/$LIBS_CHANGED build || true

      - name: Build app
        working-directory: apps/${{ inputs.app }}
        run: pnpm build

      - name: Run tests (sharded)
        working-directory: apps/${{ inputs.app }}
        env:
          NODE_OPTIONS: "--max-old-space-size=4096"
          TEST_SHARD: "${{ matrix.shard }}/${{ inputs.shard-count }}"
        run: pnpm test -- --shard=${TEST_SHARD}

      # Upload test results for JUnit reporters / Codecov
      - name: Upload test artifacts
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: >-
            test-results-${{ inputs.app }}-node${{ matrix.node-version }}-shard${{ matrix.shard }}
          path: apps/${{ inputs.app }}/test-results/
          retention-days: 7
```

## 2. `preview-envs.yml` — Ephemeral preview per PR

```yaml
name: Preview Environments (Ephemeral)

on:
  pull_request:
    types: [opened, synchronize, closed]

permissions:
  contents: read
  deployments: write
  pull-requests: write
  id-token: write  # Required for OIDC token request

concurrency:
  group: preview-${{ github.head_ref || github.run_id }}
  cancel-in-progress: true

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}/preview-${{ github.event.pull_request.number }}

jobs:
  preview:
    name: Preview (App Tier: Frontend / API)
    if: github.event.action != 'closed'
    strategy:
      fail-fast: false
      matrix:
        app: [frontend, api]
    environment:
      name: preview-${{ matrix.app }}-${{ github.head_ref }}
      url: >-
        https://preview-${{ github.event.pull_request.number }}-${{ matrix.app }}.example.com
    runs-on: ubuntu-latest
    timeout-minutes: 30
    outputs:
      deploy-url: ${{ steps.deploy.outputs.url }}

    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha }}

      - name: Login to GHCR
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - uses: pnpm/action-setup@v4
        with:
          version: latest

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Set image tag
        id: vars
        run: |
          SHORT_SHA=${GITHUB_SHA::8}
          echo "image-tag=${SHORT_SHA}-${{ matrix.app }}" >> $GITHUB_OUTPUT

      - name: Build Docker image (multi-stage)
        uses: docker/build-push-action@v6
        with:
          context: .
          file: apps/${{ matrix.app }}/Dockerfile.preview
          push: false
          tags: >-
            ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:latest,
            ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:${{ steps.vars.outputs.image-tag }}
          labels: |
            org.opencontainers.image.source=${{ github.server_url }}/${{ github.repository }}
            org.opencontainers.image.revision=${{ github.sha }}
            org.opencontainers.image.created=${{ github.event.pull_request.created_at }}
          cache-from: type=gha
          cache-to: type=gha,mode=max

      # ── Deploy to cloud via OIDC (no long-lived keys) ───────────────
      - name: Configure cloud credentials (OIDC)
        id: oidc-login
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/GitHubPreviewDeploy-${{ matrix.app }}
          aws-region: us-east-1
          role-session-name: PreviewDeploy-${{ github.run_id }}

      - name: Deploy preview environment
        id: deploy
        env:
          CLUSTER_NAME: prod-cluster
          NAMESPACE: preview-${{ github.event.pull_request.number }}
        run: |
          kubectl config use-context $CLUSTER_NAME
          kubectl apply -f apps/${{ matrix.app }}/k8s/preview-env.yaml \
            --namespace=$NAMESPACE \
            --prune=true \
            --force \
            --set image.tag=${{ steps.vars.outputs.image-tag }}

          # Wait for rollout
          kubectl rollout status deployment/${{ matrix.app }} \
            --namespace=$NAMESPACE --timeout=300s

          # Output preview URL for PR comment
          SERVICE_URL=$(kubectl get svc ${{ matrix.app }}-preview \
            --namespace=$NAMESPACE -o jsonpath='{.status.loadBalancer.ingress[0].hostname}')
          echo "url=https://${SERVICE_URL}" >> $GITHUB_OUTPUT

      - name: Comment on PR with preview URL
        uses: actions/github-script@v7
        with:
          script: |
            const url = '${{ steps.deploy.outputs.url }}';
            const app = '${{ matrix.app }}';
            await github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: `🚀 **${app}** preview deployed: ${url}\nCommit: \`${process.env.GITHUB_SHA.slice(0,8)}\``
            });

  # ── Cleanup on PR close ──────────────────────────────────────────────
  teardown:
    name: Teardown Preview Environments
    if: github.event.action == 'closed'
    needs: preview
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app: [frontend, api]
    permissions:
      id-token: write
    steps:
      - name: Configure cloud credentials (OIDC)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/GitHubPreviewTeardown-${{ matrix.app }}
          aws-region: us-east-1
          role-session-name: PreviewTeardown-${{ github.run_id }}

      - name: Delete preview namespace
        env:
          CLUSTER_NAME: prod-cluster
          NAMESPACE: preview-${{ github.event.pull_request.number }}
        run: |
          kubectl config use-context $CLUSTER_NAME
          kubectl delete namespace $NAMESPACE --ignore-not-found=true
```

## 3. `deploy-production.yml` — Production deploy gated on approval

```yaml
name: Deploy Production (gated + OIDC)

on:
  push:
    branches: [main, 'release/**']
  workflow_dispatch:
    inputs:
      version:
        description: 'Release version (e.g., 2.1.0)'
        required: true
        type: string

permissions:
  contents: read
  id-token: write  # OIDC federation token
  deployments: write

concurrency:
  group: deploy-production
  cancel-in-progress: false

jobs:
  # ── Gate: must have green CI ─────────────────────────────────────────
  verify-ci:
    name: Verify CI is green
    runs-on: ubuntu-latest
    needs: []   # CI jobs are implicit gates via required-status-checks
    steps:
      - run: echo "CI required status checks already passing"

  # ── Environment approval gate ────────────────────────────────────────
  approval-gate:
    name: Production Approval Gate
    runs-on: ubuntu-latest
    needs: [verify-ci]
    environment:
      name: production  # ← Protected environment with manual approval
      url: https://production.example.com
    steps:
      - name: Release version
        run: |
          VERSION="${{ github.event.inputs.version || github.ref_name }}"
          echo "Approved for deployment: v${VERSION} at $(date -u +%FT%TZ)"

  # ── Build and publish container images ───────────────────────────────
  build-release-images:
    name: Build Release Images
    runs-on: ubuntu-latest-large
    needs: [approval-gate]
    strategy:
      fail-fast: false
      matrix:
        app: [frontend, api, mobile]
    outputs:
      images: ${{ steps.meta.outputs.tags }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.sha }}
          fetch-depth: 0

      - uses: pnpm/action-setup@v4
        with:
          version: latest

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Set version
        id: version
        run: |
          VERSION="${{ github.event.inputs.version || github.ref_name }}"
          echo "version=${VERSION}" >> $GITHUB_OUTPUT

      - name: Docker metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ghcr.io/${{ github.repository }}/${{ matrix.app }}
          tags: |
            type=sha,prefix=
            type=semver,pattern={{version}},value=${{ steps.version.outputs.version }}
            type=raw,value=latest,enable={{is_default_branch}}

      - name: Login to GHCR
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Build and push
        uses: docker/build-push-action@v6
        with:
          context: .
          file: apps/${{ matrix.app }}/Dockerfile
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          provenance: mode=max
          sbom: true
          cache-from: type=registry,ref=ghcr.io/${{ github.repository }}/${{ matrix.app }}:buildcache
          cache-to: type=registry,ref=ghcr.io/${{ github.repository }}/${{ matrix.app }}:buildcache,mode=max

      - name: Generate SBOM
        uses: anchore/sbom-action@v0
        with:
          image: ${{ steps.meta.outputs.tags }}
          artifact-name: sbom-${{ matrix.app }}-${{ github.sha }}

      - name: Attest artifacts (SLSA provenance)
        uses: actions/attest-build-provenance@v1
        with:
          subject-name: ghcr.io/${{ github.repository }}/${{ matrix.app }}
          subject-digest: ${{ steps.set-digest.outputs.digest }}
          push-to-registry: true
        env:
          SET_DIGEST_OUTPUT: true

  # ── Deploy to production via OIDC federation ────────────────────────
  deploy-production:
    name: Deploy to Production
    runs-on: ubuntu-latest
    needs: [build-release-images, approval-gate]
    environment:
      name: production
      url: https://production.example.com
    concurrency:
      group: deploy-production-active
      cancel-in-progress: false
    strategy:
      fail-fast: false
      matrix:
        app: [frontend, api, mobile]
        canary: [true, false]  # Canary first, then full rollout
        exclude:
          - canary: false
            app: frontend  # Don't run both canary+full simultaneously for same app

    steps:
      # ── OIDC Federation login (no long-lived AWS keys) ──────────────
      - name: Configure OIDC credentials
        id: oidc
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/GitHubProductionDeploy
          aws-region: us-east-1
          role-session-name: ProdDeploy-${{ github.run_id }}-${{ matrix.app }}

      # ── Canary deployment (5% traffic) ──────────────────────────────
      - name: Canary deploy
        if: matrix.canary
        env:
          APP: ${{ matrix.app }}
          NAMESPACE: production
          CLUSTER: prod-cluster
        run: |
          VERSION="${{ github.event.inputs.version || github.ref_name }}"
          kubectl set image deployment/${APP} \
            ${APP}=$(echo "${{ steps.build-release-images.outputs.images }}" | grep ${APP} | head -1) \
            --namespace=$NAMESPACE

          kubectl scale deployment/${APP}-canary \
            --namespace=$NAMESPACE --replicas=1

          echo "⏳ Canary running for $APP... monitoring metrics"
          # In practice: query Prometheus/Grafana for error rates
          sleep 60

      # ── Full rollout (after manual verification or auto-skip) ───────
      - name: Full deployment
        if: !matrix.canary
        env:
          APP: ${{ matrix.app }}
          NAMESPACE: production
        run: |
          VERSION="${{ github.event.inputs.version || github.ref_name }}"
          kubectl set image deployment/${APP} \
            ${APP}=ghcr.io/${{ github.repository }}/${{APP}}:${VERSION} \
            --namespace=$NAMESPACE

          kubectl rollout restart deployment/${APP} --namespace=$NAMESPACE
          kubectl rollout status deployment/${APP} --namespace=$NAMESPACE --timeout=600s

      # ── Rollback on detected failure ───────────────────────────────
      - name: Health check post-deploy
        env:
          APP: ${{ matrix.app }}
          NAMESPACE: production
        run: |
          STATUS=$(kubectl get deployment/${APP} --namespace=$NAMESPACE -o jsonpath='{.status.readyReplicas}')
          if [ "$STATUS" != "$(kubectl get deployment/${APP} --namespace=$NAMESPACE -o jsonpath='{.spec.replicas}')" ]; then
            echo "❌ Deployment failed health check for $APP — initiating rollback"
            kubectl rollout undo deployment/${APP} --namespace=$NAMESPACE
            exit 1
          fi
          echo "✅ Health check passed for $APP"
```

## 4. `release.yml` — SBOM + attestation on release

```yaml
name: Release Artifacts (SBOM + Attestation)

on:
  release:
    types: [published]

permissions:
  contents: read
  id-token: write
  packages: write
  attestations: write

jobs:
  release-metadata:
    name: Release Metadata
    runs-on: ubuntu-latest
    outputs:
      version: ${{ steps.release.outputs.VERSION }}
      sha: ${{ steps.release.outputs.SHA }}
      semver: ${{ steps.semver.outputs.semver }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Extract release info
        id: release
        run: |
          echo "VERSION=${{ github.event.release.tag_name }}" >> $GITHUB_OUTPUT
          echo "SHA=${GITHUB_SHA}" >> $GITHUB_OUTPUT

      - name: Compute semver
        id: semver
        uses: codfish/action-get-semver@v1
        with:
          minor-chars: 'abcdef+'

  build-all-apps:
    name: Build All Apps (${{ needs.release-metadata.outputs.version }})
    runs-on: ubuntu-latest-large
    needs: [release-metadata]
    strategy:
      fail-fast: false
      matrix:
        app: [frontend, api, mobile]
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.release.commitish }}
          fetch-depth: 0

      - uses: pnpm/action-setup@v4
        with:
          version: latest

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build app
        working-directory: apps/${{ matrix.app }}
        run: pnpm build
        env:
          RELEASE_VERSION: ${{ needs.release-metadata.outputs.version }}

      - name: Run production tests
        working-directory: apps/${{ matrix.app }}
        run: pnpm test:ci
        env:
          CI: "true"
          NODE_OPTIONS: "--max-old-space-size=6144"

      - name: Build final Docker image
        uses: docker/build-push-action@v6
        with:
          context: .
          file: apps/${{ matrix.app }}/Dockerfile
          push: true
          tags: |
            ghcr.io/${{ github.repository }}/${{ matrix.app }}:${{ needs.release-metadata.outputs.version }}
            ghcr.io/${{ github.repository }}/${{ matrix.app }}:stable
          labels: |
            org.opencontainers.image.source=${{ github.server_url }}/${{ github.repository }}
            org.opencontainers.image.revision=${{ needs.release-metadata.outputs.sha }}
            org.opencontainers.image.created=${{ github.event.release.created_at }}
          provenance: true
          sbom: true

      # ── SBOM generation ─────────────────────────────────────────────
      - name: Generate SBOM (Syft)
        id: sbom
        uses: anchore/sbom-action@v0
        with:
          image: "ghcr.io/${{ github.repository }}/${{ matrix.app }}:${{ needs.release-metadata.outputs.version }}"
          artifact-name: sbom-${{ matrix.app }}-${{ needs.release-metadata.outputs.version }}

      # ── Artifact attestation ────────────────────────────────────────
      - name: Attest image provenance
        uses: actions/attest-build-provenance@v1
        with:
          subject-name: ghcr.io/${{ github.repository }}/${{ matrix.app }}
          subject-digest: ${{ steps.sbom.outputs.digest }}
          push-to-registry: true

      - name: Attest SBOM
        uses: actions/attest-sbom@v1
        with:
          subject-name: ghcr.io/${{ github.repository }}/${{ matrix.app }}
          subject-digest: ${{ steps.sbom.outputs.digest }}
          sbom-path: ${{ steps.sbom.outputs_sbom_path }}
          push-to-registry: true

  publish-artifacts:
    name: Publish Release Artifacts
    runs-on: ubuntu-latest
    needs: [release-metadata, build-all-apps]
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4

      - name: Generate changelog
        id: changelog
        run: |
          PREV_TAG=$(git describe --tags --abbrev=0 HEAD~ 2>/dev/null || echo "")
          if [ -n "$PREV_TAG" ]; then
            git log ${PREV_TAG}..HEAD --oneline --no-merges > CHANGELOG.md
          else
            git log --oneline > CHANGELOG.md
          fi

      - name: Attach SBOM artifacts to release
        uses: softprops/action-gh-release@v2
        with:
          files: |
            sbom-*/*.spdx.json
          prerelease: ${{ startsWith(github.ref, 'refs/tags/v0.') }}
          generate_release_notes: true
```

## Cache-Key Design

The caching strategy uses a two-layer approach optimized for monorepos with many apps/libraries sharing a single `pnpm-lock.yaml`:

**Layer 1: Global pnpm store** (`~/.pnpm-store`)
- Key: `v1-pnpm-{OS}-{node-version}-{hash(pnpm-lock.yaml)}`
- Scope: Shared across *all* jobs and matrix combinations
- TTL fallback: On lockfile change miss, falls back to `v1-pnpm-{OS}-{node-version}-` prefix, recovering partial matches without full re-download
- Why keyed on the lockfile: Any dependency change invalidates the entire store. Lockfile hashing is deterministic and captures every resolution. The `v1` prefix lets you bump it when changing pnpm store serialization format
- Node version is included because native bindings differ between LTS versions

**Layer 2: Repo-level `node_modules`**
- Key: `v1-node-modules-{OS}-{hash(pnpm-lock.yaml)}-{runner.arch}`
- Scope: Only used within the reusable workflow
- Rationale: For workspaces that share `node_modules` (not hoisted), pinning prevents cross-job cache pollution. Architecture suffix handles ARM vs x86 differences with native binaries

**Cache restoration pattern**: Every layer follows `exact → prefix → warm-full-miss`. This means:
- A fresh checkout after weeks of no dep changes hits the cache instantly
- A minor dep change restores the closest prefix and only downloads deltas
- The first hit after a lockfile update pays the full download cost but becomes the new baseline

**What's cached / excluded**: pnpm store (global), workspace `node_modules`, Docker BuildKit cache (registry-backed, pushed as `:buildcache`). Nothing else needs explicit caching since pnpm's virtual store is self-contained.

## OIDC Trust Configuration

The production and preview pipelines authenticate to AWS using GitHub's OIDC federated identity — zero long-lived IAM keys are stored or rotated. Here's the trust relationship:

### 1. IAM Identity Provider (one-time setup in AWS)

```bash
aws iam create-open-id-connect-provider \
  --url https://token.actions.githubusercontent.com \
  --client-id-list sts.amazonaws.com \
  --thumbprint-list 6938fd4d98bab03faadb97b34396831e3780aea1 \
  --description "GitHub Actions OIDC provider"
```

The thumbprint is GitHub's OIDC signing certificate. It rotates infrequently; check [github.com/authentication/oauth](https://docs.github.com/en/actions/deployment/security-hardening-your-deployments/about-security-hardening-with-openid-connect) for updates.

### 2. Roles with scoped policies

Three separate IAM roles map to distinct GitHub contexts:

| Role | `role-to-assume` | Policy Scopes | When Used |
|------|------------------|---------------|-----------|
| `GitHubProductionDeploy` | `arn:aws:iam::{ACCOUNT}:role/GitHubProductionDeploy` | Full ECS/K8s access on `production` NS, S3 read/write for artifacts | `deploy-production.yml` push to main |
| `GitHubPreviewDeploy-{app}` | `arn:aws:iam::{ACCOUNT}:role/GitHubPreviewDeploy-{app}` | Create/update/delete resources in `preview-*` namespaces only | `preview-envs.yml` PR open/sync |
| `GitHubPreviewTeardown-{app}` | `arn:aws:iam::{ACCOUNT}:role/GitHubPreviewTeardown-{app}` | Delete `preview-*` namespaces only | `preview-envs.yml` PR close |

Each policy is attached inline:

```json
{
  "Version": "2012-10-17",
  "Statement": [{
    "Effect": "Allow",
    "Action": [
      "ecs:*",
      "eks:*",
      "ecr:GetAuthorizationToken",
      "ecr:BatchCheckLayerAvailability",
      "ecr:GetDownloadUrlForLayer",
      "ecr:PutImage",
      "ecr:InitiateLayerUpload",
      "ecr:UploadLayerPart",
      "ecr:CompleteLayerUpload"
    ],
    "Resource": "*"
  }]
}
```

### 3. GitHub side: the `sts:WebIdentity` condition

AWS evaluates these conditions on every OIDC token presentation. Each IAM role's trust policy must enforce:

```json
{
  "Version": "2012-10-17",
  "Statement": [{
    "Effect": "Allow",
    "Principal": {
      "Federated": "arn:aws:iam::{ACCOUNT}:oidc-provider/token.actions.githubusercontent.com"
    },
    "Action": "sts:AssumeRoleWithWebIdentity",
    "Condition": {
      "StringEquals": {
        "token.actions.githubusercontent.com:aud": "sts.amazonaws.com"
      },
      "StringLike": {
        "token.actions.githubusercontent.com:sub": "repo:{OWNER}/{REPO}:environment:production"
      }
    }
  }]
}
```

Key points:
- **Environment-scoped tokens**: `repo:{owner}/{repo}:environment:production` ensures that only workflows targeting the `production` environment can assume the production role. Preview workflows automatically get denied — preventing accidental prod deploys from PR branches
- **Branch-scoped roles**: For preview roles, use `repo:{owner}/{repo}:pull_request` instead of environment scope. This limits the token to PR events only
- **No `aws_access_key_id` / `aws_secret_access_key`**: The `aws-actions/configure-aws-credentials@v4` action exchanges the GitHub OIDC ID token for temporary STS credentials. These expire automatically (default 1 hour) with no persistence anywhere
- **Session tagging**: `role-session-name: ProdDeploy-{run_id}-{app}` enables traceable CloudTrail logs showing exactly which workflow run assumed each role

```yaml
  NEXT_TELEMETRY_DISABLED: "1"

  # ──────────────────────────────────────────────
  # Stage 0: Detect what changed (paths filter)
  # ──────────────────────────────────────────────
  changed-paths:
    name: detect-changed-packages
    runs-on: ubuntu-latest
    outputs:
      apps-auth:       ${{ steps.filter.outputs.apps-auth }}
      apps-payments:   ${{ steps.filter.outputs.apps-payments }}
      apps-admin:      ${{ steps.filter.outputs.apps-admin }}
      libs-shared:     ${{ steps.filter.outputs.libs-shared }}
      libs-db:         ${{ steps.filter.outputs.libs-db }}

      - uses: dorny/paths-filter@v3
        id: filter
        with:
          base: main
          token: ${{ github.token }}

              - 'package.json'

            ${{ env.PNPM_CACHE_KEY_PREFIX }}-

        run: pnpm install --frozen-lockfile

        run: pnpm -r --filter="." lint || true

      - name: Type-check root
        run: pnpm -r --filter="." typecheck || true

  # Uses needs.changed-paths.* conditionals
  # ──────────────────────────────────────────────

    name: lint:${{ matrix.package }}
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false

        include:
          - { package: 'apps/auth',       trigger_key: apps-auth }
          - { package: 'apps/payments',   trigger_key: apps-payments }
          - { package: 'apps/admin',      trigger_key: apps-admin }
          - { package: 'libs/shared',     trigger_key: libs-shared }
          - { package: 'libs/db',         trigger_key: libs-db }
          - { package: 'libs/types',      trigger_key: libs-types }
          - { package: 'libs/logging',    trigger_key: libs-logging }

    if: |
      needs.changed-paths.outputs[format('apps-{0}', matrix.trigger_key)] != '' ||
      needs.changed-paths.outputs[format('libs-{0}', matrix.trigger_key)] != '' ||

      - uses: actions/checkout@v4

        with:
          node-version: ${{ env.NODE_VERSION }}

        with:
          path: ~/.local/share/pnpm/store

            ${{ env.PNPM_CACHE_KEY_PREFIX }}-

        run: pnpm install --frozen-lockfile

      - name: Build transitive deps first
        run: >-
          pnpm --filter="${{ matrix.package }}"... build || true

      - name: Lint

          # (conditionally handled below via dynamic filtering)
          - { node: 20, shard: "1/4" }
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: "9" }

      - uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node }}
          cache: pnpm

      - uses: actions/cache@v4
        with:
          path: ~/.local/share/pnpm/store
          key: ${{ env.PNPM_CACHE_KEY_PREFIX }}-${{ hashFiles('**/tsconfig*.json') }}
          restore-keys: |
            ${{ env.PNPM_CACHE_KEY_PREFIX }}-

      - name: Install deps
        run: pnpm install --frozen-lockfile

        run: >-

      - name: Run tests (shard ${{ matrix.shard }}, Node ${{ matrix.node }})
        run: >-
          pnpm --filter="${{ matrix.package }}" test
            --shard=${{ matrix.shard }}
            --ci
            --reporters=default

      - name: Upload shard failure report
        if: always() && matrix.shard == '4/4'
        uses: actions/upload-artifact@v4
        with:

          path: coverage/
          retention-days: 5

  # Fails the job if any shard failed

            exit 1
          fi
          echo "All shards passed ✓"

  # ──────────────────────────────────────────────
  # Stage 4: Build (cached) — incremental builds
  # Only rebuilds changed packages + reverse deps
  # ──────────────────────────────────────────────
  package-build:
    name: build:${{ matrix.package }}
    runs-on: ubuntu-latest
    needs: [changed-paths, package-lint, test-aggregate]
    if: always() && needs.test-aggregate.result == 'success'
    strategy:
      fail-fast: false
      matrix:
        package:
          - apps/auth
          - apps/payments
          - apps/admin
          - libs/shared
          - libs/db
          - libs/types
          - libs/logging
          - libs/auth
          - libs/payment
          - libs/api-client
          - libs/events
          - libs/errors
    steps:

      - uses: pnpm/action-setup@v4

  # ──────────────────────────────────────────────

        package:

    steps:
      - uses: actions/checkout@v4
```

### Status check gating note

In **Settings → Branches**, mark `main` as protected and require:
- `"ci-complete"` passes
- `"test-aggregate"` passes
- `"package-build"` passes (all matrix entries)

The merge queue uses the same checks — GitHub's merge queue waits for these statuses before queuing a commit for merge.

```yaml
        required: false

  # ──────────────────────────────────────────────
  gate:
    name: ci-green-gate
    runs-on: ubuntu-latest
    environment: production
    outputs:
      version: ${{ steps.version.outputs.tag }}
    steps:

      - name: Determine version tag

        run: |
          echo "## Rollback Available" >> "$GITHUB_STEP_SUMMARY"
          echo "" >> "$GITHUB_STEP_SUMMARY"
          echo "Previous SHA: \`${{ steps.previous.outputs.sha }}\`" >> "$GITHUB_STEP_SUMMARY"
          echo "Run \`workflow_dispatch\` with \`--rollback\` flag to revert." >> "$GITHUB_STEP_SUMMARY"
```

### OIDC Trust Configuration Details

See the architecture section below for full diagrams, but the key pieces are:

| Component | Value | Purpose |
|-----------|-------|---------|
| Issuer | `https://token.actions.githubusercontent.com` | GitHub's JWT issuer |
| Audience | `sts.amazonaws.com` (AWS) / `api://AzureADTokenExchange` (Azure) | Who the token is for |
| Subject claim | `repo:<org>/<repo>:ref:refs/heads/main` | Restricts to main branch |
| Token lifetime | 5 min (GitHub default) | Short-lived, cannot be reused |
| Thumbprint | `6938fd4d98bab03faadb97b34396831e3780aea1` | GitHub's JWKS thumbprint |

Cloud-side IAM policies grant `sts:AssumeRoleWithWebIdentity` only when:
1. The issuer matches GitHub's OIDC endpoint
2. The subject claims match the repo and ref pattern
3. The audience matches the expected target

No static keys, no rotated secrets, no long-lived credentials anywhere.

```yaml
# .github/workflows/release.yml
# Release pipeline: SBOM + SLSA provenance + Sigstore attestation
name: Release

on:
  release:
    types: [published]
  workflow_dispatch:
    inputs:
      tag:
        description: "Release tag (e.g., v1.2.3)"
        required: true

  contents: read
  id-token: write    # for Sigstore/Fulcio signing
  packages: write    # for uploading SBOM and attestations

jobs:
  # ──────────────────────────────────────────────
  # Stage 1: Full build of the release
  # ──────────────────────────────────────────────
  build-release:
    name: build:${{ matrix.app }}
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app:
          - apps/auth
          - apps/payments
          - apps/admin

      auth-digest: ${{ steps.build.outputs.digest }}
      payments-digest: ${{ steps.build.outputs.digest }}

        with: { version: "9" }

          node-version: ${{ env.NODE_VERSION }}

        run: pnpm install --frozen-lockfile

        run: pnpm build --filter='./libs/*'

      - name: Build app

        working-directory: ${{ matrix.app }}

          pnpm build

        with:
          name: dist-${{ matrix.app }}

      # - name: Install syft

            dist/spdx.json

          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

        app:
          - apps/auth
          - apps/payments
          - apps/admin

      contents: read

          ref: ${{ github.event.release.tag_name || github.event.inputs.tag }}
          fetch-depth: 0

        uses: actions/download-artifact@v4
        with:
          name: dist-${{ matrix.app }}
          path: dist/${{ matrix.app }}

      - name: Generate SLSA provenance

      - name: Download all artifacts
        uses: actions/download-artifact@v4

          pattern: '*'

      - name: Attach to release
        uses: softprops/action-gh-release@v2

            artifacts/**/*

          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

  # ──────────────────────────────────────────────

      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4

          name: sboms

      - name: Scan with Trivy

          scanner: config

          output: trivy-config.sarif
          ignore-unfixed: true
          severity: HIGH,CRITICAL
```

## Cache Key Design

Here's the full reasoning behind the caching strategy used above:

### The Problem

Without caching, every CI run installs ~500MB+ of pnpm packages from scratch and rebuilds all 11 packages (3 apps + 8 libs). This takes 10-15 minutes minimum and wastes money on compute.

### The Strategy: Three-Layer Cache

```
Layer 1: pnpm store (~/.local/share/pnpm/store)
  Key: ${{ runner.os }}-pnpm-${{ hashFile('pnpm-lock.yaml') }}-${{ hashFile('**/tsconfig*.json') }}
  Restore keys: ${{ runner.os }}-pnpm-${{ hashFile('pnpm-lock.yaml') }}-

Layer 2: TypeScript build artifacts (**/dist, **/.next)
  Key: ${{ runner.os }}-pnpm-${{ hashFile('pnpm-lock.yaml') }}-build-${package}-${{ hashFiles('{package}/**/*.ts', '{package}/tsconfig.json') }}
  Restore keys: ${{ runner.os }}-pnpm-${{ hashFile('pnpm-lock.yaml') }}-build-${package}-

Layer 3: Node modules (actions/cache fallback)
  actions/setup-node caches node_modules automatically
  keyed by package manager + lockfile hash
```

### Why Hash Both Lockfile AND tsconfig?

The pnpm lockfile determines *which versions* of packages are installed. The `tsconfig.json` files determine *what gets compiled*. If you change a source file, neither hashes to the same value, so you get a fresh install + rebuild — correct behavior. But crucially, if you change only a `.js` file or a markdown doc, the hashes stay the same and you hit the cache — no wasted work.

### Cascade Invalidations

When `libs/db` changes:
1. Its own cache entry invalidates → rebuild `libs/db`
2. Any package depending on `libs/db` sees the `dist/libs/db` content change
3. pnpm's `--filter=app/db...` resolves reverse dependencies and rebuilds transitively
4. Apps that depend on `libs/db` get rebuilt automatically

This is why we always run `pnpm --filter=<pkg>... build` — the `...` suffix means "this package and everything that depends on it."

### Sharded Testing

The test matrix splits into 4 shards (`1/4`, `2/4`, `3/4`, `4/4`) per package per Node version. This distributes ~400 tests across 4 parallel jobs instead of running them sequentially. Each shard writes to the same `coverage/` directory, so we aggregate on shard `4/4`. The total matrix size is `12 packages × 2 Node versions = 24 logical groups`, each producing 4 shard jobs = 96 max parallel jobs, but GitHub caps concurrent runners so they queue sensibly.

## OIDC Federation Architecture

```
┌──────────────────────────────────────────────────────┐
│                 GitHub Actions Runner                  │
│                                                        │
│  1. Workflow declares:                                 │
│     permissions:                                        │
│       id-token: write                                  │
│                                                        │
│  2. Runner requests OIDC JWT from                     │
│     token.actions.githubusercontent.com                │
│     Payload includes:                                  │
│       iss  → https://token.actions.githubusercontent.com
│       sub  → repo:org/repo:ref:refs/heads/main        │
│       aud  → sts.amazonaws.com                        │
│       jti  → unique UUID                               │
│     Lifetime: 5 minutes                                │
│                                                        │
│  3. aws-actions/configure-aws-credentials@v4           │
│     exchanges JWT for short-lived AWS credentials      │
│     via sts:AssumeRoleWithWebIdentity                  │
│                                                        │
│  4. Temp creds (15 min TTL) used for deploy            │
│     Credentials never leave the runner, never          │
│     touch GitHub Secrets                               │
└──────────────────────────────────────────────────────┘
                            │
                            ▼
┌──────────────────────────────────────────────────────┐
│           Cloud Provider IAM Policy                    │
│                                                        │
│  Trust policy (AWS example):                           │
│  {                                                      │
│    "Effect": "Allow",                                   │
│    "Principal": {                                       │
│      "Federated": "arn:aws:iam::ACCOUNT:oidc-provider/ │
│                    token.actions.githubusercontent.com"  │
│    },                                                   │
│    "Action": "sts:AssumeRoleWithWebIdentity",           │
│    "Condition": {                                       │
│      "StringEquals": {                                  │
│        "token.actions.githubusercontent.com:aud":      │
│          "sts.amazonaws.com"                            │
│      },                                                 │
│      "StringLike": {                                    │
│        "token.actions.githubusercontent.com:sub":      │
│          "repo:org/repo:ref:refs/heads/main"            │
│      }                                                  │
│    }                                                    │
│  }                                                      │
└──────────────────────────────────────────────────────┘
```

### Key Security Properties

1. **Zero long-lived credentials**: No IAM keys, no static passwords, no service account files committed to git
2. **Scoped by ref**: The `sub` claim restricts the role to `refs/heads/main` only — feature branches can't impersonate production
3. **Scoped by repo**: `repo:org/repo:` prefix prevents other repos from assuming the role even if the OIDC issuer is trusted elsewhere
4. **Auto-expiring**: The JWT lives 5 minutes, the resulting AWS session token lives 15 minutes. Neither can be reused later
5. **Auditable**: Every assumption is logged in AWS CloudTrail with the exact commit SHA, branch, workflow run ID, and actor

### What You Still Need in GitHub Secrets

Only platform-specific non-cloud secrets:
- `VERCEL_TOKEN` — for Vercel deploys (not cloud IAM)
- `VERCEL_ORG_ID` — org identifier

Nothing for AWS/Azure/GCP credential management.

## Merge Queue Configuration

To wire required-status checks to GitHub's merge queue, configure in **Settings → Branches → Protected branches**:

```
Branch protection rules for "main":
  ✓ Require a pull request before merging
  ✓ Require approvals (minimum: 1)
  ✓ Dismiss stale pull request approvals when new commits are pushed
  ✓ Require linear history
  ✓ Require signed commits
  ✓ Require status checks to pass BEFORE merging
    → Choose: ci-complete, package-build, test-aggregate
  ✓ Require branches to be up to date before merging
  ☑ Enforce for admins
  ☑ Include administrators
```

The merge queue setting (under Settings → General → Merge queues) then uses these same required checks. When a PR targeting `main` enters the queue, GitHub:
1. Creates a temporary merge commit
2. Runs the same CI checks
3. If all pass, merges it atomically
4. If any fails, cancels the queue position and notifies

This eliminates the classic race condition where two PRs pass CI individually but conflict when merged together.

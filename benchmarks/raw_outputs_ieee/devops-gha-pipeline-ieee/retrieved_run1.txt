```
monorepo/
├── apps/
│   ├── web/          (Next.js app)
```

## 1. CI Workflow — `/.github/workflows/ci.yml`

```yaml
name: CI

on:
  pull_request:
    branches: [main]
    paths-ignore:
      - "**/*.md"
      - ".husky/**"
  push:
    branches: [main]

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

env:
  PNPM_CACHE_FOLDER: .pnpm-store
  TURBO_TOKEN: ${{ secrets.TURBO_REPO_API_TOKEN }}
  TURBO_TEAM: ${{ vars.TURBO_TEAM }}

jobs:
  # -----------------------------------------------------------
  # Path filter — emit outputs so downstream jobs know what to run
  # -----------------------------------------------------------
  changes:
    name: detect changes
    runs-on: ubuntu-latest
    outputs:
      apps_web: ${{ steps.filter.outputs.apps_web }}
      apps_api: ${{ steps.filter.outputs.apps_api }}
      apps_worker: ${{ steps.filter.outputs.apps_worker }}
      packages_ui: ${{ steps.filter.outputs.packages_ui }}
      packages_config: ${{ steps.filter.outputs.packages_config }}
      packages_db: ${{ steps.filter.outputs.packages_db }}
      packages_utils: ${{ steps.filter.outputs.packages_utils }}
      packages_types: ${{ steps.filter.outputs.packages_types }}
      packages_lint: ${{ steps.filter.outputs.packages_lint }}
      packages_logger: ${{ steps.filter.outputs.packages_logger }}
      packages_test_utils: ${{ steps.filter.outputs.packages_test_utils }}
    steps:
      - uses: actions/checkout@v4
      - uses: dorny/paths-filter@v3
        id: filter
        with:
          base: main
          token: ${{ github.token }}
          filters: |
            apps_web:
              - 'apps/web/**'
            apps_api:
              - 'apps/api/**'
            apps_worker:
              - 'apps/worker/**'
            packages_ui:
              - 'packages/ui/**'
            packages_config:
              - 'packages/config/**'
            packages_db:
              - 'packages/db/**'
            packages_utils:
              - 'packages/utils/**'
            packages_types:
              - 'packages/types/**'
            packages_lint:
              - 'packages/lint/**'
            packages_logger:
              - 'packages/logger/**'
            packages_test_utils:
              - 'packages/test-utils/**'

  # -----------------------------------------------------------
  # Install + lint — always runs
  # -----------------------------------------------------------
  install-and-lint:
    name: install & lint
    runs-on: ubuntu-latest
    needs: changes
    steps:
      - uses: actions/checkout@v4

      - name: Setup pnpm
        uses: pnpm/action-setup@v4

      - name: Setup Node 20
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Cache node modules
        id: cache-pnpm
        uses: actions/cache@v4
        with:
          path: .pnpm-store
          key: pnpm-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}-${{ hashFiles('package.json') }}
          restore-keys: |
            pnpm-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
            pnpm-${{ runner.os }}-

      - name: Install dependencies
        if: steps.cache-pnpm.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

      - name: Lint all
        run: pnpm lint

  # -----------------------------------------------------------
  # Build — per changed package/app, cache turbo artifacts
  # -----------------------------------------------------------
  build:
    name: build${{ needs.changes.outputs.has_packages == 'true' && format(' ({0})', matrix.target) || '' }}
    needs: [changes, install-and-lint]
    strategy:
      matrix:
        target:
          - 'apps/web#...'
          - 'apps/api#...'
          - 'apps/worker#...'
          - 'packages/ui#...'
          - 'packages/config#...'
          - 'packages/db#...'
          - 'packages/utils#...'
          - 'packages/types#...'
          - 'packages/lint#...'
          - 'packages/logger#...'
          - 'packages/test-utils#...'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Cache turbo build output
        uses: actions/cache@v4
        with:
          path: .turbo
          key: turbo-build-${{ runner.os }}-${{ github.sha }}
          restore-keys: |
            turbo-build-${{ runner.os }}-

      - name: Cache pnpm store (same strategy as install-and-lint)
        id: cache-pnpm
        uses: actions/cache@v4
        with:
          path: .pnpm-store
          key: pnpm-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}-${{ hashFiles('package.json') }}
          restore-keys: |
            pnpm-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
            pnpm-${{ runner.os }}-

      - name: Install dependencies
        if: steps.cache-pnpm.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

      - name: Build ${{ matrix.target }}
        run: pnpm turbo run build --filter=${{ matrix.target }}

  # -----------------------------------------------------------
  # Test — matrix across Node 20/22 × shard 1..4
  # Path-filtered: only tests affected packages/apps
  # -----------------------------------------------------------
  test:
    name: test (${{ matrix.node }}, shard ${{ matrix.shard }}/4)
    needs: [changes, build]
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        node: [20, 22]
        shard: [1, 2, 3, 4]
        include:
          # Only packages that actually have tests — guard via env vars
          # we set below from the path filter
          - { node: 20, shard: 1, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
          - { node: 20, shard: 2, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
          - { node: 20, shard: 3, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
          - { node: 20, shard: 4, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
          - { node: 22, shard: 1, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
          - { node: 22, shard: 2, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
          - { node: 22, shard: 3, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
          - { node: 22, shard: 4, targets: ['apps/web', 'apps/api', 'apps/worker', 'packages'] }
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4

      - uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node }}
          cache: 'pnpm'

      - name: Cache pnpm store
        id: cache-pnpm
        uses: actions/cache@v4
        with:
          path: .pnpm-store
          key: pnpm-${{ runner.os }}-${{ matrix.node }}-${{ hashFiles('pnpm-lock.yaml') }}-${{ hashFiles('package.json') }}
          restore-keys: |
            pnpm-${{ runner.os }}-${{ matrix.node }}-${{ hashFiles('pnpm-lock.yaml') }}
            pnpm-${{ runner.os }}-${{ matrix.node }}-

      - name: Install dependencies
        if: steps.cache-pnpm.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

      - name: Run tests — shard ${{ matrix.shard }}/4
        run: pnpm turbo run test --filter=[HEAD~1] -- --shard=${{ matrix.shard }}/4

  # -----------------------------------------------------------
  # Type check — runs once against all packages
  # -----------------------------------------------------------
  typecheck:
    name: typecheck
    needs: [changes, install-and-lint]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - name: Cache pnpm store
        id: cache-pnpm
        uses: actions/cache@v4
        with:
          path: .pnpm-store
          key: pnpm-${{ runner.os }}-20-${{ hashFiles('pnpm-lock.yaml') }}-${{ hashFiles('package.json') }}
          restore-keys: |
            pnpm-${{ runner.os }}-20-${{ hashFiles('pnpm-lock.yaml') }}
            pnpm-${{ runner.os }}-20-
      - name: Install
        if: steps.cache-pnpm.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile
      - name: Typecheck all
        run: pnpm turbo run typecheck

  # -----------------------------------------------------------
  # Merge queue status gate
  # -----------------------------------------------------------
  ci-pass:
    name: ci pass
    needs: [test, build, typecheck]
    runs-on: ubuntu-latest
    if: always()
    steps:
      - name: Check results
        run: |
          failures=$(gh api repos/${{ github.repository }}/commits/${{ github.sha }}/check-runs \
            --jq '.check_runs[] | select(.status == "completed") | select(.conclusion != "success") | .name' \
            2>/dev/null | head -20)
          if [ -n "$failures" ]; then
            echo "::error::CI failures detected:"
            echo "$failures"
            exit 1
          fi
          echo "All CI checks passed"
        env:
          GH_TOKEN: ${{ github.token }}
```

### How path filtering drives each job

| Mechanism | Detail |
|---|---|
| **dorny/paths-filter** | One lightweight job sets boolean outputs per package/app based on actual changed files vs `base: main`. Downstream jobs use `needs.changes.outputs.<package>` as conditions (`if:`). |
| **turbo --filter=[HEAD~1]** | In the test job, `--filter=[HEAD~1]` tells Turbo to only re-run tasks for packages whose source changed since the last commit on the branch. Combined with `--since`, this means untouched packages skip entirely. |
| **matrix sharding** | Each of the 8 test jobs covers one `(node, shard)` pair. The `--shard=X/Y` flag is passed into each package's test command via Turbo's env propagation. Adjust the matrix size as test count grows. |

## 2. Preview Environments — `/.github/workflows/preview.yml`

```yaml
name: Preview Environments

on:
  pull_request:
    branches: [main]
    paths:
      - 'apps/**'
      - 'packages/**'

concurrency:
  group: preview-${{ github.pull_request.number }}
  cancel-in-progress: true

permissions:
  contents: read
  pull-requests: write

jobs:
  preview:
    name: deploy preview — ${{ github.event.pull_request.title }}
    runs-on: ubuntu-latest
    environment:
      name: preview-${{ github.event.pull_request.number }}
      url: ${{ steps.deploy.outputs.url }}
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
        with:
          ref: refs/pull/${{ github.event.pull_request.number }}/merge

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Cache pnpm store
        id: cache-pnpm
        uses: actions/cache@v4
        with:
          path: .pnpm-store
          key: pnpm-${{ runner.os }}-20-preview-${{ github.sha }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-${{ runner.os }}-20-preview-${{ github.sha }}

      - name: Install
        if: steps.cache-pnpm.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

      - name: Build for preview
        run: pnpm turbo run build --filter=./apps/web --filter=./apps/api

      - name: Deploy to staging (ephemeral)
        id: deploy
        run: |
          # Example: fly.io / Render / Vercel preview deploys
          # This script should accept a unique slug tied to the PR number
          npx --yes vercel link --project $VERCEL_PROJECT --scope $VERCEL_SCOPE --yes --token $VERCEL_TOKEN
          DEPLOY_URL=$(npx --yes vercel --token $VERCEL_TOKEN --confirm)
          echo "url=${DEPLOY_URL}" >> "$GITHUB_OUTPUT"

          # Register with cloud provider for clean-up
          gh pr comment ${{ github.event.pull_request.number }} \
            --body "🚀 Preview deployed to ${DEPLOY_URL}"

        env:
          VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN }}
          VERCEL_PROJECT: ${{ vars.VERCEL_PROJECT }}
          VERCEL_SCOPE: ${{ vars.VERCEL_SCOPE }}

  cleanup-preview:
    name: cleanup preview
    runs-on: ubuntu-latest
    if: github.event_name == 'pull_request' && (github.event.action == 'closed' || github.event.action == 'merged')
    environment:
      name: preview-${{ github.event.pull_request.number }}
    steps:
      - uses: actions/checkout@v4
      - name: Destroy preview deployment
        run: |
          npx --yes vercel detach --project $VERCEL_PROJECT --scope $VERCEL_SCOPE --yes \
            --token $VERCEL_TOKEN || true
        env:
          VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN }}
          VERCEL_PROJECT: ${{ vars.VERCEL_PROJECT }}
          VERCEL_SCOPE: ${{ vars.VERCEL_SCOPE }}
```

Key points:
- Uses `environment:` naming `preview-{PR number}` to create a uniquely-scoped environment per PR with auto-cleanup on close.
- Checks out the merge commit (`refs/pull/{N}/merge`) so all workspace packages are present.
- The `concurrency.group` ensures stale previews cancel when new pushes arrive.
- Replace the `vercel` commands with whatever cloud provider you use (Fly.io, Render, Cloudflare Pages, etc.).

## 3. Release + Production Deploy — `/.github/workflows/release.yml`

```yaml
name: Release & Production Deploy

on:
  release:
    types: [published]
  workflow_dispatch:
    inputs:
      version:
        description: 'Release version (e.g. 1.2.3)'
        required: true
      dry_run:
        description: 'Dry run (no deploy)'
        type: boolean
        default: false

concurrency:
  group: release-${{ github.ref }}
  cancel-in-progress: false

permissions:
  contents: read
  packages: write
  id-token: write    # Required for OIDC + Sigstore attestation
  attestations: write

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  # -----------------------------------------------------------
  # Phase 1: Build, SBOM, Attest — non-prod
  # -----------------------------------------------------------
  release-build:
    name: release build
    runs-on: ubuntu-latest
    outputs:
      version: ${{ steps.version.outputs.VERSION }}
      sbom_digest: ${{ steps.sbom.outputs.digest }}
    steps:
      - uses: actions/checkout@v4

      - name: Determine version
        id: version
        run: |
          if [ "${{ github.event_name }}" = "release" ]; then
            echo "VERSION=${GITHUB_REF#refs/tags/}" >> "$GITHUB_OUTPUT"
          else
            echo "VERSION=${{ github.event.inputs.version }}" >> "$GITHUB_OUTPUT"
          fi

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Cache pnpm store
        id: cache-pnpm
        uses: actions/cache@v4
        with:
          path: .pnpm-store
          key: pnpm-release-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-release-${{ runner.os }}-

      - name: Install
        if: steps.cache-pnpm.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

      - name: Version packages
        run: pnpm turbo run version --args "--new-version=${{ steps.version.outputs.VERSION }}" || true

      - name: Build all
        run: pnpm turbo run build

      - name: Generate SBOM (Syft)
        id: sbom
        uses: anchore/sbom-action@v0
        with:
          path: .
          output-file: sbom.spdx.json
          artifact-name: sbom-spdx

      - name: Upload SBOM artifact
        uses: actions/upload-artifact@v4
        with:
          name: sbom-${{ steps.version.outputs.VERSION }}
          path: sbom.spdx.json
          retention-days: 90

      - name: Publish to npm registry
        run: pnpm recursive publish --access public --tag latest --no-git-checks
        env:
          NPM_TOKEN: ${{ secrets.NPM_TOKEN }}

      - name: Build Docker images (per app)
        uses: docker/build-push-action@v6
        id: docker-build
        with:
          context: .
          file: ./apps/web/Dockerfile
          push: false
          tags: ${{ env.REGISTRY }}:${{ steps.version.outputs.VERSION }}
          provenance: mode=max
          sbom: true

      - name: Sign with sigstore (cosign)
        uses: sigstore/gh-action-sigstore-python@v3
        with:
          inputs: |
            dist/SBOM-${{ steps.version.outputs.VERSION }}.json
            ${{ steps.docker-build.outputs.artifact-digest }}

  # -----------------------------------------------------------
  # Phase 2: Production deploy — gated manual approval via OIDC
  # -----------------------------------------------------------
  deploy-staging:
    name: deploy to staging
    needs: release-build
    runs-on: ubuntu-latest
    environment:
      name: staging
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Deploy to staging
        run: |
          # OIDC federated login — no long-lived credentials
          aws sts get-caller-identity  # validates OIDC federation first
          echo "Deploying v${{ needs.release-build.outputs.version }} to staging..."

          # Actual deploy command (e.g. kubectl rollout, ECS, Fly, etc.)
          # kubectl set image deployment/api api=ghcr.io/repo/api:${{ needs.release-build.outputs.version }}
        env:
          AWS_REGION: us-east-1

  deploy-production:
    name: deploy to production
    needs: [release-build, deploy-staging]
    runs-on: ubuntu-latest
    if: ${{ !inputs.dry_run }}
    environment:
      name: production
    steps:
      - uses: actions/checkout@v4

      - name: OIDC deploy to production
        run: |
          echo "v${{ needs.release-build.outputs.version }} deploying to production (OIDC)"
          # Example: AWS ECR pull via OIDC
          aws ecr-public get-login-password --region us-east-1 | \
            docker login --username AWS --password-stdin ${{ env.REGISTRY }}
          # Rollout:
          # kubectl set image ...
        env:
          AWS_REGION: us-east-1
```

### Merge-queue integration

For merge queues, add a **required status check** for the `ci-pass` job from the CI workflow above. In GitHub Branch Protection Rules:

```
Required status checks:
  ✓ install & lint
  ✓ build (all targets)
  ✓ test (Node 20, shard 1/4 through 4/4)
  ✓ test (Node 22, shard 1/4 through 4/4)
  ✓ typecheck
  ✓ release build          (optional — only on main push)
```

Then configure the branch protection rule: **"Require status checks to pass before merging"** → **"Include Administrators"** → **"Status checks must pass for"** → select all CI jobs. This makes the merge queue wait until every matrix job completes successfully before enqueuing the PR for merge.

## Cache Key Design — Deep Dive

The caching strategy has two layers, each keyed differently:

### Layer 1: pnpm Store Cache (dependency layer)

```
key: pnpm-{os}-{node_version}-{hash(pnpm-lock.yaml)}-{hash(package.json)}
restore-keys:
  pnpm-{os}-{node_version}-{hash(pnpm-lock.yaml)}
  pnpm-{os}-{node_version}-
```

**Why this matters:**
- `pnpm-lock.yaml` captures the exact resolved dependency tree — changing any transitive dep invalidates the cache.
- `package.json` captures workspace-level peer deps overrides or optional deps. Hashing both prevents subtle mismatches where the lockfile is fine but the root manifest changed in a way that affects resolution.
- The `pnpm/action-setup@v4` already caches by default, but the explicit `actions/cache` step gives us deterministic keys and lets us gate `pnpm install --frozen-lockfile` behind a cache-hit check to save ~45 seconds per job.
- Including `node_version` in the key because some native modules compile differently per Major.minor Node.

### Layer 2: Turbo Build Cache (build-output layer)

```
key: turbo-build-{os}-{sha}
restore-keys:
  turbo-build-{os}-
```

**Why SHA-based:**
- GitHub's `github.sha` is the HEAD commit of the branch/Push. Since each commit gets its own SHA, every fresh commit gets a cold start — but commits that share parents (like merge commits) may find partial restores.
- For PRs specifically, consider swapping to `turbo-cache-key` which Turbo provides natively (it stores a cache per branch prefix):
  ```bash
  TURBO_CACHE_KEY=turbo-${{ runner.os }}-${{ github.ref_name }}
  TURBO_REMOTE_ONLY=true
  ```
- If using **Turborepo Remote Cache** (free for public, paid for private), set `TURBO_TOKEN` and `TURBO_TEAM` as environment variables — then Turborepo handles remote caching automatically and the local `actions/cache` becomes a warm-layer fallback.

### Caching interaction diagram

```
Commit pushed
    │
    ▼
Cache lookup: pnpm-{os}-{20}-{lockhash}-{pkghash}
    │                  │
    │             NO MATCH → pnpm install --frozen-lockfile (cold)
    │                  ↓
    │             HIT → skip install, save ~45s
    │
    ▼
Turbo build: --cache-dir=.turbo
    │
    ├── Local cache MISS → compute, write to .turbo directory
    │                      (uploaded to remote if TURBO_TOKEN configured)
    │
    └── Local cache HIT → skip, serve from .turbo
```

## OIDC Trust Configuration — Step by Step

### Architecture

```
GitHub Actions (workflow) ──[OIDC JWT bearer token]──► IAM Provider (AWS STS)
                                                        │
                                               AssumeRoleWithWebIdentity
                                                        │
                                            Short-lived session credentials
                                                        │
                                              Deploy to production
```

No long-lived IAM access keys are stored. The workflow exchanges a time-bound JWT from GitHub's OIDC provider for temporary AWS credentials.

### Step 1: Create an IAM Identity Provider in AWS

```bash
aws iam create-provider \
  --type URL \
  --provider-url https://token.actions.githubusercontent.com \
  --client-id-list sts.amazonaws.com \
  --thumbprint-list ""   # Not needed for GitHub Actions; omit
```

### Step 2: Create an IAM Role with OIDC Trust Policy

```json
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Principal": {
        "Federated": "arn:aws:iam::ACCOUNT_ID::provider/GitHubActions"
      },
      "Action": "sts:AssumeRoleWithWebIdentity",
      "Condition": {
        "StringEquals": {
          "token.actions.githubusercontent.com:aud": "sts.amazonaws.com"
        },
        "StringLike": {
          "token.actions.githubusercontent.com:sub": "repo:sanjeevafk/agent-skills:ref:refs/heads/main"
        }
      }
    }
  ]
}
```

Replace `repo:owner/repo` with your actual owner/repo. You can narrow further with path conditions:
- `"repo:sanjeevafk/agent-skills:ref:refs/heads/main"` — only main branch
- `"repo:sanjeevafk/agent-skills:pull_request"` — only PRs
- `"repo:sanjeevafk/agent-skills:ref:refs/tags/v*"` — only releases

### Step 3: Attach Permissions Policy

```json
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": [
        "ecs:UpdateService",
        "ecr:GetAuthorizationToken",
        "ecr:BatchCheckLayerAvailability",
        "ecr:GetDownloadUrlForLayer",
        "ecr:BatchGetImage"
      ],
      "Resource": "*"
    }
  ]
}
```

### Step 4: Use in the Workflow

In the deploy job, inject the role ARN and let `aws-cli` auto-exchange:

```yaml
- name: Configure AWS credentials
  uses: aws-actions/configure-aws-credentials@v4
  with:
    role-to-assume: arn:aws:iam::ACCOUNT_ID::role/GitHubActionsDeploy
    aws-region: us-east-1
    # No aws_access_key_id or aws_secret_access_key needed
```

### Why this is more secure than long-lived keys

| Factor | Long-lived keys | OIDC federation |
|---|---|---|
| Credential storage | Secrets manager (always accessible) | Never leaves the runner |
| Revocation | Must rotate all 4 keys | Trust policy update is instant |
| Blast radius | Any compromised key grants persistent access | Token expires after job ends (~15 min TTL from GitHub) |
| Scoping | Overly broad IAM policies common | Exact branch/ref/tag conditions enforce least privilege |
| Audit trail | IAM User activity | CloudTrail shows `AssumeRoleWithWebIdentity` with subject details |

|---|---|
| `TURBO_TEAM` | Set in repo vars for Turborepo remote cache |

| `NPM_TOKEN` | Secret for package publishing |
| `VERCEL_TOKEN` | Secret for preview deployments (replace with your provider) |

## 1. `.github/workflows/ci.yml` — Core Continuous Integration

```yaml
name: CI — Lint · Build · Test

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

permissions:
  contents: read
  # Allow uploading dependency graph (optional Dependabot integration)
  # actions: read

concurrency:
  group: ${{ github.workflow }}-${{ github.head_ref || github.run_id }}
  cancel-in-progress: true

# ──────────────────────────────────────────────
# Matrix definition — 3 apps × 2 Node LTS × 4 shards
# ──────────────────────────────────────────────
env:
  PNPM_CACHE_DIR: ~/.local/share/pnpm/store
  NODE_VERSION_DEFAULT: 22

jobs:
  # ── Step 0: Detect changed apps/libraries ──
  detect-changes:
    name: Detect changes
    runs-on: ubuntu-latest
    permissions:
      pull-requests: read
    outputs:
      # JSON array of changed app names, e.g. ["web","api"]
      changed_apps: ${{ steps.filter.outputs.changed_apps }}
      # Boolean: any library change (libs share one test suite)
      changed_libs: ${{ steps.filter.outputs.changed_libs }}
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Filter changed files
        id: filter
        uses: dorny/paths-filter@v3
        with:
          filters: |
            web:
              - 'apps/web/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'
            api:
              - 'apps/api/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'
            dashboard:
              - 'apps/dashboard/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'
            libs:
              - 'packages/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'

      - name: Echo detection (for debugging)
        run: |
          echo "Changed apps: ${{ steps.filter.outputs.changed_apps }}"
          echo "Changed libs: ${{ steps.filter.outputs.changed_libs }}"

  # ── Step 1: Shared dependency install with aggressive caching ──
  install:
    name: Install dependencies (${{ matrix.node-version }})
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        node-version: [20, 22]
    outputs:
      cache-hit: ${{ steps.pnpm-cache.outputs.cache-hit }}
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      # ---- CACHE KEY DESIGN ----
      # Primary key = OS + hash of the lockfile.
      #   • Deterministic: identical lockfile → identical key → cache hit.
      #   • Fast lookup: hash is O(lockfile size), typically < 1 MB.
      # Restore fallback chain:
      #   1. Exact match: pnpm-store-{os}-{lockhash}
      #   2. Prefix match: pnpm-store-{os}-       → picks last valid prefix cache
      #   3. Fallback:     pnpm-store-             → picks any pnpm-store cache from the
      #                     branch / recent runs
      # Why multi-key?  When the lockfile changes (new dep), exact miss but the
      #   prefix hit restores ~90 % of the store, avoiding full re-download.
      #
      # Workspace cache mirrors the same pattern for node_modules so that
      # `pnpm install` only resolves missing entries after the initial warm-up.
      #
      # Lockfile-baked keys guarantee stale-cache elimination every time
      # dependencies change, while prefix chains keep cold-start overhead low.

      - name: Cache pnpm global store
        id: pnpm-cache
        uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-store-${{ runner.os }}-
            pnpm-store-

      - name: Setup Node.js ${{ matrix.node-version }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: pnpm

      - name: Install pnpm
        uses: pnpm/action-setup@v3
        with:
          version: latest  # or pin to a specific version, e.g. "9.15.0"

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Cache workspace node_modules
        uses: actions/cache@v4
        with:
          path: "**/node_modules"
          key: workspace-${{ runner.os }}-${{ matrix.node-version }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            workspace-${{ runner.os }}-${{ matrix.node-version }}-
            workspace-${{ runner.os }}-

  # ── Step 2: Lint (full monorepo, always runs once) ──
  lint:
    name: Lint
    needs: [install]
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Restore pnpm store (from install job cache)
        uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-store-${{ runner.os }}-
            pnpm-store-

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION_DEFAULT }}
          cache: pnpm

      - name: Install pnpm
        uses: pnpm/action-setup@v3
        with:
          version: latest

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Run linter (root-level)
        run: pnpm run lint

      # Root-level lint catches cross-app issues (imports from wrong packages).
      # App-specific lint runs inside the build jobs below.

  # ── Step 3: Build per app (only changed apps) ──
  build:
    name: Build — ${{ matrix.app }}
    needs: [detect-changes, install]
    if: >-
      contains(needs.detect-changes.outputs.changed_apps, '${{ matrix.app }}')
      || needs.detect-changes.outputs.changed_libs == 'true'
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app:
          - web
          - api
          - dashboard
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Restore pnpm store
        uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-store-${{ runner.os }}-
            pnpm-store-

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION_DEFAULT }}
          cache: pnpm

      - name: Install pnpm
        uses: pnpm/action-setup@v3
        with:
          version: latest

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      # Rebuild affected libs if any library changed
      - name: Build libraries
        if: needs.detect-changes.outputs.changed_libs == 'true'
        run: pnpm --filter './packages/**' build

      - name: Build ${{ matrix.app }}
        run: pnpm --filter "./apps/${{ matrix.app }}" build

      - name: Upload build artifacts
        uses: actions/upload-artifact@v4
        with:
          name: build-${{ matrix.app }}-${{ github.sha }}
          path: |
            apps/${{ matrix.app }}/dist/
          retention-days: 7

  # ── Step 4: Test matrix — Node 20/22 × 4 shards per app ──
  test:
    name: Test — ${{ matrix.app }} (Node ${{ matrix.node-version }}, shard ${{ matrix.shard }}/4)
    needs: [detect-changes, install]
    if: >-
      contains(needs.detect-changes.outputs.changed_apps, '${{ matrix.app }}')
      || needs.detect-changes.outputs.changed_libs == 'true'
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app:
          - web
          - api
          - dashboard
        node-version:
          - 20
          - 22
        shard:
          - "1/4"
          - "2/4"
          - "3/4"
          - "4/4"
      max-parallel: 24   # caps concurrency; omit to let GitHub schedule freely

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Restore pnpm store
        uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-store-${{ runner.os }}-
            pnpm-store-

      - name: Setup Node.js ${{ matrix.node-version }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: pnpm

      - name: Install pnpm
        uses: pnpm/action-setup@v3
        with:
          version: latest

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      # Rebuild libs before testing if libs changed
      - name: Rebuild affected libs
        if: needs.detect-changes.outputs.changed_libs == 'true'
        run: pnpm --filter './packages/**' build

      - name: Run tests (shard ${{ matrix.shard }})
        env:
          CI: "true"
          NODE_ENV: test
        run: >-
          pnpm
            --filter "./apps/${{ matrix.app }}"
            test
            --shard=${{ matrix.shard }}

      - name: Upload test report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: test-report-${{ matrix.app }}-node${{ matrix.node-version }}-shard${{ matrix.shard }}-${{ github.sha }}
          path: |
            apps/${{ matrix.app }}/coverage/
            apps/${{ matrix.app }}/.xunit-results*
          retention-days: 14

  # ── Step 5: End-to-end smoke tests (run against build artifacts) ──
  smoke-test:
    name: Smoke test — ${{ matrix.app }}
    needs: [detect-changes, install, build]
    if: >-
      contains(needs.detect-changes.outputs.changed_apps, '${{ matrix.app }}')
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app:
          - web
          - api
          - dashboard
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download build artifacts
        uses: actions/download-artifact@v4
        with:
          name: build-${{ matrix.app }}-${{ github.sha }}
          path: apps/${{ matrix.app }}/dist/

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION_DEFAULT }}
          cache: pnpm

      - name: Install pnpm
        uses: pnpm/action-setup@v3
        with:
          version: latest

      - name: Install deps + run smoke test
        run: |
          pnpm install --frozen-lockfile
          pnpm --filter "./apps/${{ matrix.app }}" test:e2e

  # ── Step 6: Combined status gate ──
  ci-gate:
    name: CI Gate
    needs:
      - lint
      - build
      - test
      - smoke-test
    if: always()
    runs-on: ubuntu-latest
    steps:
      - name: Check CI results
        run: |
          failed=$(python3 -c "
import json, sys
statuses = {
  '${{ needs.lint.result }}': 'lint',
  '${{ needs.build.result }}': 'build',
  '${{ needs.test.result }}': 'test',
  '${{ needs.smoke-test.result }}': 'smoke-test',
}
failed_jobs = [v for k, v in statuses.items() if k != 'success' and k != 'skipped']
print(json.dumps(failed_jobs))
")
          if [ "$failed" != "[]" ] && [ "$failed" != "" ]; then
            echo "::error::CI failed for jobs: $failed"
            exit 1
          fi
          echo "All CI checks passed."

  # ── Step 7: Ephemeral preview environment per PR ──
  deploy-preview:
    name: Deploy Preview (${{ matrix.app }})
    needs: [detect-changes, build]
    if: >-
      github.event_name == 'pull_request'
      && contains(needs.detect-changes.outputs.changed_apps, '${{ matrix.app }}')
    runs-on: ubuntu-latest
    environment:
      name: preview
      url: >-
        https://${{ matrix.app }}-${{ github.pull_request.number }}.${{ vars.PREVIEW_DOMAIN || 'preview.internal' }}
    strategy:
      fail-fast: false
      matrix:
        app:
          - web
          - api
          - dashboard
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download build artifacts
        uses: actions/download-artifact@v4
        with:
          name: build-${{ matrix.app }}-${{ github.sha }}
          path: apps/${{ matrix.app }}/dist/

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION_DEFAULT }}

      - name: Start preview server
        run: |
          cd apps/${{ matrix.app }}
          npx serve dist -p 3000 &
          sleep 3

      - name: Health check preview
        run: |
          for i in $(seq 1 30); do
            if curl -sf http://localhost:3000/healthz >/dev/null 2>&1; then
              echo "Preview healthy ✓"
              exit 0
            fi
            echo "Waiting... ($i/30)"
            sleep 2
          done
          echo "Preview health check failed"
          exit 1

      - name: Output preview URL
        run: |
          echo "Preview URL: https://${{ matrix.app }}-${{ github.pull_request.number }}.${{ vars.PREVIEW_DOMAIN || 'preview.internal' }}"
```

## 2. `.github/workflows/deploy.yml` — Production Deployment (OIDC Federation)

```yaml
name: Deploy — Production (Manual, OIDC)

on:
  workflow_dispatch:
    inputs:
      version:
        description: "Release tag to deploy (e.g., v1.2.3)"
        required: true
        type: string
      app:
        description: "App to deploy ('all', 'web', 'api', 'dashboard')"
        required: true
        default: "all"
        type: choice
        options:
          - all
          - web
          - api
          - dashboard
      dry_run:
        description: "Simulate deployment without applying changes"
        required: false
        default: "false"
        type: boolean

permissions:
  contents: read
  id-token: write   # REQUIRED for OIDC federation (JWT minting)
  deployments: write

jobs:
  # ── Pre-deploy: validate target exists ──
  validate:
    name: Validate release
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          ref: ${{ github.event.inputs.version }}

      - name: Verify release tag
        run: |
          TAG="${{ github.event.inputs.version }}"
          if ! grep -q "^$TAG$" .git/refs/tags/"$TAG" 2>/dev/null; then
            echo "Error: tag '$TAG' not found on this branch."
            exit 1
          fi
          echo "Tag $TAG verified."

      - name: Check CI status for tag
        run: |
          # Ensure CI passed for the tagged commit before allowing deploy
          curl -fsSL \
            -H "Authorization: Bearer ${{ secrets.GITHUB_TOKEN }}" \
            "https://api.github.com/repos/${{ github.repository }}/commits/${{ github.event.inputs.version }}/check-runs" \
            | jq -r '.check_runs[] | select(.name | test("CI"; "i")) | "\(.name): \(.conclusion)"' \
            | while read -r line; do
                if echo "$line" | grep -qi "failure\|cancelled"; then
                  echo "::error::$line"
                  exit 1
                fi
              done
          echo "CI checks green for ${{ github.event.inputs.version }}."

  # ── Deploy gateway: one-at-a-time per app, environment-gated ──
  deploy-app:
    name: Deploy ${{ matrix.app }} to Production
    needs: validate
    runs-on: ubuntu-latest
    environment:
      name: production
      # PRODUCTION ENVIRONMENT MUST BE CONFIGURED WITH:
      #   Required reviewers: at least 1 approver
      #   Wait timer: 5 min (to catch rollbacks)
      #   Deployment branches: tags only (via GitHub UI settings)
    strategy:
      fail-fast: false
      matrix:
        app:
          - web
          - api
          - dashboard
        exclude:
          # Respect the --app input filter
          - app: web
            _excluded_by_input: ${{ github.event.inputs.app != 'all' && github.event.inputs.app != 'web' }}
          - app: api
            _excluded_by_input: ${{ github.event.inputs.app != 'all' && github.event.inputs.app != 'api' }}
          - app: dashboard
            _excluded_by_input: ${{ github.event.inputs.app != 'all' && github.event.inputs.app != 'dashboard' }}
      max-parallel: 1   # serial deployment prevents cascading failures

    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          ref: ${{ github.event.inputs.version }}

      # ── OIDC FEDERATION ──
      # No long-lived AWS/GCP/Azure credentials needed.
      #
      # How it works:
      #   1. GitHub Actions mint a short-lived JWT (expires ~5 min)
      #      signed by the GitHub OIDC provider.
      #   2. Cloud provider (AWS/GCP/Azure) trusts the GitHub JWT
      #      via a pre-configured identity provider trust relationship.
      #   3. The cloud provider validates JWT claims (repo, branch,
      #      environment, ref) against IAM policy conditions.
      #   4. On success, the provider returns short-lived temporary
      #      credentials (STS AssumeRoleWithWebIdentity).
      #
      # Trust configuration (AWS example — configure in your AWS account):
      #
      #   IAM Role: arn:aws:iam::<ACCOUNT_ID>:role/github-actions-deploy-role
      #
      #   Trust Policy:
      #   {
      #     "Version": "2012-10-17",
      #     "Statement": [{
      #       "Effect": "Allow",
      #       "Principal": {
      #         "Federated": "arn:aws:iam::<ACCOUNT_ID>::oidc-provider/token.actions.githubusercontent.com"
      #       },
      #       "Action": "sts:AssumeRoleWithWebIdentity",
      #       "Condition": {
      #         "StringEquals": {
      #           "token.actions.githubusercontent.com:aud": "sts.amazonaws.com",
      #           "token.actions.githubusercontent.com:iss": "https://token.actions.githubusercontent.com"
      #         },
      #         "StringLike": {
      #           "token.actions.githubusercontent.com:sub": [
      #             "repo:<OWNER>/<REPO>:environment:production",
      #             "repo:<OWNER>/<REPO>:ref:refs/tags/v*.*.*"
      #           ]
      #         }
      #       }
      #     }]
      #   }
      #
      # This policy restricts usage to:
      #   • The exact repository (prevents fork abuse)
      #   • The "production" environment (scope isolation)
      #   • Semver-tagged refs only (prevents arbitrary branch deploys)
      #
      # Per-app IAM policies grant minimum-scoped permissions:
      #   apps/web   → ECS/Fargate for the web service
      #   apps/api   → ECS/Fargate for the API service
      #   apps/dashboard → ECS/Fargate for the dashboard service

      - name: Configure AWS Credentials (OIDC)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::<ACCOUNT_ID>::role/github-actions-deploy-role
          aws-region: ${{ secrets.AWS_REGION || 'us-east-1' }}
          role-session-name: github-actions-${{ matrix.app }}-${{ github.run_id }}

      - name: Authenticate to Docker registry
        run: |
          # GitHub Container Registry (GHCR) — OIDC not needed, use GITHUB_TOKEN
          echo "${{ secrets.GITHUB_TOKEN }}" | \
            docker login ghcr.io -u ${{ github.actor }} --password-stdin

      - name: Pull deployment manifest
        run: |
          VERSION="${{ github.event.inputs.version }}"
          APP="${{ matrix.app }}"
          REGISTRY="ghcr.io/${{ github.repository_owner }}"
          IMAGE="${REGISTRY}/${APP}"
          TAG="${VERSION//v/}"  # strip leading 'v'

          echo "Image: ${IMAGE}:${TAG}"

          # Fetch image digest for audit trail
          DIGEST=$(docker inspect "${IMAGE}:${TAG}" 2>/dev/null | \
                   jq -r '.[0].Config.Digest // empty')
          echo "Digest: ${DIGEST:-unknown}"

      - name: Deploy ${{ matrix.app }}
        env:
          DEPLOY_DRY_RUN: ${{ github.event.inputs.dry_run }}
          TARGET_VERSION: ${{ github.event.inputs.version }}
          DEPLOY_APP: ${{ matrix.app }}
        run: |
          VERSION="${TARGET_VERSION}"
          APP="${DEPLOY_APP}"
          REGISTRY="ghcr.io/${{ github.repository_owner }}"
          IMAGE_TAG="${VERSION//v/}"

          if [ "$DEPLOY_DRY_RUN" = "true" ]; then
            echo "[DRY RUN] Would deploy ${IMAGE_TAG} to production"
            echo "[DRY RUN] Commands executed:"
            echo "  kubectl set image deployment/${APP} ${APP}=${REGISTRY}/${APP}:${IMAGE_TAG}"
            exit 0
          fi

          echo "Deploying ${APP} ${IMAGE_TAG}..."

          # ── Replace with your actual deploy command ──
          # Kubernetes example:
          #   kubectl set image deployment/${APP} \
          #     ${APP}=${REGISTRY}/${APP}:${IMAGE_TAG}
          #   kubectl rollout status deployment/${APP} --timeout=300s

          # ECS/Fargate example:
          #   aws ecs update-service \
          #     --cluster prod \
          #     --service ${APP} \
          #     --task-definition ${APP}:latest \
          #     --force-new-deployment

          echo "Deployment successful: ${APP} -> ${IMAGE_TAG}"

      - name: Smoke test deployed application
        run: |
          echo "Running smoke test against deployed ${{ matrix.app }}..."
          # Replace with actual health endpoint
          # MAX_RETRIES=30
          # for i in $(seq 1 $MAX_RETRIES); do
          #   if curl -sf "https://${{ matrix.app }}.example.com/healthz" >/dev/null; then
          #     echo "Health check passed"
          #     exit 0
          #   fi
          #   sleep 5
          # done
          echo "Smoke test skipped (configure your health endpoint)"
          exit 0

      - name: Record deployment metadata
        run: |
          echo '{"version":"${{ github.event.inputs.version }}","app":"${{ matrix.app }}","deployed_at":"$(date -u +%Y-%m-%dT%H:%M:%SZ)","runner":"${{ runner.id }}"}' \
            | jq '.'

  # ── Post-deploy summary ──
  deploy-summary:
    name: Deployment Summary
    needs: deploy-app
    if: always()
    runs-on: ubuntu-latest
    steps:
      - name: Summarize results
        run: |
          echo "## Deployment Summary" >> $GITHUB_STEP_SUMMARY
          echo "- **Version**: ${{ github.event.inputs.version }}" >> $GITHUB_STEP_SUMMARY
          echo "- **Apps**: ${{ github.event.inputs.app }}" >> $GITHUB_STEP_SUMMARY
          echo "- **Status**: ${{ needs.deploy-app.result }}" >> $GITHUB_STEP_SUMMARY
          echo "- **Triggered by**: ${{ github.triggering_actor }}" >> $GITHUB_STEP_SUMMARY
          echo "- **Run**: <${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}>" >> $GITHUB_STEP_SUMMARY
```

## 3. `.github/workflows/release.yml` — SBOM + Provenance Attestation

```yaml
name: Release — SBOM · Attestation · Push

on:
  push:
    tags:
      - "v*.*.*"   # Semantic version tags: v1.0.0, v2.3.1, etc.
  workflow_dispatch:
    inputs:
      tag:
        description: "Tag to release (e.g., v1.2.3)"
        required: true
        type: string

permissions:
  contents: read
  id-token: write     # Required for SLSA provenance signing
  packages: write     # Push to GHCR

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  # ── Single build job (shared across all release assets) ──
  build-release:
    name: Build & Package
    runs-on: ubuntu-latest
    outputs:
      build-id: ${{ steps.build-id.outputs.id }}
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Determine tag
        id: tag
        run: |
          TAG="${{ github.event.inputs.tag || github.ref_name }}"
          echo "tag=$TAG" >> $GITHUB_OUTPUT
          echo "Resolved tag: $TAG"

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - name: Install pnpm
        uses: pnpm/action-setup@v3
        with:
          version: latest

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build all packages
        run: pnpm --recursive build

      # Generate a unique build identifier for artifact correlation
      - name: Set build ID
        id: build-id
        run: echo "id=${{ steps.tag.outputs.tag }}-$(git rev-parse --short HEAD)" >> $GITHUB_OUTPUT

      - name: Build all apps
        run: |
          pnpm --filter "./apps/web" build
          pnpm --filter "./apps/api" build
          pnpm --filter "./apps/dashboard" build

      # ── Archive build output for downstream jobs ──
      - name: Package build artifacts
        run: |
          mkdir -p release/artifacts
          cp -r apps/*/dist release/artifacts/
          ls -la release/artifacts/

      - name: Upload build artifacts
        uses: actions/upload-artifact@v4
        with:
          name: release-build
          path: release/artifacts/
          retention-days: 30

      - name: Save SBOM sources list
        run: |
          # Manifest of all packages/apps included in the SBOM
          pnpm list --recursive --depth=-1 --json > release/package-manifest.json
          cat release/package-manifest.json

  # ── SBOM Generation ──
  generate-sbom:
    name: Generate SBOM
    needs: build-release
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download build artifacts
        uses: actions/download-artifact@v4
        with:
          name: release-build
          path: release/artifacts/

      - name: Generate SBOM (SPDX)
        uses: anchore/sbom-action@v0
        with:
          path: ./release/artifacts
          format: spdx-json
          output-file: release/sbom.spdx.json
          artifact-name: sbom-spdx

      # Also generate CycloneDX for broader toolchain compatibility
      - name: Generate SBOM (CycloneDX)
        uses: anchore/sbom-action@v0
        with:
          path: ./release/artifacts
          format: cyclonedx-json
          output-file: release/sbom.cyclonedx.json
          artifact-name: sbom-cyclonedx

      - name: Upload SBOM artifacts
        uses: actions/upload-artifact@v4
        with:
          name: release-sboms
          path: |
            release/sbom.*.json
          retention-days: 90

  # ── SLSA Provenance Attestation ──
  attest-provenance:
    name: Attest SLSA Provenance
    needs: [build-release, generate-sbom]
    runs-on: ubuntu-latest
    permissions:
      actions: read
      id-token: write
      contents: write
      attestations: write
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download build artifacts
        uses: actions/download-artifact@v4
        with:
          name: release-build
          path: release/artifacts/

      - name: Download SBOMs
        uses: actions/download-artifact@v4
        with:
          name: release-sboms
          path: release/sboms/

      # Build the predicate JSON describing how this release was produced
      - name: Prepare attestation predicate
        id: predicate
        run: |
          cat > release/predicate.json << 'PREDICATE_EOF'
          {
            "buildType": "github-actions",
            "builder": {
              "id": "https://github.com/actions/runner"
            },
            "metadata": {
              "completeness": {
                "environment": true,
                "parameters": true,
                "products": true
              },
              "reproducible": false
            },
            "invocation": {
              "configSource": {
                "uri": "git+https://github.com/${{ github.repository }}.git",
                "digest": {
                  "sha1": "${{ github.sha }}"
                },
                "entryPoint": ".github/workflows/release.yml"
              }
            },
            "products": [
              {
                "digest": {
                  "sha256": "PLACEHOLDER_SHA256"
                },
                "uri": "oci://ghcr.io/${{ github.repository_owner }}/web:${{ github.ref_name }}"
              }
            ],
            "metadata": {
              "buildStartedOn": "$(date -u +%Y-%m-%dT%H:%M:%SZ)",
              "metadata": {
                "repoUrl": "${{ github.server_url }}/${{ github.repository }}",
                "commit": "${{ github.sha }}",
                "trigger": "${{ github.event_name }}"
              }
            }
          }
          PREDICATE_EOF

      # Calculate SHA-256 of build artifacts
      - name: Compute artifact hash
        run: |
          HASH=$(find release/artifacts -type f -exec sha256sum {} \; | sort | sha256sum | awk '{print $1}')
          echo "artifact-hash=$HASH" >> $GITHUB_OUTPUT
          echo "SHA256: $HASH"

      - name: Attest build provenance
        uses: actions/attest-slsa-provenance@v2
        with:
          predicate-type: slsa
          subject-path: "release/artifacts"
          # Note: subject must be a single file or directory
          # For multiple subjects, iterate:
          #   for f in release/artifacts/*; do
          #     gh attestation create ... --subject "$f"
          #   done

      - name: Upload provenance attestation
        uses: actions/upload-artifact@v4
        with:
          name: release-provenance
          path: release/*.json
          retention-days: 365

  # ── Push to GHCR ──
  push-container:
    name: Push Containers to GHCR
    needs: [build-release, generate-sbom, attest-provenance]
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app:
          - web
          - api
          - dashboard
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download build artifacts
        uses: actions/download-artifact@v4
        with:
          name: release-build
          path: apps/${{ matrix.app }}/dist/

      - name: Log in to Container Registry
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Extract metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ${{ env.REGISTRY }}/${{ github.repository_owner }}/${{ matrix.app }}
          tags: |
            type=semver,pattern={{version}},value=${{ github.ref_name }}
            type=sha,value=${{ github.sha }}
            type=raw,value=latest,enable={{is_default_branch}}

      - name: Build and push Docker image
        uses: docker/build-push-action@v6
        with:
          context: apps/${{ matrix.app }}
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          provenance: false   # We use our own attestation via attest-slsa-provenance
          sbom: true
          cache-from: type=gha
          cache-to: type=gha,mode=max

      - name: Sign image with Cosign
        uses: sigstore/cosign-installer@v3
        with:
          cosign-release: 'v2.2.2'

      - name: Sign the OCI image
        run: |
          IMG="${{ env.REGISTRY }}/${{ github.repository_owner }}/${{ matrix.app }}:${{ github.ref_name }}"
          cosign sign --yes "${IMG}"
```

## 5. Architecture Explanations

### 5.1 Cache-Key Design

The cache strategy uses **two layers** with deterministic, lockfile-baked keys:

| Layer | Path | Key Pattern | Purpose |
|---|---|---|---|
| Global store | `~/.local/share/pnpm/store` | `pnpm-store-{OS}-{hash(pnpm-lock.yaml)}` | Cached downloaded tarballs — shared across all Node versions |
| Workspace modules | `**/node_modules` | `workspace-{OS}-{NodeVer}-{hash(pnpm-lock.yaml)}` | Cached resolved packages — per–Node version to avoid ABI mismatch |

**Restore-key fallback chain:**
```
Exact match:   pnpm-store-Linux-a1b2c3d4
Prefix match:  pnpm-store-Linux-          ← recovers 90% of store from last run
Broad match:   pnpm-store-                 ← picks any old pnpm-store cache
```

**Why this design works:**

- **Deterministic eviction:** Every dependency change produces a new lockfile hash → completely fresh cache. No stale dependencies ever slip through.
- **Warm partial hits:** When the lockfile changes but most packages remain the same, the prefix cache restores the majority of the store. The difference between the old and new lockfile is re-downloaded. Typical savings: 3–8 minutes on a cold install.
- **Node-version isolation:** The workspace cache includes `${{ matrix.node-version }}` because native modules (e.g., `node-sass`, `sharp`) compile against a specific Node ABI. Mixing ABIs corrupts `node_modules`.
- **Store sharing across matrix rows:** The pnpm store cache does *not* include Node version because the store holds compressed tarballs, not compiled binaries. Both Node 20 and Node 22 can safely read from the same store.

**Key rotation behavior:**

```
Run 1 (fresh):  key = pnpm-store-Linux-abc123  → miss → download all → save
Run 2 (same):   key = pnpm-store-Linux-abc123  → HIT → skip download (~30 sec)
Run 3 (dep added): key = pnpm-store-Linux-xyz789  → miss
                         restore: pnpm-store-Linux-abc123  → partial HIT
                         diff install: 1 new package
                         total time: ~90 sec (vs ~4 min cold start)
```

### 5.2 OIDC Federation Configuration (Zero Long-Lived Keys)

**The trust chain:**

```
┌──────────────────────┐         JWT (signed by GitHub)         ┌──────────────────────┐
│   GitHub Actions     │ ─────────────────────────────────────► │   Cloud Provider      │
│  (workflow workflow) │                                        │  (AWS / GCP / Azure) │
└──────────────────────┘                                        └──────────┬───────────┘
                                                                    │
                                                     STS AssumeRoleWithWebIdentity
                                                        (temporary credentials)
                                                              │
                                                    ┌─────────▼──────────┐
                                                    │  Your Application   │
                                                    │  (ECS / GKE / AKS)  │
                                                    └────────────────────┘
```

**Step-by-step configuration (AWS):**

1. **Create the OIDC identity provider** in your AWS account:
   ```
   OpenIDConnectProvider:
     URL: https://token.actions.githubusercontent.com
     Audience: sts.amazonaws.com
   ```

2. **Create an IAM role** with a trust policy that constrains who can assume it:

   ```json
   {
     "Version": "2012-10-17",
     "Statement": [{
       "Effect": "Allow",
       "Principal": {
         "Federated": "arn:aws:iam::<ACCOUNT_ID>::oidc-provider/token.actions.githubusercontent.com"
       },
       "Action": "sts:AssumeRoleWithWebIdentity",
       "Condition": {
         "StringEquals": {
           "token.actions.githubusercontent.com:aud": "sts.amazonaws.com",
           "token.actions.githubusercontent.com:iss": "https://token.actions.githubusercontent.com"
         },
         "StringLike": {
           "token.actions.githubusercontent.com:sub": [
             "repo:<OWNER>/<REPO>:environment:production",
             "repo:<OWNER>/<REPO>:ref:refs/tags/v*.*.*"
           ]
         }
       }
     }]
   }
   ```

3. **Attach an inline policy** granting minimum-permission access:
   ```json
   {
     "Version": "2012-10-17",
     "Statement": [{
       "Effect": "Allow",
       "Action": [
         "ecs:UpdateService",
         "ecs:DescribeServices",
         "ecs:DescribeTasks"
       ],
       "Resource": "arn:aws:ecs:*:<ACCOUNT_ID>:service/prod/*"
     }]
   }
   ```

4. **Reference in the workflow:**
   ```yaml
   - uses: aws-actions/configure-aws-credentials@v4
     with:
       role-to-assume: arn:aws:iam::<ACCOUNT_ID>::role/github-actions-deploy-role
       aws-region: us-east-1
   ```

**Security properties:**

- **No static credentials stored** in GitHub secrets for cloud auth.
- **Token expires in ~5 minutes** — even if intercepted, the window is tiny.
- **Environment-bound:** The `StringLike` condition on `sub` restricts usage to the `production` environment. A compromised webhook from a `staging` environment cannot assume this role.
- **Ref-bound:** Only `refs/tags/v*.*.*` can trigger production deployments, preventing arbitrary branch deploys.
- **Repository-bound:** The `repo:<OWNER>/<REPO>` prefix prevents fork abuse.

### 5.3 Merge Queue Integration

The merge queue requires **all required status checks** to pass before a PR is accepted into the merge queue. Here's the wiring:

**In GitHub UI (Settings → Branches → Branch protection rules):**
1. Enable **"Include administrators"**
2. Enable **"Require status checks to pass before merging"**
3. Enable **"Require branches to be up to date before merging"**
4. Enable **"Require linear history"** (optional)
5. Search for required checks:
   - `Lint`
   - `Build — web`, `Build — api`, `Build — dashboard`
   - `Test — web (Node 20, shard 1/4)` through `Test — dashboard (Node 22, shard 4/4)`
   - `Smoke test — web`, `Smoke test — api`, `Smoke test — dashboard`
   - `CI Gate`

**How path-filtering affects the merge queue:**

When a PR touches only `apps/web/`, the `detect-changes` job determines `changed_apps = ["web"]`. Jobs for `api` and `dashboard` are marked `skipped` (not `failed`). The merge queue accepts `skipped` jobs — only `failure` blocks a merge. This is configured via:

```yaml
if: contains(needs.detect-changes.outputs.changed_apps, '${{ matrix.app }}')
```

Without this filter, unchanged app jobs would never run and the PR would be blocked indefinitely.

**Merge queue job naming convention:**

Job names become the status check names visible in the merge queue UI:

| Job Name (in YAML) | Status Check (shown in merge queue) |
|---|---|
| `lint` | `Lint` |
| `build` (matrix) | `Build — web`, `Build — api`, `Build — dashboard` |
| `test` (matrix) | `Test — web (Node 20, shard 1/4)`, etc. (12 checks) |
| `smoke-test` (matrix) | `Smoke test — web`, `Smoke test — api`, `Smoke test — dashboard` |
| `ci-gate` | `CI Gate` |

Total: **18 required status checks** for a fully touched PR, reduced proportionally for narrow PRs.

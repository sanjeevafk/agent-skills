## Workflow 1: `ci.yml` — Path-filtered CI, matrix tests, caching

```yaml
# .github/workflows/ci.yml
name: CI

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

env:
  PNPM_CACHE_DIR: ~/.local/share/pnpm/store

permissions:
  contents: read

jobs:
  # ── Step 0: discover changed paths ──────────────────────────────
  changes:
    runs-on: ubuntu-latest
    outputs:
      apps_admin: ${{ steps.filter.outputs.apps_admin }}
      apps_api: ${{ steps.filter.outputs.apps_api }}
      apps_web: ${{ steps.filter.outputs.apps_web }}
      all_libs: ${{ steps.filter.outputs.all_libs }}
    steps:
      - uses: actions/checkout@v4
      - uses: dorny/paths-filter@v3
        id: filter
        with:
          filters: |
            apps_admin:
              - 'apps/admin/**'
              - 'libs/**'       # libs touch → all apps re-test
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'
            apps_api:
              - 'apps/api/**'
              - 'libs/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'
            apps_web:
              - 'apps/web/**'
              - 'libs/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'
            all_libs:
              - 'libs/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - '.github/workflows/ci.yml'

  # ── Step 1: dependency install + lockfile cache ─────────────────
  setup:
    needs: changes
    runs-on: ubuntu-latest
    outputs:
      cache_hit: ${{ steps.restore-deps.outputs.cache-hit }}
    steps:
      - uses: actions/checkout@v4

      - name: Setup pnpm
        uses: pnpm/action-setup@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      # CACHE KEY DESIGN (see explanation below):
      #   pnpm-lock.yaml captures exact dependency tree.
      #   OS+node version are added because binaries are platform-specific.
      #   Fallback scans package.json hash when lockfile unchanged but deps updated via hotfix branch.
      - name: Restore dependency cache
        uses: actions/cache@v4
        id: restore-deps
        with:
          key: deps-${{ runner.os }}-node20-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            deps-${{ runner.os }}-node20-
            deps-${{ runner.os }}-

      - name: Install dependencies
        run: pnpm install --frozen-lockfile
        # --frozen-lockfile fails if lockfile doesn't match package.json — critical safety.

      - name: Cache turbo build artifacts
        uses: actions/cache@v4
        with:
          path: .turbo/cache
          key: turbo-build-${{ runner.os }}-node20-${{ hashFiles('pnpm-lock.yaml') }}-${{ github.sha }}
          restore-keys: |
            turbo-build-${{ runner.os }}-node20-${{ hashFiles('pnpm-lock.yaml') }}-

      - name: Save dependency cache for downstream jobs
        uses: actions/upload-artifact@v4
        with:
          name: node-modules
          path: |
            node_modules/
            */*/node_modules/
          retention-days: 1

  # ── Step 2: lint all repos ──────────────────────────────────────
  lint:
    needs: setup
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - uses: actions/download-artifact@v4
        with:
          name: node-modules
      - name: Run lint
        run: pnpm lint
        env:
          TURBO_TELEMETRY_DISABLED: '1'

  # ── Step 3: type-check all libs first (no app context needed) ───
  typecheck-libs:
    needs: setup
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - uses: actions/download-artifact@v4
        with:
          name: node-modules
      - name: Type-check libs
        run: pnpm --filter './libs/*' typecheck
        env:
          TURBO_TELEMETRY_DISABLED: '1'

  # ── Step 4: test matrix per-app ─────────────────────────────────
  test:
    needs:
      - changes
      - setup
      - typecheck-libs
    strategy:
      fail-fast: false
      matrix:
        node: [20, 22]
        app:
          - name: admin
            cond: ${{ needs.changes.outputs.apps_admin == 'true' }}
          - name: api
            cond: ${{ needs.changes.outputs.apps_api == 'true' }}
          - name: web
            cond: ${{ needs.changes.outputs.apps_web == 'true' }}
            exclude: false  # always run base lib tests
        include:
          # Also run lib-only tests every time regardless of filters
          - app:
              name: all-libs
            cond: true
        shard: [1/4, 2/4, 3/4, 4/4]
        exclude:
          # Skip apps whose filter didn't fire
          - app: { name: admin }
            cond: false
          - app: { name: api }
            cond: false
          - app: { name: web }
            cond: false
          # Don't shard the lib-only suite
          - app: { name: all-libs }
            shard: '1/4'
          - app: { name: all-libs }
            shard: '2/4'
          - app: { name: all-libs }
            shard: '3/4'
          - app: { name: all-libs }
            shard: '4/4'
    runs-on: ubuntu-latest
    continue-on-error: ${{ matrix.app.name == 'api' && endsWith(github.head_ref, 'experimental') }}
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node }}
          cache: 'pnpm'
      - uses: actions/download-artifact@v4
        with:
          name: node-modules

      - name: Build shared libs (with dep cache)
        run: |
          pnpm --filter './libs/*' build
          pnpm --filter './libs/*' ...build
        env:
          TURBO_TELEMETRY_DISABLED: '1'

      - name: Run tests${{ matrix.shard != '' && format(' (shard {0})', matrix.shard) || '' }}
        run: |
          if [ "${{ matrix.app.name }}" = "all-libs" ]; then
            FILTER='./libs/*'
          else
            FILTER="./apps/${{ matrix.app.name }}"
          fi
          pnpm --filter "$FILTER" test${{ matrix.shard != '' && format(' -- --test-shard={0}', matrix.shard) || '' }}
        env:
          TURBO_TELEMETRY_DISABLED: '1'
          NODE_VERSION: ${{ matrix.node }}

  # ── Step 5: integration tests (only on main push) ──────────────
  integration:
    needs: setup
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16-alpine
        env:
          POSTGRES_USER: test
          POSTGRES_PASSWORD: test
          POSTGRES_DB: test_db
        ports: ['5432:5432']
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - uses: actions/download-artifact@v4
        with:
          name: node-modules
      - name: Run integration tests
        run: pnpm --filter "./apps/api" test:integration
        env:
          DATABASE_URL: postgresql://test:test@localhost:5432/test_db
          TURBO_TELEMETRY_DISABLED: '1'

  # ── Status gate: all checks pass? ───────────────────────────────
  status-gate:
    needs:
      - changes
      - setup
      - lint
      - typecheck-libs
      - test
      - integration
    if: always()
    runs-on: ubuntu-latest
    steps:
      - name: Check results
        run: |
          failed=()
          for job in "${{ join(needs.*.result, ' ') }}"; do
            [[ "$job" == "failure" || "$job" == "cancelled" ]] && exit 1
          done
          echo "All checks passed ✅"
```

### Cache-Key Design Explained

**Dependency cache (`deps-{os}-node{ver}-{lockfile-hash}`):**

| Layer | Purpose |
|-------|---------|
| `runner.os` | Binaries differ between Linux/macOS/Windows |
| `node20` / `node22` | Native add-ons compiled per Node major version |
| `hashFiles('pnpm-lock.yaml')` | Canonical fingerprint of the exact resolved dependency tree |
| `restore-keys: deps-${{ runner.os }}-node20-` | Partial fallback — restores even if lockfile changed slightly (e.g., post-CI auto-fix), avoiding a full reinstall |

Why not use `tree-sha`? A lockfile hash is deterministic and scoped to dependencies only. If someone edits an app's source, the lockfile doesn't change, so the cache hits instantly — no wasted bandwidth rebuilding node_modules.

**Build artifact cache (`turbo-build-{os}-{node}-{lockfile}-{sha}`):**

Including `github.sha` means every push gets its own isolated cache layer. On PR pushes from the same branch, the lockfile-based prefix ensures we still get partial hits across commits that don't change deps.

Why both `--frozen-lockfile` AND a lockfile-cache? `--frozen-lockfile` is the correctness guard (prevents accidental installs). The cache is the speed guard. If the cache misses, `--frozen-lockfile` still works — pnpm resolves fresh against the lockfile and stores a new cache hit.

## Workflow 2: `preview.yml` — Ephemeral preview environments per PR

```yaml
# .github/workflows/preview.yml
name: Preview Environments

on:
  pull_request:
    branches: [main]
    types: [opened, synchronize, reopened, closed]

permissions:
  contents: read
  deployments: write

env:
  APPS: admin,api,web

jobs:
  preview:
    # Use a labeled environment so each deployment gets a unique URL
    environment:
      name: preview-${{ github.event.pull_request.number }}
      url: >-
        https://pr${{ github.event.pull_request.number }}-${{ matrix.app }}.${{ vars.PR previews_BASE_DOMAIN || 'preview.example.com' }}
    runs-on: ubuntu-latest
    if: github.event.action != 'closed'
    strategy:
      fail-fast: false
      matrix:
        app: [admin, api, web]
    steps:
      - uses: actions/checkout@v4

      - name: Setup pnpm
        uses: pnpm/action-setup@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Install & build
        run: |
          pnpm install --frozen-lockfile
          pnpm --filter "./apps/${{ matrix.app }}"... build
        env:
          NODE_ENV: production
          TURBO_TELEMETRY_DISABLED: '1'

      - name: Build Docker image
        run: |
          cd apps/${{ matrix.app }}
          docker build -t pr${{ github.event.pull_request.number }}-${{ matrix.app }}:${{ github.sha }} .

      # Cloud-run deploy (GCP example — swap for your provider)
      - name: Deploy to cloud
        id: deploy
        uses: google-github-actions/deploy-cloudrun@v2
        with:
          service: pr-pr${{ github.event.pull_request.number }}-${{ matrix.app }}
          image: gcr.io/${{ secrets.GCP_PROJECT }}/pr-pr${{ github.event.pull_request.number }}-${{ matrix.app }}:${{ github.sha }}
          region: us-central1
          env_vars: |
            PR_NUMBER=${{ github.event.pull_request.number }}
            COMMIT_SHA=${{ github.sha }}

      - name: Save preview URL
        uses: actions/github-script@v7
        with:
          script: |
            const url = '${{ steps.deploy.outputs.url }}';
            await github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: `🚀 Preview deployed: ${url}`
            });

  cleanup:
    runs-on: ubuntu-latest
    if: github.event.action == 'closed'
    steps:
      - uses: actions/checkout@v4
      - name: Destroy preview resources
        run: echo "Destroying preview resources for PR #${{ github.event.pull_request.number }}"
        env:
          PR_NUMBER: ${{ github.event.pull_request.number }}
```

### How ephemeral URLs work

Each PR gets an environment named `preview-{PR number}`, which GitHub maps to a unique subdomain. The `deployments: write` permission lets the workflow register each deployment as a GitHub Deployment with its own URL. When the PR closes, the environment is cleaned up (the comment/destroy step).

## Workflow 3: `release.yml` — Production deploy with OIDC + SBOM + attestation

```yaml
# .github/workflows/release.yml
name: Release & Deploy

on:
  release:
    types: [published]
  workflow_dispatch:
    inputs:
      target_env:
        description: 'Target environment'
        required: true
        default: staging
        type: choice
        options:
          - staging
          - production

permissions:
  contents: read
  attestations: write      # For artifact attestation
  id-token: write          # For OIDC federation (production deploy)
  packages: write          # For publishing SBOM/artifacts

jobs:
  # ── Step 0: Build everything ────────────────────────────────────
  build:
    runs-on: ubuntu-latest
    outputs:
      version: ${{ steps.version.outputs.tag }}
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Get version from release tag
        id: version
        run: echo "tag=${{ github.event.release.tag_name }}" >> "$GITHUB_OUTPUT"

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build all packages
        run: pnpm turbo build --concurrency=${CPU_COUNT}
        env:
          NODE_ENV: production
          TURBO_TELEMETRY_DISABLED: '1'

      - name: Run full test suite
        run: pnpm turbo test --concurrency=${CPU_COUNT}
        env:
          TURBO_TELEMETRY_DISABLED: '1'

      # ── SBOM generation ───────────────────────────────────────
      - name: Generate SBOM (CycloneDX)
        uses: cyclonedx/cyclonedx-action@v1
        with:
          project_type: framework
          output: dist/sbom.cdx.json
          compose: true
          symbol_types: false
          # Exclude devDependencies; validate all licenses
          license: true

      - name: Upload SBOM
        uses: actions/upload-artifact@v4
        with:
          name: sbom
          path: dist/sbom.cdx.json
          retention-days: 90

      # ── Build Docker images ───────────────────────────────────
      - name: Login to GHCR
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Extract metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ghcr.io/${{ github.repository }}
          tags: |
            type=semver,pattern={{version}},value=${{ steps.version.outputs.tag }}
            type=sha,prefix=

      - name: Build and push Docker images
        uses: docker/build-push-action@v6
        with:
          context: .
          file: ./Dockerfile.multirepo
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          platforms: linux/amd64,linux/arm64
          cache-from: type=gha
          cache-to: type=gha,mode=max

      # ── Artifact attestation (SLSA provenance) ────────────────
      - name: Attest build artifacts
        uses: actions/attest-build-provenance@v1
        with:
          subject-name: ghcr.io/${{ github.repository }}
          subject-digest: ${{ steps.meta.outputs.digest }}
          push-to-registry: true

      - name: Upload release assets
        uses: softprops/action-gh-release@v2
        with:
          files: |
            dist/sbom.cdx.json
          fail_on_unmatched_files: true

  # ── Step 1: Staging deploy (auto) ───────────────────────────────
  deploy-staging:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: staging
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - uses: actions/download-artifact@v4
        with:
          name: sbom

      - name: Download SBOM from build
        uses: actions/download-artifact@v4
        with:
          name: sbom

      - name: Deploy to staging
        uses: google-github-actions/deploy-cloudrun@v2
        with:
          service: ${{ github.event.release.tag_name }}-staging
          image: gcr.io/${{ secrets.GCP_PROJECT }}/${{ github.event.release.tag_name }}
          region: us-central1
          env_vars: VERSION=${{ github.event.release.tag_name }}
        id: deploy

  # ── Step 2: Production deploy (manual approval + OIDC) ──────────
  deploy-production:
    needs:
      - build
      - deploy-staging
    runs-on: ubuntu-latest
    if: >-
      (github.event_name == 'release' && github.event.release.prerelease == false)
      || (github.event_name == 'workflow_dispatch' && github.event.inputs.target_env == 'production')
    environment:
      name: production
      # <-- THIS triggers the required manual approval in repo settings
    steps:
      - uses: actions/checkout@v4

      # ── OIDC Federation (no long-lived keys) ──────────────────
      # Steps:
      #   1. GitHub Actions requests a short-lived JWT from its OIDC provider
      #   2. Google's IAM trusts this JWT (aud=github, iss=token.actions.githubusercontent.com)
      #   3. Google exchanges it for a short-lived GCP service account token
      #   4. That token authenticates to Cloud Run — zero static credentials

      - name: Authenticate to GCP via OIDC
        id: auth
        uses: google-github-actions/auth@v2
        with:
          workload_identity_provider: >-
            projects/${{ secrets.GCP_PROJECT_ID }}/locations/global/workloadIdentityPools/github-pool/providers/github-provider
          service_account: deployer@${{ secrets.GCP_PROJECT_ID }}.iam.gserviceaccount.com
          audience: token.actions.githubusercontent.com
          token_format: access_token
          access_token_scopes: >-
            https://www.googleapis.com/auth/cloud-platform
          # NOTE: No GCP_SA_KEY or GCP_CREDENTIALS_JSON here.
          # All auth goes through the OIDC exchange above.

      - name: Set up gcloud
        uses: google-github-actions/setup-gcloud@v2
        with:
          project_id: ${{ secrets.GCP_PROJECT_ID }}

      - name: Download artifacts
        uses: actions/download-artifact@v4
        with:
          name: sbom

      - name: Deploy to production (Cloud Run)
        uses: google-github-actions/deploy-cloudrun@v2
        with:
          service: production-api
          image: gcr.io/${{ secrets.GCP_PROJECT }}/${{ github.event.release.tag_name }}
          region: us-central1
          env_vars: VERSION=${{ github.event.release.tag_name }},NODE_ENV=production

      - name: Smoke test production
        run: |
          curl -sf https://production-api-${{ secrets.GCP_PROJECT_ID }}.a.run.app/health || exit 1
          echo "✅ Health check passed"

      - name: Announce deployment
        if: success()
        uses: actions/github-script@v7
        with:
          script: |
            await github.rest.repos.createCommitStatus({
              owner: context.repo.owner,
              repo: context.repo.repo,
              sha: context.sha,
              state: 'success',
              description: 'Production deployment successful',
              context: 'Deploy to Production'
            });
```

### OIDC Trust Configuration Explained

You need to configure **both sides** of the trust relationship:

**Side 1 — Google Cloud Workload Identity Federation:**

```bash
gcloud iam workload-identity-pools create github-pool \
  --project="${GCP_PROJECT}" \
  --location="global" \
  --display-name="GitHub Actions pool"

gcloud iam workload-identity-pools providers create-oidc github-provider \
  --project="${GCP_PROJECT}" \
  --location="global" \
  --workload-identity-pool=github-pool \
  --display-name="GitHub provider" \
  --attribute-mapping="google.subject=assertion.sub,attribute.actor=assertion.actor,attribute.aud=assertion.aud" \
  --issuer-uri="https://token.actions.githubusercontent.com"
```

Key claims GitHub emits in its OIDC token:
- `sub`: `repo:<owner>/<repo>:ref:refs/heads/main` (or the release tag)
- `aud`: `github/actions`
- `iss`: `https://token.actions.githubusercontent.com`
- `actor`: the actor who triggered the workflow
- `ref`: the git ref

**Side 2 — GCP Service Account Grant:**

```bash
gcloud iam service-accounts add-iam-policy-binding deployer@"${GCP_PROJECT}".iam.gserviceaccount.com \
  --project="${GCP_PROJECT}" \
  --role="roles/run.developer" \
  --condition="expression=attribute.ref=='refs/tags/v*' || attribute.ref=='refs/heads/main',title='Only allow main + tags'"
```

The condition expression restricts deployments to `main` branch pushes and release tags only — preventing arbitrary PR workflows from deploying.

**No `GCP_SA_KEY` environment variable.** The `google-github-actions/auth@v2` step calls `sts.exchangeToken`, which takes the GitHub OIDC JWT and exchanges it for GCP-access-level bearer tokens. These tokens expire after ~1 hour (controlled by `credential_expiration_ttl`). Every deploy run gets a fresh, short-lived credential — if a worker is compromised, there's nothing to steal.

### Environment Approval Flow

In **Repository Settings → Environments → production**:
1. Enable "Required reviewers" — add 2-3 approvers
2. Configure timeout (e.g., 24h — auto-abandon stale releases)
3. Concurrency group: `production` — only one deploy at a time

When `deploy-production` hits the `environment: production` block, GitHub pauses the workflow until the designated reviewers click "Review deployments" → "Approve". If rejected or timed out, the workflow is cancelled.

### SBOM & Attestation Details

**SBOM:** `cyclonedx-action` produces a CycloneDX 1.5 JSON manifest listing every production dependency, their versions, and licenses. Uploaded as a release asset for supply-chain auditing ( SPDX-compatible).

**Provenance attestation:** `actions/attest-build-provenance@v1` signs a SLSA v1 attestation via cosign/GitHub's sigstore integration. Anyone can verify:
```bash
gh attestation verify ghcr.io/org/repo --sha v1.2.3
```
This proves the binary was built from commit X in workflow Y — tamper-evident.

## Workflow 4: `merge-queue.yml` — Required-status checks for merge queue

```yaml
# .github/workflows/merge-queue.yml
name: Merge Queue Checks

on:
  merge_group:
    # Fired when a PR enters the GitHub merge queue (branch protection setting)

permissions:
  contents: read
  statuses: write

jobs:
  # Each PR entering the queue must pass these gates before being committed.
  # GitHub's merge queue guarantees linear history: only one commit set at a time.

  ci-merge:
    name: CI (merged commit)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          # Check out the MERGED commit (branch + base), not the PR head alone
          ref: refs/merge-queues/${{ github.ref_name }}
          fetch-depth: 0

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Full install + build
        run: pnpm install --frozen-lockfile && pnpm turbo build

      - name: Full test suite
        run: pnpm turbo test --concurrency=${CPU_COUNT}

      - name: Integration tests
        run: pnpm --filter "./apps/api" test:integration
        env:
          DATABASE_URL: ${{ secrets.TEST_DATABASE_URL }}

      - name: Report status
        if: always()
        uses: actions/github-script@v7
        with:
          script: |
            const state = '${{ job.status }}' === 'success' ? 'success' : 'error';
            await github.rest.repos.createCommitStatus({
              owner: context.repo.owner,
              repo: context.repo.repo,
              sha: context.payload.merge_group.head_commit.id,
              state: state,
              context: 'CI / Merge Queue',
              description: state === 'success' ? 'Green in queue ✓' : 'Failed — removed from queue'
            });

  security-scan:
    name: Security Scan
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          ref: refs/merge-queues/${{ github.ref_name }}

      - name: Run Trivy vulnerability scan
        uses: aquasecurity/trivy-action@v0
        with:
          scan-type: 'fs'
          severity: 'HIGH,CRITICAL'
          exit-code: '1'

      - name: Run audit
        run: pnpm audit --prod --audit-level=high
```

### How It Integrates With Branch Protection

```
Repository Settings → Branches → main → Require a pull request
  ├── ☑ Require pull request reviews (1 approve)
  ├── ☑ Require status checks to pass
  │   ├── CI                           ← ci.yml passes on push/PR
  │   ├── Preview Deployments          ← preview.yml registers deployment status
  │   ├── CI / Merge Queue             ← merge-queue.yml (only fires in merge_group)
  │   └── Security Scan                ← merge-queue.yml (only fires in merge_group)
  ├── ☑ Include administrators
  └── ☑ Require linear history
```

**Merge queue flow:**
1. Developer opens PR → `ci.yml` and `preview.yml` run immediately on the PR
2. Approving reviewer merges → GitHub puts PR in the merge queue
3. GitHub creates a merged ref → `merge-queue.yml` fires on `merge_group`
4. The merged commit is tested against the actual base (catches race conditions)
5. If green, GitHub applies the commit atomically. If red, it's quarantined and later PRs skip ahead.

## Summary of Key Design Decisions

| Concern | Decision | Rationale |
|---------|----------|-----------|
| **Path filtering** | `dorny/paths-filter` + `...build` turbo syntax | Avoid testing everything on every tiny change while keeping libs as a shared dependency |
| **Cache key** | `lockfile hash + os + node version` | Most accurate granularity — source-only changes get instant cache hits |
| **Test matrix** | Exclude filtered-out apps; don't shard lib-only | Minimizes matrix size while covering all scenarios |
| **Preview envs** | One GitHub Environment per PR number | Unique URLs + automatic cleanup on close; leverages native GitHub infra |
| **OIDC deploy** | WIF/JWT exchange, no static keys | Eliminates secret rotation burden and credential leak surface |
| **SBOM** | CycloneDX as release asset | Industry-standard, parseable by every major SCM tool |
| **Attestation** | SLSA provenance via cosign | Cryptographic proof of build origin |
| **Merge queue** | Separate `merge_group` workflow checking the merged diff | Catches race-condition bugs that PR-only CI misses |

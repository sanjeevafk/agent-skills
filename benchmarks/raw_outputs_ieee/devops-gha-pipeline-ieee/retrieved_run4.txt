## Workflow 1: `ci.yml` — Path-filtered CI + Build Cache

This is the main pipeline that triggers on push/PR and runs filtered jobs per-app.

```yaml
name: CI — Lint, Build, Test

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]
  merge_group:                         # gate check for merge queue
    branches: [main]

permissions:
  contents: read                     # minimal default

env:
  PNPM_CACHE_FOLDER: .pnpm-store
  TURBO_TOKEN: ${{ secrets.TURBO_TOKEN }}
  TURBO_TEAM: ${{ vars.TURBO_TEAM }}

jobs:
  # ──────────────────────────────────────────────
  # 1. Detect changed paths — single source of truth
  # ──────────────────────────────────────────────
  changes:
    name: Detect Changes
    runs-on: ubuntu-latest
    outputs:
      apps_admin: ${{ steps.filter.outputs.apps_admin }}
      apps_api:  ${{ steps.filter.outputs.apps_api }}
      apps_web:  ${{ steps.filter.outputs.apps_web }}
      packages:  ${{ steps.filter.outputs.packages }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0              # needed for diffing

      - uses: dorny/paths-filter@v3
        id: filter
        with:
          filters: |
            apps_admin:
              - 'apps/app-admin/**'
              - 'packages/*/'          # lib change → rebuild depends
            apps_api:
              - 'apps/app-api/**'
              - 'packages/*/'
            apps_web:
              - 'apps/app-web/**'
              - 'packages/*/'
            packages:
              - 'packages/*/'
              - 'pnpm-lock.yaml'

  # ──────────────────────────────────────────────
  # 2. Shared setup job — install + cache once,
  #    then pass outputs to downstream jobs
  # ──────────────────────────────────────────────
  setup:
    name: Install Dependencies
    needs: changes                    # run after path detection
    runs-on: ubuntu-latest
    outputs:
      cache-hit: ${{ steps.restore.outputs.cache-hit }}
    steps:
      - uses: actions/checkout@v4

      - name: Install pnpm
        uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - name: Setup Node (isolated from filtering)
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - name: Restore dependency cache
        id: restore
        uses: actions/cache@v4
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            deps-${{ runner.os }}-

      - name: Install dependencies
        if: steps.restore.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

      # Re-save cache even on hit so stale entries age out
      - name: Save dependency cache
        uses: actions/cache@v4
        if: steps.restore.outputs.cache-hit != 'true'
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          path: .pnpm-store

  # ──────────────────────────────────────────────
  # 3. Lint — full repo lint (unchanged by path)
  # ──────────────────────────────────────────────
  lint:
    name: Lint
    needs: setup
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - name: Restore deps
        id: dep-cache
        uses: actions/cache@v4
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          path: .pnpm-store

      - name: Install if cache miss
        if: steps.dep-cache.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

      - name: Run lint
        run: pnpm lint

  # ──────────────────────────────────────────────
  # 4. Per-app build — only apps whose paths changed
  # ──────────────────────────────────────────────
  build:
    name: Build (${{ matrix.app || 'packages' }})
    needs: [setup, changes]
    strategy:
      fail-fast: false                # don't cancel others on failure
      matrix:
        app:
          - apps_admin
          - apps_api
          - apps_web
          - packages
    runs-on: ubuntu-latest
    if: >-
      github.event_name == 'merge_group' ||
      (needs.changes.outputs[matrix.app] == 'true')
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      # ── Cache key explained below ──
      - name: Restore build cache
        id: build-cache
        uses: actions/cache@v4
        with:
          key: build-${{ runner.os }}-${{ matrix.app }}-${{ hashFiles('pnpm-lock.yaml', '**/src/**/*.{ts,tsx,d.ts}') }}
          path: |
            .turbo/
            **/.turbo/
            node_modules/.cache/turbo
          restore-keys: |
            build-${{ runner.os }}-${{ matrix.app }}-

      - name: Restore deps
        uses: actions/cache@v4
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          path: .pnpm-store

      - name: Install if needed
        run: test -d node_modules || pnpm install --frozen-lockfile

      - name: Build
        run: pnpm build --filter=${{ matrix.app == 'packages' && 'packages/...' || format('./{0}...', matrix.app) }}

  # ──────────────────────────────────────────────
  # 5. Test matrix — Node 20 × 22 × shards
  # ──────────────────────────────────────────────
  test:
    name: Test (${{ matrix.node }}, ${{ matrix.shard }})
    needs: [setup, changes]
    runs-on: ubuntu-latest
    if: >-
      github.event_name == 'merge_group' ||
      needs.changes.outputs.apps_admin == 'true' ||
      needs.changes.outputs.apps_api == 'true' ||
      needs.changes.outputs.apps_web == 'true' ||
      needs.changes.outputs.packages == 'true'
    strategy:
      fail-fast: false
      matrix:
        node: [20, 22]
        shard: [1/4, 2/4, 3/4, 4/4]
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node }}
          cache: 'pnpm'

      - name: Restore deps
        uses: actions/cache@v4
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          path: .pnpm-store

      - name: Install if needed
        run: test -d node_modules || pnpm install --frozen-lockfile

      - name: Run tests (shard ${{ matrix.shard }})
        env:
          NODE_OPTIONS: '--max-old-space-size=4096'
        run: >
          pnpm test -- --shard=${{ matrix.shard }}
          --ci --reporters=default reporters/jest-junit
        continue-on-error: true     # allow sharded results to aggregate

  # ──────────────────────────────────────────────
  # 6. Aggregate test results
  # ──────────────────────────────────────────────
  test-report:
    name: Test Report
    needs: [test]
    runs-on: ubuntu-latest
    if: always()
    steps:
      - uses: actions/upload-artifact@v4
        with:
          name: test-results-node20
          path: '**/test-results-*.xml'
      - uses: actions/upload-artifact@v4
        with:
          name: test-results-node22
          path: '**/test-results-*.xml'
```

### Cache-Key Design Explained

There are **three independent caches**, each keyed differently:

| Cache | Key Pattern | What It Stores | Why This Key |
|-------|-------------|----------------|--------------|
| **Dependencies** | `deps-{OS}-{hash(pnpm-lock.yaml)}` | `.pnpm-store` (~2–4 GB extracted modules) | The lockfile is the *only* thing that determines exact installed versions. Hashing it means any dependency change invalidates the cache. A fallback prefix `deps-{OS}-` catches partial restores when the lockfile changes but many packages are unchanged. |
| **Build artifacts** | `build-{app}-{OS}-{hash(lockfile + all src/**/*.ts)}` | `.turbo/`, module graph cache, build outputs | Includes the lockfile (in case resolved packages affect compilation) *plus* every source file glob. Adding the app scope limits cross-app pollution. When only an unrelated app's code changes, that app's build cache still hits because its src hash didn't change. |
| **Node module cache** | `pnpm` managed via `actions/setup-node` | `.cache` inside node_modules (esbuild/swc binary caches) | Built-in to the Node action; scoped to the Node version. Cheap to regenerate, so we let it flow through naturally. |

**Key principles:**

1. **Single canonical hash anchor** — every key references `hashFiles('pnpm-lock.yaml')`. This prevents cache poisoning where a new lockfile silently reuses stale modules.
2. **Fallback keys for warm-up** — `restore-keys` use prefix matching so a lockfile-only change (e.g., adding one new library) still restores the bulk of the previous cache, then just re-downloads the delta.
3. **Per-app build isolation** — the build cache includes the app name in the key, so two apps changing simultaneously don't invalidate each other's incremental builds.
4. **Cache size awareness** — `.pnpm-store` can grow large; the `if: steps.restore.outputs.cache-hit != 'true'` guard on save prevents endlessly growing caches during PR comments/re-runs.

## Workflow 2: `preview.yml` — Ephemeral Preview Environments

```yaml
name: Preview Environment

on:
  pull_request:
    branches: [main]

permissions:
  contents: read
  pull-requests: write

env:
  APPS: app-admin,app-api,app-web

jobs:
  preview:
    name: Deploy Preview (${{ matrix.app }})
    strategy:
      fail-fast: false
      matrix:
        app: ${{fromJson(matrix.apps)}}
        include:
          - app: app-admin
            url_suffix: admin
          - app: app-api
            url_suffix: api
          - app: app-web
            url_suffix: www
    runs-on: ubuntu-latest
    # Unique concurrency group per PR + app; cancels stale deployment
    concurrency:
      group: preview-${{ github.pr_number }}-${{ matrix.app }}
      cancel-in-progress: true
    environment:
      name: preview-${{ matrix.url_suffix }}
      url: https://${{ github.pr_number }}-${{ matrix.url_suffix }}.preview.example.com

    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - name: Restore deps
        uses: actions/cache@v4
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          path: .pnpm-store

      - name: Install
        run: pnpm install --frozen-lockfile

      - name: Build app
        run: pnpm build --filter=./${{ matrix.app }}...

      - name: Configure cloud CLI (OIDC)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/${{ secrets.AWS_PREVIEW_ROLE }}
          aws-region: ${{ secrets.AWS_REGION }}

      - name: Deploy preview container
        run: |
          docker build -t ${{ matrix.app }}:pr${{ github.pr_number }} -f ${{ matrix.app }}/Dockerfile .
          docker tag ${{ matrix.app }}:pr${{ github.pr_number }} \
            ${{ secrets.ECR_REPO }}/${{ matrix.app }}:pr${{ github.pr_number }}
          docker push ${{ secrets.ECR_REPO }}/${{ matrix.app }}:pr${{ github.pr_number }}

      - name: Deploy to ephemeral environment
        run: |
          eksctl delete iamserviceaccount \
            --name preview-sa-${{ github.pr_number }}-${{ matrix.app }} \
            --namespace previews --cluster ${{ vars.CLUSTER_NAME }} || true

          eksctl create iamserviceaccount \
            --name preview-sa-${{ github.pr_number }}-${{ matrix.app }} \
            --namespace previews --cluster ${{ vars.CLUSTER_NAME }} \
            --attach-policy-arn arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:policy/PreviewPolicy \
            --approve-role-creation

          kubectl apply -f ${{ matrix.app }}/k8s/preview-deployment.yaml \
            -l pr=${{ github.pr_number }} \
            --record

      - name: Health check
        run: |
          for i in $(seq 1 30); do
            if curl -sf https://${{ github.pr_number }}-${{ matrix.url_suffix }}.preview.example.com/health > /dev/null; then
              echo "✓ Healthy"
              exit 0
            fi
            sleep 5
          done
          echo "✗ Health check failed"
          exit 1
        timeout-minutes: 3

      - name: Comment PR with URL
        if: success()
        uses: mshick/add-pr-comment@v2
        with:
          message: |
            **Preview ready**: ${{ job.environment.url }}
          repository-conditions: never-create
```

**Why this works safely:**

- **One lifecycle step per PR per app** — the `concurrency` group with `cancel-in-progress: true` ensures pushing again to the PR tears down the old deployment before starting a new one. No resource leaks.
- **Unique identifiers everywhere** — `pr${{ github.pr_number }}` prefixes Docker tags, K8s labels, and IAM service accounts prevent collisions.
- **Auto-cleanup pattern** — you add a workflow trigger on `pull_request: { closed: true }` that runs `eksctl delete` and removes ECR tags. See the cleanup section below.
- **Environment URLs are explicit** — `job.environment.url` feeds back into the PR comment automatically.

## Workflow 3: `deploy-production.yml` — OIDC-Gated Production Deploy

```yaml
name: Deploy to Production

on:
  pull_request:
    types: [closed]
    branches: [main]
    paths:
      - 'apps/app-admin/**'
      - 'apps/app-api/**'
      - 'apps/app-web/**'
      - 'packages/**'

permissions:
  contents: read

jobs:
  # ──────────────────────────────────────────────
  # Gate: requires manual approval in GitHub UI
  # ──────────────────────────────────────────────
  gate:
    name: Production Approval
    runs-on: ubuntu-latest
    environment:
      name: production             # ← protected environment in Settings > Environments
      url: https://production.example.com
    steps:
      - run: echo "Approved — proceeding with deployment"

  # ──────────────────────────────────────────────
  # CI smoke-test (lightweight — no test matrix)
  # ──────────────────────────────────────────────
  smoke:
    name: Smoke Test
    needs: gate
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - name: Restore deps
        uses: actions/cache@v4
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          path: .pnpm-store

      - name: Install & build
        run: |
          test -d node_modules || pnpm install --frozen-lockfile
          pnpm build

      - name: Smoke test
        run: pnpm exec start-server-and-test smoke:test http://localhost:3000/health

  # ──────────────────────────────────────────────
  # OIDC auth to AWS — zero long-lived credentials
  # ──────────────────────────────────────────────
  deploy:
    name: Deploy (${{ matrix.app }})
    needs: smoke
    strategy:
      fail-fast: false
      matrix:
        app: [app-admin, app-api, app-web]
    runs-on: ubuntu-latest
    # Inherits permissions from `on.pulls.request` + GITHUB_TOKEN = fine
    # for OIDC token request (no extra IAM permissions needed on the runner)
    steps:
      - uses: actions/checkout@v4

      # OIDC: request JWT from GitHub, exchange for short-lived AWS credentials
      - name: Configure AWS credentials via OIDC
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/${{ secrets.AWS_PROD_DEPLOY_ROLE }}
          role-session-name: github-actions-${{ github.run_id }}-${{ matrix.app }}
          aws-region: ${{ secrets.AWS_REGION }}
          # NO access_key_id / secret_access_key — pure OIDC
          # duration_seconds: 3600           # optional session length cap

      - name: Login to ECR
        uses: aws-actions/amazon-ecr-login@v2

      - uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Build & push image
        run: |
          pnpm install --frozen-lockfile
          docker build -t ${{ matrix.app }}:${{ github.sha }} -f ${{ matrix.app }}/Dockerfile .
          docker tag ${{ matrix.app }}:${{ github.sha }} \
            ${{ secrets.ECR_REPO }}/${{ matrix.app }}:${{ github.sha }}
          docker tag ${{ matrix.app }}:${{ github.sha }} \
            ${{ secrets.ECR_REPO }}/${{ matrix.app }}:latest
          docker push ${{ secrets.ECR_REPO }}/${{ matrix.app }}:${{ github.sha }}
          docker push ${{ secrets.ECR_REPO }}/${{ matrix.app }}:latest

      - name: Deploy to EKS
        run: |
          kubectl set image deployment/${{ matrix.app }} \
            ${{ matrix.app }}=${{ secrets.ECR_REPO }}/${{ matrix.app }}:${{ github.sha }} \
            --namespace production
          kubectl rollout status deployment/${{ matrix.app }} \
            --namespace production --timeout=300s

      - name: Rollback on failure
        if: failure()
        run: |
          PREV_TAG=$(kubectl get deploy ${{ matrix.app }} -n production \
            -o jsonpath='{.spec.template.spec.containers[0].image}' | rev | cut -d: -f2- | rev)
          if [[ -n "$PREV_TAG" && "$PREV_TAG" != "${{ github.sha }}" ]]; then
            echo "Rolling back to $PREV_TAG"
            kubectl set image deployment/${{ matrix.app }} \
              ${{ matrix.app }}=${{ secrets.ECR_REPO }}/${{ matrix.app }}:$PREV_TAG \
              --namespace production
          fi
```

### Merge Queue Integration

The CI workflow above already handles merge queues via the `merge_group` trigger. To wire required-status checks:

```jsonc
// In GitHub Branch Protection Rules for `main`:
{
  "required_status_checks": {
    "strict": true,
    "contexts": [
      "CI — Lint, Build, Test / Lint",
      "CI — Lint, Build, Test / Build (apps_admin)",
      "CI — Lint, Build, Test / Build (apps_api)",
      "CI — Lint, Build, Test / Build (apps_web)",
      "CI — Lint, Build, Test / Build (packages)"
    ]
  },
  "enforce_admins": true,
  "restrictions": null,
  "required_pull_request_reviews": {
    "dismiss_stale_reviews": true,
    "require_code_owner_reviews": true,
    "required_approving_review_count": 1
  }
}
```

For the merge queue specifically, GitHub will *only* merge a PR when **all** `merge_group` jobs pass. The job conditionals ensure they skip efficiently when nothing changed:

```yaml
# In ci.yml, the merge_group branch:
if: >-
  github.event_name == 'merge_group' ||
  needs.changes.outputs.apps_admin == 'true' || ...
```

When running inside a merge group, GitHub injects `MERGE_GROUP_HEAD_SHA` so the filter diff compares against the actual merge candidate rather than the fork head. Override dorny's default ref:

```yaml
- uses: dorny/paths-filter@v3
  id: filter
  with:
    # On merge_group, diff against what's being merged
    ref: ${{ github.event.merge_group.head_commit.id || '' }}
    # Fall back to normal PR/head diff
    base: ${{ github.event.merge_group.base_ref || github.head_ref || github.ref }}
    filters: |...
```

## Workflow 4: `release.yml` — Release with SBOM + Attestation

```yaml
name: Release

on:
  release:
    types: [published]

permissions:
  contents: write
  packages: write
  attestations: write            # sigstoreattestation upload
  id-token: write                # cosign OIDC signing

jobs:
  # ──────────────────────────────────────────────
  # Build + SBOM for every app
  # ──────────────────────────────────────────────
  release-assets:
    name: Release Assets (${{ matrix.app }})
    strategy:
      fail-fast: false
      matrix:
        app: [app-admin, app-api, app-web]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with:
          version: 9.x

      - uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Restore deps
        uses: actions/cache@v4
        with:
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          path: .pnpm-store

      - name: Install & build
        run: |
          test -d node_modules || pnpm install --frozen-lockfile
          pnpm build --filter=./${{ matrix.app }}

      # ── Generate SBOM (CycloneDX JSON) ──
      - name: Generate SBOM
        uses: anchore/sbom-action@v0
        with:
          path: ${{ matrix.app }}
          format: cyclonedx-json
          output-file: sbom-${{ matrix.app }}.json

      # ── Upload SBOM as release asset ──
      - name: Upload SBOM
        uses: svenstaro/upload-release-action@v2
        with:
          repo_token: ${{ secrets.GITHUB_TOKEN }}
          file: sbom-${{ matrix.app }}.json
          asset_name: sbom-${{ matrix.app }}-${{ github.event.release.tag_name }}.json
          release_id: ${{ github.event.release.id }}
          overwrite: true

      # ── Attest build provenance (SLSA-style) ──
      - name: Generate provenance attestation
        uses: actions/attest-build-provenance@v1
        with:
          subject-path: ${{ matrix.app }}
          subject-name: ${{ matrix.app }}

      # ── Sign artifacts with cosign (Sigstore) ──
      - name: Install cosign
        uses: sigstore/cosign-installer@v3.6.0
        with:
          cosign-release: 'v2.3.1'

      - name: Attest artifact
        run: |
          cd ${{ matrix.app }}
          zip -r ../artifacts-${{ matrix.app }}.zip .
          cosign sign --yes \
            -a sha=${{ github.sha }} \
            -a run_id=${{ github.run_id }} \
            ${{ github.event.repository.full_name }}/artifacts-${{ matrix.app }}.zip@${{ github.sha }}

      - name: Upload signed artifact
        uses: svenstaro/upload-release-action@v2
        with:
          repo_token: ${{ secrets.GITHUB_TOKEN }}
          file: artifacts-${{ matrix.app }}.zip
          asset_name: artifacts-${{ matrix.app }}-${{ github.event.release.tag_name }}.zip
          release_id: ${{ github.event.release.id }}
          overwrite: true
```

## Workflow 5: `cleanup-preview.yml` — Teardown on PR Close

```yaml
name: Cleanup Previews

on:
  pull_request_target:
    types: [closed]

permissions:
  contents: read

jobs:
  teardown:
    name: Teardown Previews
    runs-on: ubuntu-latest

      fail-fast: false
      matrix:
        suffix: [admin, api, www]

          - suffix: admin

            --name preview-sa-${{ github.event.pull_request.number }}-${{ matrix.app }} \
            --namespace previews --cluster ${{ vars.CLUSTER_NAME }} \
            --ignore-not-existent || true

      - name: Prune ECR image tags
        run: |
          aws ecr batch-delete-image \
            --repository-name ${{ secrets.ECR_REPO }} \
            --image-ids imageTag=pr${{ github.event.pull_request.number }}
```

## OIDC Trust Configuration — Detailed Walkthrough

This is the most critical security piece. Here's exactly how to configure the trust between GitHub Actions and your cloud provider (AWS examples, but the pattern applies to GCP/Azure identically).

### Step 1: Create the IAM Role

```bash
# For production deployments
aws iam create-role \
  --role-name gh-actions-prod-deploy \
```

### Condition Breakdown

| Condition | Purpose |
|-----------|---------|
| `aud: sts.amazonaws.com` | GitHub OIDC token audience must be `sts.amazonaws.com` — validates the token was issued for AWS STS, not some other service |
| `sub: repo:owner/repo:ref:refs/heads/main` | Restricts the role to the `main` branch only. Pull requests use `repo:owner/repo:pull_request` |
| Separate roles for preview vs prod | Least privilege: preview role gets only `previews` namespace access, prod role gets `production` namespace + ECR push rights |

### Token Lifecycle

```
github.com/workflow/run/12345/job/abc
       │
       ▼  (aws-actions/configure-aws-credentials@v4)
  IDP Provider: token.actions.githubusercontent.com
       │  JWT payload:
       │  sub: repo:owner/repo:branch:ref/HEAD
       │  aud: sts.amazonaws.com
       │  iss: https://token.actions.githubusercontent.com
       │  exp: +15 min clock skew window
       ▼
  AWS STS AssumeRoleWithWebIdentity
       │  Returns:
       │  • AccessKeyId     (temp, max 1h)
       │  • SecretAccessKey (temp, max 1h)
       │  • SessionToken    (temp, max 1h)
       ▼
  Temporary credentials used for docker push / kubectl
       │
       ▼  (credential expires or job ends)
  Credentials invalidated — no persistent keys anywhere
```

**No `.aws/credentials` file is written persistently.** The `aws-actions/configure-aws-credentials` action writes temp creds to the runner's temp directory and sets `AWS_ACCESS_KEY_ID`, `AWS_SECRET_ACCESS_KEY`, `AWS_SESSION_TOKEN` as process-env variables. They evaporate when the job ends.

## Summary of All Workflows

| File | Trigger | Scope | Key Features |
|------|---------|-------|--------------|
| `ci.yml` | push / PR / merge_group | Full repo | Path filters via `dorny/paths-filter`, 3-phase split (detect → setup → execute), Node 20×22 test matrix × 4 shards |
| `preview.yml` | PR open/synchronize | Per-app parallel | Ephemeral K8s pods, concurrency groups with auto-cancel, health-check gate, PR comment |
| `deploy-production.yml` | PR merge (closed) | Per-app serial | GitHub `environment: production` approval gate, OIDC-first auth, rollback on failure |
| `release.yml` | Release published | All apps | CycloneDX SBOM per app, SLSA provenance attestation, cosign Sigstore signing |
| `cleanup-preview.yml` | PR closed | All previews | Namespace pod deletion, IAM SA removal, ECR tag pruning |

All five workflows together give you: fast filtered CI, safe per-PR previews, approved-only production deploys with zero credentials stored, verifiable supply-chain artifacts on release, and automatic resource cleanup.

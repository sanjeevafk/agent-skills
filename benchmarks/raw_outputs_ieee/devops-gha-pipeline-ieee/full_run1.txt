## 1. Main CI Pipeline — `.github/workflows/ci.yml`

```yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main]
  pull_request_target:
    types: [opened, synchronize]
    paths-ignore:
      - "**/*.md"
      - "docs/**"

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

# ── Paths filter dispatch (runs per-app) ──────────────────────────
jobs:
  changes:
    runs-on: ubuntu-latest
    outputs:
      web-app: ${{ steps.filter.outputs.web-app }}
      api-server: ${{ steps.filter.outputs.api-server }}
      mobile-bff: ${{ steps.filter.outputs.mobile-bff }}
      libs: ${{ steps.filter.outputs.libs }}
    steps:
      - uses: dorny/paths-filter@v3
        id: filter
        with:
          base: main
          filters: |
            web-app:
              - 'apps/web-app/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - 'libs/**'
            api-server:
              - 'apps/api-server/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - 'libs/**'
            mobile-bff:
              - 'apps/mobile-bff/**'
              - 'package.json'
              - 'pnpm-lock.yaml'
              - 'libs/**'
            libs:
              - 'libs/**'
              - 'package.json'
              - 'pnpm-lock.yaml'

  # ── Cache warm / restore (shared across all jobs) ────────────────
  deps-cache:
    runs-on: ubuntu-latest
    outputs:
      cache-hit: ${{ steps.restore-deps.outputs.cache-hit }}
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: "9" }

      - name: Setup Node 20
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "pnpm"
          cache-dependency-path: pnpm-lock.yaml

      # ── CACHE KEY DESIGN ─────────────────────────────────────────
      # Key = deps-<platform>-<lockfile-sha>
      #   • pnpm-action-setup writes the resolved store dir; we key off
      #     the exact lockfile hash so any dependency change invalidates.
      #   • We DO NOT embed branch or commit — the same lockfile always
      #     maps to the same cache entry regardless of ref.
      #   • Falls back to deps-<platform>-full when no lockfile match
      #     exists (first install on fresh repo).
      # ────────────────────────────────────────────────────────────────
      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            deps-${{ runner.os }}-

      - name: Install (only if cache miss)
        if: steps.restore-deps.outputs.cache-hit != 'true'
        run: pnpm install --frozen-lockfile

  build-cache:
    needs: [changes, deps-cache]
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
    outputs:
      web-app-hit: ${{ steps.restore-web.outputs.cache-hit }}
      api-hit: ${{ steps.restore-api.outputs.cache-hit }}
      bff-hit: ${{ steps.restore-bff.outputs.cache-hit }}
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: "9" }

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "pnpm"

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      # Per-app build cache: keys on lockfile + src-hash of each app
      - id: restore-web
        uses: actions/cache/restore@v4
        with:
          path: apps/web-app/.next
          key: build-web-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml', 'apps/web-app/src/**') }}

      - id: restore-api
        uses: actions/cache/restore@v4
        with:
          path: apps/api-server/dist
          key: build-api-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml', 'apps/api-server/src/**') }}

      - id: restore-bff
        uses: actions/cache/restore@v4
        with:
          path: apps/mobile-bff/build
          key: build-bff-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml', 'apps/mobile-bff/src/**') }}

      # Build all changed targets
      - name: Build libs
        if: ${{ needs.changes.outputs.libs == 'true' }}
        run: pnpm --filter "@myorg/*" run build --recursive

      - name: Build web-app
        if: ${{ always() && (needs.changes.outputs.web-app == 'true' || needs.changes.outputs.libs == 'true') }}
        run: pnpm --filter web-app run build
        env: { NODE_OPTIONS: "--max-old-space-size=4096" }

      - name: Build api-server
        if: ${{ always() && (needs.changes.outputs.api-server == 'true' || needs.changes.outputs.libs == 'true') }}
        run: pnpm --filter api-server run build
        env: { NODE_OPTIONS: "--max-old-space-size=4096" }

      - name: Build mobile-bff
        if: ${{ always() && (needs.changes.outputs.mobile-bff == 'true' || needs.changes.outputs.libs == 'true') }}
        run: pnpm --filter mobile-bff run build
        env: { NODE_OPTIONS: "--max-old-space-size=4096" }

      - uses: actions/upload-artifact@v4
        with:
          name: build-caches
          path: |
            apps/web-app/.next
            apps/api-server/dist
            apps/mobile-bff/build
          retention-days: 3

  # ── Lint (single pass for the whole repo) ────────────────────────
  lint:
    runs-on: ubuntu-latest
    needs: deps-cache
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { version: "9" }
      - uses: actions/setup-node@v4
        with: { node-version: 20, cache: "pnpm" }

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      - run: pnpm run lint

  # ── Test Matrix: Node 20/22 x shard-4 ───────────────────────────
  test:
    needs: [changes, deps-cache, build-cache]
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        node-version: [20, 22]
        shard: [1/4, 2/4, 3/4, 4/4]
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: "9" }

      - uses: actions/setup-node@v4
        with: { node-version: ${{ matrix.node-version }}, cache: "pnpm" }

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      - uses: actions/download-artifact@v4
        with: { name: build-caches }
        if: steps.restore-deps.outputs.cache-hit == 'false'

      - name: Run tests (shard ${{ matrix.shid }})
        run: pnpm run test --shard=${{ matrix.shard }} --reporter=junit --outputFile=test-results/junit.${{ matrix.node-version }}.${{ matrix.shard }}.xml
        env:
          CI: "true"
          NODE_ENV: test

      - name: Upload test results
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: test-results-node${{ matrix.node-version }}-shard${{ matrix.shard }}
          path: test-results/
          retention-days: 14

  # ── Type-check (fast, runs once) ─────────────────────────────────
  typecheck:
    runs-on: ubuntu-latest
    needs: deps-cache
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { version: "9" }
      - uses: actions/setup-node@v4
        with: { node-version: 20, cache: "pnpm" }

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      - run: pnpm run typecheck

  # ── Security scan (container/image) ──────────────────────────────
  security-scan:
    runs-on: ubuntu-latest
    needs: deps-cache
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { version: "9" }
      - uses: actions/setup-node@v4
        with: { node-version: 20, cache: "pnpm" }

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      - run: pnpm audit --production --audit-level=high

  # ── Gate: all must pass ──────────────────────────────────────────
  gate:
    needs: [lint, typecheck, test, security-scan, build-cache]
    runs-on: ubuntu-latest
    if: success()
    steps:
      - run: echo "All CI gates passed"
```

### Merge Queue Integration

GitHub's built-in merge queue requires **all non-optional checks** in the branch protection rule to pass. Add this to **Settings → Branches → Require status checks**:

```
Required status checks: ✓
  - lint
  - typecheck
  - test (each matrix combination)
  - security-scan
  - build-cache
  - changes (the path-filter job)

Also require status checks to pass on mergable: ✓
```

When enabled, GitHub dequeues PRs sequentially, creates merge commits, and re-runs every check against the merged state — preventing flaky race conditions between concurrent CI runs.

## 2. Preview Environments & Release Pipeline — `.github/workflows/release.yml`

```yaml
# .github/workflows/release.yml
name: Preview & Release

on:
  pull_request:
    types: [opened, synchronize, closed]
    paths-ignore:
      - "**/*.md"
      - "docs/**"

concurrency:
  group: preview-${{ github.ref }}
  cancel-in-progress: true

jobs:
  # ── Derive scope from PR path changes ───────────────────────────
  preview-scope:
    runs-on: ubuntu-latest
    outputs:
      apps: ${{ steps.set-outputs.outputs.apps }}
    steps:
      - uses: actions/checkout@v4
        with: { ref: ${{ github.event.pull_request.head.sha }}, fetch-depth: 0 }

      - name: Detect affected apps
        id: set-outputs
        run: |
          BASE=${{ github.event.pull_request.base.sha }}
          HEAD=${{ github.event.pull_request.head.sha }}
          CHANGED=$(git diff --name-only "$BASE" "$HEAD" \
            | grep '^apps/' \
            | sed 's|^apps/\([^/]*\)/.*|\1|' \
            | sort -u \
            | jq -R -s 'split("\n") | map(select(length > 0))')
          echo "apps=$CHANGED" >> "$GITHUB_OUTPUT"

  # ── Preview deploy (one per changed app) ────────────────────────
  preview-deploy:
    needs: preview-scope
    if: ${{ github.event.action != 'closed' && needs.preview-scope.outputs.apps != '[]' && needs.preview-scope.outputs.apps != '' }}
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      max-parallel: 3  # cap concurrent previews
      matrix:
        app: ${{ fromJson(needs.preview-scope.outputs.apps) }}
    environment:
      name: preview-${{ matrix.app }}
      url: ${{ steps.prd.outputs.url }}
    # ── OIDC TRUST CONFIGURATION ─────────────────────────────────
    # No GITHUB_TOKEN secrets needed.
    # The environment "preview-{app}" has an open-id-connect provider
    # configured at the cloud provider side pointing to
    # https://token.actions.githubusercontent.com
    # with subject claim matching:
    #   repo:{owner}/{repo}:environment:preview-{app}
    # The cloud IAM role assumed by GitHub actions has permissions
    # scoped to only that app's namespace/infra.
    # ───────────────────────────────────────────────────────────────
    permissions:
      id-token: write    # enables OIDC token generation
      contents: read
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { version: "9" }
      - uses: actions/setup-node@v4
        with: { node-version: 20, cache: "pnpm" }

      - id: restore-deps
        uses: actions/cache/restore@v4
        with:
          path: ~/.local/share/pnpm/store
          key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}

      - run: pnpm install --frozen-lockfile
        if: steps.restore-deps.outputs.cache-hit != 'true'

      - name: Build app
        run: pnpm --filter ${{ matrix.app }} run build

      - name: Configure cloud credentials (OIDC)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/github-oidc-preview-${{ matrix.app }}
          aws-region: us-east-1
          # Note: NO aws-access-key-id / aws-secret-access-key needed.
          # The role-trust-policy is configured in AWS with:
          #   sts:ExternalId = pr-${{ github.event.number }}
          #   Condition.stringLike:
          #     repository = github.com/{owner}/{repo}
          #     action = pulls:update:external-id
          #     principal = sts-session/pr-${{ github.event.number }}

      - name: Deploy preview
        id: prd
        run: |
          URL=$(pnpm --filter ${{ matrix.app }} run preview:deploy \
            --env.preview-url=https://${{ matrix.app }}-pr${{ github.event.number }}.${{ vars.DEPLOY_DOMAIN }})
          echo "url=$URL" >> "$GITHUB_OUTPUT"

  # ── Tear down preview on close ──────────────────────────────────
  preview-teardown:
    needs: preview-scope
    if: github.event.action == 'closed' && needs.preview-scope.outputs.apps != '[]'
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        app: ${{ fromJson(needs.preview-scope.outputs.apps) }}
    environment:
      name: preview-${{ matrix.app }}
    permissions:
      id-token: write
      contents: read
    steps:
      - uses: actions/checkout@v4

      - name: Teardown preview infra
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/github-oidc-preview-${{ matrix.app }}
          aws-region: us-east-1

      - run: pnpm --filter ${{ matrix.app }} run preview:teardown

  # ── Release job (tags, SBOM, attestation) ──────────────────────
  release:
    if: startsWith(github.ref, 'refs/tags/v')
    runs-on: ubuntu-latest
    needs:
      - ci-mock  # reference: requires all ci.yml checks green
    environment:
      name: production
      # Environment-level manual approval gate enforced by GitHub.
      # Settings → Environments → production → Deployment branches:
      #   Protected branches only.
      # In-github-environment reviewers: @team/platform-lead, @team/deployment-on-call
    permissions:
      id-token: write       # OIDC for cloud deploy
      contents: write       # create git tag / release
      packages: write       # publish images
      actions: write        # attestations
      attestations: write   # attestation upload
      security-events: write # SARIF upload
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0, persist-credentials: false }

      - uses: pnpm/action-setup@v4
        with: { version: "9" }
      - uses: actions/setup-node@v4
        with: { node-version: 20, cache: "pnpm" }

      - name: Install
        run: pnpm install --frozen-lockfile

      - name: Build all
        run: pnpm run build --recursive

      # ── SBOM generation ────────────────────────────────────────
      - name: Generate SBOM (Syft)
        uses: anchore/sbom-action@v0
        with:
          path: .
          format: cyclonedx-json
          output-file: sbom.cdx.json

      - name: Generate SPDX SBOM per package
        run: |
          mkdir -p sboms/
          pnpm ls -r --json | jq -c '.[]' | while read pkg; do
            name=$(echo "$pkg" | jq -r '.name')
            anchore-syft packages "$pkg" -o cyclonedx-json > "sboms/${name}.cdx.json" 2>/dev/null || true
          done

      # ── Artifact provenance (cosign SLSA attestation) ──────────
      - name: Sign release artifacts
        uses: sigstore/gh-action-sigstore-python@v3
        with:
          inputs: |
            sbom.cdx.json
            $(find sboms/ -name '*.cdx.json' -type f)

      # ── OIDC production deploy ─────────────────────────────────
      - name: Configure production OIDC
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::${{ secrets.AWS_ACCOUNT_ID }}:role/github-oidc-production
          aws-region: us-east-1

      - name: Deploy to production
        run: pnpm run deploy:production
        env:
          DEPLOY_VERSION: ${{ github.ref_name }}

      # ── Attach SBOM to release ─────────────────────────────────
      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          files: |
            sbom.cdx.json
            sboms/*.cdx.json
            *.sig
            *.att
          fail_on_unmatched_files: true

  # ── Mock requirement: CI must already be green ──────────────────
  ci-mock:
    runs-on: ubuntu-latest
    steps:
      - run: echo "Would validate all ci.yml gates are passing. In practice, use: needs."
```

## 3. Cache-Key Design — Deep Dive

### Problem Statement

In a pnpm monorepo there are **two independent caches** to manage:

1. **Dependency cache** — the global pnpm store (`~/.local/share/pnpm/store`) containing extracted packages
2. **Build cache** — per-app build artifacts (`.next`, `dist/`, `build/`)

Both must be invalidated precisely when their inputs change, never reused incorrectly, and shareable across PRs and branches.

### Dependency Cache Key

```
deps-{OS}-{hash(pnpm-lock.yaml)}
```

**Why lockfile-only:**
- `pnpm-lock.yaml` encodes every transitive resolution. If it changes, *any* package may have changed. Hashing it is deterministic and O(1) file read.
- We intentionally do **not** include `src/` hashes because dependency install is independent of source code.
- On first-ever run with no cache hit, `restore-keys` provides a partial fallback (`deps-{OS}-`) so subsequent installs still benefit from partial matches until the full key is written.

**Cache write-back logic:**
```yaml
- id: restore-deps
  uses: actions/cache/restore@v4
  with:
    path: ~/.local/share/pnpm/store
    key: deps-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
    restore-keys: |
      deps-${{ runner.os }}-
```

No separate `save` step needed — the `restore` action auto-saves on job exit when the key hasn't matched exactly.

### Build Cache Keys

Each app gets its own build cache keyed on **lockfile + source hash**:

```
build-web-{OS}-{hash(pnpm-lock.yaml, apps/web-app/src/**)}
build-api-{OS}-{hash(pnpm-lock.yaml, apps/api-server/src/**)}
build-bff-{OS}-{hash(pnpm-lock.yaml, apps/mobile-bff/src/**)}
```

**Why both lockfile AND src:**
- A lockfile change invalidates build caches too (new dependency could break a build), even if the app's own `src/` didn't change.
- A source-only change skips full rebuilds when the cache already reflects that `src/` state.
- Using `actions/cache/restore` with multiple glob patterns produces a SHA of all matched files concatenated.

**Artifact sharing:**
After `build-cache` completes, artifacts (`.next`, `dist`, `build`) are uploaded via `actions/upload-artifact` so downstream `test` matrix jobs can download them instead of rebuilding. This cuts test runtime significantly since the heavy compilation step was already done in a single `build-cache` job.

### Cache Lifecycle Summary

```
PR opened → deps-cache restores/misses
            ↓
         build-cache restores/misses (reads deps cache)
            ↓
      test matrix (16 combos: 2 node × 4 shards) downloads artifacts
            ↓
      All pass → PR green, caches updated for next run
```

When `pnpm-lock.yaml` changes → dependency cache invalidates → full reinstall → new deps key saved.
When only `libs/shared/` changes → all app path-filters trigger → `build-cache` rebuilds affected apps → new build keys saved.
When only docs change → `paths-ignore` skips everything.

## 4. OIDC Trust Configuration

### Architecture Overview

```
GitHub Actions Runner
    │
    │  POST /login/octpus?audience=api://AzureADTokenExchange
    │  id_token: JWT signed by GitHub's OIDC provider
    │  claims: { sub: "repo:owner/repo:environment:production", ... }
    ▼
AWS STS AssumeRoleWithWebIdentity
    │  role-to-assume: arn:aws:iam::{account}:role/github-oidc-production
    │  TokenProvider: actions/configure-aws-credentials@v4 (uses id-token:write permission)
    ▼
AWS Temporary Credentials (5-15 min TTL)
    │  Used only for the deployment job duration
    ▼
Cloud Provider API (ECS/EKS/GCP/Azure)
```

### What You Configure in AWS

#### 1. IAM Identity Provider (run once per account)

```bash
aws iam create-open-id-connect-provider \
  --url https://token.actions.githubusercontent.com \
  --client-id-list sts.amazonaws.com \
  --thumbprint-list 6938fd4d98bab03faadb97b34396831e3780aea1  \
                       1c58a3a8518e8759bf075b76b750d4f2df264fcd
```

This thumbprint is GitHub's static OIDC CA cert fingerprint (verifies at [token.actions.githubusercontent.com](https://token.actions.githubusercontent.com)).

#### 2. Trust-Policy Documents

**Production role** — restricts to the repo's main branch and protected environment:

```json
{
  "Version": "2012-10-17",
  "Statement": [{
    "Effect": "Allow",
    "Principal": {
      "Federated": "arn:aws:iam::{ACCOUNT_ID}:oidc-provider/token.actions.githubusercontent.com"
    },
    "Action": "sts:AssumeRoleWithWebIdentity",
    "Condition": {
      "StringEquals": {
        "token.actions.githubusercontent.com:aud": "sts.amazonaws.com"
      },
      "StringLike": {
        "token.actions.githubusercontent.com:sub": "repo:{OWNER}/{REPO}:ref:refs/heads/main"
      }
    }
  }]
}
```

**Preview role** — scoped per-environment so PR `{N}` can only access the preview resource:

```json
{
  "Version": "2012-10-17",
  "Statement": [{
    "Effect": "Allow",
    "Principal": {
      "Federated": "arn:aws:iam::{ACCOUNT_ID}:oidc-provider/token.actions.githubusercontent.com"
    },
    "Action": "sts:AssumeRoleWithWebIdentity",
    "Condition": {
      "StringEquals": {
        "token.actions.githubusercontent.com:aud": "sts.amazonaws.com"
      },
      "StringLike": {
        "token.actions.githubusercontent.com:sub": "repo:{OWNER}/{REPO}:environment:preview-{app}"
      }
    }
  }]
}
```

**No long-lived keys stored anywhere.** The `actions/configure-aws-credentials@v4` action reads the `id-token: write` permission granted by GitHub, obtains a short-lived JWT from the GH metadata endpoint (`http://169.254.169.254/latest/api/token` + `/meta-data/identity`), exchanges it via `AssumeRoleWithWebIdentity`, and sets `AWS_ACCESS_KEY_ID`, `AWS_SECRET_ACCESS_KEY`, `AWS_SESSION_TOKEN` as environment variables scoped to the job.

### Per-App Role Isolation

Each preview app gets its own role:
- `github-oidc-preview-web-app` — only can deploy to `web-app-*` ECS service
- `github-oidc-preview-api-server` — only can deploy to `api-server-*` ECS service  
- `github-oidc-preview-mobile-bff` — only can deploy to `mobile-bff-*` ECS service

Role policies restrict to specific CloudFormation stacks, ECS services, or Kubernetes namespaces. A compromised PR cannot reach another app's infrastructure.

### Security Properties

| Property | Mechanism |
|---|---|
| No stored secrets | JWT exchange is ephemeral, generated per-job |
| Repo-scoped | `StringLike: sub` includes owner/repo |
| Branch-scoped | `ref:refs/heads/main` prevents dev-branch deploys |
| Environment-scoped | `environment:preview-{app}` limits preview roles |
| Time-limited | STS tokens expire in 15 minutes max |
| Auditable | CloudTrail logs every `AssumeRoleWithWebIdentity` call |

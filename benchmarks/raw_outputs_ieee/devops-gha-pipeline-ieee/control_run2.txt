## Workflow File #1: `.github/workflows/ci.yml` — Path-Filtered CI Per App

```yaml
name: CI — Path-Filtered per App

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

# Cancel duplicate runs on the same branch/PR
concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

env:
  PNPM_CACHE_DIR: ~/.local/share/pnpm/store

jobs:
  # ──────────────────────────────────────────────
  # Dependency resolution — only when lockfile or
  # workspace graph actually changes
  # ──────────────────────────────────────────────
  resolve-deps:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 2          # need diff against parent for change detection

      - name: Determine changed apps/libs
        id: changes
        run: |
          git diff --quiet HEAD~1..HEAD . || echo "full=changed" >> $GITHUB_OUTPUT
          
          if [[ "${{ github.event.pull_request }}" != "" ]]; then
            BASE=${{ github.event.pull_request.base.sha }}
            MERGE=$(git rev-parse HEAD)
            DIFF_BASE="$BASE..$MERGE"
          else
            DIFF_BASE="HEAD~1..HEAD"
          fi

          APPS="webapp api-server worker"
          LIBS="core types db-client logger auth validate test-utils config"

          for lib in $LIBS; do
            if git diff --quiet $DIFF_BASE -- "packages/$lib/" || [[ -z "$(git diff $DIFF_BASE -- "packages/$lib/")" ]]; then
              : # unchanged
            else
              echo "lib-$lib=changed" >> $GITHUB_OUTPUT
            fi
          done

          for app in $APPS; do
            if git diff --quiet $DIFF_BASE -- "apps/$app/" ; then
              : # unchanged
            else
              echo "app-$app=changed" >> $GITHUB_OUTPUT
            fi
            # Also flag the app if any of its local deps changed
            for lib in $LIBS; do
              if git diff --quiet $DIFF_BASE -- "packages/$lib/"; then
                :
              else
                echo "app-$app=changed" >> $GITHUB_OUTPUT
              fi
            done
          done

      - name: Resolve deps only if workspace-wide input changed
        if: ${{ steps.changes.outputs.full == 'changed' }}
        uses: pnpm/action-setup@v4
        with:
          version: 9

      - name: Install dependencies
        if: ${{ steps.changes.outputs.full == 'changed' }}
        run: pnpm install --frozen-lockfile

      - uses: actions/upload-artifact@v4
        if: ${{ steps.changes.outputs.full == 'changed' }}
        with:
          name: node-modules-cache
          path: |
            node_modules/
            .pnpm-store/
          retention-days: 1
          if-no-files-found: ignore

  # ──────────────────────────────────────────────
  # Type-check entire monorepo once (cheap, fast)
  # ──────────────────────────────────────────────
  tsc-check:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with: { version: 9 }

      - uses: actions/setup-node@v4
        with:
          node-version-file: ".nvmrc"     # pinned via file
          cache: pnpm

      - name: Cache pnpm store (keyed on lockfile hash)
        uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-store-${{ runner.os }}-

      - run: pnpm install --frozen-lockfile

      - name: Build all packages first (tsc needs compiled deps)
        run: pnpm --filter "./packages/**" build

      - name: Type-check all apps
        run: pnpm --filter "./apps/**" check-types

  # ──────────────────────────────────────────────
  # Per-app CI jobs — spawned conditionally
  # ──────────────────────────────────────────────
  ci-webapp:
    needs: [resolve-deps, tsc-check]
    if: always() && contains(needs.*.result, 'success') && !contains(needs.resolve-deps.outputs.*, '=changed') || contains(github.event.head_commit.modified, 'apps/webapp/') || always() && contains(fromJson('["skipped", "success"]'), needs.resolve-deps.result)
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4 { version: 9 }
      - uses: actions/setup-node@v4 { node-version-file: '.nvmrc', cache: pnpm }
      
      - name: Restore deps cache
        uses: actions/cache@v4
        with:
          path: node_modules
          key: mods-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            mods-${{ runner.os }}-

      - name: Restore pnpm store
        uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-store-${{ runner.os }}-

      - run: pnpm install --frozen-lockfile
      - run: pnpm --filter "./packages/**" build --filter webapp...
      - run: pnpm --filter webapp lint
      - run: pnpm --filter webapp build

  ci-api-server:
    needs: [resolve-deps, tsc-check]
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4 { version: 9 }
      - uses: actions/setup-node@v4 { node-version-file: '.nvmrc', cache: pnpm }

      - uses: actions/cache@v4
        with:
          path: node_modules
          key: mods-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: | mods-${{ runner.os }}-

      - uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: | pnpm-store-${{ runner.os }}-

      - run: pnpm install --frozen-lockfile
      - run: pnpm --filter "./packages/**" build --filter api-server...
      - run: pnpm --filter api-server lint
      - run: pnpm --filter api-server build

  ci-worker:
    needs: [resolve-deps, tsc-check]
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4 { version: 9 }
      - uses: actions/setup-node@v4 { node-version-file: '.nvmrc', cache: pnpm }

      - uses: actions/cache@v4
        with:
          path: node_modules
          key: mods-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: | mods-${{ runner.os }}-

      - uses: actions/cache@v4
        with:
          path: ${{ env.PNPM_CACHE_DIR }}
          key: pnpm-store-${{ runner.os }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: | pnpm-store-${{ runner.os }}-

      - run: pnpm install --frozen-lockfile
      - run: pnpm --filter "./packages/**" build --filter worker...
      - run: pnpm --filter worker lint
      - run: pnpm --filter worker build

  # Final gate: aggregate all results
  ci-result:
    needs: [resolve-deps, tsc-check, ci-webapp, ci-api-server, ci-worker]
    if: always()
    runs-on: ubuntu-latest
    steps:
      - name: Check overall status
        run: |
          failed=$(jq -r '[needs.*.result[] | select(. == "failure")] | length' <<< '${{ toJson(needs) }}')
          errored=$(jq -r '[needs.*.result[] | select(. == "cancelled" || . == "skipped")] | length' <<< '${{ toJson(needs) }}')
          if (( failed > 0 )); then
            echo "::error::${failed} job(s) failed"
            exit 1
          fi
          if (( errored > 0 )); then
            echo "::warning::${errored} job(s) were skipped or cancelled"
          fi
```

### Cleaner Approach: Use `paths-filter` Action

The hand-rolled diff above works but is brittle. The production-ready version uses the official filter action:

```yaml
# In the real file, replace resolve-deps step 2 with:
      - uses: dorny/paths-filter@v3
        id: filters
        with:
          filters: |
            webapp:
              - 'apps/webapp/**'
              - 'packages/*/src/**'           # any lib src change affects all
            api-server:
              - 'apps/api-server/**'
              - 'packages/*/src/**'
            worker:
              - 'apps/worker/**'
              - 'packages/*/src/**'
            docs-only:
              - 'docs/**'
              - '*.md'
            shared:
              - 'pnpm-lock.yaml'
              - '.github/workflows/*'
              - 'tsconfig*.json'
              - 'pnpm-workspace.yaml'
```

Then wire each CI app job to run only when its filter fires:

```yaml
  ci-webapp:
    needs: [path-filter]
    if: needs.path-filter.outputs.webapp == 'true'
```

This is significantly more maintainable than git-diff shell logic. I recommend the `dorny/paths-filter@v3` approach.

## Workflow File #2: `.github/workflows/test-matrix.yml` — Shard Test Across Node 20/22

```yaml
name: Test Matrix — Node × Shards

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

concurrency:
  group: tests-${{ github.ref }}
  cancel-in-progress: true

env:
  CI: "true"
  NODE_ENV: test

jobs:
  test:
    strategy:
      fail-fast: false
      matrix:
        node-version: [20, 22]
        shard-index: [1, 2, 3, 4]
        include:
          - node-version: 20
            os: ubuntu-latest
          - node-version: 22
            os: ubuntu-latest

    runs-on: ${{ matrix.os }}
    timeout-minutes: 30

    steps:
      - uses: actions/checkout@v4

      - name: Setup pnpm
        uses: pnpm/action-setup@v4
        with: { version: 9 }

      - name: Setup Node ${{ matrix.node-version }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: pnpm

      # ── Two-layer cache strategy ──────────────
      # Layer 1: pnpm store (global packages)
      - uses: actions/cache@v4
        id: store-cache
        with:
          path: ~/.local/share/pnpm/store
          key: pnpm-store-${{ runner.os }}-${{ matrix.node-version }}-${{ hashFiles('pnpm-lock.yaml') }}
          restore-keys: |
            pnpm-store-${{ runner.os }}-${{ matrix.node-version }}-

      # Layer 2: per-project node_modules
      - uses: actions/cache@v4
        id: mods-cache
        with:
          path: |
            **/node_modules/
            !node_modules/.cache/
          key: mods-${{ runner.os }}-${{ matrix.node-version }}-${{ hashFiles('pnpm-lock.yaml', 'pnpm-workspace.yaml') }}
          restore-keys: |
            mods-${{ runner.os }}-${{ matrix.node-version }}-

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      # ── Build shared libs once (cached between shards) ─
      - name: Build all packages
        run: pnpm --filter "./packages/**" build --parallel

      # ── Run vitest/jest with shard flags ──────
      # Each shard runs ~25% of test files.
      # The total-shard count must match the matrix size (8).
      # We split by Node version first, then shard within each.
      - name: Run tests — Node ${{ matrix.node-version }} shard ${{ matrix.shard-index }}/4
        run: |
          TEST_TOTAL=4
          TOTAL_SHARDS=$(( TEST_TOTAL * 2 ))  # 2 Node versions × 4 shards
          SHARD=$(( (matrix.shard-index - 1) + ((matrix.node-version == 22) * 4) + 1 ))
          TOTAL=$TOTAL_SHARDS

          pnpm --recursive --filter "!./packages/test-utils" \
            test --shard="${SHARD}/${TOTAL}"

      - name: Upload coverage
        if: ${{ matrix.node-version == 22 && matrix.shard-index == 1 }}
        uses: actions/upload-artifact@v4
        with:
          name: coverage-node-${{ matrix.node-version }}
          path: |
            **/coverage/lcov.info
          retention-days: 7

  test-coverage-report:
    needs: test
    if: always()
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Download coverage artifacts
        uses: actions/download-artifact@v4
        with:
          pattern: coverage-*
          path: coverage-artifacts
          merge-multiple: true

      - name: Fail if coverage dropped below threshold
        run: |
          # Require ≥ 80% line coverage
          npx --yes c8 report --check-coverage --lines 80
```

### Sharding Math Explained

We have **8 total shard positions** (2 Node × 4 shards). Vitest/Jest use `--shard N/M` where M = total shards. Each runner executes exactly one quarter of the test suite. After all 8 complete, the coverage numbers are combined.

## Workflow File #3: `.github/workflows/pr-preview.yml` — Ephemeral Preview Environments

```yaml
name: PR Preview Environments

on:
  pull_request:
    branches: [main]
    types: [opened, synchronize, reopened]

concurrency:
  group: preview-${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  preview-webapp:
    runs-on: ubuntu-latest
    environment:
      name: preview-webapp
      url: ${{ steps.deploy.outputs.url }}
    timeout-minutes: 20
    outputs:
      url: ${{ steps.deploy.outputs.url }}
    steps:
      - uses: actions/checkout@v4

      - name: Determine if this PR touches webapp
        uses: dorny/paths-filter@v3
        id: filters
        with:
          filters: |
            affected:
              - 'apps/webapp/**'
              - 'packages/*/src/**'

      - name: Skip if no relevant changes
        if: steps.filters.outputs.affected != 'true'
        run: echo "No webapp-related changes detected, skipping preview."

      - name: Setup Node
        if: steps.filters.outputs.affected == 'true'
        uses: actions/setup-node@v4
        with: { node-version-file: '.nvmrc' }

      - name: Setup pnpm
        if: steps.filters.outputs.affected == 'true'
        uses: pnpm/action-setup@v4
        with: { version: 9 }

      - name: Install & Build
        if: steps.filters.outputs.affected == 'true'
        run: |
          pnpm install --frozen-lockfile
          pnpm --filter "./packages/**" build
          pnpm --filter webapp build

      - name: Deploy preview (Cloudflare Pages example)
        if: steps.filters.outputs.affected == 'true'
        id: deploy
        run: |
          URL=https://pr-${{ github.event.pull_request.number }}-webapp.preview.example.com
          # Replace with actual deploy command:
          # cloudflared preview ... or vercel --token ...
          echo "url=$URL" >> $GITHUB_OUTPUT

      - name: Comment with preview URL
        if: steps.filters.outputs.affected == 'true'
        uses: actions/github-script@v7
        with:
          script: |
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: `🚀 **Preview deployed!**\n\n[Webapp preview → ${{ steps.deploy.outputs.url }}](${{ steps.deploy.outputs.url }})`
            })

  preview-api:
    runs-on: ubuntu-latest
    environment:
      name: preview-api
      url: ${{ steps.deploy.outputs.url }}
    timeout-minutes: 20
    outputs:
      url: ${{ steps.deploy.outputs.url }}
    steps:
      - uses: actions/checkout@v4
      - uses: dorny/paths-filter@v3 { id: filters }
        with:
          filters: |
            affected:
              - 'apps/api-server/**'
              - 'packages/*/src/**'
      - if: steps.filters.outputs.affected != 'true'
        run: echo "Skipping — no API-related changes"
      - uses: actions/setup-node@v4
        with: { node-version-file: '.nvmrc' }
      - uses: pnpm/action-setup@v4 { version: 9 }
      - run: pnpm install --frozen-lockfile && pnpm --filter ./packages/** build && pnpm --filter api-server build
      - id: deploy
        run: |
          URL=https://api-pr-${{ github.event.pull_request.number }}.preview.example.com
          echo "url=$URL" >> $GITHUB_OUTPUT
```

Key properties:
- **Destructive concurrency group** (`preview-${{ github.ref }}`) cancels stale previews when a new commit arrives.
- **Per-PR GitHub Environment** creates isolated secrets scoping — even though previews usually don't need secrets, environments gate visibility.
- **Path filter** prevents preview spinning up for doc-only PRs.
- **Auto-comment** drops the preview URL on the PR for instant access.

```yaml
            await github.rest.repos.createCommitStatus({
              owner: context.repo.owner,
              repo: context.repo.repo,
              sha: context.payload.merge_group?.head_sha || context.sha,
              state: 'success',
              description: 'All CI checks passed. Ready to merge.',
              context: 'merge-queue/gatekeeper',
            })
```

## Cache-Key Design Deep Dive

We use a **two-layer cache** keyed on content hashes, not timestamps. This avoids stale cache pollution entirely.

### Layer 1: `pnpm` Store Cache

```
pnpm-store-{OS}-{NodeMajor}-{hash(pnpm-lock.yaml)}
```

| Component | Why |
|---|---|
| `{OS}` | Linux vs macOS binary compatibility differs; skip cross-contamination |
| `{NodeMajor}` | Node ABI breaks between major versions (20 vs 22); native addons recompile |
| `hash(pnpm-lock.yaml)` | The lockfile encodes exact resolved version + integrity hash for every dep. Any version bump or add/remove changes the SHA-256, producing a unique key |

**Fallback chain**: If the exact key misses, restore from the nearest ancestor key (same OS, same Node, older lockfile hash). This guarantees the cache is usable even on partial misses — the `pnpm install --frozen-lockfile` will fill gaps incrementally because most packages already exist in the partially-restored store.

**Why not cache with just `hashFiles('pnpm-lock.yaml')`?** Without OS/Node prefix, the key would be too narrow and cause unnecessary invalidations. Adding the prefix makes each variant independent while still being fully deterministic.

### Layer 2: `node_modules` Cache

```
mods-{OS}-{NodeMajor}-{hash(pnpm-lock.yaml, pnpm-workspace.yaml)}
```

Same pattern, but includes `pnpm-workspace.yaml` in the hash because changing workspace membership (adding/removing packages) alters the dependency graph regardless of the lockfile content.

### Cache Behavior Table

| Scenario | Exact miss? | Effectiveness of restore key |
|---|---|---|
| New lockfile (dep bumped) | Yes | Partial — restores old packages, incremental download fills gaps |
| Same lockfile, new Node | Yes (different prefix) | Partial — restores same-OS+older-Node packages |
| Same lockfile, same Node | No hit | Full cache hit, instant install |
| First-ever run | Complete miss | Falls through to full install |

### Why We Don't Cache Built Artifacts

Built `dist/` directories contain transitive computed outputs that depend on many factors (source maps, build args, platform-specific binaries). A miscompilation is far worse than a rebuild. We keep the cache focused on **inputs that never change at runtime** — the lockfile and workspace graph.

## OIDC Federation Trust Configuration

The OIDC approach eliminates long-lived AWS IAM credentials entirely. Here's how the trust relationship is constructed.

### Architecture Diagram

```
GitHub Actions Runner ──JWT signed by GitHub's Fulcio-like IdP──► AWS STS AssumeRoleWithWebIdentity
       │                                                           │
       │  POST https://token.actions.githubusercontent.com        │
       │  ?audience=sts.amazonaws.com                              │
       │  headers: Authorization: Bearer jwt                     │
       │                                                           │
       ▼                                                           ▼
  GitHub Issues Token Issuer                              AWS STS
       │                                                     │
       │  Returns signed JWT with claims:                    │
       │  - sub: repo:sanjeev/agent-skills:ref:refs/tags/v1 │
       │  - aud: sts.amazonaws.com                           │
       │  - iss: https://token.actions.githubusercontent.com │
       │  - ref, sha, actor, environment, …                 │
       │                                                     │
       └──────────────── JWT ───────────────────────────────►│
                                                             │
                                                     Returns
                                                  temporary creds
                                                   (1hr TTL)
```

### AWS IAM Role Policy (to be created via Terraform/CDK/Console)

```json
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Principal": {
        "Federated": "arn:aws:iam::123456789012:oidc-provider/token.actions.githubusercontent.com"
      },
      "Action": "sts:AssumeRoleWithWebIdentity",
      "Condition": {
        "StringEquals": {
          "token.actions.githubusercontent.com:aud": "sts.amazonaws.com"
        },
        "StringLike": {
          "token.actions.githubusercontent.com:sub": "repo:sanjeev/agent-skills:*"
        }
      }
    }
  ]
}
```


### Steps to Register the OIDC Provider in AWS (one-time setup)

```bash
# 1. Find your GitHub OIDC issuer URL
ISSUER="https://token.actions.githubusercontent.com"

# 2. Add the provider — thumbprint is GitHub's static TLS cert fingerprint
#    This never changes, so you can safely add it to IaC
aws iam create-open-id-connect-provider \
  --url $ISSUER \
  --client-id-list sts.amazonaws.com \
  --thumbprint-list 6938fd4d98bab03faadb97b34396831e3780aea1 \
  --description "GitHub Actions OIDC provider"
```

### Condition Keys That Make This Secure

| Key | Purpose | Example Value |
|---|---|---|
| `token.actions.githubusercontent.com:aud` | Audience claim — must target AWS STS | `sts.amazonaws.com` |
| `token.actions.githubusercontent.com:sub` | Subject — scoped to specific repo | `repo:sanjeev/agent-skills:environment:production` |
| `token.actions.githubusercontent.com:ref` | Tag/branch gating | `refs/tags/v*`, `refs/heads/main` |
| `token.actions.githubusercontent.com:sha` | Commit pinning | specific SHA for auditability |
| `token.actions.githubusercontent.com:environment` | Env scoping | `production`, `staging` |

You tighten the policy by adding more conditions:

```json
// Only allow production deployments from tags, not from main branch pushes
"StringLike": {
  "token.actions.githubusercontent.com:sub": "repo:sanjeev/agent-skills:ref:refs/tags/v*",
  "token.actions.githubusercontent.com:environment": "production"
}
```

### Why No Long-Lived Keys Are Needed

The `aws-actions/configure-aws-credentials@v4` action with `role-to-assume` performs the entire flow internally:
1. Requests a short-lived JWT from GitHub's well-known endpoint (`/.github/actions/configuration`)
2. Calls `sts:AssumeRoleWithWebIdentity` presenting that JWT
3. Exports `AWS_ACCESS_KEY_ID`, `AWS_SECRET_ACCESS_KEY`, `AWS_SESSION_TOKEN` as env vars with auto-expiring session credentials (default 1 hour)
4. These are single-use tokens — even if intercepted, they expire quickly and are scoped to the exact role + session

This means: zero secrets stored in GitHub Secrets for AWS, zero risk of key leakage from a compromised repository, and full audit trail through CloudTrail logs showing exactly which commit/tag triggered which credential assumption.

### Stage 1 — Smoke Test (Read-Only Crawl)

**Objective:** Confirm the dashboard loads without errors, passes Core Web Vitals, and makes no failed requests.

**Scriptable checks:**

```javascript
// Run inside puppeteer/playwright page context
async function smokeTest(page, url) {
  const logs = [];
  const responses = new Map();

  // Capture console
  page.on('console', msg => {
    if (msg.type() === 'error' || msg.type() === 'warning') {
      logs.push({ type: msg.type(), text: msg.text() });
    }
  });

  // Capture network
  page.on('response', resp => responses.set(resp.url(), resp));

  await page.goto(url, { waitUntil: 'networkidle0', timeout: 30000 });

  // Console audit — filter known noise
  const criticalErrors = logs.filter(l =>
    l.type === 'error' &&
    !l.text.match(/(analytics|segment|hotjar|intercom|gtag)/i)   // skip third-party
  );

  // Network audit
  const failedReqs = [...responses.values()].filter(r => r.status() >= 400);

  // Core Web Vitals via Performance API
  const vitals = await page.evaluate(() => {
    return new Promise(resolve => {
      setTimeout(async () => {
        const entries = performance.getEntriesByType('paint');
        const lcpEntry = await new Promise(r => {
          new PerformanceObserver(list => r(list.getEntries()[list.getEntries().length - 1])).observe({ type: 'largest-contentful-paint', buffered: true });
        });
        resolve({
          lcp: lcpEntry ? lcpEntry.startTime : null,
          cls: performance.layoutShiftSum || 0,
          inp: null // collected via event observer below
        });
      }, 3000);
    });
  });

  // INP via Interaction observer
  let bestInp = Infinity;
  await page.evaluate(() => {
    return new Promise(resolve => {
      new PerformanceObserver(list => {
        for (const entry of list.getEntries()) {
          if (entry.duration < bestInp) bestInp = entry.duration;
        }
      }).observe({ type: 'event', buffered: true });
      setTimeout(resolve, 5000);
    });
  });

  return {
    criticalConsoleErrors: criticalErrors,
    warnings: logs.filter(l => l.type === 'warning'),
    failedNetworkRequests: failedReqs.map(r => ({ status: r.status(), url: r.url() })),
    coreWebVitals: {
      lcp: vitals.lcp,
      cls: vitals.cls,
      inp: bestInp === Infinity ? null : bestInp
    },
    pass: {
      lcpPass: !vitals.lcp || vitals.lcp <= 2500,
      clsPass: !vitals.cls || vitals.cls <= 0.1,
      inpPass: !bestInp || bestInp <= 200
    }
  };
}
```

**Exit criteria for Stage 1:**
- ✓ No critical console errors
- ✓ Zero 4xx/5xx network responses (CDN/favicon preload noise excluded)
- ✓ LCP ≤ 2.5s · CLS ≤ 0.1 · INP ≤ 200ms (all three must pass)
- ✗ One failure → DO NOT PROCEED to later stages until fixed

#### 2A — Navigation Links

```javascript
async function crawlNavLinks(page, initialUrl) {
  const seen = new Set();
  const results = [];

  await page.goto(initialUrl, { waitUntil: 'networkidle0' });

  const links = await page.$$eval('nav a[href], [role="navigation"] a[href]', els =>
    els.map(el => el.href).filter(h => h.startsWith(initialUrl.split('/')[0] + '//') && !h.includes('#'))
  );

  for (const href of links) {
    if (seen.has(href)) continue;
    seen.add(href);
    try {
      const resp = await page.evaluate(async () => {
        const r = await fetch(new URL(location.pathname, location.origin).href, { method: 'GET', keepalive: true });
        return { ok: r.ok, status: r.status };
      });
      // Note: cross-origin will fail CORS — acceptable noise
      results.push({ url: href, status: resp?.status ?? 'cross-origin-skip' });
    } catch (e) {
      results.push({ url: href, status: 'navigate-error', error: e.message });
    }
  }
  return results;
}
```

#### 2B — Filters (the critical path for an analytics dashboard)

```javascript
async function testFilters(page, url) {
  const steps = [
    // Step 1: Load dashboard
    { action: 'goto', target: url },
    // Step 2: Select a date range filter
    { action: 'click', selector: '[data-filter="date-range"] button[aria-label="Last 30 days"]' },
    // Step 3: Wait for chart/table re-render
    { action: 'wait-for-network-idle', timeout: 10000 },
    // Step 4: Verify chart updated — check SVG paths or canvas changed
    { action: 'assert', check: async (pg) => {
      const chartPaths = await pg.$$eval('.chart svg path', els => els.length);
      return chartPaths > 0; // chart re-rendered, didn't break
    }},
    // Step 5: Apply secondary filter (category dropdown)
    { action: 'select', selector: '[data-filter="category"] select', value: 'organic' },
    { action: 'wait-for-network-idle', timeout: 10000 },
    // Step 6: Verify table filtered — row count changes
    { action: 'assert', check: async (pg) => {
      const rows = await pg.$$eval('[data-table="analytics"] tbody tr', els => els.length);
      return rows > 0 && rows < 500; // narrowed from unfiltered state
    }},
    // Step 7: Reset filters — should return to original state
    { action: 'click', selector: '[data-filter-reset]' },
    { action: 'wait-for-network-idle', timeout: 10000 },
    // Step 8: Verify clean state
    { action: 'assert', check: async (pg) => {
      const activeFilters = await pg.$$('button[aria-pressed="true"]');
      return activeFilters.length === 0;
    }},
  ];

  const result = { steps: [], failedStep: null };
  for (const step of steps) {
    const ctx = { step, time: Date.now() };
    try {
      switch (step.action) {
        case 'goto': await page.goto(step.target, { waitUntil: 'networkidle0' }); break;
        case 'click': await page.click(step.selector); break;
        case 'wait-for-network-idle': await page.waitForNetworkIdle({ idleTimeout: 1000 }); break;
        case 'select': await page.select(step.selector, step.value); break;
        case 'assert': {
          const passed = await step.check(page);
          if (!passed) throw new Error('Assertion failed');
          break;
        }
      }
      result.steps.push({ ...ctx, status: 'pass' });
    } catch (e) {
      result.steps.push({ ...ctx, status: 'fail', error: e.message });
      result.failedStep = result.steps[result.steps.length - 1];
      break;
    }
  }
  return result;
}
```

### Stage 3 — Visual Regression

**Approach:** Screenshot at 3 breakpoints × light/dark mode = 6 screenshots per key page. Mask time-dependent widgets. Diff against committed baselines at 0.1% tolerance.

```javascript
async function visualRegression(page, url, breakpoints = [1440, 768, 390]) {
  const screenshots = [];

  for (const width of breakpoints) {
    await page.setViewport({ width, height: width >= 1440 ? 900 : width >= 768 ? 1024 : 844 });

    for (const theme of ['light', 'dark']) {
      await page.goto(url, { waitUntil: 'networkidle0' });

      // Toggle theme
      if (theme === 'dark') {
        await page.evaluate(() => {
          document.documentElement.classList.add('dark');
          localStorage.setItem('theme', 'dark');
        });
        await page.waitForTimeout(300); // allow CSS transition
      } else {
        await page.evaluate(() => {
          document.documentElement.classList.remove('dark');
          localStorage.setItem('theme', 'light');
        });
        await page.waitForTimeout(300);
      }

      // Mask time-sensitive regions
      await page.evaluate(() => {
        document.querySelectorAll('[data-mask-time-sensitive]').forEach(el => {
          el.style.setProperty('opacity', '0.3', 'important');
        });
      });

      const filename = `${url.replace(/[^a-z0-9]/gi, '_')}-${width}-${theme}.png`;
      const buffer = await page.screenshot({ fullPage: true });
      screenshots.push({ filename, width, theme, timestamp: Date.now() });
    }
  }

  return screenshots;
}

function compareAgainstBaselines(screenshots, baselineDir) {
  const results = [];
  for (const ss of screenshots) {
    const baselinePath = path.join(baselineDir, ss.filename);
    if (!fs.existsSync(baselinePath)) {
      results.push({ file: ss.filename, verdict: 'INCONCLUSIVE', reason: 'no committed baseline' });
      continue;
    }
    // Structured similarity (SSIM) or perceptual hash diff
    const current = sharp(ss.buffer);
    const baseline = sharp(baselinePath);
    // Compare — flag if diff exceeds 0.1% of pixels
    const diffRatio = computePerceptualDiff(current, baseline);
    const exceedsThreshold = diffRatio > 0.001; // 0.1%
    results.push({
      file: ss.filename,
      diffPercent: (diffRatio * 100).toFixed(4) + '%',
      thresholdExceeded: exceedsThreshold,
      verdict: exceedsThreshold ? 'FAIL' : 'PASS'
    });
  }
  return results;
}
```

**Layout shift detection (>5px):**

```javascript
async function detectLayoutShifts(page) {
  const shifts = await page.evaluate(() => {
    return new Promise(resolve => {
      const observed = [];
      new ResizeObserver(entries => {
        for (const entry of entries) {
          const { x, y, width, height } = entry.contentRect;
          if (Math.abs(x) > 5 || Math.abs(y) > 5) {
            observed.push({ element: entry.target.tagName, shiftX: x, shiftY: y, w: width, h: height });
          }
        }
      }).observe(document.body, { attributeFilter: ['style'] });
      setTimeout(() => resolve(observed), 5000);
    });
  });
  return shifts;
}
```

### Stage 4 — Accessibility Pass

```javascript
async function accessibilityAudit(page, url) {
  // Inject axe-core
  await page.addScriptTag({ url: 'https://cdn.jsdelivr.net/npm/axe-core@4.8.3/dist/axe.min.js' });

  const results = await page.evaluate(async () => {
    return await axe.run(document, {
      runOnly: ['wcag2a', 'wcag2aa', 'cat.keyboard', 'cat.color', 'cat.forms'],
      resultTypes: ['violations', 'passes']
    });
  });

  // Categorize violations
  const violations = results.violations.map(v => ({
    id: v.id,
    impact: v.impact, // 'critical' | 'serious' | 'moderate' | 'minor'
    description: v.description,
    nodes: v.nodes.length,
    helpUrl: v.helpUrl,
    wcagCriteria: v.tags?.map(t => t.slice(5)) || [] // e.g., ['color-contrast', 'label-title-only']
  }));

  // Keyboard nav verification
  const keyboardIssues = await page.evaluate(() => {
    const focusable = document.querySelectorAll('a, button, input, select, textarea, [tabindex]:not([tabindex="-1"])');
    const outlineRemoved = [];
    // Check for removed outlines without alternative focus indicator
    const styleSheets = Array.from(document.styleSheets);
    for (const sheet of styleSheets) {
      try {
        for (const rule of sheet.cssRules) {
          if (rule.selectorText?.includes(':focus') && rule.style?.outline === 'none' && !rule.style?.outlineStyle) {
            outlineRemoved.push(rule.selectorText);
          }
        }
      } catch (e) { /* CORS */ }
    }
    return {
      totalFocusable: focusable.length,
      likelyOutlineRemovals: outlineRemoved
    };
  });

  // ARIA roles on chart tooltips
  const tooltipAccessibility = await page.evaluate(() => {
    const tooltips = document.querySelectorAll('[role="tooltip"], [aria-describedby]');
    const chartElements = document.querySelectorAll('.chart, svg');
    const hasLiveRegion = !!document.querySelector('[role="log"], [role="status"], [aria-live]');
    return {
      tooltipsCount: tooltips.length,
      chartElementsWithARIA: Array.from(chartElements).filter(el =>
        el.hasAttribute('role') || el.closest('[role="img"]')
      ).length,
      hasLiveRegion,
      missingLiveRegion: !hasLiveRegion ? 'Chart updates not announced to screen readers' : null
    };
  });

  return { violations, keyboardIssues, tooltipAccessibility, timestamp: Date.now() };
}
```

**Sign-off decision:**
- **SHIP** — all criteria met, zero open issues
- **SHIP WITH FIXES** — minor/non-blocking issues filed, fix owners assigned, PR template linked
- **DO NOT SHIP** — any critical violation, Core Web Vital failure on desktop, blocked by blast-radius mutation risk
- **INCONCLUSIVE** — baseline missing for ≥1 key page screenshot

### Stage 1 — Smoke Crawl (Read-Only, Non-Interactive)

**Goal:** Verify the page loads without console errors or failed requests. Zero interaction.

#### Test Steps

```
1. Launch headless Chromium, navigate to dashboard URL
2. Capture all console messages for 5 seconds post-load
3. Filter noise patterns: analytics, mixpanel, segment, hotjar, sentry.io, 
   google-analytics.com, cdn.jsdelivr.net, unpkg.com (library CDNs)
4. Collect all network responses → flag any 4xx/5xx EXCEPT:
   - 304 Not Modified (cache hit)
   - 429 Rate limit within threshold (< 3/min)
   - /health or /ping endpoints returning non-200 during startup
5. Take screenshots at all 3 viewports ABOVE THE FOLD
6. Measure Core Web Vitals programmatically via Performance API
```

```typescript
// stage1-smoke-crawl.spec.ts
import { test, expect } from '@playwright/test';

const VIEWPORTS = [
  { name: 'desktop', width: 1440, height: 900 },
  { name: 'tablet',  width: 768,  height: 1024 },
  { name: 'mobile',  width: 390,  height: 844 },
];

// Console noise filter patterns
const SILENT_PATTERNS = [
  'mixpanel', 'segment', 'hotjar', 'googletagmanager', 'facebookpixel',
  'cdn\\.jsdelivr', 'unpkg\\.com', 'cloudflareinsights', 'widget\\.userreplay',
  'performance.now', // internal perf markers, not LCP/CLS/INP
];

function isSilenced(msg: string): boolean {
  return SILENT_PATTERNS.some(p => new RegExp(p, 'i').test(msg));
}

test.describe('Stage 1: Smoke Crawl', () => {
  const dashboardUrl = process.env.DASHBOARD_URL ?? 'http://localhost:3000/dashboard';

  for (const vp of VIEWPORTS) {
    test(`${vp.name} viewport (${vp.width}x${vp.height})`, async ({ page }) => {
      await page.setViewportSize(vp);

      // --- Console error collection ---
      const consoleErrors: string[] = [];
      page.on('console', msg => {
        if (msg.type() === 'error' && !isSilenced(msg.text())) {
          consoleErrors.push(`[${msg.type()}] ${msg.text()}`);
        }
      });

      // --- Network failure collection ---
      const failedRequests: { url: string; status: number }[] = [];
      page.on('response', async resp => {
        const status = resp.status();
        if ((status >= 400 && status < 600) && 
            !resp.url().includes('/health') &&
            !resp.url().includes('/ping')) {
          failedRequests.push({ url: resp.url(), status });
        }
      });

      // --- Navigate + settle ---
      await page.goto(dashboardUrl, { waitUntil: 'networkidle', timeout: 30_000 });
      
      // Wait for performance entries to be populated
      await page.waitForTimeout(5_000);

      // --- Validate: zero silenced console errors ---
      expect(consoleErrors).toEqual([]);

      // --- Validate: zero failed requests ---
      expect(failedRequests.length).toBe(0);

      // --- Screenshots ---
      await page.screenshot({ path: `stage1/${vp.name}-above-the-fold.png`, fullPage: false });

      // --- Core Web Vitals via JS injection ---
      const cwv = await page.evaluate(async (): Promise<{
        lcp: number | null; cls: number; inp: number | null;
      }> => {
        const lcpEntries = (performance.getEntriesByType('largest-contentful-paint') as PerformanceEntry[])
          .filter(e => e.startTime > 0);
        const lcp = lcpEntries.length ? lcpEntries[lcpEntries.length - 1].startTime : null;

        let cls = 0;
        for (const entry of performance.getEntriesByType('layout-shift') as PerformanceEntry[]) {
          if (!(entry as any).hadRecentInput) cls += (entry as any).value;
        }

        // INP approximation: use event delay from longtask-like events
        // In Playwright we get best-effort via PerformanceObserver data
        const navTiming = performance.timing as any;
        const inp = navTiming.domInteractive 
          ? Math.min(navTiming.domInteractive - navTiming.requestStart, 500) 
          : null;

        return { lcp: lcp ? +(lcp / 1000).toFixed(3) : null, cls: +cls.toFixed(4), inp };
      });

      console.log(`${vp.name}: LCP=${cwv.lcp}s CLS=${cwv.cls} INP=${cwv.inp}ms`);

      // --- Threshold assertions ---
      if (cwv.lcp !== null) {
        expect(cwv.lcp).toBeLessThan(2.5);
      }
      expect(cwv.cls).toBeLessThan(0.1);
      if (cwv.inp !== null) {
        expect(cwv.inp).toBeLessThan(200);
      }
    });
  }
});
```

### Stage 3 — Visual Regression with Masking

**Goal:** Catch unintended layout/style drift while tolerating time-dependent widgets.

#### Strategy

```
Threshold: 0.1% pixel diff allowed (not area-based, but global image diff percentage)

Masking layers applied BEFORE comparison:
┌─────────────────────────┐
│ 1. Timestamp widget     │  Remove from screenshot (overlay black rect)
│ 2. Live clock/gauge     │  Same treatment
│ 3. Date-range header    │  Text-only, allow variance
│ 4. Loading skeleton     │  Auto-retry until resolved or 10s timeout
│ 5. Auth avatar          │  Crop to fixed region, compare rest
└─────────────────────────┘
Baseline approach: 
  • First successful run creates baseline images (with masks baked in)
  • Stored in visual-baseline/{viewport}/{page}-{component}.png.masked.png
  • On CI: baseline from main branch; on local: generate fresh
  • Dark mode: SEPARATE baseline directory _dark suffix
```

```typescript
// stage3-visual.spec.ts
import { test, expect } from '@playwright/test';
import { pixelmatch } from 'pixelmatch';
import { PNG } from 'pngjs';

const BASELINE_DIR = './visual-baseline';
const CURRENT_DIR  = './visual-current';
const DIFF_DIR     = './visual-diffs';

function maskRegion(buffer: Buffer, mask: { x: number; y: number; w: number; h: number }): Buffer {
  const src = PNG.sync.read(buffer);
  const dst = new PNG({ width: src.width, height: src.height });
  
  for (let y = 0; y < src.height; y++) {
    for (let x = 0; x < src.width; x++) {
      const srcIdx = (y * src.width + x) << 2;
      const dstIdx = (y * src.width + x) << 2;
      
      if (x >= mask.x && x < mask.x + mask.w && y >= mask.y && y < mask.y + mask.h) {
        // Paint masked region solid gray (#888)
        dst.data[dstIdx]     = 136; // R
        dst.data[dstIdx + 1] = 136; // G
        dst.data[dstIdx + 2] = 136; // B
        dst.data[dstIdx + 3] = 255; // A
      } else {
        // Copy source pixel
        dst.data[dstIdx]     = src.data[srcIdx];
        dst.data[dstIdx + 1] = src.data[srcIdx + 1];
        dst.data[dstIdx + 2] = src.data[srcIdx + 2];
        dst.data[dstIdx + 3] = src.data[srcIdx + 3];
      }
    }
  }
  return PNG.sync.write(dst);
}

function compareImages(baseline: Buffer, current: Buffer): number {
  const b = PNG.sync.read(baseline);
  const c = PNG.sync.read(current);
  const diff = new PNG({ width: b.width, height: b.height });
  const pixelsDiff = pixelmatch(b.data, c.data, diff.data, b.width, b.height, { threshold: 0.1 });
  return pixelsDiff / (b.width * b.height); // Normalized diff ratio
}
```

**Key design decisions:**
- **Masking happens on the screenshot buffer itself** before pixelmatch comparison, not as CSS overlay hiding (which could miss layout shifts behind the hidden area)
- **0.1% threshold**: means 1 out of every 1000 pixels can differ (sub-pixel rendering, anti-aliasing differences)
- **Time-widget masking coordinates are approximate** — need calibration per screen size; better to detect them via DOM element position than hardcoded pixels
- **Separate `_dark` baseline directory** keeps light/dark configs cleanly isolated

**Exit from Stage 3:** Diff ratio ≤ 0.1% across all 9 combinations (3 viewports × 3 dark modes × baseline variants). Any failure writes diff image to report directory for manual review.

```typescript
    console.log(`Light mode: ${lightResult.passes} pass, ${lightResult.fails} fail`);

    // Dark mode
    await page.emulateMedia({ colorScheme: 'dark' });
    await page.waitForTimeout(1_000);
    const darkResult = await checkContrast();
    console.log(`Dark mode: ${darkResult.passes} pass, ${darkResult.fails} fail`);

    expect(lightResult.fails).toBe(0);
    expect(darkResult.fails).toBe(0);
  });

  test('prefers-reduced-motion respected', async ({ page }) => {
    await page.goto(dashboardUrl);
    await page.emulateMedia({ reducedMotion: 'reduce' });
    await page.waitForTimeout(5_000);

    // Check that no CSS transitions are actively running on animated elements
    const transitioning = await page.evaluate(() => {
      const animated = document.querySelectorAll('.chart-animate, .fade-in, *[class*="animate"]');
      return Array.from(animated).filter(el => {
        const style = window.getComputedStyle(el);
        return style.transitionDuration !== '0s' || style.animationDuration !== '0s';
      }).length;
    });

    expect(transitioning).toBe(0);
  });
});
```

### Dark Mode Baseline Strategy

The dark mode baseline requires special handling because:

1. **Two independent baseline trees** — `baseline/light/` and `baseline/dark/`
2. **CSS variable swaps happen at runtime** — test toggles `data-theme="dark"` on `<html>`, waits for paint, then captures
3. **Chart libraries often ignore dark themes** — must assert that tooltip backgrounds turn white-on-dark or use CSS variable colors
4. **Tables often have hardcoded borders** — visually regression-sensitive

```
Baseline directory structure:
visual-baseline/
├── 1440x900/
│   ├── charts.png.masked          ← light baseline
│   ├── charts.png.masked_dark     ← dark baseline
│   └── tables.png.masked_dark
├── 768x1024/
│   ├── charts.png.masked
│   └── charts.png.masked_dark
└── 390x844/
    ├── charts.png.masked
    └── charts.png.masked_dark

CI flow:
  main branch baseline → compare → diff threshold → pass/fail
  Feature branch → uses main baseline as comparison target
  New feature → generates new baseline (requires --review flag)
```

## Phase 1: Smoke Test — Console & Network Crawl

### What it does

Single-page read-only crawl of `/dashboard` at 1440px desktop viewport. Collects console errors and HTTP status codes, screens above-the-fold content, measures Core Web Vitals.

### Playwright script (run against staging)

```js
// smoke-test-dashboard.js
const { chromium } = require('playwright');

async function runSmokeTest() {
  const browser = await chromium.launch({ headless: true });
  const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });

  const consoleErrors = [];
  const networkFailures = [];

  page.on('console', msg => {
    if (msg.type() === 'error') {
      const text = msg.text();
      // Filter known noise
      if (/analytics|gtag|fbq|hotjar|intercom/i.test(text)) return;
      consoleErrors.push({ type: msg.type(), text: text, location: msg.location().url });
    }
  });

  page.on('response', resp => {
    if (resp.status() >= 400) {
      networkFailures.push({
        url: resp.url(),
        status: resp.status(),
        method: resp.request().method(),
      });
    }
  });

  console.log('[smoke] Navigating to dashboard...');
  const response = await page.goto(process.env.DASHBOARD_URL || 'https://staging.yourapp.com/dashboard', {
    waitUntil: 'networkidle',
    timeout: 30_000,
  });

  console.log(`[smoke] Page status: ${response.status()}`);
  console.log(`[smoke] Console errors collected: ${consoleErrors.length}`);
  console.log(`[smoke] Network failures collected: ${networkFailures.length}`);

  // Core Web Vitals via Performance API
  const cwv = await page.evaluate(() => {
    return new Promise(resolve => {
      const entries = performance.getEntriesByType('paint');
      const lcp = new Promise(r => {
        new PerformanceObserver(list => r(list.getEntries()[list.getEntries().length - 1]?.startTime)).observe({ type: 'largest-contentful-paint', buffered: true });
        setTimeout(() => r(null), 2000);
      });
      const layoutShifts = performance.getEntriesByType('layout-shift').filter(e => !e.hadRecentInput);
      const cls = layoutShifts.reduce((sum, e) => sum + (e.value || 0), 0);
      resolve({ entries: Object.fromEntries(entries.map(e => [e.name, e.startTime])), cls });
    });
  });

  console.log('[smoke] Core Web Vitals:', JSON.stringify(cwv, null, 2));

  // Screenshot above-the-fold
  await page.screenshot({ path: 'screenshots/smoke-desktop.png', fullPage: false });

  // Mobile viewport check
  const mobile = await browser.newPage({ viewport: { width: 390, height: 844 } });
  await mobile.goto(page.url(), { waitUntil: 'networkidle' });
  await mobile.screenshot({ path: 'screenshots/smoke-mobile.png', fullPage: false });

  await browser.close();

  // Fail fast if critical issues found
  const criticalConsoleErrors = consoleErrors.filter(e =>
    !/analytics|gtag|fbq|hotjar|intercom|fontawesome|googleapis/.test(e.text)
  );
  if (criticalConsoleErrors.length > 0) {
    console.error('[FAIL] Critical console errors:', criticalConsoleErrors);
    process.exit(1);
  }

  const serverErrors = networkFailures.filter(n => n.status >= 500);
  if (serverErrors.length > 0) {
    console.error('[FAIL] Server errors:', serverErrors);
    process.exit(1);
  }

  // CWV thresholds
  if (cwv.cls > 0.1) {
    console.error(`[FAIL] CLS ${cwv.cls} exceeds threshold 0.1`);
    process.exit(1);
  }

  console.log('[PASS] Smoke test passed.');
}

runSmokeTest().catch(err => {
  console.error('[CRASH]', err.message);
  process.exit(1);
});
```

## Phase 3: Visual Regression — Multi-Viewport Screenshots

### Strategy

Capture screenshots at 3 breakpoints, compare against committed baselines with a **0.1% diff threshold**. Mask time-dependent widgets (timestamps, live counters, animated charts) before comparison.

### Incompatible visual-regression library

```
npm install --save-dev @bramus/visual-diff@latest
```

Or use **Playwright's built-in screenshot comparison**:

```js
// visual-regression-dashboard.js
const { chromium } = require('playwright');
const pixelmatch = require('pixelmatch');
const PNG = require('pngjs').PNG;
const fs = require('fs');
const path = require('path');

const BASELINES_DIR = './baselines';
const SCREENSHOTS_DIR = './screenshots';
const BREAKPOINTS = [
  { name: 'desktop',   width: 1440, height: 900 },
  { name: 'tablet',    width: 768,  height: 1024 },
  { name: 'mobile',    width: 390,  height: 844 },
];
const DIFF_THRESHOLD = 0.001; // 0.1%

function maskTimeSensitiveElements(imageData) {
  // Overlay a semi-transparent rectangle over known dynamic widget areas.
  // Coordinates are approximate and should be calibrated per project.
  // This masks: live timestamp badges, spinning loaders, animated counters.
  const w = imageData.width;
  const h = imageData.height;
  // Simple approach: darken pixels in top-right quadrant where timestamps usually live
  for (let y = 0; y < Math.floor(h * 0.15); y++) {
    for (let x = Math.floor(w * 0.7); x < w; x++) {
      const idx = (y * w + x) * 4;
      imageData.data[idx]     = 0;    // R
      imageData.data[idx + 1] = 0;    // G
      imageData.data[idx + 2] = 0;    // B
      imageData.data[idx + 3] = 128;  // A (semi-transparent)
    }
  }
  return imageData;
}

function rgbToHex(r, g, b) {
  return '#' + [r, g, b].map(c => c.toString(16).padStart(2, '0')).join('');
}

async function runVisualRegression() {
  const browser = await chromium.launch({ headless: true });
  const url = process.env.DASHBOARD_URL || 'https://staging.yourapp.com/dashboard';
  const results = [];

  for (const bp of BREAKPOINTS) {
    console.log(`\n[bp:${bp.name}] Running visual regression at ${bp.width}x${bp.height}...`);
    const page = await browser.newContext({ viewport: bp }).newPage();
    await page.goto(url, { waitUntil: 'networkidle', timeout: 30_000 });
    await page.waitForTimeout(1000); // wait for animations to settle

    const shotPath = path.join(SCREENSHOTS_DIR, `regress-${bp.name}.png`);
    await page.screenshot({ path: shotPath, fullPage: true });

    const baselinePath = path.join(BASELINES_DIR, `regress-${bp.name}.png`);

    if (!fs.existsSync(baselinePath)) {
      results.push({ viewport: bp.name, pass: false, diffPct: -1,
        details: 'NO BASELINE — capture first run as new baseline' });
      console.log(`  [INCONCLUSIVE] No baseline at ${baselinePath}`);
      continue;
    }

    // Read images
    const currentRaw = PNG.sync.read(fs.readFileSync(shotPath));
    const baseRaw = PNG.sync.read(fs.readFileSync(baselinePath));
    const { width, height } = currentRaw;
    const diff = new PNG({ width, height });

    const mismatchPx = pixelmatch(currentRaw.data, baseRaw.data, diff.data, width, height, {
      threshold: 0.01, // perceptual threshold per pixel
    });
    const diffPct = mismatchPx / (width * height);

    // Now mask time-sensitive regions and re-compute
    const maskedCurrent = maskTimeSensitiveElements(PNG.sync.read(fs.readFileSync(shotPath)).data);
    const maskedBase = maskTimeSensitiveElements(PNG.sync.read(fs.readFileSync(baselinePath)).data);

    const diffMasked = new PNG({ width, height });
    const mismatchPxMasked = pixelmatch(maskedCurrent, maskedBase, diffMasked.data, width, height, { threshold: 0.01 });
    const diffPctMasked = mismatchPxMasked / (width * height);

    const pass = diffPctMasked <= DIFF_THRESHOLD;
    results.push({ viewport: bp.name, pass, diffPct: parseFloat((diffPctMasked * 100).toFixed(3)),
      details: pass ? `masked diff ${diffPctMasked.toFixed(4)} (${(diffPctMasked*100).toFixed(2)}%) ≤ 0.1%`
        : `masked diff ${(diffPctMasked*100).toFixed(2)}% exceeds threshold` });

    // Save diff image for review
    const diffPath = path.join(SCREENSHOTS_DIR, `diff-${bp.name}.png`);
    fs.writeFileSync(diffPath, PNG.sync.write(diffMasked));

    const icon = pass ? '[✓]' : '[✗]';
    console.log(`  ${icon} ${bp.name}: ${diffPctMasked.toFixed(4)} diff${pass ? ' within threshold' : ' EXCEEDS THRESHOLD'}`);

    // Check for specific layout issues
    const overflow = await page.evaluate(() => {
      const html = document.documentElement;
      return html.scrollWidth > html.clientWidth || html.scrollHeight > html.clientHeight;
    });
    if (overflow) {
      results.push({ viewport: bp.name, pass: false, detail: 'scroll overflow detected' });
      console.log(`  [✗] ${bp.name}: horizontal/vertical scroll overflow`);
    }
  }

  await browser.close();

  const totalPass = results.filter(r => r.pass && r.diffPct !== -1).length;
  const inconclusive = results.filter(r => r.diffPct === -1).length;
  const failures = results.filter(r => !r.pass && r.diffPct !== -1).length;

  console.log(`\nVisual regression: ${totalPass} passed, ${failures} failed, ${inconclusive} inconclusive`);
  fs.writeFileSync('results/visual-regression-results.json', JSON.stringify(results, null, 2));
  process.exit(failures > 0 ? 1 : 0);
}

runVisualRegression().catch(err => {
  console.error('[CRASH]', err.message);
  process.exit(1);
});
```

## Phase 4: Accessibility Pass

### axe-core integration

```js
// accessibility-dashboard.js
const { chromium } = require('playwright');

async function runAccessibilityAudit() {
  const browser = await chromium.launch({ headless: true });
  const url = process.env.DASHBOARD_URL || 'https://staging.yourapp.com/dashboard';
  const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });

  // Inject axe-core
  await page.addInitScript(() => {
    const script = document.createElement('script');
    script.src = 'https://cdn.jsdelivr.net/npm/axe-core@4.10.0/axe.min.js';
    document.head.appendChild(script);
  });
  await page.goto(url, { waitUntil: 'networkidle' });
  await page.waitForTimeout(1000);

  // 4a. Run axe-core WCAG 2.2 AA audit
  const violations = await page.evaluate(async () => {
    const result = await axe.run(document, {
      runOnly: ['wcag2a', 'wcag2aa', 'wcag21aa'],
      resultTypes: ['violations', 'incomplete'],
    });
    return result.violations.map(v => ({
      id: v.id,
      impact: v.impact,
      description: v.description,
      nodes: v.nodes.length,
      helpUrl: v.helpUrl,
      examples: v.nodes.slice(0, 3).map(n => ({
        target: n.target,
        html: n.html?.substring(0, 200), // truncate for readability
      })),
    }));
  });

  console.log(`[a11y] Violations found: ${violations.length}`);
  violations.forEach(v => {
    console.log(`  [${v.impact?.toUpperCase()}] ${v.id}: ${v.description} (${v.nodes} instances)`);
    console.log(`       Help: ${v.helpUrl}`);
  });

  // 4b. Contrast ratio check on key elements
  const contrastIssues = await page.evaluate(() => {
    const elements = document.querySelectorAll('[class], [style]');
    const issues = [];
    for (const el of elements) {
      const bg = getComputedStyle(el).backgroundColor;
      const fg = getComputedStyle(el).color;
      if (!fg || fg === 'rgba(0, 0, 0, 0)' || !bg || bg === 'rgba(0, 0, 0, 0)') continue;
      // Parse hex colors and compute luminance difference
      const parseColor = (c) => {
        const match = c.match(/[\d]+/g)?.map(Number);
        if (!match || match.length < 3) return null;
        return { r: match[0], g: match[1], b: match[2] };
      };
      const fgC = parseColor(fg);
      const bgC = parseColor(bg);
      if (!fgC || !bgC) continue;

      const luminance = ({ r, g, b }) => {
        const [rs, gs, bs] = [r, g, b].map(c => {
          const srgb = c / 255;
          return srgb <= 0.03928 ? srgb / 12.92 : Math.pow((srgb + 0.055) / 1.055, 2.4);
        });
        return 0.2126 * rs + 0.7152 * gs + 0.0722 * bs;
      };

      const L1 = luminance(fgC);
      const L2 = luminance(bgC);
      const ratio = (Math.max(L1, L2) + 0.05) / (Math.min(L1, L2) + 0.05);

      if (ratio < 4.5) {
        issues.push({
          target: el.className?.baseVal || el.id || '(inline element)',
          fg, bg,
          ratio: ratio.toFixed(2),
          requirement: 'WCAG 2.2 AA requires ≥ 4.5:1 for normal text',
        });
      }
    }
    return issues.slice(0, 20); // cap output
  });

  console.log(`\n[a11y] Contrast ratio issues (< 4.5:1): ${contrastIssues.length}`);
  contrastIssues.slice(0, 10).forEach(ci => {
    console.log(`  ${ci.target}: ratio=${ci.ratio} (fg=${ci.fg}, bg=${ci.bg})`);
  });

  // 4c. Keyboard navigation test
  const tabOrder = await page.evaluate(() => {
    const focusable = Array.from(document.querySelectorAll(
      'a[href], button, input, select, textarea, [tabindex]:not([tabindex="-1"])'
    ));
    return focusable.map((el, i) => ({
      index: i,
      tag: el.tagName.toLowerCase(),
      role: el.getAttribute('role') || '',
      label: el.textContent?.trim().substring(0, 50) || '',
      ariaLabel: el.getAttribute('aria-label') || '',
    }));
  });

  console.log(`\n[a11y] Focusable elements: ${tabOrder.length}`);
  // Verify logical order — skip non-chart elements for brevity
  const missingLabels = tabOrder.filter(el =>
    !el.label && !el.ariaLabel && el.role !== 'presentation'
  ).slice(0, 10);
  if (missingLabels.length > 0) {
    console.log('[a11y] Elements missing accessible labels:');
    missingLabels.forEach(el => {
      console.log(`  <${el.tag}> at index ${el.index} (role="${el.role}")`);
    });
  }

  // 4d. ARIA roles on chart tooltips
  const tooltipCheck = await page.evaluate(() => {
    // Hover a chart element to trigger tooltip
    const bars = document.querySelectorAll('[data-testid="chart-bar"], .chart-series__rect');
    if (bars.length === 0) return [{ issue: 'No chart bars found for tooltip testing' }];

    // Simulate hover on first bar
    const firstBar = bars[0];
    const rect = firstBar.getBoundingClientRect();
    firstBar.dispatchEvent(new MouseEvent('mouseover', {
      clientX: rect.left + rect.width / 2,
      clientY: rect.top + rect.height / 2,
      bubbles: true,
    }));

    return new Promise(resolve => {
      setTimeout(() => {
        const tooltips = document.querySelectorAll('[role="tooltip"], .chart-tooltip, [data-testid="chart-tooltip"]');
        const issues = [];
        tooltips.forEach(tip => {
          const role = tip.getAttribute('role');
          const hasAriaHidden = tip.getAttribute('aria-hidden');
          if (role !== 'tooltip') {
            issues.push({ tooltip: tip.className, role: role || 'none',
              note: 'expected role="tooltip"' });
          }
          // Tooltips should have aria-describedby or contain descriptive text
          const textContent = tip.textContent?.trim();
          if (!textContent) {
            issues.push({ tooltip: tip.className, note: 'empty tooltip — no visible text' });
          }
        });
        if (tooltips.length === 0) {
          issues.push({ issue: 'No tooltip rendered after chart hover' });
        }
        resolve(issues);
      }, 500);
    });
  });

  console.log(`\n[a11y] Chart tooltip ARIA check:`);
  tooltipCheck.forEach(t => {
    console.log(`  ✗ ${t.issue || t.note}`);
  });
  if (tooltipCheck.length === 0) {
    console.log('  ✓ Tooltips have correct role="tooltip" and contain text');
  }

  // Screenshot with violation highlights
  await page.screenshot({ path: 'screenshots/a11y-desktop.png', fullPage: true });

  await browser.close();

  // Summarize
  const totalViolations = violations.reduce((sum, v) => sum + v.nodes, 0);
  const criticalViolations = violations.filter(v => v.impact === 'critical' || v.impact === 'serious');

  console.log(`\n=== A11y Summary ===`);
  console.log(`Total violation instances: ${totalViolations}`);
  console.log(`Critical/serious: ${criticalViolations.length}`);
  console.log(`Contrast issues: ${contrastIssues.length}`);
  console.log(`Missing labels: ${missingLabels.length}`);
  console.log(`Tooltip ARIA issues: ${tooltipCheck.length}`);

  const report = {
    axeViolations: violations,
    contrastIssues,
    missingLabels,
    tooltipIssues: tooltipCheck,
    timestamp: new Date().toISOString(),
  };
  require('fs').writeFileSync('results/accessibility-results.json', JSON.stringify(report, null, 2));

  // Exit with failure if critical violations exist
  if (criticalViolations.length > 0) {
    console.error('\n[FAIL] Critical/serious violations found — DO NOT SHIP');
    process.exit(1);
  }
}

runAccessibilityAudit().catch(err => {
  console.error('[CRASH]', err.message);
  process.exit(1);
});
```

## Baseline Strategy Summary

| Breakpoint | Light Mode | Dark Mode | Storage |
|---|---|---|---|
| 1440px (desktop) | `baselines/regress-desktop.png` | `baselines-dark/regress-desktop.png` | Git-tracked |
| 768px (tablet) | `baselines/regress-tablet.png` | `baselines-dark/regress-tablet.png` | Git-tracked |
| 390px (mobile) | `baselines/regress-mobile.png` | `baselines-dark/regress-mobile.png` | Git-tracked |

- **Diff threshold:** 0.1% masked pixel difference
- **Masking overlay:** top-right quadrant (~15% height × 30% width) covers timestamps, live counters, refresh indicators
- **Chart masking:** charts are inherently animated; we accept up to 5% raw diff but after masking time-sensitive regions, final diff must be ≤ 0.1%
- **New baseline capture:** only when designers formally approve the new design in staging; never auto-generated from CI

## Exit Criteria

The dashboard passes QA and gets the green light only when ALL of the following are true:

| Gate | Condition |
|---|---|
| **Smoke Test** | 0 critical console errors, 0 server (5xx) failures, CLS ≤ 0.1 |
| **Core Web Vitals** | LCP < 2.5s, CLS < 0.1, INP < 200ms (all 3 must pass) |
| **Interactions** | All filter flows pass, drill-down renders tables, no broken nav links, dark mode toggle functions |
| **Visual Regression** | All 3 breakpoints at both light/dark modes: masked diff ≤ 0.1%, no scroll overflow |
| **Accessibility** | 0 critical/serious WCAG violations, all chart tooltips have `role="tooltip"`, contrast ≥ 4.5:1 on text elements |
| **Manual A11y** | Keyboard navigation verified end-to-end, screen reader landmarks confirmed |
| **Security** | No PII/tokens in screenshots (redaction verified), no mutating journeys executed on prod |

### Verdict options

- **SHIP** — all gates pass, zero blocking issues
- **SHIP WITH FIXES** — non-blocking issues only (cosmetic, low-severity a11y warnings), documented and tracked in tickets
- **DO NOT SHIP** — any blocker: critical a11y violations, visual regression fails on any viewport, Core Web Vitals out of threshold, security concerns
- **INCONCLUSIVE** — no baseline exists yet; cannot determine regressions

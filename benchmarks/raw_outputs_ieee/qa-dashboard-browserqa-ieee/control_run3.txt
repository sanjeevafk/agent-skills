### 2. Stage 1 — Read-Only Smoke Crawl

**Goal:** Collect every unhandled error and failed network call *without triggering any side-effects.*

```js
// src/qa/smoke-crawl.js
import { chromium } from 'playwright';

/**
 * Launches headless browser(s), navigates to /dashboard, collects
 * console messages and network errors, exits cleanly.
 * Runs once per viewport via --viewport parameter.
 */
export async function smokeCrawl({ url, viewport }) {
  const browser = await chromium.launch({ headless: true });
  const context = await browser.newContext({
    viewport,
    // Ensure consistent rendering — disables GPU accel for reproducible visuals
    deviceScaleFactor: 1,
  });

  const logs = [];
  const errors = [];

  const page = await context.newPage();

  // Capture every log line
  page.on('console', (msg) => {
    if (msg.type() === 'error' || msg.type() === 'warning') {
      logs.push({ severity: msg.type(), text: msg.text(), location: msg.location().url });
    }
  });

  // Track all responses; flag non-2xx / CORS failures
  const failedRequests = [];
  page.on('response', async (res) => {
    if (!res.ok()) {
      failedRequests.push({ url: res.url(), status: res.status(), method: res.request().method() });
    }
  });
  page.on('requestfailed', (req) => {
    failedRequests.push({ url: req.url(), status: 0, reason: req.failure()?.errorText, method: req.method() });
  });

  try {
    await page.goto(url, { waitUntil: 'networkidle', timeout: 30_000 });
    // Extra stability margin — lets lazy-loaded charts hydrate
    await page.waitForTimeout(5_000);
  } finally {
    // Always collect even on navigation failure
    try {
      for (const msg of await page.evaluate(() => [...document.querySelectorAll('[data-test-id="console-fatal"]')] ?? [])) {
        // Framework-specific fallback markers
      }
    } catch {}
  }

  await browser.close();

  return {
    viewport,
    totalLogs: logs.length,
    consoleErrors: logs.filter(l => l.severity === 'error'),
    consoleWarnings: logs.filter(l => l.severity === 'warning'),
    failedRequests,
    overallStatus: logs.some(l => l.severity === 'error') ? 'FAIL' : 'PASS',
  };
}
```

**What we collect:**
- `console.error` lines with stack traces
- `console.warn` about deprecated APIs or missing translations
- Every HTTP `< 400` response (real failures, not intentional 404 probes)
- Service Worker registration errors
- CSP violations logged via `onmessage` handler
- Font loading timeouts (`document.fonts.ready` promise rejection)

```js
/**
 * Configuration for the analytics dashboard visual regression suite.
 * 
 * Key design decisions explained:
 *
 * • DIFF_THRESHOLD = 0.001 (0.1%)
 *   Charts rendered via Canvas/SVG can vary pixel-by-pixel due to font
 *   hinting or anti-aliasing. 0.1% is tight enough to catch layout shifts
 *   but wide enough to tolerate sub-pixel rounding differences between
 *   CI machines. Anything above this almost certainly indicates a regression.
 *
 * • MASK_STRATEGY
 *   Three classes of time-dependent or non-deterministic DOM must be
 *   blacked out before comparison: live clock widgets, auto-updating
 *   counters, and random seed decorations (sparkline jitter, gradient
 *   noise).
 *
 * • BASELINE_REGENERATION
 *   First run against a clean repo creates baselines (--record=baseline).
```

**Masking logic (applied before diff):**

```js
// src/qa/utils/mask-time-widgets.js
/**
 * Takes a Page object, reads computed styles of matched elements,
 * and replaces them with solid-color placeholders matching their
 * box model. Prevents time-sensitive content from causing diffs.
 */
export async function maskTimeWidgets(page, selectors) {
  await page.addStyleTag({
    content: selectors.map(sel => `
      ${sel} {
        background: none !important;
        border: 2px dashed rgba(255, 0, 0, 0.4) !important;
      }
    `).join('\n'),
  });
}
```

**Baseline strategy:**
- **Light mode**: captured once against a controlled fixture page (seeded data, frozen timestamps via `Date.now()` stub). Stored in `tests/visual/baseline/light/`.
- **Dark mode**: captured separately from light baseline. Dark mode does NOT reuse the light image with a color-inversion filter — it needs its own baselines because shadow colors, border opacities, and text contrast differ structurally under `#1a1a1a` vs `#ffffff`.
- Each baseline filename encodes viewport + theme + section ID: `desktop-1440-light-chart-overview.png`.
- If someone legitimately changes a chart's axis labels (which alters dimensions), they must update only that single region — run with `--rebasline --region chart-overview`.

```js
// src/qa/tests/accessibility/chart-a11y.test.js
import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

test.describe('Dashboard Accessibility', () => {
  test('WCAG 2.1 AA: zero violations on fresh load', async ({ page }) => {
    await page.setViewportSize({ width: 1440, height: 900 });
    await page.goto('/dashboard');
    await page.waitForLoadState('networkidle');

    const axeResults = await new AxeBuilder({ page })
      .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa'])
      .analyze();

    // Strict: 0 violations at AA level
    const aaViolations = axeResults.violations.filter(v => v.impact === 'critical' || v.impact === 'serious');
    expect(aaViolations.length).toBe(0);

    // Warnings allowed for transient issues (e.g., aria-describedby on lazy chart canvas)
    const warnings = axeResults.violations.filter(v => v.impact === 'moderate');
    expect(warnings.length).toBeLessThanOrEqual(3); // grace period for chart libs
  });

  test('Color contrast: all text meets 4.5:1 (normal) / 3:1 (large)', async ({ page }) => {
    await page.setViewportSize({ width: 1440, height: 900 });
    await page.goto('/dashboard');

    const contrastIssues = await page.evaluate(() => {
      const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
        acceptNode(node) {
          const style = window.getComputedStyle(node.parentElement);
          if (style.color === 'transparent' || style.color === 'inherit') {
            return NodeFilter.FILTER_REJECT;
          }
          return NodeFilter.FILTER_ACCEPT;
        }
      });

      const results = [];
      let node;
      while ((node = walker.nextNode())) {
        const text = node.textContent?.trim();
        if (!text || text.length < 2) continue;
        
        // Get computed color and background for contrast check
        // Using simplified luminance calculation
        const textColor = getTextColor(node.parentElement);
        const bgColor = getBackgroundColor(node.parentElement);
        
        if (textColor && bgColor) {
          const ratio = contrastRatio(textColor, bgColor);
          if (ratio < 4.5) {
            results.push({
              text: text.substring(0, 50),
              ratio: ratio.toFixed(2),
              required: '4.5:1',
              element: node.parentElement.tagName,
            });
          }
        }
      }
      return results;

      function getTextColor(el) {
        const style = window.getComputedStyle(el);
        return style.color !== 'inherit' ? parseColor(style.color) : null;
      }

      function getBackgroundColor(el) {
        let current = el;
        while (current && current !== document.body) {
          const bg = window.getComputedStyle(current).backgroundColor;
          if (bg && bg !== 'rgba(0, 0, 0, 0)' && bg !== 'transparent') {
            return parseColor(bg);
          }
          current = current.parentElement;
        }
        return { r: 255, g: 255, b: 255 }; // assume white background
      }

      function parseColor(colorStr) {
        // Simplified — real impl would handle rgb(), rgba(), hex(), hsl()
        const match = colorStr.match(/(\d+)/g);
        if (!match) return null;
        return {
          r: parseInt(match[0]),
          g: parseInt(match[1]),
          b: parseInt(match[2]),
        };
      }

      function contrastRatio(c1, c2) {
        const l1 = relativeLuminance(c1);
        const l2 = relativeLuminance(c2);
        const lighter = Math.max(l1, l2);
        const darker = Math.min(l1, l2);
        return (lighter + 0.05) / (darker + 0.05);
      }

      function relativeLuminance(c) {
        const sRGB = [c.r / 255, c.g / 255, c.b / 255];
        const linear = sRGB.map(c => c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4));
        return 0.2126 * linear[0] + 0.7152 * linear[1] + 0.0722 * linear[2];
      }
    });

    expect(contrastIssues.length).toBe(0);
  });

  test('Focus order: tab key traverses logical reading order', async ({ page }) => {
    await page.setViewportSize({ width: 1440, height: 900 });
    await page.goto('/dashboard');

    // Tab through all focusable elements
    const focusOrder = [];
    for (let i = 0; i < 30; i++) {
      await page.keyboard.press('Tab');
      const focusedEl = await page.evaluate(() => {
        const el = document.activeElement;
        if (!el || el === document.body) return null;
        return {
          tag: el.tagName.toLowerCase(),
          role: el.getAttribute('role'),
          ariaLabel: el.getAttribute('aria-label'),
          text: el.textContent?.substring(0, 40).trim(),
          className: el.className?.substring?.(0, 60),
        };
      });
      if (focusedEl) focusOrder.push(focusedEl);
    }

    // Verify header nav comes before main content
    const headerIdx = focusOrder.findIndex(f => f.tag === 'nav' && f.role === 'navigation');
```

### 6. Blast-Radius Rules (What NEVER Gets Clicked)

These are guardrails baked into the test harness to prevent data corruption, billing charges, or unintended state mutations:

| Category | Rule | Rationale |
|---|---|---|
| **Billing / Subscription** | Never navigate to `/settings/billing` or click `[data-testid="upgrade-plan"]` | Could trigger trial expiry or charge events |
| **Destructive Actions** | Never click `Delete`, `Disconnect`, `Revoke`, `Purge`, `Drop` buttons | Irreversible data loss |
| **Production Data Sources** | Never trigger `Sync Now` on production source connectors | Real API rate-limits, potential data duplication |
| **External Redirects** | Block all redirects to `*.stripe.com`, `*.github.com/oauth` | OAuth tokens, payment flow entry |
| **Email Triggers** | Never click `Send Report` / `Share` buttons | Sends real emails to configured recipients |
| **API Write Endpoints** | Intercept and reject `POST/PUT/PATCH` to `/api/*/write`, `/api/*/sync` | Guards against accidental mutations |
| **Admin Functions** | Never navigate to `/admin/*` paths | Access control bypass risk |
| **Real Notifications** | Never open toast messages containing `🔔 Notification` with `view` action | Consumes unread counts |

**Enforcement mechanism:**

```js
// In the Playwright test setup — global route interceptor
page.route('**/api/*/write*', route => route.abort('blockedbyclient'));
page.route('**/api/*/sync*', route => route.abort('blockedbyclient'));
page.route('**/settings/billing', route => route.abort('blockedbyclient'));
page.route('**/admin/**', route => route.abort('blockedbyclient'));
```

### 7. Dark Mode Baseline Strategy

```
baselines/
├── light/
│   ├── desktop-1440/
│   │   ├── chart-overview.png
│   │   ├── kpi-row.png
│   │   └── ...
│   ├── tablet-768/
│   └── mobile-390/
└── dark/
    ├── desktop-1440/
    │   ├── chart-overview.png
    │   ├── kpi-row.png
    │   └── ...
    ├── tablet-768/
    └── mobile-390/
```

**Rules:**
1. Dark baselines are captured separately — no color-inversion post-processing. This is critical because CSS variables like `--shadow-color`, `--border-opacity`, and `--text-secondary` change non-linearly between light/dark themes.
2. Both baselines use the same frozen seed data and the same timestamp (`2026-08-25T00:00:00Z`), swapped out via `--frozen-time` CLI flag.
3. To re-base dark mode: set `VITE_THEME=dark` env var and run with `--rebasline`. Compare the result against the existing `dark/` baseline — if unexpected differences appear, that's a signal that the dark theme wasn't properly adjusted for the current code change.
4. System preference test: After capturing baselines, also verify `prefers-color-scheme: dark` media query triggers correctly when the browser/system setting is toggled — but this isn't baseline-requiring, just a sanity check.

| 9 | **Accessibility** — contrast ratio ≥ 4.5:1 everywhere text appears | min ratio ≥ 4.5 | Stage 4 contrast audit |

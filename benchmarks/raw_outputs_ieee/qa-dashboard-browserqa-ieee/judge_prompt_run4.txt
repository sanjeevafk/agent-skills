### What it does

### Playwright script (run against staging)

```js
// smoke-test-dashboard.js
const { chromium } = require('playwright');

async function runSmokeTest() {
  const browser = await chromium.launch({ headless: true });
  const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });

  const consoleErrors = [];
  const networkFailures = [];

  page.on('console', msg => {
    if (msg.type() === 'error') {
      const text = msg.text();
      // Filter known noise
      if (/analytics|gtag|fbq|hotjar|intercom/i.test(text)) return;
      consoleErrors.push({ type: msg.type(), text: text, location: msg.location().url });
    }
  });

  page.on('response', resp => {
    if (resp.status() >= 400) {
      networkFailures.push({
        url: resp.url(),
        status: resp.status(),
        method: resp.request().method(),
      });
    }
  });

  console.log('[smoke] Navigating to dashboard...');
  const response = await page.goto(process.env.DASHBOARD_URL || 'https://staging.yourapp.com/dashboard', {
    waitUntil: 'networkidle',
    timeout: 30_000,
  });

  console.log(`[smoke] Page status: ${response.status()}`);
  console.log(`[smoke] Console errors collected: ${consoleErrors.length}`);
  console.log(`[smoke] Network failures collected: ${networkFailures.length}`);

  // Core Web Vitals via Performance API
  const cwv = await page.evaluate(() => {
    return new Promise(resolve => {
      const entries = performance.getEntriesByType('paint');
      const lcp = new Promise(r => {
        new PerformanceObserver(list => r(list.getEntries()[list.getEntries().length - 1]?.startTime)).observe({ type: 'largest-contentful-paint', buffered: true });
        setTimeout(() => r(null), 2000);
      });
      const layoutShifts = performance.getEntriesByType('layout-shift').filter(e => !e.hadRecentInput);
      const cls = layoutShifts.reduce((sum, e) => sum + (e.value || 0), 0);
      resolve({ entries: Object.fromEntries(entries.map(e => [e.name, e.startTime])), cls });
    });
  });

  console.log('[smoke] Core Web Vitals:', JSON.stringify(cwv, null, 2));

  // Screenshot above-the-fold
  await page.screenshot({ path: 'screenshots/smoke-desktop.png', fullPage: false });

  // Mobile viewport check
  const mobile = await browser.newPage({ viewport: { width: 390, height: 844 } });
  await mobile.goto(page.url(), { waitUntil: 'networkidle' });
  await mobile.screenshot({ path: 'screenshots/smoke-mobile.png', fullPage: false });

  await browser.close();

  // Fail fast if critical issues found
  const criticalConsoleErrors = consoleErrors.filter(e =>
    !/analytics|gtag|fbq|hotjar|intercom|fontawesome|googleapis/.test(e.text)
  );
  if (criticalConsoleErrors.length > 0) {
    console.error('[FAIL] Critical console errors:', criticalConsoleErrors);
    process.exit(1);
  }

  const serverErrors = networkFailures.filter(n => n.status >= 500);
  if (serverErrors.length > 0) {
    console.error('[FAIL] Server errors:', serverErrors);
    process.exit(1);
  }

  // CWV thresholds
  if (cwv.cls > 0.1) {
    console.error(`[FAIL] CLS ${cwv.cls} exceeds threshold 0.1`);
    process.exit(1);
  }

  console.log('[PASS] Smoke test passed.');
}

runSmokeTest().catch(err => {
  console.error('[CRASH]', err.message);
  process.exit(1);
});
```

```javascript
function maskElements(page, selectors) {
  await page.evaluate((sels) => {
    sels.forEach(sel => {
      document.querySelectorAll(sel).forEach(el => {
        el.dataset.originalBg = el.style.backgroundColor || getComputedStyle(el).backgroundColor;
        el.style.setProperty('background-color', '#CCCCCC', 'important');
      });
    });
  }, selectors);
}
```

```ts
/**
 * Masks areas that change between runs due to time-sensitive content.
 * Returns a canvas mask with black rectangles over those regions.
 */
function createTimeMask(page, viewport) {
  const selectors = [
    '[data-testid="real-time-counter"]',    // live numbers
    '[data-testid="timestamp-badge"]',       // "updated X min ago"
    '[data-testid="sparkline-inline"]',      // tiny inline sparklines
    '[data-testid="current-value-label"]',   // live values in headers
  ];
  
  const maskCanvas = createCanvas(viewport.width, viewport.height);
  const ctx = maskCanvas.getContext('2d');
  ctx.fillStyle = '#000000';
  
  for (const sel of selectors) {
    const elements = page.locator(sel).all();
    for (const el of elements) {
      const box = await el.boundingBox();
      if (box) ctx.fillRect(box.x, box.y, box.width, box.height);
    }
  }
  return maskCanvas;
}
```

### Diff Threshold & Comparison Logic

- **Pixel diff threshold**: `0.1%` (1 in 1000 pixels allowed to differ)
- **Algorithm**: Structural similarity (SSIM) preferred over raw pixel diff; fallback to pixel diff if library unavailable
- **Per-pixel tolerance**: Allow ±2 luminance delta in masked regions (anti-aliasing margin)
- **Flaky rejection**: If >5% of comparisons hit the threshold automatically flag for human review rather than hard fail

```ts
import { PixelDiff } from '@example/diff-lib';

async function compareScreenshots(ref: Buffer, comp: Buffer, mask?: Buffer): Promise<{diffPercent: number, failed: boolean}> {
  const diff = new PixelDiff({ refWidth: ref.width, refHeight: ref.height, maxColorDiff: 2 });
  const result = diff.compare(ref, comp, mask);
  
  // Expressed as percentage of total pixels exceeding threshold
  const diffPercent = (result.differentPixels / (ref.width * ref.height)) * 100;
  return { diffPercent, failed: diffPercent > 0.1 };
}
```

### Contrast Verification

Check every text/background color combination against WCAG AA minimums:

```ts

 * Extracts all computed style text elements and verifies contrast ratios.
 * Requires a headed browser instance for accurate CSSOM resolution.
 */
async function auditContrast(page) {
  const violations = await page.evaluate(() => {
    const elements = Array.from(document.querySelectorAll('*'));
    const results = [];
    
    for (const el of elements) {
      const style = window.getComputedStyle(el);
      if (style.color === 'transparent' || style.visibility === 'hidden') continue;
      
      const fg = rgbToHex(style.color);
      const bg = getEffectiveBackground(el);
      
      if (!fg || !bg) continue;
      
      const ratio = contrastRatio(fg, bg);
      const minRequired = isLargeText(el) ? 3.0 : 4.5; // WCAG AA
      
      if (ratio < minRequired) {
        results.push({
          element: `${el.tagName}.${el.className.split(' ').join('.')}`,
          fg, bg, ratio,
          required: minRequired,
          node: el.outerHTML.substring(0, 200),
        });
      }
    }
    return results;
  });
  
  return violations;
}
```

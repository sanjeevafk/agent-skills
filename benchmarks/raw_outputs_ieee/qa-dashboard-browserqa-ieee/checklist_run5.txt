# Analytics Dashboard QA Plan — Staged Verification Blueprint

## Stage 0: Blast-Radius Rules (Guardrails)

These are hard boundaries — nothing in this scope gets clicked under any circumstance:

| No-Click Zone | Rationale |
|---|---|
| `/api/export` endpoints | Side-effectful; only verify network presence |
| Admin routes (`/admin/*`) | Not in scope; accidental navigation triggers data mutation |
| Any route with query param `?debug=1` | Dev-only path; must be excluded from viewport captures |
| Modal overlays with "Confirm" buttons | Irreversible actions — visibility check only |
| Third-party embeds (Heatmaps, Intercom) | External domain; out of blast radius |

**Navigation invariant**: After any filter interaction, the URL hash/query must update but the page must NOT navigate away (SPA behavior confirmed).

## Stage 1: Smoke Crawl (Read-Only)

### Objective
Hit every route the dashboard touches, collect console errors, failed network requests, and unhandled promise rejections — zero-interaction.

### Implementation — Playwright + Puppeteer hybrid

```javascript
// smoke_crawl.js — read-only visit of all dashboard views
const { chromium } = require('playwright');
const { Client } = require('@opentelemetry/api');

const VIEWS = [
  '/analytics/overview',
  '/analytics/revenue',
  '/analytics/users',
  '/analytics/funnels',
  '/analytics/cohort',
  '/analytics/events',
];

const VIEWPORTS = {
  desktop: { width: 1440, height: 900 },
  tablet:  { width: 768,  height: 1024 },
  mobile:  { width: 390,  height: 844 },
};

async function smokeCrawl(urlBase = 'http://localhost:3000') {
  const browser = await chromium.launch({ headless: true });
  const results = { consoleErrors: [], networkFailures: [], promises: [] };

  for (const view of VIEWS) {
    for (const [device, vp] of Object.entries(VIEWPORTS)) {
      const page = await browser.newPage({ viewport: vp });

      // Collect console messages
      page.on('console', msg => {
        if (msg.type() === 'error' || msg.type() === 'warning') {
          results.consoleErrors.push({
            device, view, type: msg.type(),
            text: msg.text(), source: msg.location().url,
          });
        }
      });

      // Collect failed network requests
      page.on('response', async resp => {
        if (resp.status() >= 400) {
          results.networkFailures.push({
            device, view, status: resp.status(),
            url: resp.url(), method: resp.request().method(),
          });
        }
      });

      // Collect unhandled rejection
      page.on('pageerror', err => {
        results.promises.push({ device, view, error: err.message });
      });

      await page.goto(`${urlBase}${view}`, { waitUntil: 'networkidle' });
      await page.waitForSelector('[data-testid="dashboard-root"]', { timeout: 15000 });
      // Small pause to let lazy-loaded charts hydrate
      await page.waitForTimeout(3000);

      await page.close();
    }
  }

  await browser.close();
  return results;
}

// Expose exit codes: 0 = clean, 1 = warnings, 2 = failures
function classify(results) {
  const netFail = results.networkFailures.filter(
    r => !r.url.includes('_next/static') && r.status !== 304
  );
  if (netFail.length > 0) return { level: 'fail', details: netFail };
  if (results.consoleErrors.length > 0) return { level: 'warn', count: results.consoleErrors.length };
  return { level: 'pass' };
}

module.exports = { smokeCrawl, classify, VIEWPORTS, VIEWS };
```

### Acceptance Criteria

- **Zero** 4xx/5xx responses on API calls originating from dashboard paths
- **Zero** console `error` entries from app code (third-party SDK warnings excluded via `source` filter)
- All `VIEWS` load within 3s at LCP metric threshold
- No hydration mismatch warnings in React dev-server output

### Local Diff Assertion (no external service needed)

```javascript
// visual_regression/diff_assertion.js
const sharp = require('sharp');

async function compareWithBaseline(currentPath, baselinePath, threshold = 0.001) {
  const current = await sharp(currentPath).toBuffer();
  const baseline = await sharp(baselinePath).toBuffer();

  const [meta, img] = await Promise.all([
    sharp(current).metadata(),
    sharp(current).toColorMatrix(),
  ]);

  // Resize baseline to match current dimensions
  const resized = await sharp(baselinePath)
    .resize(meta.width, meta.height)
    .toBuffer();

  const baseImg = await sharp(resized).toColorMatrix();

  // Compute RMSE between color matrices
  let sumSq = 0;
  const len = Math.min(img.length, baseImg.length);
  for (let i = 0; i < len; i++) {
    const diff = img[i] - baseImg[i];
    sumSq += diff * diff;
  }
  const rmse = Math.sqrt(sumSq / len);

  // Convert RMSE to percentage diff
  const pctDiff = (rmse / 255) * 100;

  return {
    pass: pctDiff <= (threshold * 100),
    percentDifference: pctDiff.toFixed(4),
    rmse: rmse.toFixed(4),
    threshold: (threshold * 100).toFixed(4),
  };
}

module.exports = { compareWithBaseline };
```

## Stage 4: Accessibility Pass

### Scope

| Check | Tool | Threshold |
|---|---|---|
| WCAG AA contrast ratios | axe-core / `@axe-core/playwright` | Level AA minimum |
| Focus order (Tab / Shift+Tab) | Manual assertion via `page.keyboard.press('Tab')` | Logical DOM order |
| ARIA roles on chart tooltips | Query selector assertions | Required roles present |
| Keyboard accessibility | Full Tab traversal | All interactive elements reachable |
| Screen reader text | `aria-label`, `role="img"` + `<title>` pairing | Non-empty descriptive text |

### Implementation

```javascript
// accessibility/pass.spec.js
import { test, expect } from '@playwright/test';
import { injectAxe, checkA11y } from 'axe-playwright';

const DEVICES = [
  { name: 'desktop', width: 1440, height: 900 },
  { name: 'tablet',  width: 768,  height: 1024 },
  { name: 'mobile',  width: 390,  height: 844 },

test.describe('Accessibility', () => {
  for (const bp of DEVICES) {
    test(`${bp.name}: no violations`, async ({ page }) => {
      await page.setViewportSize(bp);
      await page.goto(process.env.STAGING_URL + '/analytics/overview');
      await page.waitForSelector('[data-testid="chart-canvas"]', { timeout: 10000 });
      await page.waitForTimeout(1000);

      // Inject axe-core and run full WCAG-AA scan
      await injectAxe(page);

      // Configure to ignore known third-party exceptions
      await page.configureAxe({
        rule: {
          'color-contrast': { enabled: true },
          'link-name':  { enabled: true },
          'button-name': { enabled: true },
          'image-alt':  { enabled: true },
          'region':     { enabled: true },
        },
      });

      await checkA11y(page, null, { detailedReport: true });
    });
  }

  // Focus order validation
  test('focus traverses in logical reading order', async ({ page }) => {
    await page.goto(process.env.STAGING_URL + '/analytics/overview');
    await page.waitForSelector('[data-testid="chart-canvas"]', { timeout: 10000 });

    // Get the tab order of focused elements over 10 tabs
    const focused = [];
    for (let i = 0; i < 10; i++) {
      await page.keyboard.press('Tab');
      const handle = await page.evaluateHandle(() => document.activeElement);
      const tagName = await handle.evaluate(el => el.tagName.toLowerCase());
      const text = await handle.evaluate(el =>
        (el.textContent || '').substring(0, 40)
      );
      const role = await handle.evaluate(el => el.getAttribute('role'));
      focused.push({ tag: tagName, text, role });
      handle.dispose();
    }

    // Verify focus enters the main filter area first
    expect(focused[0].tag).toMatch(/(input|button|a)/i);

    // Focus should not skip past visible chart area
    const focusCountBeforeChart = focused.findIndex(
      f => f.tag === 'canvas' || f.role === 'img'
    );
    expect(focusCountBeforeChart).toBeGreaterThan(1);
  });

  // Chart tooltip ARIA roles
  test('chart tooltips have correct ARIA roles', async ({ page }) => {
    await page.goto(process.env.STAGING_URL + '/analytics/overview');
    await page.waitForSelector('[data-testid="chart-canvas"]', { timeout: 10000 });

    // Hover over a chart point to trigger tooltip
    const canvas = page.locator('[data-testid="chart-canvas"]');
    await canvas.dispatchEvent('mouseenter', { position: { x: 100, y: 100 } });
    await page.waitForTimeout(500);

    const tooltip = page.locator('[data-testid="chart-tooltip"]');
    await expect(tooltip).toBeVisible();

    // Tooltip must have proper role
    const role = await tooltip.getAttribute('role');
    expect(['tooltip', 'region', 'alert']).toContain(role ?? null);

    // Must have aria-labelledby or aria-describedby linking to content
    const labelledBy = await tooltip.getAttribute('aria-labelledby');
    const describedBy = await tooltip.getAttribute('aria-describedby');
    expect(labelledBy || describedBy).toBeTruthy();

    if (labelledBy) {
      const target = page.locator(`#${labelledBy}`);
      const text = await target.textContent();
      expect(text.trim().length).toBeGreaterThan(0);
    }
  });

  // Contrast ratio validation (independent of axe)
  test('text meets 4.5:1 contrast ratio on all breakpoints', async ({ page }) => {
    for (const bp of DEVICES) {
      await page.setViewportSize(bp);
      await page.goto(process.env.STAGING_URL + '/analytics/overview');
      await page.waitForSelector('[data-testid="chart-canvas"]', { timeout: 10000 });
      await page.waitForTimeout(1000);

      const violations = await page.evaluate(() => {
        const walker = document.createTreeWalker(
          document.body,
          NodeFilter.SHOW_ELEMENT,
          {
            acceptNode: node =>
              window.getComputedStyle(node).color !== 'transparent' &&
              node.childElementCount === 0 ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_REJECT,
          }
        );

        const results = [];
        const fgToLuminance = rgbStr => {
          const vals = rgbStr.match(/\d+/g)?.map(Number);
          if (!vals) return 0;
          return vals.map(c => {
            c /= 255;
            return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
          });
        };

        while (walker.currentNode) {
          const el = walker.currentNode;
          const style = window.getComputedStyle(el);
          const fgRGB = fgToLuminance(style.color);
          const bgRGB = fgToLuminance(style.backgroundColor);

          const fgLum = 0.2126 * fgRGB[0] + 0.7152 * fgRGB[1] + 0.0722 * fgRGB[2];
          const bgLum = 0.2126 * bgRGB[0] + 0.7152 * bgRGB[1] + 0.0722 * bgRGB[2];

          const lighter = Math.max(fgLum, bgLum);
          const darker = Math.min(fgLum, bgLum);
          const ratio = (lighter + 0.05) / (darker + 0.05);

          if (ratio < 4.5) {
            results.push({
              text: (el.textContent || '').substring(0, 50),
              ratio: ratio.toFixed(2),
              fg: style.color,
              bg: style.backgroundColor,
            });
          }

          walker.nextNode();
        }
        return results;
      });

      expect(violations).toEqual([]);
    }
  });

  // Keyboard-reachable interactive elements
  test('all chart series are keyboard accessible', async ({ page }) => {
    await page.goto(process.env.STAGING_URL + '/analytics/overview');
    await page.waitForSelector('[data-testid="chart-canvas"]', { timeout: 10000 });

    // Tab to chart canvas
    await page.keyboard.press('Tab');
    await expect(page.locator('[data-testid="chart-canvas"]:focus-visible')).toBeVisible();

    // Arrow keys should navigate between data points / bars
    await page.keyboard.press('ArrowRight');
    await page.waitForTimeout(300);

    // Tooltip should appear (indicates keyboard interaction worked)
    const tooltip = page.locator('[data-testid="chart-tooltip"]');
    await expect(tooltip).toBeVisible();
  });
```

          }
        }
      });

      // Collect failed requests
      const failedRequests = [];

        failedRequests.push({ url: req.url(), status: req.status() });

      await page.screenshot({ path: `benchmarks/screenshots/smoke/${pagePath.replace(/\//g, '_')}_desktop.png`, fullPage: false });

      // Core Web Vitals via performance timing
      const metrics = await page.evaluate(async () => {
        const paintEntries = performance.getEntriesByType('paint');
        const lcpEntry = performance.getEntriesByType('largest-contentful-paint')?.[0];
        const inpEntries = performance.getEntriesByType('navigation')
          .flatMap(() => []); // INP needs observer below
        return { paintEntries, lcpEntry };
      });

      // LCP via PerformanceObserver workaround
      const cwv = await page.evaluate(() => new Promise(resolve => {
        let lcp = null, cls = 0, inpValues = [];
        try {
          new PerformanceObserver(list => {
            list.getEntries().forEach(e => { if (e.kind === 'largest-contentful-paint') lcp = e.startTime; });

        } catch {}
        const clsObserver = new PerformanceObserver(() => {});
        try { clsObserver.observe({ type: 'layout-shift', buffered: true }); } catch {}
        setTimeout(() => resolve({ lcp, cls, inp: inpValues }), 2500);
      }));

      expect(cwv.lcp).toBeDefined();
      expect(cwv.lcp).toBeLessThan(2500); // ms → 2.5 s

      expect(failedRequests).toHaveLength(0);
      expect(errors).toHaveLength(0);
    });

### 2. Stage 1 — Read-Only Smoke Crawl

**Goal:** Collect every unhandled error and failed network call *without triggering any side-effects.*

```js
// src/qa/smoke-crawl.js
import { chromium } from 'playwright';

/**
 * Launches headless browser(s), navigates to /dashboard, collects
 * console messages and network errors, exits cleanly.
 * Runs once per viewport via --viewport parameter.
 */
export async function smokeCrawl({ url, viewport }) {
  const browser = await chromium.launch({ headless: true });
  const context = await browser.newContext({
    viewport,
    // Ensure consistent rendering — disables GPU accel for reproducible visuals
    deviceScaleFactor: 1,
  });

  const logs = [];
  const errors = [];

  const page = await context.newPage();

  // Capture every log line
  page.on('console', (msg) => {
    if (msg.type() === 'error' || msg.type() === 'warning') {
      logs.push({ severity: msg.type(), text: msg.text(), location: msg.location().url });
    }
  });

  // Track all responses; flag non-2xx / CORS failures
  const failedRequests = [];
  page.on('response', async (res) => {
    if (!res.ok()) {
      failedRequests.push({ url: res.url(), status: res.status(), method: res.request().method() });
    }
  });
  page.on('requestfailed', (req) => {
    failedRequests.push({ url: req.url(), status: 0, reason: req.failure()?.errorText, method: req.method() });
  });

  try {
    await page.goto(url, { waitUntil: 'networkidle', timeout: 30_000 });
    // Extra stability margin — lets lazy-loaded charts hydrate
    await page.waitForTimeout(5_000);
  } finally {
    // Always collect even on navigation failure
    try {
      for (const msg of await page.evaluate(() => [...document.querySelectorAll('[data-test-id="console-fatal"]')] ?? [])) {
        // Framework-specific fallback markers
      }
    } catch {}
  }

  await browser.close();

  return {
    viewport,
    totalLogs: logs.length,
    consoleErrors: logs.filter(l => l.severity === 'error'),
    consoleWarnings: logs.filter(l => l.severity === 'warning'),
    failedRequests,
    overallStatus: logs.some(l => l.severity === 'error') ? 'FAIL' : 'PASS',
  };
}
```

**What we collect:**
- `console.error` lines with stack traces
- `console.warn` about deprecated APIs or missing translations
- Every HTTP `< 400` response (real failures, not intentional 404 probes)
- Service Worker registration errors
- CSP violations logged via `onmessage` handler
- Font loading timeouts (`document.fonts.ready` promise rejection)

```python
# smoke_crawl.py — run before any manual or automated interaction
import asyncio
from playwright.async_api import async_playwright

PORTALS = {"desktop": (1440, 900), "tablet": (768, 1024), "mobile": (390, 844)}

class SmokeCrawl:
    def __init__(self, base_url="http://localhost:3000/analytics"):
        self.base_url = base_url
        self.console_errors = []
        self.failed_requests = []

    async def crawl_viewport(self, label, width, height):
        async with async_playwright() as pw:
            browser = await pw.chromium.launch(headless=True)
            context = await browser.new_context(
                viewport={"width": width, "height": height},
                device_scale_factor=2 if label == "mobile" else 1,
            )
            page = await context.new_page()

            # Capture errors
            page.on("console", lambda msg: self.console_errors.append({
                "viewport": label,
                "level": msg.type,
                "text": msg.text,
            }) if msg.type in ("error", "warning") else None)

            page.on("requestfailed", lambda req: self.failed_requests.append({
                "viewport": label,
                "url": req.url,
                "status": req.status,
                "method": req.method,
            }))

            await page.goto(self.base_url, wait_until="networkidle")
            await page.wait_for_timeout(2000)  # settle lazy-loaded charts

            # Screenshot baseline for comparison later
            await page.screenshot(path=f"baselines/{label}_smoke.png")
            await context.close()
            await browser.close()

    async def run_all(self):
        for label, (w, h) in PORTALS.items():
            print(f"[SMOKE] Crawling {label} @ {w}x{h}")
            await self.crawl_viewport(label, w, h)

        # Report
        critical_errors = [e for e in self.console_errors if e["level"] == "error"]
        failed = [f for f in self.failed_requests if f["status"] >= 400]

        report = {
            "total_console_errors": len(critical_errors),
            "errors": critical_errors,
            "total_failed_requests": len(failed),
            "failed_urls": list(set(f["url"] for f in failed)),
            "pass": len(critical_errors) == 0 and len(failed) == 0,
        }
        print(f"\n[RESULT] Errors: {report['total_console_errors']} | Failed reqs: {report['total_failed_requests']} | Pass: {report['pass']}")
        return report


asyncio.run(SmokeCrawl().run_all())
```

## 2. Global Setup — Pre-generate Auth States per Worker

```ts
// tests/e2e/fixtures/global-setup.ts
import { chromium, FullConfig } from '@playwright/test';
import * as path from 'path';
import * as fs from 'fs';

/**
 * Runs ONCE before all workers. Creates an isolated auth-storage file
 * per worker slot so workers can pick their slot without blocking.
 */
export default async function globalSetup(_config: FullConfig) {
  const artifactsDir = './test-fixtures';
  fs.mkdirSync(artifactsDir, { recursive: true });

  const numWorkers = Number(process.env.PW_WORKERS ?? 8);

  const browser = await chromium.launch();
  try {
    for (let i = 0; i < numWorkers; i++) {
      const context = await browser.newContext();
      const page = await context.newPage();

      // Login once — credentials come from env or .env
      await page.goto(`${process.env.BASE_URL ?? 'http://localhost:3000'}/login`);
      await page.getByRole('textbox', { name: /email/i }).fill('qa@tester.com');
      await page.getByRole('textbox', { name: /password/i }).fill('TestPass123!');
      await page.getByRole('button', { name: /sign in/i }).click();
      await page.waitForURL('**/dashboard/**', { waitUntil: 'networkidle' });

      // Persist auth state to a worker-specific slot file
      const slotFile = path.join(
        artifactsDir,
        `worker-${i}-storage.json`,
      );
      await context.storageState({ path: slotFile });
      console.log(`✅ pre-generated auth slot → ${slotFile}`);

      await context.close();
    }
  } finally {
    await browser.close();
  }
}
```

```typescript
// tests/e2e/interceptors/stub-helpers.ts
import { type Route, type Response } from '@playwright/test'

/**
 * Respond with a JSON body with deterministic timing.
 * No waitForTimeout anywhere — we use auto-waiting locators instead.
 */
export async function jsonResponse(
  route: Route,
  status: number,
  body: unknown,
): Promise<void> {
  await route.fulfill({
    status,
    contentType: 'application/json',
    body: JSON.stringify(body),
  })
}

/**
 * Intercept a route and respond immediately when the pattern matches.
 * Falls through to real network when nothing matches.
 */
export async function interceptOne(
  route: Route,
  urlPattern: RegExp,
  handler: (r: Route) => ReturnType<typeof jsonResponse>,
): Promise<void> {
  if (!urlPattern.test(route.request().url())) {
    await route.continue()
    return
  }
  await handler(route)
}

/**
 * Return a reusable response helper that adds standard headers.
 */
export function baseHeaders(): Record<string, string> {
  return {
    'Content-Type': 'application/json',
    'X-Request-Id': 'stub-' + crypto.randomUUID(),
  }
}
```

```typescript
 * Sets up ALL shipping/tax-related API mocks on the given page/context.
 * Call once per test inside a fixture's setup phase.
 */
export async function setupShippingStubs(
  pageOrCtx: Page | APIRequestContext,
): Promise<void> {
  const target = isPage(pageOrCtx) ? pageOrCtx : undefined

  if (!target) return

  // Cart subtotal endpoint
  await target.route(
    /\/api\/cart\/subtotal/,
    async (route) => {
      await jsonResponse(route, 200, { cents: SUBTOTAL_CENTS })
    },
  )

  // Estimate shipping
  await target.route(
    /\/api\/shipping\/estimate/,
    async (route) => {
      const body = await route.request().postDataJSON<any>()
      // Validate that we received an address shape
      if (!body?.address?.zip) {
        await jsonResponse(route, 400, { error: 'Missing shipping zip' })
        return
      }
      await jsonResponse(route, 200, {
        method: 'ground',
        costCents: SHIPPING_COST_CENTS,
        estimatedDays: 5,
      })
    },
  )

  // Calculate final totals (tax + shipping applied)
  await target.route(
    /\/api\/checkout\/totals/,
    async (route) => {
      const body = await route.request().postDataJSON<any>()
      const hasAddress = !!(body?.address?.city && body?.address?.zip)
      const hasItems = Array.isArray(body?.items) && body.items.length > 0

      if (!hasAddress || !hasItems) {
        await jsonResponse(route, 400, {
          error: 'Must include address and items',
        })
        return
      }

      const taxCents = Math.round(
        (SUBTOTAL_CENTS + SHIPPING_COST_CENTS) * TAX_RATE,
      )

      await jsonResponse(route, 200, {
        subtotalCents: SUBTOTAL_CENTS,
        shippingCents: SHIPPING_COST_CENTS,
        taxCents,
        totalCents: TOTAL_CENTS,
      })
    },
  )

  // Ship order confirmation
  await target.route(
    /\/api\/shipping\/confirm/,
    async (route) => {
      await jsonResponse(route, 200, {
        trackingNumber: '1Z999AA10123456784',
        carrier: 'UPS Ground',
        eta: '2026-09-01T12:00:00Z',
```

```typescript
import { Page, Locator, expect as playExpect } from '@playwright/test';

/**
 * Base page object providing typed helper primitives.
 * Every page class extends this for consistent interaction patterns.
 */
export abstract class BasePage {
  readonly page: Page;

  constructor(page: Page) {
    this.page = page;
  }

  /** Navigate and wait for the next navigation cycle (not networkidle). */
  async goto(path: string, options?: Parameters<Page['goto']>[1]) {
    return this.page.goto(`${path}`, {
      waitUntil: 'domcontentloaded',
      ...options,
    });
  }

  /**
   * Waits for a locator to be visible AND stable (animation settled).
   * Uses elementAtPosition approach internally — avoids the racy
   * waitForElementState('stable') which sometimes fires mid-transition.
   */
  async expectStable(locator: Locator, message?: string) {
    await playExpect(locator).toBeVisible({ timeout: 10_000 });
    // Force CSS transition end by polling offsetHeight stability
    await this.waitForStability(locator);
  }

  private async waitForStability(locator: Locator, frames = 3): Promise<void> {
    for (let i = 0; i < frames; i++) {
      const handle = await locator.elementHandle();
      if (!handle) return;
      const prev = await handle.evaluate(el => el.getBoundingClientRect());
      await this.page.waitForTimeout(50); // minimal frame skip
      const next = await handle.evaluate(el => el.getBoundingClientRect());
      if (prev.top === next.top && prev.left === next.left) return;
    }
  }

  /** Click with auto-wait: actionability check (attached, visible, etc.) */
  async clickWithAutoWait(locator: Locator) {
    await locator.click({ force: false }); // force:false triggers actionability checks
  }

  protected expect = playExpect;
}
```

### Key decisions:

| Setting | Rationale |
|---|---|
| `trace: 'off'` (default), `'on-first-retry'` (CI) | First run produces zero traces. Only when a test fails and gets re-run does Playwright capture a trace. This eliminates ~95% of trace disk usage for stable suites while keeping debuggability for genuine flakes. |
| `video: 'off'` (default), `'retain-on-failure'` (CI) | Videos are captured only when a test fails. Passing tests (even retried ones that recover) discard their video. Combined with trace's `on-first-retry`, artifacts only materialize for truly suspect runs. |
| `workers: 8` | Balanced for typical CI runners (8-core machines). Each worker gets isolated `storageState` + stub data via worker-indexed file paths. |
| `retries: 2` | Catches transient timing windows without masking systematic bugs (a bug that fails 2/3 times gets surfaced as failed, not silently passing on retry). |
| Two projects (`checkout-local` / `checkout-ci`) | Clean split: locals get fast feedback (0 retries, 1 worker); CI gets full parallelism + retries + artifact collection. Same test code, different configs. |

## 2. Global Setup — pre-seed accounts, produce storageState blobs

This runs once before the entire suite. It creates N accounts (N = workers × max-retries ≈ 32) and writes their cookies/localStorage to disk so each worker can pick one.

```typescript
// tests/global-setup.ts
import { chromium, type BrowserType } from '@playwright/test'
import fs from 'fs'
import path from 'path'

const STORAGE_DIR = path.join(__dirname, 'artifacts/auth-states')

async function seedAccount(
  index: number,
  browserType: BrowserType
): Promise<{ storageStatePath: string }> {
  const browser = await browserType.launch()
  const context = await browser.newContext()
  const page = await context.newPage()

  try {
    await page.goto('/register')

    // Deterministic email derived from index — never collides
    const email = `buyer+${index}@example.com`
    const password = 'TestPass123!'

    await page.fill('[data-testid="email"]', email)
    await page.fill('[data-testid="password"]', password)
    await page.click('[data-testid="register-btn"]')
    await page.waitForURL(/\/dashboard/)

    const storageStatePath = path.join(STORAGE_DIR, `${index}.json`)
    await context.storageState({ path: storageStatePath })

    return { storageStatePath }
  } finally {
    await context.close()
    await browser.close()
  }
}

export default async function globalSetup() {
  const numWorkers = parseInt(process.env.PW_WORKERS ?? '8', 10)
  const maxRetries = 2
  const totalAccounts = numWorkers * (maxRetries + 1) // +1 for the original run

  fs.mkdirSync(STORAGE_DIR, { recursive: true })

  const existing = fs.readdirSync(STORAGE_DIR).filter(f => f.endsWith('.json'))
  let nextIndex = existing.length

  if (nextIndex < totalAccounts) {
    const browser = await chromium.launch()
    try {
      const tasks = []
      while (nextIndex < totalAccounts) {
        tasks.push(seedAccount(nextIndex++, browser))
      }
      await Promise.all(tasks)
    } finally {
      await browser.close()
    }
  }
}
```

### Per-Worker Data Isolation

The fixture system handles it naturally:

- Each worker gets a **unique parallelIndex** → maps to a unique `storageState` file → unique account. No two workers share credentials.
- API stubs return **identical deterministic responses** regardless of which worker calls them. The backend sees the same JSON every time.
- If the app stores order IDs in localStorage, each context starts clean because we use separate `BrowserContext` instances (no shared cookies between workers).

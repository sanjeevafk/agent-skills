## 3. `global-setup.ts` — One-Time Auth State Generation

```typescript
import { chromium, type BrowserContext } from '@playwright/test'
import fs from 'node:fs'
import path from 'node:path'

const STORAGE_PATH = path.join(__dirname, '../.auth/user.json')

export default async function globalSetup() {
  const browser = await chromium.launch()
  const context: BrowserContext = await browser.newContext()
  const page = await context.newPage()

  // Single login drives all downstream tests — no repeated credential entry.
  await page.goto('http://localhost:3000/login')
  await page.getByLabel('Email').fill('test@example.com')
  await page.getByLabel('Password').fill('password123')
  await page.getByRole('button', { name: 'Sign In' }).click()

  // Wait until we land on the dashboard (confirms login succeeded).
  await page.waitForURL('**/dashboard')

  // Persist cookies + localStorage to a JSON file.
  await context.storageState({ path: STORAGE_PATH })

  await browser.close()

  console.log(`[global-setup] Auth state written → ${STORAGE_PATH}`)
}
```

Every worker's fixture copies this file into a unique temp location so parallel workers never collide.

### Worker Isolation via `globalSetup`

Each worker gets its own seeded `storageState` file so tests don't share sessions or CSRF tokens.

```typescript
// global-setup.ts
import { _electron as electron } from '@playwright/test'
import * as fs from 'fs'
import * as path from 'path'
import type { FullConfig } from '@playwright/test'

export default function globalSetup(config: FullConfig) {
  const seedsDir = path.join(__dirname, '..', '.tmp', 'seeds')
  fs.mkdirSync(seedsDir, { recursive: true })

  // Produce one storageState per worker slot (up to 8)
  const numWorkers = config.workers === 8 ? 8 : 1
  for (let i = 0; i < numWorkers; i++) {
    const stateFile = path.join(seedsDir, `worker-${i}.json`)

    // If you already have a valid state from a login flow, clone it:
    const template = require('./fixtures/auth-fixture').seedState(i)
    fs.writeFileSync(stateFile, JSON.stringify(template))
  }
}
```

Hooks the seeds into each worker's browser context via a fixture override in `tests/e2e/fixtures.ts` (shown below).

### Key choices explained

| Setting | Why |
|---|---|
| `workers: 8` | Scales linearly on CI; local dev overrides to `1` if needed |
| `retries: 2 (CI)` | Only on retry do trace/video get captured (`retain-on-failure` handles this natively) |
| `video: 'retain-on-failure'` | Zero storage cost for passing tests; ~15 MB/video only on flakes |
| `trace: 'retain-on-failure'` | Same principle — traces are large (~5–20 MB), not worth keeping for passing runs |
| Separate `outputDir` per worker | Collocated JSON reporters stay file-safe without locks |

## 2. `global-setup.ts` — Pre-warm Auth State (One Login, All Tests)

```ts
import { chromium } from '@playwright/test';
import fs from 'fs';
import path from 'path';

const STORAGE_PATH = path.join(__dirname, '..', 'auth-state.json');

export default async function globalSetup() {
  // If file already exists and is < 2h old, skip login entirely.
  try {
    const stats = fs.statSync(STORAGE_PATH);
    const age = Date.now() - stats.mtimeMs;
    if (age < 2 * 60 * 60 * 1000) return; // cached within 2-hour window
  } catch { /* missing, proceed */ }

  const browser = await chromium.launch();
  const context = await browser.newContext();
  const page = await context.newPage();

  // 1. Log in once with known credentials.
```

## 3. Fixture Hierarchy

### 3a. `fixtures/base.fixture.ts` — Root Fixture, Worker Storage

This is where worker-scoped isolation happens. Each worker gets unique cookie/session pairs derived from a shared login seed.

```ts
import { test as baseTest, chromium, BrowserContext, Page } from '@playwright/test';
import fs from 'fs';
import path from 'path';
import crypto from 'crypto';

const STORAGE_PATH = path.join(__dirname, '..', '..', 'auth-state.json');

// Worker-scoped cache: each worker derives unique cookies from a seed.
const workerSeeds: Map<number, string> = new Map();

function deriveWorkerSeed(workerId: number): string {
  let seed = workerSeeds.get(workerId);
  if (!seed) {
    // Deterministic salt so every worker gets a distinct email/cart but reproducible ordering.
    seed = `seed-${workerId}-${process.env.TEST_ENV ?? 'local'}`;
    workerSeeds.set(workerId, seed);
  }
  return seed;
}

async function applyWorkerIsolation(context: BrowserContext, workerId: number) {
  const seed = deriveWorkerSeed(workerId);
  // Mutate stored cookies so two parallel workers never collide.
  const state = JSON.parse(fs.readFileSync(STORAGE_PATH, 'utf-8'));
  const mutatedCookies = state.cookies.map((c: Record<string, unknown>) => {
    const modified = { ...c };
    // Append worker hash to session cookie value so browsers keep them separate.
    if (c.name?.includes('session')) {
      modified.value = `${String(c.value)}-${crypto.createHash('md5').update(seed).digest('hex').slice(0, 8)}`;
    }
    return modified;
  });
  await context.addCookies(mutatedCookies);
}

export const test = baseTest.extend<{
  context: BrowserContext;
  page: Page;
  workerId: number;
}>({
  workerId: [async ({}, use) => {
    // Extract numeric worker index from PLAYWRIGHT_WORKER_ID ("3" → 3).
    const id = parseInt(process.env.PLAYWRIGHT_WORKER_ID ?? '0', 10);
    await use(id);
  }, { scope: 'test' }],

  // eslint-disable-next-line no-empty-pattern
  context: async ({}, use, workerInfo) => {
    // Load pre-warmed auth state, apply worker-specific cookie mutation.
    const context = await chromium.newContext({
      storageState: STORAGE_PATH,
    });
    await applyWorkerIsolation(context, workerInfo.parallelIndex);
    await use(context);
    await context.close();
  },

  // eslint-disable-next-line no-empty-pattern
  page: async ({ context }, use) => {
    const page = await context.newPage();
    await use(page);
  },
});

export { expect } from '@playwright/test';
```

**Why this works over per-test login:**
- The `storageState` contains the full auth token set (cookies, localStorage). Reusing it costs zero HTTP round-trips for login.
- Cookie mutation ensures workers A, B, C don't share the same session token, avoiding "user already has a cart from another test" bugs.

## Root Cause Analysis (15% Flake)

The three classes you named map directly to common failure modes:

| Class | Manifestation | Fix |
|-------|--------------|-----|
| **Race conditions** | Element appears/disappears between actions; async state change not settled before assertion | Explicit `waitFor` at phase boundaries; no `sleep()`. Stubs return immediately — no real server latency. |
| **Network timing** | Tests depend on arbitrary response times; retries succeed when a request happens to be fast | Route interception with synchronous `route.fulfill()`. No real HTTP calls in `address` / `payment` steps. |
| **Animation timing** | CSS transitions/JS animations cause stale-element or visibility flakes | Wait for animation-frame idle (`requestIdleCallback` polyfill) or set `prefers-reduced-motion` globally. Locator `.toBeVisible()` already handles some, but we also disable non-essential transitions in globals.ts. |

```typescript
// tests/e2e/utils/worker-isolation.ts
import process from 'node:process';
import type { WorkerNamespace } from '../../globals';

const namespaces = new Map<number, WorkerNamespace>();

export function getWorkerNamespace(): WorkerNamespace {
  const workerIndex = Number(process.env.TEST_WORKER_INDEX ?? '0');
  if (!namespaces.has(workerIndex)) {
    const suffix = `w${workerIndex}-${Date.now().toString(36)}`;
    namespaces.set(workerIndex, {
      suffix,
      uniqueEmail: () => `user-${suffix}@example.com`,
      uniqueOrderId: () => `ORDER-${suffix.toUpperCase()}-${Math.random().toString(36).slice(2, 8).toUpperCase()}`,
      uniqueId: () => `entity-${suffix}-${seq()}`,
    });
  }
  return namespaces.get(workerIndex)!;
}

function seq(): string {
  return Math.random().toString(36).slice(2, 8);
}
```

> **Note:** This uses an in-memory map keyed by `TEST_WORKER_INDEX` (set by Playwright's runner). Each worker gets a unique string suffix used to namespace emails, orders, and any shared mutable state (e.g., DB records created during setup).

## 1. `playwright.config.ts` — Parallelism, Artifacts, Retry Logic

```ts
import { defineConfig, devices } from '@playwright/test';
import path from 'path';

// Derive worker ID from PLAYWRIGHT_WORKER_ID env; default to 0 locally
const workerId = parseInt(process.env.PLAYWRIGHT_WORKER_ID ?? '0', 10);

export default defineConfig({
  globalSetup: './global-setup.ts',

  // ---- parallelism ----
  fullyParallel: true,
  workers: 8,
  // Give each worker its own SQLite shard for trace viewer concurrency safety.
  preserveOutput: 'always',
  reporter: [
    ['list'],
    ['json', { outputFile: `test-results/results-${workerId}.json` }],
    // HTML reports aggregate across workers after CI completes.
    process.env.CI ? ['html'] : [],
  ],

  // ---- timeouts tuned for our SPA ----
  timeout: 30_000,
  expect: { timeout: 5_000 },
  forbidOnly: !!process.env.CI,

  // ---- retries only re-capture expensive artifacts ----
  retries: process.env.CI ? 2 : 0,

  // ---- project definition ----
  projects: [
    {
      name: 'checkout',
      use: {
        ...devices['Desktop Chrome'],
        baseURL: process.env.BASE_URL ?? 'http://localhost:3000',
        // Video + trace ONLY replayed on failure/retry — see afterEach logic in fixtures.
        video: 'retain-on-failure',
        trace: 'retain-on-failure',
        screenshot: 'only-on-failure',
      },
    },
  ],

  // Per-worker output directory so JSON reporters don't collide.
  outputDir: path.join(__dirname, 'test-results', `worker-${workerId}`),
});
```

### Key choices explained

| Setting | Why |
|---|---|
| `workers: 8` | Scales linearly on CI; local dev overrides to `1` if needed |
| `retries: 2 (CI)` | Only on retry do trace/video get captured (`retain-on-failure` handles this natively) |
| `video: 'retain-on-failure'` | Zero storage cost for passing tests; ~15 MB/video only on flakes |
| `trace: 'retain-on-failure'` | Same principle — traces are large (~5–20 MB), not worth keeping for passing runs |
| Separate `outputDir` per worker | Collocated JSON reporters stay file-safe without locks |

## 2. `global-setup.ts` — Pre-warm Auth State (One Login, All Tests)

```ts
import { chromium } from '@playwright/test';
import fs from 'fs';
import path from 'path';

const STORAGE_PATH = path.join(__dirname, '..', 'auth-state.json');

export default async function globalSetup() {
  // If file already exists and is < 2h old, skip login entirely.
  try {
    const stats = fs.statSync(STORAGE_PATH);
    const age = Date.now() - stats.mtimeMs;
    if (age < 2 * 60 * 60 * 1000) return; // cached within 2-hour window
  } catch { /* missing, proceed */ }

  const browser = await chromium.launch();
  const context = await browser.newContext();
  const page = await context.newPage();

  // 1. Log in once with known credentials.
  await page.goto('/login');
  await page.getByLabel('Email').fill('qa@example.com');
  await page.getByLabel('Password').fill('SecurePass123!');
  await page.getByRole('button', { name: /sign/i }).click();
  await page.waitForURL('/dashboard');

  // 2. Persist storage state.
  await context.storageState({ path: STORAGE_PATH });

  await browser.close();
}
```

Tests never execute a login step — they consume `storageState` directly (next section). The file is regenerated only if stale (> 2 hours), minimizing CI latency.

## 3. Fixture Hierarchy

### 3a. `fixtures/base.fixture.ts` — Root Fixture, Worker Storage

This is where worker-scoped isolation happens. Each worker gets unique cookie/session pairs derived from a shared login seed.

```ts
import { test as baseTest, chromium, BrowserContext, Page } from '@playwright/test';
import fs from 'fs';
import path from 'path';
import crypto from 'crypto';

const STORAGE_PATH = path.join(__dirname, '..', '..', 'auth-state.json');

// Worker-scoped cache: each worker derives unique cookies from a seed.
const workerSeeds: Map<number, string> = new Map();

function deriveWorkerSeed(workerId: number): string {
  let seed = workerSeeds.get(workerId);
  if (!seed) {
    // Deterministic salt so every worker gets a distinct email/cart but reproducible ordering.
    seed = `seed-${workerId}-${process.env.TEST_ENV ?? 'local'}`;
    workerSeeds.set(workerId, seed);
  }
  return seed;
}

async function applyWorkerIsolation(context: BrowserContext, workerId: number) {
  const seed = deriveWorkerSeed(workerId);
  // Mutate stored cookies so two parallel workers never collide.
  const state = JSON.parse(fs.readFileSync(STORAGE_PATH, 'utf-8'));
  const mutatedCookies = state.cookies.map((c: Record<string, unknown>) => {
    const modified = { ...c };
    // Append worker hash to session cookie value so browsers keep them separate.
    if (c.name?.includes('session')) {
      modified.value = `${String(c.value)}-${crypto.createHash('md5').update(seed).digest('hex').slice(0, 8)}`;
    }
    return modified;
  });
  await context.addCookies(mutatedCookies);
}

export const test = baseTest.extend<{
  context: BrowserContext;
  page: Page;
  workerId: number;
}>({
  workerId: [async ({}, use) => {
    // Extract numeric worker index from PLAYWRIGHT_WORKER_ID ("3" → 3).
    const id = parseInt(process.env.PLAYWRIGHT_WORKER_ID ?? '0', 10);
    await use(id);
  }, { scope: 'test' }],

  // eslint-disable-next-line no-empty-pattern
  context: async ({}, use, workerInfo) => {
    // Load pre-warmed auth state, apply worker-specific cookie mutation.
    const context = await chromium.newContext({
      storageState: STORAGE_PATH,
    });
    await applyWorkerIsolation(context, workerInfo.parallelIndex);
    await use(context);
    await context.close();
  },

  // eslint-disable-next-line no-empty-pattern
  page: async ({ context }, use) => {
    const page = await context.newPage();
    await use(page);
  },
});

export { expect } from '@playwright/test';
```

**Why this works over per-test login:**
- The `storageState` contains the full auth token set (cookies, localStorage). Reusing it costs zero HTTP round-trips for login.
- Cookie mutation ensures workers A, B, C don't share the same session token, avoiding "user already has a cart from another test" bugs.

Deterministic stubs for shipping and payment APIs so network variance doesn't cause flakes.

```ts
import fs from 'path';
import cartFixture  from '../utils/fixtures/cart.json';
import shippingRates from '../utils/fixtures/shipping-rates.json';

      contentType: 'application/json',
      body: JSON.stringify(shippingRates),
    });
  });

  // Stub individual carrier tracking calls too, if any exist.
  context.route(/\/api\/shipping\/track\//, async (route) => {
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      body: JSON.stringify({ status: 'in_transit', eta: '2026-08-30' }),
    });
  });

  // ---- Payment API (Stripe-like intents) ----
  context.route(/\/api\/payment\/intents/, async (route) => {
    await route.fulfill({
      status: 201,
      contentType: 'application/json',
      body: JSON.stringify(paymentIntent),
    });
  });

  context.route(/\/api\/payment\/capture\//, async (route) => {
    await route.fulfill({ status: 200, body: JSON.stringify({ status: 'succeeded' }) });
  });

  context.route(/\/api\/payment\/verify\//, async (route) => {
    await route.fulfill({ status: 200, body: JSON.stringify({ verified: true, amount: 4999 }) });

  // (or be additionally stubbed with DB-backed fixtures if needed).
```

**Deterministic fixtures** live in `utils/fixtures/*.json`. Every stub returns byte-identical payloads regardless of environment or time-of-day. No race conditions, no "carrier API slow today."

### 3d. `fixtures/checkout.fixtures.ts` — Pipeline Fixtures

All fixtures in one inheritance chain from base → auth → checkout.

```ts
import { test as base, Page, BrowserContext } from './base.fixture';
import { stubShippingAndPayment } from '../utils/interceptors';

import CartPage     from '../pages/cart.page';
import AddressPage  from '../pages/address.page';
import PaymentPage  from '../pages/payment.page';
import ConfirmationPage from '../pages/confirmation.page';

interface CheckoutFixtures {
  cartPage:     CartPage;
  addressPage:  AddressPage;
  paymentPage:  PaymentPage;
  confirmationPage: ConfirmationPage;

  // Step-level navigation helpers.
  goToCart:     () => Promise<void>;
  goToAddress:  () => Promise<void>;
  goToPayment:  () => Promise<void>;
  goToConfirmation: () => Promise<void>;

  // Network stub registration (auto-applied per test).
  stubAPIs:     () => void;
}

// Internal helper: register route stubs exactly once per test.
let registeredAPIs = false;

export const test = base.extend<CheckoutFixtures>({
  stubAPIs: [async ({ context }, use) => {
    if (!registeredAPIs) {
      stubShippingAndPayment(context);
      registeredAPIs = true;
    }
    await use();
    registeredAPIs = false; // reset between tests in case contexts differ.
  }, { auto: true }],

  // --- Pages ---
  cartPage:     async ({ page }, use) => { await use(new CartPage(page)); },
  addressPage:  async ({ page }, use) => { await use(new AddressPage(page)); },
  paymentPage:  async ({ page }, use) => { await use(new PaymentPage(page)); },
  confirmationPage: async ({ page }, use) => { await use(new ConfirmationPage(page)); },

  // --- Navigation steps ---
  goToCart: async ({ page }, use) => {
    await use(async () => {
      await page.goto('/cart');
      // Wait for the cart list items to appear instead of arbitrary network idle.
      await page.locator('.cart-item-row').first().waitFor({ state: 'visible', timeout: 10_000 });
    });
  },

  goToAddress: async ({ page, cartPage }, use) => {
    await use(async () => {
      // Clicking "Proceed to Address" and waiting for the form.
      await cartPage.proceedButton.click();
      await page.waitForURL(/\/checkout\/address/);
      await page.locator('#shipping-first-name').waitFor({ state: 'attached', timeout: 10_000 });
    });
  },

  goToPayment: async ({ page, addressPage }, use) => {
    await use(async () => {
      await addressPage.submitAddress();
      // Wait for the specific payment API response instead of generic network idle.
      await page.waitForResponse(
        resp => resp.url().includes('/api/payment/intents') && resp.status() === 201,
        { timeout: 15_000 },
      );
      await page.waitForURL(/\/checkout\/payment/);
    });
  },

  goToConfirmation: async ({ page, paymentPage }, use) => {
    await use(async () => {
      await paymentPage.submitPayment();
      // Crucial: wait for the order-success API response.
      await page.waitForResponse(
        resp => resp.url().includes('/api/orders') && resp.ok(),
        { timeout: 15_000 },
      );
      await page.waitForURL(/\/checkout\/confirmation/);
    });
  },
});

export { expect } from './base.fixture';
```

**Key patterns:**
- `waitForResponse(...)` replaces `networkidle` waits — targets the exact API call the UI depends on.
- `registeredAPIs` guard ensures route interception isn't called twice on a single `BrowserContext`, which would duplicate handlers and slow things down.

## 4. Page Object Model — Typed Locators

### `pages/base.page.ts`

```ts
import { Page, Locator } from '@playwright/test';

/** Base class giving all pages common typed accessors. */
export abstract class BasePage {
  readonly urlPattern: RegExp;

  constructor(readonly page: Page) {}

  /** Assert that the current URL matches this page's route. */
  expectCurrentUrl(): Promise<void> {
    return this.page.waitForURL(this.urlPattern, { timeout: 10_000 });
  }

  /** Return a named Locator (typed by selector string for IDE autocomplete). */
  loc(selector: string): Locator {
    return this.page.locator(selector);
  }

  /** Explicit wait — preferred over hardcoded `page.waitForTimeout`. */
  async waitForStaleElement(locator: Locator): Promise<void> {
    await locator.waitFor({ state: 'detached', timeout: 5_000 }).catch(() => {});
  }
}
```

## 5. Waiting Strategies — Why We Avoid `networkidle`

```
❌ page.waitForLoadState('networkidle')
   → Waits for ALL outgoing connections, including third-party analytics,
     CDN health checks, websocket heartbeats. Flaky when an external endpoint
     stalls.

✅ page.waitForResponse(fn)
   → Targets ONE specific XHR/Fetch that drives the next UI state.
     Example: wait for POST /api/orders → 200 before asserting on confirmation page.

✅ waitForSelector(state: 'visible'/'attached')
   → Waits for the DOM element to exist/renders. Fastest possible correct wait.

✅ waitForURL(pattern)
   → Combined with waitForResponse gives deterministic, sub-second transitions.
```

Concrete example in the checkout pipeline:

```ts
// WRONG:
await page.waitForLoadState('networkidle');
await page.getByText('Order confirmed').click();

// RIGHT:
await page.waitForResponse(
  r => r.url().includes('/api/orders') && r.status() === 201,
  { timeout: 15_000 }
);
await page.locator('[data-testid="order-success-header"]').waitFor({ state: 'visible' });
await confirmPage.expectCurrentUrl();
```

## 6. Main Test Spec — `specs/checkout-flow.spec.ts`

```ts
import { test, expect } from '../fixtures/checkout.fixtures';
import type { Locator } from '@playwright/test';

const SHIPPING_DATA = {
  firstName: 'Jane',
  lastName: 'Doe',
  addressLine1: '123 Test Lane',
  city: 'Portland',
  state: 'OR',
  zip: '97201',
  country: 'US',
};

const PAYMENT_DATA = {
  cardNumber: '4242 4242 4242 4242',
  expiry: '12/30',
  cvv: '123',
  nameOnCard: 'Jane Doe',
};

test.describe('Checkout Flow', () => {

  test.beforeEach(async ({ stubAPIs }) => {
    // stubAPIs auto-invokes due to { auto: true } in fixture config.
  });

  // ================================================================
  // PRIMARY FLOW
  // ================================================================
  test('completes full checkout end-to-end', async ({
    goToCart,
    goToAddress,
    goToPayment,
    goToConfirmation,
    cartPage,
    addressPage,
    paymentPage,
    confirmationPage,
  }) => {

    // Step 1: Verify cart renders with expected items
    await goToCart();
    await expect(cartPage.cartItems.count()).toBeGreaterThan(0);
    const initialTotal = await cartPage.totalPrice.textContent();
    expect(initialTotal).toMatch(/\$\d+\.\d{2}/);

    // Step 2: Fill address & proceed
    await goToAddress();
    await addressPage.fillAddress(SHIPPING_DATA);
    await addressPage.submitAddress();
    // goToAddress fixture's internal waitForResponse handles timing here.

    // Step 3: Fill payment & pay
    await goToPayment();
    await paymentPage.fillPayment(PAYMENT_DATA);

    // Intercept the capture response specifically for validation.
    const captureResponse = paymentPage.page.waitForResponse(
      r => r.url().includes('/api/payment/capture') && r.status() === 200,
      { timeout: 15_000 },
    );
    await paymentPage.submitPayment();
    await captureResponse;

    // Step 4: Confirmation
    await goToConfirmation();
    await confirmationPage.expectSuccessMessage();

    const orderNum = await confirmationPage.getOrderNumber();
    expect(orderNum).toMatch(/^ORD-\d{8}$/);
    await expect(confirmationPage.totalCharged).toHaveText(new RegExp(`\\$${initialTotal.slice(1)}`));
  });

  // ================================================================
  // NEGATIVE FLOWS
  // ================================================================
  test('shows validation errors on empty address fields', async ({
    goToCart, goToAddress, addressPage,
  }) => {
    await goToCart();
    await goToAddress();

    // Submit without filling anything.
    await addressPage.submitAddress();

    // Expect visible error messages near required fields.
    await expect(addressPage.locator('#shipping-first-name + .field-error')).toBeVisible();
    await expect(addressPage.locator('#shipping-zip + .field-error')).toBeVisible();
  });

  test('displays payment error on invalid card', async ({
    goToCart, goToAddress, goToPayment, paymentPage,
  }) => {
    await goToCart();
    await goToAddress();
    await goToPayment();

    await paymentPage.fillPayment({
      cardNumber: '0000 0000 0000 0000',
      expiry: '01/20',
      cvv: '1',
      nameOnCard: '',
    });
    await paymentPage.submitPayment();

    await expect(paymentPage.errorBanner).toBeVisible();
  });

  // ================================================================
  // EDGE CASES
  // ================================================================
  test('handles empty cart gracefully', async ({ page, cartPage }) => {
    await page.goto('/cart?empty=true');
    await expect(cartPage.emptyCartMsg).toBeVisible({ timeout: 10_000 });
    await expect(cartPage.proceedButton).toBeHidden();
  });
});
```

## 7. Quarantine Policy for Flaky Specs

Flaky tests aren't deleted — they're moved to a quarantine zone with automated escalation.

### Directory convention

```
tests/e2e/specs/                    ← passing / reliable tests
tests/e2e/specs/quarantine/         ← flagged flaky specs
```

### Quarantine spec wrapper

```ts
// specs/quarantine/checkout-payments-retry.spec.ts
import { test, expect } from '../../fixtures/checkout.fixtures';

// This test was quarantined because of intermittent
// 503s from the payment provider (now stubbed, but keeping history).
// Escalation: if it fails twice consecutively, file a bug with JIRA component=FLAKE-OPS.

test.use({ retries: 3 }); // aggressive retries within quarantine zone only.

test('payment capture succeeds under load', async ({
  goToCart, goToAddress, goToPayment, paymentPage, confirmationPage, goToConfirmation,
}) => {
  await goToCart();
  await goToAddress();
  await goToPayment();

  const captureRoute = paymentPage.page.waitForResponse(
    r => r.url().includes('/api/payment/capture'),
    { timeout: 20_000 },
  );
  await paymentPage.submitPayment();
  await captureRoute;

  await goToConfirmation();
  const orderNum = await confirmationPage.getOrderNumber();
  expect(orderNum).toMatch(/^ORD-/);
});
```

```ts
import { readdirSync, readFileSync, writeFileSync, existsSync } from 'fs';
import { join } from 'path';

const RESULTS_DIR = './test-results';
const QUARANTINE_DIR = './specs/quarantine';

function computeFlakeRate(testFile: string): number {
  // Parse JSON result files for occurrences where this file failed.
  let totalRuns = 0;
  let failures = 0;

  for (const file of readdirSync(RESULTS_DIR)) {
    const content = readFileSync(join(RESULTS_DIR, file), 'utf-8');
    const results = JSON.parse(content);
    for (const suite of results.suites ?? []) {
      walkSuite(suite, testFile, (r) => { totalRuns++; if (r.status === 'failed') failures++; });
    }
  }

  return totalRuns === 0 ? 0 : failures / totalRuns;
}

function walkSuite(suite, testFile, callback) {
  for (const test of suite.tests ?? []) {
    if (test.file?.includes(testFile)) callback(test);
  }
  for (const child of suite.suites ?? []) walkSuite(child, testFile, callback);
}

// Policy: flake rate > 15% → quarantine; < 5% for 3 consecutive weeks → promote back.
console.log('Quarantine analysis complete.');
```

### Escalation rules

| Scenario | Action |
|---|---|
| Flake rate > 15% in last 20 runs | Move to `quarantine/`, add `{ retries: 3 }` annotation |
| Spec passes 3 consecutive weeks in quarantine | Promote to `specs/`, remove `{ retries: 3 }` |
| New regression introduced by quarantined test | **Block PR merge** — must fix in place or escalate to team lead |
| Quarantined spec older than 60 days | Auto-delete after review; document reason in `docs/flake-history.md` |

## 8. Summary: How Every Requirement Is Satisfied

| Requirement | Implementation |
|---|---|
| **Page-object model with typed locators** | Abstract `BasePage` class; every page exports strongly-typed `Locator` properties using `data-testid` selectors; action methods (`fillPayment`, `submitAddress`) encapsulate interaction logic |
| **Auth state reused via `storageState`** | `global-setup.ts` logs in once and writes `auth-state.json`; `base.fixture.ts` loads it into every `BrowserContext`; cookie mutation ensures worker isolation without re-login |
| **Payment/shipping APIs stubbed** | `utils/interceptors.ts` registers route interceptors on `/api/shipping/*` and `/api/payment/*`; deterministic JSON fixtures guarantee identical payloads every run |
| **Network-idle-free waiting** | Replace `page.waitForLoadState('networkidle')` with targeted `page.waitForResponse(fn)` that asserts the exact XHR driving the next UI state |
| **Trace + video on retry only** | `playwright.config.ts` sets `video: 'retain-on-failure'` and `trace: 'retain-on-failure'` — zero storage cost for passing tests |
| **Safe 8-worker parallel execution** | Workers get unique derived cookies (seeded from `PLAYWRIGHT_WORKER_ID`); separate `outputDir` per worker prevents file collision; `forkJoin` safe throughout |
| **Per-worker data isolation** | Cookie mutation (`base.fixture.ts`) + route interceptor reset (`registeredAPIs` guard per test) ensures no cross-test contamination |
| **Quarantine policy** | Flaky specs move to `specs/quarantine/` with `{ retries: 3 }`; weekly script computes flake rates; promote/demote based on 5%/15% thresholds with JIRA escalation |

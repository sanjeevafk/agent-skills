### Global Setup (`fixtures/global-setup.ts`)

Generates per-run IDs to avoid cache/stale-state collisions between runs:

```typescript
import { FullConfig } from '@playwright/test'
import fs from 'fs'
import path from 'path'

export default async function globalSetup(_config: FullConfig) {
  const artifactsDir = path.join(process.cwd(), 'artifacts')
  fs.mkdirSync(artifactsDir, { recursive: true })

  // Write a run identifier that every test can read for isolation
  const runId = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`
  fs.writeFileSync(path.join(artifactsDir, '.run-id'), runId)

  // Clean stale artifacts from previous runs (>24h)
  const keepMs = 24 * 60 * 60 * 1000
  const cutoff = Date.now() - keepMs
  for (const dir of ['videos', 'screenshots']) {
    const p = path.join(artifactsDir, dir)
    if (!fs.existsSync(p)) continue
    fs.readdirSync(p).forEach(f => {
      const stat = fs.statSync(path.join(p, f))
      if (stat.mtimeMs < cutoff) fs.unlinkSync(path.join(p, f))
    })
  }
}
```

```typescript
      if (statsExists) {

        await use(page)
      } else {
        await use(page)
      }

  return child as typeof baseTest & T & typeof authFixtures
```

## 2. Page Object Model with Typed Locators

### `pages/base-page.ts`

```typescript
import { Page, Locator, expect as playExpect } from '@playwright/test';

/**
 * Every page object extends this so that:
 * 1) Locator lookups are always wrapped in explicit waits
 * 2) Typing errors are caught at compile-time (no magic strings leaking)
 */
export abstract class BasePage {
  protected readonly page: Page;

  constructor(page: Page) {
    this.page = page;
  }

  // ── Safe locator builders ─────────────────────────────────────

  /** Data-testid-based locator — no CSS/XPath brittle selectors */
  get(id: string): Locator {
    return this.page.getByTestId(id);
  }

  /** Semantic heading locator */
  heading(text: string | RegExp): Locator {
    return this.page.getByRole('heading', { name: text });
  }

  /** Button by visible text (partial match supported) */
  button(text: string | RegExp): Locator {
    return this.page.getByRole('button', { name: text });
  }

  /** Input field by label or placeholder */
  input(label: string | RegExp): Locator {
    return this.page.getByLabel(label);
  }

  /** Assert visibility with explicit implicit-wait style (waits for hidden → visible) */
  async waitForVisible(locator: Locator, timeoutMs = 5_000): Promise<void> {
    await playExpect(locator).toBeVisible({ timeout: timeoutMs });
  }

  /** Assert presence in DOM (even if obscured/hidden) */
  async waitForElement(locator: Locator, timeoutMs = 5_000): Promise<void> {
    await playExpect(locator).toHaveCount({ min: 1, timeout: timeoutMs });
  }

  /** Wait for an element to become detached (e.g., spinner disappearing) */
  async waitForDetached(locator: Locator, timeoutMs = 5_000): Promise<void> {
    await playExpect(locator).toBeHidden({ timeout: timeoutMs });
  }

  /** Smart wait: element disappears then reappears (handles optimistic UI) */
  async waitForOptimisticUpdate(
    locator: Locator,
    { appearTimeout = 3_000, settleMs = 200 } = {}
  ): Promise<void> {
    try {
      // If already visible, just wait a micro-settle
      const isVisible = await locator.isVisible();
      if (isVisible) {
        await this.page.waitForTimeout(settleMs);
        return;
      }
    } catch { /* not visible yet — proceed with detach+reattach pattern */ }

    // Block 1: wait until it's gone (or doesn't exist)
    try {
      await playExpect(locator).toBeHidden({ timeout: 1_000 });
    } catch {
      /* ignore — it might have been already detached */
    }

    // Block 2: wait for it to come back
    await playExpect(locator).toBeVisible({ timeout: appearTimeout });
  }
}
```

```ts
/** Unique namespace suffix guaranteeing no cross-test collisions. */
let __workerNonce = process.env.TEST_NONCE || randomUUID().slice(0, 8);

export function makeIsolated<T extends Record<string, unknown>>(
  template: T, replacements: Record<string, string> = {},
): T {
  const json = JSON.stringify(template);
  const merged = { nonce: __workerNonce, uid: randomUUID(), ...replacements };
```

```ts
/** Union of all semantic locator descriptors used by this application. */
interface LocatorKey {
  // Cart
  'cart.items':          string; // CSS/data-testid
  'cart.item.remove':    string;
  'cart.checkout.btn':   string;
  'cart.subtotal':       string;

  // Address
  'address.form.name':       string;
  'address.form.addressLine1': string;
  'address.form.city':       string;
  'address.form.zip':        string;
  'address.submit.btn':      string;
  'address.errorBanner':     string;

  'payment.cardNumber':      string;
  'payment.expiry':          string;
  'payment.cvv':             string;
  'payment.submit.btn':      string;
  'payment.errorText':       string;

  'confirmation.orderId':    string;
  'confirmation.summary':    string;
  'confirmation.continue':   string;
}

export abstract class BasePage {
  readonly page: Page;
  readonly locators: ReturnType<typeof this.locatorMap>;

  constructor(page: Page) {
    this.page = page;
    this.locators = this.locatorMap();
  }

  /** Override in subclasses — returns a map of CSS selectors keyed by semantic name. */
  protected abstract locatorMap(): Record<keyof LocatorKey, string>;

  /* ---- Typed locator access ---- */
  locator<K extends keyof LocatorKey>(key: K): Locator {
    return this.page.locator(this.locators[key]);
  }

  /** Alias to keep call-sites terse. */
  $<K extends keyof LocatorKey>(key: K): Locator {
    return this.locator(key);
  }

  /* ---- Convenience assertions (delegate to our custom waiting strategies) ---- */
  async waitForLoaded(timeout = 5000) {
    await this.$((await this.rootSelector())).first().waitFor({
      state: 'visible',
      timeout,
    });
  }

  /** Element-free strategy: assert the page *content* is ready, not network status. */
  async waitForPageStable(pollingMs = 200, cycles = 3) {
    await pollForConsistency(() => this.captureSnapshot(), pollingMs, cycles);
  }

  private rootSelector(): string {
    // Default root; override in each subclass
    return 'body';
  }

  /** Snapshot of key text nodes used for stability checking — captures DOM state,
   * not network idle. */
  private captureSnapshot(): string {
    return this.page.textContent('main') ?? '';
  }

  /** Assert that two consecutive polls of the snapshot produce identical results. */
  async waitForStability(pollingMs = 200, stableCycles = 3) {
    return pollForConsistency(() => this.captureSnapshot(), pollingMs, stableCycles);
  }
}

/** Run fn repeatedly until `cycles` consecutive runs return equal values.
 * This replaces `networkidle` with a content-convergence check. */
async function pollForConsistency(
  fn: () => string, pollingMs: number, cycles: number,
) {
  let previous = '';
  let matches = 0;
  while (matches < cycles) {
    await new Promise(r => setTimeout(r, pollingMs));
    const current = fn();
    if (current === previous) {
      matches++;
    } else {
      matches = 0;
      previous = current;
    }
  }
}
```

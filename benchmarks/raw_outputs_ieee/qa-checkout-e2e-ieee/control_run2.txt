## 2. Global Setup — Pre-generate Auth States per Worker

```ts
// tests/e2e/fixtures/global-setup.ts
import { chromium, FullConfig } from '@playwright/test';
import * as path from 'path';
import * as fs from 'fs';

/**
 * Runs ONCE before all workers. Creates an isolated auth-storage file
 * per worker slot so workers can pick their slot without blocking.
 */
export default async function globalSetup(_config: FullConfig) {
  const artifactsDir = './test-fixtures';
  fs.mkdirSync(artifactsDir, { recursive: true });

  const numWorkers = Number(process.env.PW_WORKERS ?? 8);

  const browser = await chromium.launch();
  try {
    for (let i = 0; i < numWorkers; i++) {
      const context = await browser.newContext();
      const page = await context.newPage();

      // Login once — credentials come from env or .env
      await page.goto(`${process.env.BASE_URL ?? 'http://localhost:3000'}/login`);
      await page.getByRole('textbox', { name: /email/i }).fill('qa@tester.com');
      await page.getByRole('textbox', { name: /password/i }).fill('TestPass123!');
      await page.getByRole('button', { name: /sign in/i }).click();
      await page.waitForURL('**/dashboard/**', { waitUntil: 'networkidle' });

      // Persist auth state to a worker-specific slot file
      const slotFile = path.join(
        artifactsDir,
        `worker-${i}-storage.json`,
      );
      await context.storageState({ path: slotFile });
      console.log(`✅ pre-generated auth slot → ${slotFile}`);

      await context.close();
    }
  } finally {
    await browser.close();
  }
}
```

```ts
// tests/e2e/fixtures/browser-context.ts
import { test as base } from '@playwright/test';
import * as path from 'path';
import * as crypto from 'crypto';

/**
 * Each worker gets a unique prefix string. All POMs append it to DOM
 * identifiers so parallel runs never collide.
 */
export const test = base.extend<{
  workerId: string;
  uniqueSuffix: string;
}>({
  workerId: [async ({}, use) => {
    // PLAYWRIGHT_WORKER_ID looks like "0", "1", …
    const id = process.env.PLAYWRIGHT_WORKER_ID ?? '0';
    await use(id);
  }, { scope: 'worker' }],

  uniqueSuffix: [async ({ workerId }, use) => {
    // Deterministic suffix derived from workerId + a hash seed so that
    // parallel workers targeting the same spec still get distinct values.
    const suffix = `w${workerId}-${crypto.randomBytes(3).toString('hex')}`;
    await use(suffix);
  }, { scope: 'worker' }],
});
```

## 4. Page Object Model — Typed Locators

### Base page (shared conventions)

```ts
// tests/e2e/pages/base-page.ts
import { Page, Locator, expect } from '@playwright/test';

/**
 * Every locator carries a `data-testid` annotation in comments.
 * The QA team must maintain these attributes on the frontend.
 */
export class BasePage {
  readonly page: Page;

  constructor(page: Page) {
    this.page = page;
  }

  /** Wait until no network requests are inflight for `ms` ms */
  async waitForNetworkIdle(ms = 1500): Promise<void> {
    await this.page.waitForLoadState('networkidle');
    // Extra settle time for soft animations / lazy images
    await this.page.waitForTimeout(ms);
  }

  /** Click with auto-retry; fails fast on overlay blockers */
  async safeClick(locator: Locator): Promise<void> {
    await locator.waitFor({ state: 'visible', timeout: 8_000 });
    await locator.click({ trial: false });
    // Wait for any routing/dialog triggered by click
    await this.page.waitForLoadState('domcontentloaded').catch(() => {});
  }

  /** Assert element visible with a descriptive message */
  async assertVisible(locator: Locator, message?: string): Promise<void> {
    await expect(locator).toBeVisible({ timeout: 10_000 });
  }

  /** Assert element has expected text content */
  async assertText(locator: Locator, expected: string): Promise<void> {
    await expect(locator).toHaveText(expected, { timeout: 8_000 });
  }

  navigateTo(path: string): Promise<void> {
    return this.page.goto(`${path}`, { waitUntil: 'networkidle' });
  }
}
```

```ts
// tests/e2e/pages/cart-page.ts
import { Page, Locator } from '@playwright/test';
import { BasePage } from './base-page';

export class CartPage extends BasePage {
  // -- typed locators (data-testid-based) -----------------------------------
  readonly proceedButton: Locator;
  readonly itemRow: (index: number) => Locator;
  readonly itemCountBadge: Locator;
  readonly subtotalEl: Locator;
  readonly removeButton: (productId: string) => Locator;
  readonly emptyCartMessage: Locator;

  constructor(page: Page) {
    super(page);
    this.proceedButton = page.locator('[data-testid="cart-proceed-checkout"]');
    this.itemRow = (i) => page.locator(`[data-testid="cart-item-row:${i}"]`);
    this.itemCountBadge = page.locator('[data-testid="cart-item-count"]');
    this.subtotalEl = page.locator('[data-testid="cart-subtotal"]');
    this.removeButton = (pid) => page.locator(`[data-testid="cart-remove:${pid}"]`);
    this.emptyCartMessage = page.locator('[data-testid="cart-empty-message"]');
  }

  async goTo(): Promise<void> {
    await this.navigateTo('/cart');
    await this.waitForNetworkIdle();
  }

  async getItemCount(): Promise<number> {
    return Number(await this.itemCountBadge.textContent() ?? '0');
  }

  async getSubtotal(): Promise<number> {
    const raw = (await this.subtotalEl.textContent() ?? '').replace(/[^0-9.]/g, '');
    return parseFloat(raw);
  }

  async proceedToCheckout(): Promise<void> {
    await this.safeClick(this.proceedButton);
    // Navigation handled by caller; we just clicked
  }

  async isCartEmpty(): Promise<boolean> {
    return this.emptyCartMessage.isVisible().catch(() => false);
  }
}
```

```ts
// tests/e2e/pages/checkout-flow.ts
import { Page } from '@playwright/test';
import { CartPage } from './cart-page';
import { AddressPage } from './address-page';
import { PaymentPage } from './payment-page';
import { ConfirmationPage } from './confirmation-page';
import type { CheckoutData } from '../fixtures';

/**
 * High-level choreography POM. Each step navigates, waits, and asserts
 * the transition before calling the next step. Keeps specs thin —
 * specs only declare *what* they care about asserting.
 */
export class CheckoutFlow {
  readonly cart: CartPage;
  readonly address: AddressPage;
  readonly payment: PaymentPage;
  readonly confirmation: ConfirmationPage;

  constructor(public readonly page: Page) {
    this.cart = new CartPage(page);
    this.address = new AddressPage(page);
    this.payment = new PaymentPage(page);
    this.confirmation = new ConfirmationPage(page);
  }

  async start(): Promise<void> {
    await this.cart.goTo();
  }

  async step1_proceedToCheckout(): Promise<void> {
    await this.cart.proceedToCheckout();
    await this.page.waitForURL('**/checkout/address', { timeout: 10_000 });
    await this.address.waitForNetworkIdle();
  }

  async step2_fillAddressAndContinue(data: CheckoutData['shippingAddress']): Promise<void> {
    await this.address.fillAddress(data);
    await this.address.continueToPayment();
    await this.page.waitForURL('**/checkout/payment', { timeout: 10_000 });
    await this.payment.waitForNetworkIdle();
  }

  async step3_payAndPlaceOrder(cardToken: string): Promise<void> {
    await this.payment.fillCard(cardToken);
    await this.payment.placeOrder();
    // After placing, the app redirects to confirmation
    await this.page.waitForURL('**/checkout/confirm/**', { timeout: 15_000 });
    await this.confirmation.waitForNetworkIdle();
  }

  async step4_validateConfirmation(expectedOrderId?: string): Promise<void> {
    await this.confirmation.assertVisible(
      this.confirmation.orderNumber,
      'order number should be displayed',
    );
    // Additional assertions belong in individual specs
  }

  /** Convenience method that runs the entire happy path */
  async runHappyPath(data: CheckoutData): Promise<{ orderId: string }> {
    await this.start();
    await this.step1_proceedToCheckout();
    await this.step2_fillAddressAndContinue(data.shippingAddress);
    await this.step3_payAndPlaceOrder(data.cardToken);
    await this.step4_validateConfirmation();

    const orderId = await this.confirmation.getOrderNumber();
    return { orderId };
  }
}
```

## 6. Waiting Strategies — Network-Idle Helper Library

```ts
// tests/e2e/utils/network-wait.ts
import { Page, Response } from '@playwright/test';

/**
 * Waits for all XHR/Fetch requests to settle within a time budget.
 * Falls back gracefully if some long-poll or SSE connection stays open.
 */
export async function waitForSettledNetwork(
  page: Page,
  options: { idleForMs?: number; maxWaitMs?: number } = {},
): Promise<void> {
  const { idleForMs = 1500, maxWaitMs = 10_000 } = options;

  // First pass: browser-level networkidle
  try {
    await page.waitForLoadState('networkidle', { timeout: maxWaitMs });
  } catch {
    // Ignore timeout; move to fine-grained approach
  }

  // Second pass: track active XHR/fetch promises
  const activeRequests = new Map<string, Response>();
  const requestListener = (req: import('@playwright/test').APIRequestContext) => {};

  // Third pass: settle with a short timeout for any remaining soft requests
  await page.waitForTimeout(idleForMs);
}

/**
 * Smart expectation: retries visibility checks with jitter so that
 * race conditions between JS rendering and network resolution don't
 * cause flakes.
 */
export async function expectEventually(
  locator: import('@playwright/test').Locator,
  matcher: 'toBeVisible' | 'toHaveText' | 'toHaveValue',
  value: unknown,
  opts: { timeout?: number; attempts?: number } = {},
): Promise<void> {
  const { timeout = 10_000, attempts = 3 } = opts;

  for (let i = 0; i < attempts; i++) {
    try {
      switch (matcher) {
        case 'toBeVisible':
          await locator.waitFor({ state: 'visible', timeout });
          break;
        case 'toHaveText':
          await expect(locator).toHaveText(value, { timeout });
          return; // success
        case 'toHaveValue':
          await expect(locator).toHaveValue(value, { timeout });
          return;
      }
      // Spurious visibility passed — might be stale. Give it one more frame.
      await locator.waitFor({ state: 'detached', timeout: 200 }).catch(() => {});
      await locator.waitFor({ state: 'visible', timeout });
      return;
    } catch {
      if (i === attempts - 1) throw;
      await locator.waitFor({ state: 'visible', timeout: 500 });
    }
  }
}
```

Usage in a spec:

```ts
import { expectEventually } from '../utils/network-wait';

test('item appears after async add', async ({ page }) => {
  const item = page.locator('[data-testid="cart-item-row:0"]');
  // Instead of hard sleep or blind wait:
  await expectEventually(item, 'toBeVisible', undefined, { attempts: 4 });
});
```

## 7. Per-Worker Data Isolation Strategy

Three mechanisms ensure worker 0 never stomps on worker 1:

| Mechanism | How it works |
|-----------|-------------|
| **Unique suffixes** | Every locator reference that targets a dynamically-created DOM node gets suffixed with `{uniqueSuffix}`. Generated once per worker at fixture init. |
| **Separate storageState files** | `globalSetup` writes `worker-{N}-storage.json`. Worker N reads only its own slot. No shared cookies. |
| **Isolated upload/temp dirs** | `storageState: { path: \`./test-fixtures/w${workerId}\` }` keeps file system boundaries intact. |

In practice, POMs apply suffixes automatically:

```ts
// In the combined fixture (index.ts), the `uniqueSuffix` fixture is available:
test('adds item to cart uniquely', async ({ page, uniqueSuffix }) => {
  await page.locator(`[data-testid="add-item:${uniqueSuffix}"]`).click();
  await page
    .locator(`[data-testid="cart-item-row:${uniqueSuffix}"]`)
    .waitFor({ state: 'visible' });
});
```

If the app does not support dynamic `data-testid` values, use Playwright's **scoped browser contexts**: each worker naturally gets its own cookie/storage partition because we inject `storageState` at context creation time (see below in the beforeEach override).

Override auth at the per-test level instead of globally:

```ts
// In any spec that needs explicit control:
test.use({ storageState: '<path-to-storage>' });
```

Or programmatically set it via a helper:

```ts
// tests/e2e/fixtures/setup-helpers.ts
import { test as base } from '@playwright/test';

export const testWithAuth = base.extend<{ _authInjected: boolean }>({
  _authInjected: [
    async ({ browser, storageStatePath }, use, workerInfo) => {
      const context = await browser.newContext({
        storageState: storageStatePath,
        // Ensure uploads go to worker-scoped temp dir
        acceptDownloads: true,
        viewport: { width: 1280, height: 720 },
      });
      const page = await context.newPage();

      // Override the default `page` fixture for this test
      // NOTE: this trick replaces the built-in page with our auth-scoped one
      await use(true);

      // Cleanup happens automatically; contexts close when worker ends
    },
    { auto: true, scope: 'test' },
  ],
});
```

Actually, the cleanest way in Playwright v1.35+ is to configure storageState per-project or use a fixture that creates a new context:

```ts
// tests/e2e/fixtures/auth-with-context.ts
import { test as base } from '@playwright/test';

export const test = base.extend<{
  authPage: import('@playwright/test').Page;
  authContext: import('@playwright/test').BrowserContext;
}>({
  authContext: [
    async ({ browser, storageStatePath }, use) => {
      const ctx = await browser.newContext({
        storageState: storageStatePath,
      });
      await use(ctx);
      await ctx.close();
    },
    { scope: 'test' },
  ],
  authPage: async ({ authContext }, use) => {
    const page = await authContext.newPage();
    await use(page);
  },
});
```

Tests then use `authPage` instead of the default `page`.

## 8. Quarantine Policy for Flaky Specs

### Detection phase

```ts
// tests/e2e/.playwright-flakelog.json  (generated by CI post-run script)
{
  "flaky_specs": {
    "payment.spec.ts": { "last_flake_timestamp": "2026-08-24T14:03:00Z", "count_7d": 3 },
    "end-to-end.spec.ts": { "last_flake_timestamp": "2026-08-23T09:11:00Z", "count_7d": 5 }
  }
}
```

A post-run GitHub Actions / CI script parses `test-results.json` and writes this artifact.

### Quarantine rules

| Threshold | Action |
|-----------|--------|
| ≥ 3 flaky runs in 7 days | Mark the spec as `@quarantine`; run it serially with `retries: 0` until developer triages |
| ≥ 5 flaky runs in 7 days | Block merge PRs that touch the quarantined spec unless annotated with `#skip-quarantine` |
| Zero flaky runs in 14 consecutive days | Auto-demote out of quarantine |

### Implementation — quarantine marker in tests

```ts
// tests/e2e/specs/payment.spec.ts
import { test, expect } from '../../fixtures';

/**
 * @quarantine  ← comment triggers CI quarantine detection
 * Reason: intermittent 502 from shipping stub under load
 */
test('handles partial fulfillment gracefully', async ({ page, checkoutData }) => {
  // ... test body ...
});
```

### CI gating script snippet

```bash
#!/usr/bin/env bash
# scripts/quarantine-gate.sh

FLAKY_LOG=".playwright-flakelog.json"

# Read 7-day flaky counts
COUNTS=$(jq '.flaky_specs | to_entries[] | select(.value.count_7d >= 3) | .key' "$FLAKY_LOG")

if [ -n "$COUNTS" ]; then
  echo "⚠️  Quarantined specs detected: $COUNTS"
  echo "Running them serially with retries=0..."

  # Force serial execution for quarantined files
  for spec in $COUNTS; do
    npx playwright test "$spec" --workers=1 --retries=0
  done

  # Exit non-zero if any quarantine run failed
  exit $?
fi

echo "✅ No quarantined specs pending triage."
```

### Triaging a quarantine entry

1. Developer opens the quarantined spec
2. Examines the trace artifact (`trace.zip`) from the last failure
3. Fixes the root cause (race condition, missing stub, brittle selector)
4. Removes the `@quarantine` marker comment
5. Submits a PR; CI must pass 5 consecutive green runs before re-enabling retries

## 9. Putting It All Together — What Each Test Imports

Every `.spec.ts` file starts with exactly one import:

```ts
import { test, expect, type CheckoutData } from '../fixtures';
```

That single `test` carries:
- `page` (Playwright core)
- `workerId` + `uniqueSuffix` (isolation)
- `storageStatePath` (pre-baked auth)
- `stubPaymentGateway` + `stubShippingRates` + `stubOrderConfirmation` (auto-routed API mocks)
- `checkoutData` (deterministic order payload)

No per-test login, no hardcoded sleeps, no shared mutable state.

## Summary of Flake Reduction Techniques

| Technique | Flake surface eliminated |
|-----------|------------------------|
| **storageState reuse** | Auth timing races, captcha flows, session expirations during test setup |
| **API stubs + deterministic fixtures** | External gateway timeouts, shipping calc latency, race between POST order and GET confirmation |
| **Typed `data-testid` locators** | Selector brittlement from CSS/class changes |
| **`networkidle` + settled-network polling** | Elements appearing after async renders |
| **Trace/video only on retry** | Reduces I/O contention during first-pass runs; CI stays fast |
| **Worker-scoped storageState + uniqueSuffix** | Cross-worker data collision (same cart, same cookies) |
| **Quarantine loop** | Systemic flakes caught early, triaged serially, demoted on sustained green |

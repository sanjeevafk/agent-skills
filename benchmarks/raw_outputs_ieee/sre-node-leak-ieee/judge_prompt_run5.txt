1. A listener references outer scope variables that capture request context
2. The listener is never removed (no `.off()`)
3. Or `.removeListener()` / `.once()` isn't used correctly
4. Or the emitter itself lives longer than the request lifecycle

### EventEmitter listener leak patterns

**Pattern A: Anonymous arrow functions passed directly to `.on()`**


```javascript

  logger.log(this.connectionId, chunk);

// BUT: you must also remove these later with server.off('data', handler)
```

**Pattern B: One-time listeners not triggered**

```javascript

// .once() accumulates listeners until the event fires again
client.once('response', (data) => { /* captures full request */ });
// If client reuses but 'response' was already fired, .once() adds another
```

**Fix:** Use `.addListener()` with cleanup, or `.off()` after the callback:

```javascript
function onResponse(data) {
  client.off('response', onResponse);  // guaranteed cleanup
  process(data);
}
client.on('response', onResponse);
```

**Pattern C: Error handlers accumulated per-request**

```javascript
// MOST COMMON in Express apps:
app.use(async (req, res, next) => {
  stream.on('error', handleError);  // ← added per-request, never removed!
  await transform(stream);
});
```

**Fix:** Always clean up stream/listener listeners in a `finally` block:

```javascript
app.use(async (req, res, next) => {
  try {
    await transform(stream);
  } finally {
    stream.off('error', handleError);  // guaranteed removal
  }
});
```

## Phase 3 — Confirm Whether EventEmitter Leaks Listeners

### 3.1 Live Diagnostic Script

Run this against a live pod to check listener count over time:

```javascript
// scripts/check-listener-leak.js
const { EventEmitter } = require('eventemitter3'); // or 'events'
const originalOn = EventEmitter.prototype.on;
const originalAddListener = EventEmitter.prototype.addListener;

const registry = new Map();

function trackEvents(emitter, label) {
  if (!registry.has(emitter)) {
    registry.set(emitter, { label, listeners: {}, added: 0, removed: 0 });
  }

  emitter.on = function(event, listener) {
    const info = registry.get(this);
    info.listeners[event] = (info.listeners[event] || 0) + 1;
    info.added++;
    return originalOn.apply(this, arguments);
  };

  emitter.addListener = function(event, listener) {
    return this.on(event, listener);
  };

  // Patch removeListener too
  const origRemove = emitter.removeListener;
  emitter.removeListener = function(event, listener) {
    const info = registry.get(this);
    if (info && info.listeners[event]) {
      info.listeners[event] = Math.max(0, info.listeners[event] - 1);
      info.removed++;
    }
    return origRemove.apply(this, arguments);
  };
}

module.exports = { trackEvents, registry };
```

### Cause A: EventEmitter Listener Leak (Highest Probability)

**Symptom signature:** Closure objects dominate heap delta, retainers point to `EventEmitter._eventsMap`.

**Common patterns:**


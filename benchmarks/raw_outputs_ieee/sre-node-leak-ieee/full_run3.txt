```bash
  const nodesB = new Map(snapB.nodes.ids.map((id, i) => [i, { id, name: snapB.strings[snapB.names[i]] }]));

    typeCountsA[typeName] = (typeCountsA[typeName] || 0) + 1;
  }
  for (let i = 0; i < snapB.types.length; i++) {
    const typeName = snapB.strings[snapB.typeNames[i]] || 'unknown';
    typeCountsB[typeName] = (typeCountsB[typeName] || 0) + 1;
  }

  console.log('=== Type Growth Analysis ===');
  for (const type of Object.keys(typeCountsB)) {
    const growth = (typeCountsB[type] || 0) - (typeCountsA[type] || 0);
    if (growth > 10) {
      console.log(`${type}: ${typeCountsA[type] || 0} → ${typeCountsB[type]} (+${growth})`);
    }
  }

  // Find retained objects
  const retainedBy = new Map();
  for (let i = 0; i < snapB.edges.length; i += 3) {
    const fromNode = snapB.edges[i];
    const toNode = snapB.edges[i + 1];
    if (!retainedBy.has(fromNode)) retainedBy.set(fromNode, []);
    retainedBy.get(fromNode).push(toNode);
  }
}
```

### Phase 5: Likely Root Causes and Fixes

Based on the symptoms described (RSS 300MB→4GB over 48h, closures in event-emitter registry, Buffers in response-logging middleware), here are the five most probable root causes ranked by likelihood:

#### Cause 1: Accumulating Event Listeners (HIGHEST PROBABILITY)

**Pattern:** Each HTTP request or background job registers an event listener without cleaning up.

```javascript
// ❌ BROKEN — listener accumulates forever
app.use(async (req, res, next) => {
  const telemetry = createTelemetry(req);
  
  // This listener is NEVER removed — it captures `req`, `res`, and entire closure scope
  telemetry.on('complete', (data) => {
    sendToAnalytics(data); // `data` may be large
  });
  
  // If the response ends early or error occurs, the listener stays registered
  // With millions of requests over 48h, this grows unbounded
  next();
});

// ✅ FIX: Use once() instead of on(), or ensure cleanup
app.use(async (req, res, next) => {
  const telemetry = createTelemetry(req);
  
  telemetry.once('complete', (data) => {
    sendToAnalytics(data);
  });
  
  // Explicit cleanup on response finish
  res.on('finish', () => {
    telemetry.removeAllListeners();
  });
  
  next();
});

// ✅ BETTER FIX: Don't register listeners in request handlers at all
// Use sync completion or a different pattern
```

```javascript
// Another variant: dynamic event emission without bounded registration
class RequestProcessor {
  constructor() {
    this.emitter = new EventEmitter(); // Single shared instance
  }
  
  // ❌ Every call adds another listener to the SAME emitter
  process(requestId) {

      this.emitter.on('result', (data) => {
        if (data.id === requestId) resolve(data);
        // Listener never removed — thousands accumulate

  }

  const localEmit = new EventEmitter();
  this.emitter.on('result', (data) => {
    if (data.id === requestId) {
      localEmit.emit('result', data);
      this.emitter.removeListener('result', boundHandler); // remove immediately
    }

  return new Promise(resolve => {
    const boundHandler = (data) => { localEmit.emit('result', data); };
    this.emitter.on('result', boundHandler);
  }).finally(() => this.emitter.removeListener('result', boundHandler));
}
```

**Kubernetes auto-scaling impact:** When pods restart, memory goes back down — masking the leak during development but surfacing reliably after ~48h in production.

### Most Likely Verdict

Given the two concrete observations from the heap snapshots — **closures in the event-emitter registry** and **Buffers from response-logging middleware** — the diagnosis converges on:

1. **Primary cause:** An Express middleware (likely around telemetry, logging, or auth) registers `emitter.on()` per-request without `removeListener` or `once()`. Each closure captures the full `req`/`res` scope including parsed bodies and headers, creating a classic "per-request listener leak."

2. **Secondary cause:** A response-logging utility concatenates raw response `Buffer`s into a module-level array with no capacity limit, TTL, or eviction. Over millions of responses, this alone accounts for gigabytes of Buffer retention.

3. **Compounding factor:** These two interact — the closure from cause #1 likely captures references to the logging middleware's buffer array, preventing even garbage collection of old buffers.

**Fix priority:** Address the emitter listener leak first (it's the retainer root), then fix the buffer logging middleware with bounds and eviction.

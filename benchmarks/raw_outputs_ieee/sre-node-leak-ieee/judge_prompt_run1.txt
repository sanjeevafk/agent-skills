### Step B — Trace a Single Closure Back to Root

Pick the largest closure group by "Shallow Size":

```
Click closure entry → Right panel shows:
  ┌─ Retainers
  │  ├─ Array [events] ← event-emitter internal _events array
  │  │   └─ EventEmitter instance (your registry)
  │  │       └─ Global reference (module singleton? class static?)
  │  └─ WeakMap ← possible metadata cache keyed by request ID
  │      └─ Request object (held alive longer than request lifecycle)

  └─ Closure Details
```

**Reading the chain:** Follow arrows from the leaf (your closure) upward. The moment you see a `Global`, `Module.exports`, `Process global`, or `EventEmitter._events[]` node, you've found the retention root. That root must stop holding a reference.

**Pattern:** Anonymous listeners registered per request/event, never removed.

```js
  logResponse(chunk);

// On teardown:

  dataStream.removeAllListeners('chunk');
  dataStream.destroy();
});
```

**Another variant — EventEmitter per connection/session:**

```js

    this.emitters = new Map();  // N entries, never cleaned
  }

    const ee = new EventEmitter();
    ee.on('message', msg => { /* captures session state */ });
    this.emitters.set(id, ee);
    return ee;

  }

  const ee = this.emitters.get(id);

    ee.removeAllListeners();

}
```

```javascript
// Place in monitoring health-check route or cron job
const fs = require('fs');
const path = require('path');

async function takeHeapSnapshot(label) {
  // Use vm module to avoid GC interference during snapshot
  const vm = require('vm');
  
  // Step 1: Force a full GC before snapshot for clean baseline
  if (global.gc) {
    global.gc();
    await sleep(100); // let GC settle
  } else {
    // Enable profiling GC even without --expose-gc
    process._rawDebug('No manual GC available; using forced profile mode');
    process.setMinTickBeforeForceGC(1);
    process.forceGC?.();
  }
  
  // Step 2: Write snapshot atomically
  const filePath = path.join('/tmp', `heap_${label}_${Date.now()}.heapsnapshot`);
  const writeStream = fs.createWriteStream(filePath);
  
  return new Promise((resolve, reject) => {
    process.stdout.write(`DEBUGPORT=${process.argv.find(a => a.startsWith('--inspect')) || 'not set'}`);
    
    // Use v8's built-in snapshot capability via eval in debug context
    const snapshotPath = `/tmp/heap_${label}_${process.pid}_${Date.now()}.heapsnapshot`;
    
    // Best approach: use --max-old-space-size monitoring to know WHEN to snapshot
    resolve(snapshotPath);
  });
}

function sleep(ms) {
  return new Promise(r => setTimeout(r, ms));
}
```

1. Take snapshot every 6 hours (matching your observation interval)
2. Store on persistent volume (EFS, GCS, S3 mount) not node-local disk
3. Compress with zstd to reduce storage pressure
4. Keep rolling window of last 8 snapshots (~2 days at 6h intervals)

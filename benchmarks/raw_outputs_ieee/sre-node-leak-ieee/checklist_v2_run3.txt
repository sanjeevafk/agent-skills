```bash
# Method 2: Programmatic via /health/heap endpoint (implement once, use always)
# Add to your health check route handler:
const v8 = require('v8');
const fs = require('fs');
const path = require('path');
const os = require('os');

let snapshotCounter = 0;

function takeHeapSnapshot() {
  const timestamp = Date.now();
  snapshotCounter++;
  const filename = path.join(
    os.tmpdir(),
    `heap-${timestamp}-${snapshotCounter}.heapsnapshot`
  );
  
  // writeArrayMode is critical: produces JSON that can be read by external tools
  const writer = fs.createWriteStream(filename);
  const gcHandle = v8.getHeapCodeAndSourceMaps();
  const snapshot = v8.takeHeapSnapshot({
    executableState: true,     // include source + bytecode
    trackOperations: false     // skip GC operation traces (noise)
  });
  
  // Stream it — prevents V8 from loading entire snapshot into another JS object
  const stringify = JSON.stringify(snapshot);
  // For large snapshots (>500MB), stream chunks:
  let offset = 0;
  const chunkSize = 64 * 1024 * 1024; // 64MB
  
  while (offset < stringify.length) {
    const chunk = stringify.slice(offset, offset + chunkSize);
    writer.write(chunk);
    offset += chunkSize;
  }
  writer.end();
  
  return { filename, sizeBytes: fs.statSync(filename).size };
}

// Expose as a metrics-only endpoint (auth-protected in prod)
app.get('/internal/debug/heap', async (req, res) => {
  try {
    const result = await takeHeapSnapshot();
    // Force compact before sending — but only if you have room
    // If already near OOM, skip this and just save to file
    console.log(`Heap snapshot saved: ${result.filename} (${Math.round(result.sizeBytes / 1024 / 1024)} MB)`);
    res.json({ snapshotPath: result.filename, sizeBytes: result.sizeBytes });
  } catch (err) {
    // If we fail here, the pod is about to die anyway
    console.error('Failed to take heap snapshot:', err);
    res.status(503).json({ error: 'OOM during snapshot' });
  }
});
```

```python
# /usr/local/bin/process_diff.py — parse and format inspector-diff output
import json
import sys

def process(inspector_json_path, output_dir):
    with open(inspector_json_path) as f:
        data = json.load(f)
    
    output_lines = []
    
    for category in data.get('categories', []):
        name = category['name']
        diff = category.get('diff', {})
        retained_mb = diff.get('total_retained_change', 0) / 1024 / 1024
        
        if abs(retained_mb) < 1:
            continue
            
        output_lines.append(f"\n{'='*60}")
        output_lines.append(f"Category: {name}")
        output_lines.append(f"Retained change: {retained_mb:+.1f} MB")
        output_lines.append(f"Shallow change: {diff.get('total_shallow_change', 0) / 1024 / 1024:+.1f} MB")
        output_lines.append(f"Object count change: {diff.get('object_count_change', 0):+}")
        
        for obj in category.get('growth_leaders', [])[:10]:
            cls = obj.get('constructorName', '?')
            inst = obj.get('instanceChange', 0)
            sz = obj.get('retainedSizeChange', 0) / 1024 / 1024
            output_lines.append(f"  {cls}: {inst:+,} instances (+{sz:.1f} MB retained)")
    
    output_text = '\n'.join(output_lines)
    
    with open(f'{output_dir}/analysis.txt', 'w') as f:
        f.write(output_text)
    
    print(output_text)

if __name__ == '__main__':
    process(sys.argv[1], sys.argv[2] if len(sys.argv) > 2 else '.')
```

### Key Patterns to Hunt For in the Diff

| What to look for | What it means |
|---|---|
| **Dominated by closure nodes** | Callbacks capturing scopes are being retained (likely listener leak) |
| **Buffer objects increasing** | Response bodies or log lines allocated but never released |
| **String[] in retained sets** | Log messages accumulating |
| **EventEmitter instances growing** | Registry pattern where emitters accumulate without cleanup |
| **Function entries in snapshot** | Anonymous closures being registered but unregistered |
| **WeakMap entries** | Possibly intentional caching — verify keys aren't growing forever |

```javascript
// src/monitoring/listener-health.js
// Place in your startup sequence — provides /internal/health/emitters
const EventEmitter = require('events');
const { registerMetric } = require('./metrics');

class ListenerHealthTracker {
  constructor() {
    this.trackedEmitters = new WeakMap();
    this.originalOn = EventEmitter.prototype.on;
    this.originalAddListener = EventEmitter.prototype.addListener;
    
    this.wrapAllInstances();
    this.startPolling();
  }
  
  wrapAllInstances() {
    // Intercept existing and future EventEmitters
    const tracker = this;
    
    // Patch the constructor so every new instance gets tracked
    const OriginalEE = EventEmitter;
    EventEmitter._wrapped = true;
    
    // Use a Proxy on the prototype to catch all future .on calls
    const handler = {
      apply(target, thisArg, args) {
        if (!tracker.trackedEmitters.has(thisArg)) {
          tracker.trackedEmitters.set(thisArg, new Map());
        }
        
        const events = tracker.trackedEmitters.get(thisArg);
        const eventType = args[0];
        events.set(eventType, (events.get(eventType) || 0) + 1);
        
        return Reflect.apply(target, thisArg, args);
      }
    };
    
    // We patch the method on instances as they're created
    // Best approach: override _addListener (internal hook point used by Node)
    const origAddListener = EventEmitter.prototype._addListener;
    EventEmitter.prototype._addListener = function(event, listener) {
      if (!this._listenerCounts) {
        this._listenerCounts = new Map();
      }
      this._listenerCounts.set(event, (this._listenerCounts.get(event) || 0) + 1);
      return origAddListener.call(this, event, listener);
    };
  }
  
  startPolling() {
    setInterval(() => this.drainAndReport(), 60000);
  }
  
  drainAndReport() {
    const findings = [];
    
    for (const [ee, counts] of this.trackedEmitters) {
      // Skip standard node streams and http modules
      const eeClass = ee.constructor.name;
      if (['IncomingMessage', 'OutgoingMessage', 'Socket', 'Readable', 'Writable', 'Duplex'].includes(eeClass)) {
        continue;
      }
      
      for (const [event, count] of counts) {
        const actualCount = ee.listenerCount(event);
        
        if (actualCount !== count) {
          // Mismatch — listeners were added externally or not properly tracked
          findings.push({
            className: eeClass,
            event,
            trackedAdds: count,
            actualListeners: actualCount,
            drift: actualCount - count
          });
        }
        
        if (actualCount > 50) {
          findings.push({
            className: eeClass,
            event,
            actualListeners: actualCount,
            warning: 'high_listener_count'
          });
        }
      }
    }
    
    if (findings.length > 0) {
      console.warn('[LISTENER-LEAK]', JSON.stringify(findings));
      registerMetric('emitter.leak.indicators', findings.length);
    }
  }
}

module.exports = ListenerHealthTracker;
```

```javascript
// ✅ FIX 1: Trimmed log store with LRU eviction
const LOG_MAX_ENTRIES = 5000;
const LOG_BODY_MAX_LENGTH = 512; // bytes

const logEntries = new CircularBuffer(LOG_MAX_ENTRIES); // implement or use lru-cache

function safeResponseLogger(req, res, next) {
  const startTime = process.hrtime.bigint();
  const chunks = [];
  let totalSize = 0;
  
  const originalEnd = res.end.bind(res);
  const originalWrite = res.write.bind(res);
  
  res.write = function(chunk) {
    // Cap buffer size — drop data if response is too large to log
    if (totalSize < LOG_BODY_MAX_LENGTH && typeof chunk === 'string') {
      chunks.push(Buffer.from(chunk, 'utf8'));
      totalSize += chunk.length;
    }
    return originalWrite(chunk);
  };
  
  res.end = function(chunk, ...rest) {
    if (chunk && totalSize < LOG_BODY_MAX_LENGTH) {
      chunks.push(Buffer.from(chunk, 'utf8'));
    }
    
    const duration = Number(process.hrtime.bigint() - startTime) / 1e6;
    
    // Never log raw binary — serialize safely
    const bodyPreview = chunks.length > 0
      ? Buffer.concat(chunks).toString('utf8').slice(0, LOG_BODY_MAX_LENGTH)
      : null;
    
    // Async log write — don't block response delivery
    queueMicrotask(() => {
      logger.info('response', {
        requestId: req.id,
        method: req.method,
        url: req.originalUrl,
        statusCode: res.statusCode,
        durationMs: Math.round(duration * 100) / 100,
        bodyPreview: truncate(bodyPreview, LOG_BODY_MAX_LENGTH)
        // NEVER include full responseBody unless debugging a specific issue
      });
    });
    
    // Clear our references immediately
    chunks.length = 0;
    totalSize = 0;
    
    return originalEnd(chunk, ...rest);
  };
  
  next();
}

function truncate(str, maxLen) {
  if (!str) return str;
  if (str.length <= maxLen) return str;
  return str.slice(0, maxLen) + `... (${str.length} chars)`;
}

module.exports = safeResponseLogger;
```

## Phase 5: Root Cause Summary & Likeliest Scenarios

### Most Likely Root Cause (based on evidence: closures + event emitter registry + Buffers)

```
Primary suspect: EventEmitter listener leak combined with response-body buffering

Chain of causation:

1. A middleware or utility creates EventEmitters per-request (or reuses a shared one)
   └─> Each request registers listeners (.on()) for lifecycle events
   └─> Listeners are closures that capture `req`, `res`, `chunks`, `dbResult`
   
2. The same middleware buffers response bodies for logging
   └─> Resolves to Buffer.alloc()/Buffer.from() allocations
   
3. Listeners are never removed (.off() / removeListener() absent)
   └─> EventEmitter.registry grows linearly with request count
   └─> Each closure retains its captured scope (full request/response/body)
   
4. After ~48 hours at typical API traffic:
   └─> 10K-100K leaked listeners
   └─> Each retaining 10KB–500KB of request/response context
   └─> Total retained: 100GB–50GB → exceeds container memory limit → OOMKilled
```

```javascript
// Emergency circuit breaker — detect and reset leaking emitters
// Drop this in as a temporary measure
const EMITTER_THRESHOLD = 1000; // listeners per emitter type

function monitorAndResetLeakingEmitters() {
  // Walk all active EventEmitter instances via WeakRef iteration
  // Note: Node doesn't expose an iterator over live instances,
  // so we instrument at construction time
  
  if (global.__trackedEmitters) {
    for (const [emitter, counts] of global.__trackedEmitters) {
      for (const [event, count] of counts) {
        if (count > EMITTER_THRESHOLD) {
          console.error(`[EMERGENCY-CIRCUIT-BREAKER] Resetting ${event} listeners on ${emitter.constructor.name}`);
          emitter.removeAllListeners(event);
          counts.set(event, 0);
          // This resets the leak but may cause missed events
          // Good enough to prevent OOM until proper fix deploys
        }
      }
    }
  }
}

setInterval(monitorAndResetLeakingEmitters, 60000);
```

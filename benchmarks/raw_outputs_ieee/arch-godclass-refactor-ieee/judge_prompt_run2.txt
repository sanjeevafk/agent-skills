```python
import json, hashlib
from contextlib import contextmanager

@contextmanager
def transaction_snapshot(db_conn):
    """Capture all state before and after, diff the result."""
    tables_to_audit = ["orders", "order_items", "payments", "audit_log"]
    snapshots_before = {}
    for table in tables_to_audit:
        # Hash each row deterministically: sort columns, serialize row tuple
        rows = db_conn.execute(f"SELECT * FROM {table} ORDER BY id")
        snapshots_before[table] = hash_rows(rows)
    yield snapshots_before
    # caller compares: assert snapshots_before == hash_after(db_conn)
```

Key insight: use **row-level hashing** rather than raw diffs. You want the test to fail only when business-logical data changes, not when auto-increment IDs shift. Sort by primary key, include all non-monotonic columns, produce a SHA256 per table.

For INSERT-only paths (e.g., `audit_log` appends), compare length plus content hash. For UPDATE/DELETE paths, also track deleted IDs.

class SideEffectCapture:
    """Captures SQL executed, emails sent, and audit events during a call."""

    def __init__(self, service):
        self.sql_calls = []
        self.emails_sent = []
        self.audit_events = []
        self._original_execute = service.db.execute if hasattr(service, 'db') else None
        self._original_send_email = service.mailer.send if hasattr(service, 'mailer') else None
        self._original_log_audit = service.audit.log if hasattr(service, 'audit') else None

    def attach(self):
        if self._original_execute:
            service.db.execute = lambda sql, *a, **kw: (self.sql_calls.append(sql), self._original_execute(sql, *a, **kw))[1]
        if self._original_send_email:
            service.mailer.send = lambda *a, **kw: (self.emails_sent.append((a, kw)), self._original_send_email(*a, **kw))[1]
        if self._original_log_audit:
            service.audit.log = lambda *a, **kw: (self.audit_events.append((a, kw)), self._original_log_audit(*a, **kw))[1]

    def detach(self):
        if self._original_execute:
            service.db.execute = self._original_execute
        if self._original_send_email:
            service.mailer.send = self._original_send_email
        if self._original_log_audit:

### How to capture SQL side effects

The challenge is raw SQL scattered through methods. Two approaches:

**Approach A — Database-level query interceptor (preferred)**

If using a standard ORM or connection pool, intercept at the lowest level so you don't need to know where SQL lives in the code:

```python
class SqlSnapshotInterceptor:
    """Wraps a DB connection to record every query + params + result."""
    
    def __init__(self, connection):
        self.connection = connection
        self.queries = []
    
    def execute(self, sql, params=None):
        result = self.connection.execute(sql, params)
        self.queries.append({
            'sql': self._normalize_sql(sql),
            'params': params,
            'rowcount': result.rowcount,
            'description': result.description  # column metadata
        })
        return result
    
    def fetchall(self):
        data = self.connection.fetchall()
        return data
    
    @staticmethod
    def _normalize_sql(sql):
        """Collapse whitespace, remove timestamps/hints for stable diffs."""
        import re
        sql = re.sub(r'\s+', ' ', sql).strip()
        return sql
    
    def __enter__(self):
        return self
    
    def __exit__(self, *args):
        pass
```

### DbStateSnapshot utility

```python
class DbStateSnapshot:
    """Capture full state of relevant tables before/after execution."""
    
    TABLES = ['orders', 'order_items', 'tax_entries', 'audit_log', 'pricing_rules']
    
    def __init__(self, connection):
        self.connection = connection
        self.before = {}
        self.after = {}
    
    def __enter__(self):
        for table in self.TABLES:
            self.before[table] = self._select_all(table)
        return self
    
    def capture(self):
        snapshot = {}
        for table in self.TABLES:
            snapshot[table] = self._select_all(table)
            diff = self._diff(self.before[table], snapshot[table])
            if diff:
                snapshot[f'{table}_changes'] = diff
        return snapshot
    
    @staticmethod
    def _select_all(table):
        rows = []
        cursor = self.connection.cursor()

            rows.append(dict(zip(columns, row)))
```

## 2. Characterization / Golden-Master Tests

The single biggest risk in extracting from a God class is behavioral drift. Golden-master tests lock down current behavior. I'd take this approach:

### What to Snapshot Per Public Method

For each public method (`createOrder`, `updateOrder`, `cancelOrder`, etc.), I need three kinds of outputs captured as golden data:

**B. Database state after execution** — query the same tables/order IDs that the method modifies, serialize them to JSON. This proves the SQL side effects are preserved through refactoring.

**C. Outbound messages queued/sent** — intercept emails and audit log writes, capture their content as JSON blobs.

```python
    def visit_Call(self): ...         # catch .execute(), .send_email(), etc.
```

```python
Outputs (return values, persisted state shape, side-effect payloads) are

import pytest, copy, json
from unittest.mock import MagicMock, patch
from orders.order_service import OrderService

# ── Fixtures ────────────────────────────────────────────────
@pytest.fixture
def service():
    """Inject real DB but mock external collaborators so we control side effects."""
    svc = OrderService.__new__(OrderService)
    svc.__dict__ = {
        'db_conn': MagicMock(),          # real-ish connection wrapper
        'smtp_host': 'localhost',
        'audit_logger': MagicMock(),
        # all 34 fields initialized to identity/null/default per the original init
    }

# Capture strategy: wrap every collaborator so the test records calls.
@pytest.fixture(autouse=True)
def capture_side_effects(service):
    service._captured_sql = []
    service._captured_emails = []
    service._captured_audit = []
    # Patch the three escape hatches
    orig_exec = service.db_conn.execute      # whatever name it uses
    def tracing_execute(sql, *args):
        service._captured_sql.append({'sql': sql, 'params': args})
        return orig_exec(sql, *args) if callable(orig_exec) else None
    service.db_conn.execute = tracing_execute
    yield

# ── Helper: load baseline ─────────────────────────────────
def _golden(key):
    path = f'baseline/{key}.json'
    import os
    if not os.path.exists(path):
        pytest.skip(f'Missing baseline {path} — run --regen-baseline')
    return json.load(open(path))

# ── Test matrix: one per public method ─────────────────────
@pytest.mark.parametrize('method_name,input_data', [
    ('create_order',   {'customer_id': 42, 'items': [{'sku':'X', 'qty':2}]}),
    ('submit_order',   {'order_id': 100}),
    ('apply_coupon',   {'order_id': 100, 'code': 'SAVE10'}),
    ('calculate_tax',  {'order_id': 100, 'jurisdiction': 'CA'}),
    ('cancel_order',   {'order_id': 100, 'reason': 'customer request'}),
    ('get_order',      {'order_id': 100}),
    ('list_orders',    {'customer_id': 42, 'limit': 50}),
    ('process_payment',{'order_id': 100, 'amount': 99.99, 'method': 'card'}),
])
def test_public_method_output(service, method_name, input_data):
    method = getattr(service, method_name)
    result = method(**input_data)
    
    baseline_key = f'{method_name}_output'
    baseline = _golden(baseline_key)
    
    # Structural equality: strip timestamps, auto-increment IDs, UUIDs
    cleaned = _sanitize_for_comparison(result)
    assert cleaned == baseline, (
        f'{method_name} output drifted.\n'
        f'Expected: {json.dumps(baseline, indent=2)}\n'
        f'Got:      {json.dumps(cleaned, indent=2)}'
    )


def _sanitize_for_comparison(obj):
    """Replace non-deterministic fields so golden comparison is stable."""
    if isinstance(obj, dict):
        clean = {}
        for k, v in obj.items():
            if k in ('id', 'created_at', 'updated_at', 'uuid', 'sequence'):
                continue  # drop synthetic/deterministic fields
            clean[k] = _sanitize_for_comparison(v)
        return clean
    if isinstance(obj, list):
        return [_sanitize_for_comparison(i) for i in obj]
    return obj
```

**SQL side-effects snapshot test:**

```python
# tests/characterization/test_sql_side_effects.py
"""
Records every SQL string + parameter set emitted during each public method.
Used as a checksum: if SQL changes (even formatting), the golden fails and
we investigate whether semantics truly match.
"""
@pytest.mark.parametrize('method_name,input_data,expected_tables', [
    ('create_order',   {'customer_id': 42, 'items': [{'sku':'X','qty':2}]}, ['orders','order_items']),
    ('submit_order',   {'order_id': 100},                            ['orders']),
    ('apply_coupon',   {'order_id': 100, 'code': 'SAVE10'},          ['order_coupons','orders']),
    ('cancel_order',   {'order_id': 100, 'reason': 'test'},          ['orders','order_status_log']),
    ('process_payment',{'order_id': 100, 'amount': 99.99},           ['payments','orders']),
])
def test_sql_side_effects_match_baseline(service, method_name, input_data, expected_tables):
    method = getattr(service, method_name)
    result = method(**input_data)
    
    # We captured SQL via the fixture's tracing_execute override
    sql_list = service._captured_sql
    
    # Load baseline
    baseline = _golden(f'{method_name}_sql')
    
    # Compare: same number of statements, same SQL text (normalized),
    # same param count (values may differ due to fixtures)
    assert len(sql_list) == len(baseline['statements']), \
        f'{method_name}: SQL statement count changed ({len(sql_list)} vs {len(baseline)})'
    
    for i, (actual, expected) in enumerate(zip(sql_list, baseline['statements'])):
        actual_clean = _normalize_sql(actual['sql'])
        expected_clean = _normalize_sql(expected['sql'])
        assert actual_clean == expected_clean, (
            f'SQL drift at statement {i} in {method_name}:\n'
            f'Expected: {expected_clean}\n'
            f'Actual:   {actual_clean}'
        )


def _normalize_sql(sql):
    import re
    # Remove whitespace differences, lowercase keywords
    sql = re.sub(r'\s+', ' ', sql).strip().lower()
    # Strip param markers — compare structure, not values
    sql = re.sub(r'\?|:%\w+|:[a-zA-Z_]\w*', '?', sql)
    return sql
```

### Phase 1 — Collaborator Extraction (The Seam)

The 34 fields fall into clear responsibility clusters. I extract them into interfaces in this order:

#### Step 1: Extract `OrderRepository` (Database → Persistence layer)

**Why first:** SQL is the highest-leverage noise — it's scattered across methods, tightly coupled, and blocking any other refactor. Once the DB access goes behind one interface, every method shrinks immediately.

```python
# Domain interface — written BEFORE extracting the collaborator
# src/orders/domain/repositories.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional, Sequence

@dataclass(frozen=True)
class OrderItem:
    sku: str
    quantity: int
    unit_price: float

@dataclass
class Order:
    id: Optional[int]
    customer_id: int
    items: Sequence[OrderItem]
    subtotal: float
    tax_amount: float
    discount_amount: float
    total: float
    status: str                     # 'draft','submitted','paid','cancelled'
    shipping_address: Optional[str]
    notes: Optional[str]

class OrderRepository(ABC):
    @abstractmethod
    def save(self, order: Order) -> int:  # returns new ID
        ...
    @abstractmethod
    def get_by_id(self, order_id: int) -> Optional[Order]:
        ...
    @abstractmethod
    def get_by_customer(self, customer_id: int, limit: int) -> Sequence[Order]:
        ...
    @abstractmethod
    def update_status(self, order_id: int, new_status: str, reason: Optional[str] = None):
        ...
```

**Adaptor — wrapping existing SQL (no behavior change):**

```python
# src/orders/infrastructure/sql_order_repository.py
class SqlOrderRepository(OrderRepository):
    """Thin adapter: wraps the raw SQL inside OrderService."""
    def __init__(self, db_conn):
        self.db = db_conn
    
    def save(self, order: Order) -> int:
        # Replace the inline INSERT with this single call.
        # The SQL string itself stays identical — just factored out.
        if order.id is None:
            cur = self.db.execute(
                "INSERT INTO orders (customer_id, subtotal, tax_amount, "
                "discount_amount, total, status, shipping_address, notes) "
                "VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
                order.customer_id, order.subtotal, order.tax_amount,
                order.discount_amount, order.total, order.status,
                order.shipping_address, order.notes,
            )
            new_id = cur.lastrowid
            for item in order.items:
                self.db.execute(
                    "INSERT INTO order_items (order_id, sku, quantity, unit_price) "
                    "VALUES (?, ?, ?, ?)",
                    new_id, item.sku, item.quantity, item.unit_price,
                )
            return new_id
        # UPDATE path... identical SQL to original
        
    # get_by_id, get_by_customer, update_status follow the same pattern
    # Each is literally moved from the method body into its own method.
```

#### Final Interface Composition

```python
# src/orders/application/order_service.py (NEW — thinner shell)
class OrderService:
    """Thin orchestrator — no SQL, no email templates, no pricing math."""
    
    def __init__(
        self,
        repository: OrderRepository,
        pricing: PricingCalculator,
        tax: TaxCalculator,
        notifications: OrderNotificationService,
        audit: OrderAuditLogger,
    ):
        self.repo = repository
        self.pricing = pricing
        self.tax = tax
        self.notifications = notifications
        self.audit = audit
    
    def create_order(self, customer_id: int, items: list[dict], shipping_address=None, notes=None) -> Order:
        errors = self.pricing.validate_items(items)
        if errors: raise ValidationError(errors)
        
        order_items = [OrderItem(**i) for i in items]
        subtotal = self.pricing.compute_subtotal(order_items)
        discount, err = self.pricing.apply_discount(subtotal, '')
        total = subtotal - discount
        tax_amt = self.tax.calculate(total, '', order_items)
        
        order = Order(None, customer_id, order_items, subtotal, tax_amt, discount, total + tax_amt, 'draft', shipping_address, notes)
        order.id = self.repo.save(order)
        
        self.notifications.notify(NotificationEvent.ORDER_CREATED, order, {})
        self.audit.log(order.id, 'CREATE', f'created by customer {customer_id}')
        return order
    
    # submit_order, apply_coupon, cancel_order, get_order, etc.
    # Each becomes 10-30 lines instead of 80-200.
```

### Proving Behavior Preservation at Every Commit

| Mechanism | When | What It Checks |
|-----------|------|-----------------|
| **Golden-master behavioral tests** (`test_order_service_golden.py`) | Every CI run, after every refactor commit | Every public method produces structurally identical output |
| **SQL diff tests** (`test_sql_side_effects.py`) | Every refactor commit until Week 4, then periodically | Same query count, same normalized SQL strings |
| **Email payload tests** (`test_email_side_effects.py`) | Every commit involving notification changes | Same recipients, templates, context keys |
| **Round-trip integration test** | End of each phase | Create → Submit → Pay → Cancel, then read back matches |
| **Property-based fuzzing** | Weekly during early phases | Random valid/invalid inputs produce no exceptions, consistent outputs |
| **Static analysis gate** | Every commit | `mypy`, `ruff`, no circular imports between layers |

### Risk Register & Mitigations

| Risk | Mitigation |
|------|-----------|
| Golden baselines diverge due to environmental factors (timestamps, IDs) | `_sanitize_for_comparison()` strips nondeterministic fields |
| SQL normalization misses semantic-equivalent rewrites | Normalize both `EXPLAIN PLAN` outputs, not just text |
| Someone merges without running tests | CI requires `pytest tests/characterization/` to pass before merge; branch protection rule |
| Service constructor signature change leaks externally | Public API never changes — new private constructor, DI container wires the real dependencies |
| Email template updates slip through | Every email path has its own parametrized test keyed by event name |

### Summary

The campaign follows a proven sequence: **characterize → isolate seams → strangulate vertically → decommission.** The key insight is that the repository extractor comes first because SQL is the highest-friction coupling — removing it alone reduces the class by ~15% in the first commit. Each subsequent extractor targets the next-deepest coupling. Golden tests are the invariant that keeps the whole chain honest; if they don't pass, the commit doesn't ship.

```python
# characterization/transcript_runner.py — the heart of golden-master
import json, time, hashlib, functools
from pathlib import Path
from typing import Any

OUTPUT_DIR = Path("tests/golden_snapshots")
OUTPUT_DIR.mkdir(parents=True, existists=False)

class TranscriptCollector:
    """Intercepts all DB calls, email sends, and audit log writes.
    
    Each invocation records:
      1. The exact SQL text + parameter values used
      2. The email recipient, subject, body hash, attachments metadata
      3. Audit log record keys + severity
    
    The transcript is a list of ordered "effects" that get hashed
    deterministically per-order so the same inputs always produce
    the same output file.
    """
    def __init__(self):
        self.effects: list[dict] = []
    
    def record_sql(self, operation: str, table: str, sql: str, params: dict | None):
        # Canonicalize: remove whitespace differences, sort param keys
        cleaned = ' '.join(sql.split())
        sanitized_params = dict(sorted((k, str(v)) for k, v in (params or {}).items()))
        self.effects.append({
            "type": "sql",
            "operation": operation,           # INSERT|UPDATE|DELETE|SELECT
            "table": table,
            "sql_hash": hashlib.sha256(cleaned.encode()).hexdigest()[:12],
            "cleaned_sql": cleaned,
            "param_count": len(sanitized_params),
            "params_fingerprint": hashlib.sha256(
                json.dumps(sanitized_params).encode()
            ).hexdigest()[:8],
        })
    
    def record_email(self, to: str, subject: str, body_plain: str, 
                     body_html: str | None, attachments: list[dict]):
        self.effects.append({
            "type": "email",
            "to": to,
            "subject_hash": hashlib.sha256(subject.encode()).hexdigest()[:12],
            "body_md5": hashlib.md5(body_plain.encode()).hexdigest(),
            "has_html": body_html is not None,
            "attachment_count": len(attachments),
        })
    
    def record_audit(self, action: str, entity_type: str, entity_id: str, 
                     severity: str, detail: str):
        self.effects.append({
            "type": "audit",
            "action": action,
            "entity_type": entity_type,
            "entity_id": entity_id,
            "severity": severity,
            "detail_hash": hashlib.sha256(detail.encode()).hexdigest()[:12],
        })
    
    def snapshot_key(self) -> str:
        return hashlib.sha256(
            json.dumps(self.effects, sort_keys=True).encode()
        ).hexdigest()[:16]


def freeze_order_result(order_result: Any) -> dict:
    """Convert any rich result object to pure-JSON-serializable form."""
    # If results carry datetime, Decimal, UUID — normalize here
    return {
        "order_id": str(order_result.order_id),
        "status": order_result.status,
        "total_cents": int(order_result.total_cents),
        "line_items": [
            {"sku": li.sku, "qty": li.qty, "unit_price_cents": int(li.unit_price_cents)}
            for li in order_result.line_items
        ],
        "created_at": str(order_result.created_at),
    }


def record_execution(order_service, method_name: str, args, kwargs) -> dict:
    """Execute one OrderService public method + collect full transcript.
    
    This replaces every real collaborator temporarily via monkey-patch
    or dependency injection swap (see §0b below). Returns the
    golden-master JSON payload.
    """
    collector = TranscriptCollector()
    
    # --- Patch collaborators ---
    _patch_db(order_service, collector)
    _patch_emailer(order_service, collector)
    _patch_logger(order_service, collector)
    
    # Execute
    start = time.monotonic()
    result = getattr(order_service, method_name)(*args, **kwargs)
    elapsed_ms = round((time.monotonic() - start) * 1000, 1)
    
    # Revert patches (or rely on scoped DI swap)
    _unpatch_all(order_service)
    
    frozen_result = freeze_order_result(result)
    snapshot = {
        "method": method_name,
        "input_hash": hashlib.sha256(
            json.dumps({"args": args, "kwargs": kwargs}, default=str).encode()
        ).hexdigest(),
        "result": frozen_result,
        "effects": collector.effects,
        "elapsed_ms": elapsed_ms,
        "snapshot_key": collector.snapshot_key(),
    }
    return snapshot


def _patch_db(service, collector):
    raw_conn = service._db_connection          # or however it owns its connection
    original_execute = raw_conn.execute
        
    @functools.wraps(original_execute)
    def traced_execute(sql, params=None):
        parsed = _parse_statement(sql)         # heuristic parser
        collector.record_sql(parsed["op"], parsed["tbl"], sql, params)
        return original_execute(sql, params)
    
    raw_conn.execute = traced_execute


def _patch_emailer(service, collector):
    orig_send = service._mailer.send
    @functools.wraps(orig_send)
    def traced_send(to, subject, body_plain, body_html=None, attachments=None):
        collector.record_email(to, subject, body_plain, body_html, attachments or [])
        return orig_send(to, subject, body_plain, body_html=body_html, attachments=attachments)
    service._mailer.send = traced_send


def _patch_logger(service, collector):
    orig_write = service._auditor.write
    @functools.wraps(orig_write)
    def traced_write(action, entity_type, entity_id, severity, detail):
        collector.record_audit(action, entity_type, entity_id, severity, detail)
        return orig_write(action, entity_type, entity_id, severity, detail)
    service._auditor.write = traced_write


def _parse_statement(sql: str) -> dict:
    """Naive but sufficient statement classifier."""
    upper = sql.strip().upper()
    op = upper[:6]                              # SELECT|INSERT|UPDATE|DELETE
    table_map = {
        "INSERT INTO orders": "orders",
        "UPDATE orders SET": "orders",
        "INSERT INTO order_lines": "order_lines",
        "INSERT INTO order_payments": "payments",
        "SELECT FROM payments": "payments",
    }
    tbl = "unknown"
    for key, val in table_map.items():
        if key.upper() in sql.upper():
            tbl = val
            break
    return {"op": op, "tbl": tbl}
```

### Extract Sequence (Optimal Order)

| Step | Extract | Why First? | Effort | Risk |
|------|---------|------------|--------|------|
| **1** | `DatabaseRepository` (or `OrderGateway`) | Most invasive — direct SQL scattered everywhere. Few external callers besides OrderService itself. High blast radius but clear interface. | Medium-High | High (if wrong) |
| **2** | `NotificationSender` (unified email/SMS/webhook) | Pure interface, zero internal dependencies. Can swap implementations instantly. | Low | Low |
| **3** | `OrderValidator` (extract validation rules from inline checks) | Mostly pure functions. Easiest extraction because it's likely embedded conditionals. | Low | Very Low |
| **4** | `PricingEngine` | May depend on validator; extract after validator so pricing can receive validated input cleanly. | Medium | Medium |
| **5** | `TaxCalculator` | Usually pure function (input: item, jurisdiction → output: tax). Safe to extract once pricing is clean. | Low-Medium | Low |
| **6** | `AuditLogWriter` | Simple append-only; may share DB connection with step 1 (so do after DB repo is extracted). | Low | Very Low |

### Why This Order?

1. **Database first** gives us a clean `persistOrder(Order)` and `findOrderById(id)` — every other layer now talks to an interface, not raw SQL strings. This is the highest-risk move, but also the highest-reward: it isolates the most violation of single responsibility.

2. **Notifications second** is low-risk and gives a visible early win. The interface is dead-simple: `send(Notification)`. No branching logic needed inside the interface contract.

3. **Validator third** decomposes the most tangled inline `if` blocks across all methods. Once these are separate, the main flow methods read like a pipeline.

4. **Pricing fourth** often has domain-specific rules (promotions, bulk discounts, geo-rules). With validation cleaned up, pricing receives clean DTOs.

5. **Tax fifth** is the simplest domain logic — usually a lookup table or formula. Cleanest when pulled last because it depends on finalized pricing context.

6. **Audit last** because audit logging is a cross-cutting concern that should ideally use the repository interface (step 1) rather than raw DB connections.

class DatabaseSnapshot:
    """Captures a point-in-time view of order-related tables."""

    ORDER_TABLES = {
        'orders', 'order_items', 'order_tax_lines',
        'order_payments', 'order_audit_log'
    }

    def __init__(self, connection):
        self._conn = connection

    def take(self) -> dict:
        """Return row counts + deterministic checksums per table."""
        snap = {}
        cur = self._conn.cursor()
        for table in self.ORDER_TABLES:
            try:
                # Get primary key column name (assumes standard naming)
                pk_map = {'orders': 'id', 'order_items': 'id',
                          'order_tax_lines': 'id', 'order_payments': 'id',
                          'order_audit_log': 'id'}
                pk_col = pk_map.get(table, 'id')

                # Row count
                cur.execute(f'SELECT COUNT(*) FROM {table}')
                snap[table] = {'row_count': cur.fetchone()[0], 'checksums': []}

                # Checksum of all data (excluding volatile fields)
                cur.execute(f'SELECT {pk_col} FROM {table} ORDER BY {pk_col}')
                pks = [str(r[0]) for r in cur.fetchall()]

                for pk in pks:
                    cur.execute(f'SELECT * FROM {table} WHERE {pk_col} = %s ORDER BY column_name', (pk,))
                    rows = cur.fetchall()
                    if rows:
                        col_names = [d[0] for d in cur.description]
                        # Normalize: exclude timestamps, convert bytes, sort
                        normalized = {}
                        for row in rows:
                            for i, val in enumerate(row):
                                col = col_names[i]
                                if val is None:
                                    normalized[col] = ''
                                elif isinstance(val, (datetime,)):
                                    normalized[col] = val.isoformat()  # deterministic
                                elif isinstance(val, (int, float, bool, str)):
                                    normalized[col] = str(val)
                                else:
                                    normalized[col] = repr(val)

                        ck = hashlib.sha256(
                            json.dumps(normalized, sort_keys=True).encode()
                        ).hexdigest()
                        snap[table]['checksums'].append({
                            'pk': str(pk),
                            'checksum': ck
                        })
            except Exception:
                continue  # Table may not exist in test schema
        return snap

    def describe_diff(self, before: dict, after: dict) -> list[str]:
        """Produce human-readable diffs between two snapshots."""
        diffs = []
        for table in self.ORDER_TABLES:
            b = before.get(table, {})
            a = after.get(table, {})

            bc = b.get('row_count', 0)
            ac = a.get('row_count', 0)
            if bc != ac:
                diffs.append(f'{table}: row_count changed {bc}→{ac}')

            b_ck = {x['pk']: x['checksum'] for x in b.get('checksums', [])}
            a_ck = {x['pk']: x['checksum'] for x in a.get('checksums', [])}

            for pk in set(b_ck) | set(a_ck):
                if b_ck.get(pk) != a_ck.get(pk):
                    diffs.append(f'{table}[pk={pk}]: content changed')
        return diffs

### Phase 2: Seam Extraction — Target Module Boundaries

After the characterization suite passes, define where the monolith should fracture. Here are the natural module boundaries based on responsibility:

| Current Role | Proposed Module | Rationale |
|---|---|---|
| Validation rules (input sanitization, business-rule checks) | `OrderValidator` / `validation` package | Pure functions, no I/O. Can be extracted first because it has zero dependencies and no side effects. |
| Pricing calculation (line item math, discounts, promotions) | `PricingEngine` | Self-contained arithmetic. No persistence, no networking. Replaceable formula logic. |
| Tax computation (jurisdiction lookup, exemption handling) | `TaxCalculator` | Deterministic given jurisdiction + items. Domain-specific but isolated. |
| Email dispatch (notifications, confirmations) | `OrderNotifier` (interface-based) | I/O boundary. Needs a port/adapter shape from day one. |
| Audit logging | `AuditLogger` (interface-based) | Write-side effect. Simple append operation. |
| Persistence (SQL operations, transactions) | `OrderRepository` (interface + impl) | All direct SQL goes here. Once this interface is defined, the rest of OrderService becomes orchestrator. |

#### Seam extraction order — and why:

**Step 1: Extract `OrderValidator` (pure, no-risk)**

- Grep for validation logic (`throw new ValidationError`, `if (!valid)...`) scattered through OrderService methods.
- Pull each validation check into dedicated validator methods in a new class. Parameterize with input DTO, return structured error collection.
- Inject into OrderService via constructor (add a new ctor param, wire up in DI container / factory).
- **Why first:** Zero side effects. You can refactor validators test-by-test within the characterization suite. Every validation path stays covered; you're moving code out, not changing behavior.

**Step 2: Extract `PricingEngine` and `TaxCalculator` (deterministic, high-value)**

- These are the two heaviest computation domains in the class. They share no I/O — they operate on in-memory data.
- Create interfaces now (`IPricingEngine`, `ITaxCalculator`) even if the first implementation lives inside the old class. Pass OrderService's own results through them to prove equivalence via characterization tests.
- Swap in the extracted implementations method-by-method. Each pricing/tax call site becomes `pricingEngine.calculate(item)` instead of inline math.
- **Why second:** High complexity density, pure computation = high confidence refactor. Characterization tests validate every price/tax combination. This alone removes ~300 lines from OrderService.

**Step 3: Extract `OrderRepository` (the critical seam — persists everything)**

- This is the hardest and most important extraction. Every direct SQL statement gets pulled into repository methods.
- Define the interface first (`IOrderRepository`): `save(Order)`, `findById(id)`, `findOpenByCustomer(customerId)`, `persistAudit(AuditEntry)`, `adjustInventory(itemId, qty)`.
- Create a migration adapter: the new repository's first implementation wraps the existing SQL in methods. But here's the key — **define the interface before replacing any SQL calls**. Have OrderService call `repository.save(order)` instead of executing SQL directly. Then implement `save` with the existing SQL behind the interface.
- **Why third:** Repository is the I/O bottleneck. It touches both DB and often coordinates with other collaborators (inventory, audit). You need the validation and pricing layers already stable so you can trust the in-order data flowing into persistence. Also, repository extraction forces you to confront transaction boundaries explicitly — something buried in a 1,400-line class.

**Step 4: Extract `OrderNotifier` (email, notifications)**

- Interface: `INotifier.sendOrderConfirmation(Order)`, `.sendCancellationNotice(Order)`, etc.
- Production impl wires up to SMTP. Test impl captures sent messages (as characterized in Phase 1).
- **Why fourth:** Notifier depends on the order being fully formed (priced, taxed, validated, persisted). It's the last step in the orchestration chain, so it makes sense to extract last. Email template logic is messy and low-confidence — best to deal with after the core flow is clean.

**Step 5: Extract `AuditLogger` (append-only side channel)**

- Interface: `IAuditLogger.log(AuditEvent)`. Simplest collaborator — one method signature.
- **Why fifth:** Minimal complexity, purely additive. Doesn't affect order semantics. Safe cleanup pass.

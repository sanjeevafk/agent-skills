# Refactoring Campaign: Giant OrderService → Clean Architecture

## Phase 0: Understanding the Bleeding (Day 1-2)

### Current Pain Analysis

A 1,400-line class with 34 fields touching validation, pricing, tax, persistence, email, and audit logging is violating **every single responsibility principle**. Direct SQL means coupling to a specific database engine. Email sends mean side effects are embedded in business logic. At this size, any change requires understanding 40% of the class before safely modifying 2%.

### What We Can't Touch

The entire public API surface must remain byte-for-byte identical in signature and behavior. Every method that callers depend on continues to work — we're refactoring internals only, shippable incrementally behind an unchanged facade.

```python
class SideEffectCapture:
    """Captures SQL executed, emails sent, and audit events during a call."""

    def __init__(self, service):
        self.sql_calls = []
        self.emails_sent = []
        self.audit_events = []
        self._original_execute = service.db.execute if hasattr(service, 'db') else None
        self._original_send_email = service.mailer.send if hasattr(service, 'mailer') else None
        self._original_log_audit = service.audit.log if hasattr(service, 'audit') else None

    def attach(self):
        if self._original_execute:
            service.db.execute = lambda sql, *a, **kw: (self.sql_calls.append(sql), self._original_execute(sql, *a, **kw))[1]
        if self._original_send_email:
            service.mailer.send = lambda *a, **kw: (self.emails_sent.append((a, kw)), self._original_send_email(*a, **kw))[1]
        if self._original_log_audit:
            service.audit.log = lambda *a, **kw: (self.audit_events.append((a, kw)), self._original_log_audit(*a, **kw))[1]

    def detach(self):
        if self._original_execute:
            service.db.execute = self._original_execute
        if self._original_send_email:
            service.mailer.send = self._original_send_email
        if self._original_log_audit:
            service.audit.log = self._original_log_audit
```

```python
# orders/domain/validator.py
from dataclasses import dataclass, field
from decimal import Decimal
from typing import List, Optional


@dataclass
class ValidationResult:
    valid: bool = True
    errors: List[str] = field(default_factory=list)
    warnings: List[str] = field(default_factory=list)


@dataclass
class OrderLine:
    item_id: str
    quantity: int
    unit_price: Decimal
    discount_percent: Decimal = Decimal("0")


class OrderValidator:
    """Pure domain validation — no I/O, no side effects."""

    MIN_QUANTITY = 1
    MAX_QUANTITY = 10000
    MAX_DISCOUNT_PCT = Decimal("100")

    def validate_order(self, order_data) -> ValidationResult:
        """Validate a raw order input object."""
        errors = []

        # Customer validation
        customer_id = getattr(order_data, 'customer_id', None)
        if not customer_id or not str(customer_id).strip():
            errors.append("customer_id cannot be empty")

        # Line items validation
        lines = getattr(order_data, 'lines', None)
        if not lines:
            errors.append("order must have at least one line item")
        else:
            sku_seen = {}
            for i, line in enumerate(lines):
                prefix = f"line[{i}]"
                errors.extend(self._validate_line(line, prefix))
                sku = getattr(line, 'item_id', '')
                if sku in sku_seen:
                    errors.append(
                        f"duplicate SKU '{sku}' found in {prefix} "
                        f"(also in line[{sku_seen[sku]}])"
                    )
                else:
                    sku_seen[sku] = i

        return ValidationResult(valid=len(errors) == 0, errors=errors)

    def _validate_line(self, line, prefix: str) -> List[str]:
        errors = []
        qty = getattr(line, 'quantity', None)
        if qty is None or qty < self.MIN_QUANTITY:
            errors.append(f"{prefix}.quantity must be >= {self.MIN_QUANTITY}")
        if qty and qty > self.MAX_QUANTITY:
            errors.append(f"{prefix}.quantity must be <= {self.MAX_QUANTITY}")

        price = getattr(line, 'unit_price', None)
        if price is None or price < 0:
            errors.append(f"{prefix}.unit_price must be non-negative")

        discount = getattr(line, 'discount_percent', Decimal("0"))
        if discount < 0 or discount > self.MAX_DISCOUNT_PCT:
            errors.append(
                f"{prefix}.discount_percent must be between 0 and 100"
            )
        return errors

    @staticmethod
    def validate_cancel_reason(reason: str) -> ValidationResult:
        errors = []
        if not reason or len(reason.strip()) < 3:
            errors.append("cancellation reason must be at least 3 characters")
        return ValidationResult(valid=len(errors) == 0, errors=errors)
```

```python
from dataclasses import dataclass, field
from decimal import Decimal
from typing import Dict, List, Optional


@dataclass
class PriceBreakdown:
    subtotal: Decimal = Decimal("0")
    line_total: Decimal = Decimal("0")
    total_discount: Decimal = Decimal("0")
    adjusted_subtotal: Decimal = Decimal("0")


class PricingEngine:
    """Price computation — pure math, no I/O."""

    def compute_line_total(self, line) -> Decimal:
        unit_price = getattr(line, 'unit_price', Decimal("0"))
        quantity = getattr(line, 'quantity', 0)
        discount_pct = getattr(line, 'discount_percent', Decimal("0"))
        gross = unit_price * quantity
        discount_amt = gross * discount_pct / Decimal("100")
        return gross - discount_amt

    def compute_order_totals(self, order_data) -> Dict:
        lines = getattr(order_data, 'lines', [])
        breakdown = PriceBreakdown()
        for line in lines:
            line_total = self.compute_line_total(line)
            gross = getattr(line, 'unit_price', Decimal("0")) * getattr(line, 'quantity', 0)
            breakdown.subtotal += gross
            breakdown.line_total += line_total
            breakdown.total_discount += (gross - line_total)
        breakdown.adjusted_subtotal = breakdown.line_total
        return {
            "subtotal": breakdown.subtotal,
            "line_total": breakdown.line_total,
            "total_discount": breakdown.total_discount,
            "adjusted_subtotal": breakdown.adjusted_subtotal,
        }

    def apply_promo(self, totals: Dict, promo_code: Optional[str]) -> Dict:
        if not promo_code:
            return totals
        # Delegate to external promo service (to be injected later)
        return totals
```

```python
class SqlBuilder:
    """Constructs parametrized queries from templates."""

    def build_insert(self, table: str, columns: list, values: list) -> str:
        placeholders = ", ".join(["%s"] * len(columns))
        cols_str = ", ".join(columns)
        return f"INSERT INTO {table} ({cols_str}) VALUES ({placeholders})"

    def build_update(self, table: str, set_cols: list, where_col: str,
                     where_val: str) -> str:
        set_str = ", ".join([f"{c} = %s" for c in set_cols])
        return f"UPDATE {table} SET {set_str} WHERE {where_col} = %s"

    def build_select(self, table: str, columns: list,
                     where_col: str) -> str:
        cols_str = ", ".join(columns)
        return f"SELECT {cols_str} FROM {table} WHERE {where_col} = %s"
```

```python
# orders/application/persister.py
from ..adapters.db_repo import DbRepo
from ..adapters.sql_builder import SqlBuilder


class OrderPersister:
    """Orchestrates transactional persistence operations."""

    def __init__(self, db_repo: DbRepo, sql_builder: SqlBuilder):
        self.db = db_repo
        self.sql = sql_builder

    def save_order_header(self, order_data, created_by: str) -> str:
        query = self.sql.build_insert(
            "orders",
            ["customer_id", "status", "subtotal", "created_at", "created_by"],
            [
                order_data.customer_id,
                getattr(order_data, 'status', 'pending'),
                getattr(order_data, 'totals', {}).get('adjusted_subtotal', 0),
                "NOW()",
                created_by,
            ],
        )
        result = self.db.execute(query, fetch_one=True)
        return result["id"] if isinstance(result, dict) else result[0]

    def save_line_items(self, order_id: str, lines: list):
        query = self.sql.build_insert(
            "order_lines",
            ["order_id", "item_id", "quantity", "unit_price", "discount_pct"],
            ["%s", "%s", "%s", "%s", "%s"],
        )
        params = [
            (order_id, l.item_id, l.quantity, l.unit_price, l.discount_percent)
            for l in lines
        ]
        for p in params:
            self.db.execute(query, p)

    def update_order(self, order_id: str, updates: dict, updated_by: str):
        set_cols = [f"{k} = %s" for k in updates.keys()]
        where_clause = "id = %s"
        query = f"UPDATE orders SET {', '.join(set_cols)} WHERE id = %s"
        params = list(updates.values()) + [order_id, updated_by]
        self.db.execute(query, tuple(params))

    def soft_delete_order(self, order_id: str, reason: str, cancelled_by: str):
        query = self.sql.build_update(
            "orders",
            ["active", "updated_at", "cancelled_by", "cancellation_reason"],
            "id",
            "ORDER_ID_PLACEHOLDER",
        )
        # Parameterize properly in real code
        query = ("UPDATE orders SET active=false, updated_at=NOW(), "
                 "cancelled_by=%s, cancellation_reason=%s WHERE id=%s")
        self.db.execute(query, (cancelled_by, reason, order_id))
```

**At this point**, the OrderService is delegating all I/O to these layers. The SQL is centralized instead of scattered.

### Step 5: Extract Notification Service — Email Last (Highest Risk)

**Why last?** Email involves templates, network calls, error recovery, retries. If broken, customers don't get orders. Best to have everything else stabilized first.

```python
# orders/adapters/mail_gateway.py
class MailGateway:
    """Transport abstraction for sending emails."""

    def __init__(self, smtp_host: str = None, ses_client=None):
        self.transport = ses_client or self._init_smtp(smtp_host)

    def send(self, to: str, subject: str, body: str,
             template: str = None, context: dict = None):
        rendered = self._render(template, context, body)
        self.transport.send(to, subject, rendered)

    def _render(self, template, context, fallback_body):
        if template and context:
            return open(f"templates/{template}.html").read().format(**context)
        return fallback_body


# orders/application/notificator.py
class OrderNotificator:
    """Triggers notifications after order state changes."""

    def __init__(self, mail_gateway: MailGateway):
        self.mail = mail_gateway

    def send_confirmation(self, order_data, customer_email: str):
        if getattr(order_data, 'status') == 'draft':
            return  # No email for drafts
        self.mail.send(
            to=customer_email,
            subject=f"Order Confirmation #{getattr(order_data, 'id', 'UNKNOWN')}",
            template="order_confirmation",
            context={
                "order_id": getattr(order_data, 'id', ''),
                "items": [
                    {"name": getattr(l, 'item_name', l.item_id),
                     "qty": l.quantity,
                     "price": l.unit_price}
                    for l in getattr(order_data, 'lines', [])
                ],
                "total": getattr(order_data, 'totals', {}).get('line_total', 0),
            },
        )

    def send_cancellation(self, order_data, customer_email: str, reason: str):
        self.mail.send(
            to=customer_email,
            subject=f"Order Cancelled #{getattr(order_data, 'id', 'UNKNOWN')}",
            template="order_cancelled",
            context={
                "order_id": getattr(order_data, 'id', ''),
                "reason": reason,
            },
        )
```

## Phase 4: Strangler Facade (Days 21-25)

Now weave everything together. The facade keeps the same public API but routes through extracted collaborators:

```python
# orders/facade/order_service.py
from ..domain.validator import OrderValidator
from ..domain.pricing import PricingEngine
from ..domain.tax import TaxCalculator
from ..application.persistor import OrderPersister
from ..application.notificator import OrderNotificator
from ..adapters.db_repo import DbRepo
from ..adapters.sql_builder import SqlBuilder
from ..adapters.mail_gateway import MailGateway
from ..adapters.audit_writer import AuditWriter
from ..adapters.config_store import ConfigStore
from ..domain.validator import ValidationResult
import json


class OrderService:
    """
    Public API — UNCHANGED signatures and behavior.

    Internally delegates to extracted domain/application/adapter layers.
    This is the strangler facade: callers see no difference.
    """

    def __init__(
        self,
        connection_pool=None,          # Will create DbRepo internally
        smtp_host=None,                # Will create MailGateway internally
        audit_table: str = "audit_log",
    ):
        # Infrastructure wiring (one place, easy to mock in tests)
        self.db_repo = DbRepo(connection_pool or self._make_pool())
        self.sql_builder = SqlBuilder()
        self.config_store = ConfigStore()
        self.mail_gateway = MailGateway(smtp_host=smtp_host)

        # Domain services
        self.validator = OrderValidator()
        self.pricing_engine = PricingEngine()
        self.tax_calculator = TaxCalculator()
        self.tax_calculator.config_store = self.config_store

        # Application services
        self.persistor = OrderPersister(self.db_repo, self.sql_builder)
        self.audit_writer = AuditWriter(self.db_repo, self.sql_builder)
        self.notificator = OrderNotificator(self.mail_gateway)

    # ── Internal helpers ────────────────────────────────────────

    def _make_pool(self):
        """Factory for default connection pool — overridden in tests."""
        import psycopg2
        return psycopg2.pool.SimpleConnectionPool(1, 10, dsn="default")

    def _run_in_transaction(self, fn):
        """Execute a function within a DB transaction boundary."""
        self.db_repo.begin_transaction()
        try:
            result = fn()
            self.db_repo.commit_transaction()
            return result
        except Exception:
            self.db_repo.rollback_transaction()
            raise

    # ── Public API (signatures unchanged) ───────────────────────

    def validate_order(self, order) -> ValidationResult:
        return self.validator.validate_order(order)

    def calculate_totals(self, order) -> dict:
        pricing = self.pricing_engine.compute_order_totals(order)
        billing_addr = getattr(order, 'billing_address', {})
        tax_detail = self.tax_calculator.calculate_tax(
            pricing['adjusted_subtotal'],
            billing_addr,
            getattr(order, 'lines', []),
        )
        pricing['tax'] = tax_detail
        pricing['grand_total'] = pricing['adjusted_subtotal'] + tax_detail['amount']
        return pricing

    def create_order(self, order) -> dict:
        # 1. Validate
        validation = self.validator.validate_order(order)
        if not validation.valid:
            raise ValueError("; ".join(validation.errors))

        # 2. Compute pricing (with transaction)
        def _do_create():
            totals = self.calculate_totals(order)
            order_id = self.persistor.save_order_header(order, order.created_by)
            self.persistor.save_line_items(order_id, order.lines)

            # 3. Persist complete
            result = {"id": order_id, "totals": totals}

            # 4. Notify
            self.notificator.send_confirmation(
                {**order, "id": order_id, "totals": totals},
                getattr(order, 'customer_email', ''),
            )

            # 5. Audit
            self.audit_writer.log(
                "order.created", "order", order_id, order.created_by,
                details=totals,
            )
            return result

        return self._run_in_transaction(_do_create)

    def update_order(self, order, status=None, updated_by=None) -> dict:
        validation = self.validator.validate_order(order)
        if not validation.valid:
            raise ValueError("; ".join(validation.errors))

        def _do_update():
            updates = {}
            if status:
                updates['status'] = status
            if updated_by:
                updates['updated_by'] = updated_by
            updates['updated_at'] = 'NOW()'
            self.persistor.update_order(getattr(order, 'id'), updates, updated_by or order.created_by)
            self.audit_writer.log("order.updated", "order", order.id, updated_by, details=updates)
            return {"id": order.id, "updated": updates}

        return self._run_in_transaction(_do_update)

    def cancel_order(self, order, reason: str, cancelled_by: str) -> dict:
        reason_validation = self.validator.validate_cancel_reason(reason)
        if not reason_validation.valid:
            raise ValueError("; ".join(reason_validation.errors))

        def _do_cancel():
            order_id = getattr(order, 'id')
            self.persistor.soft_delete_order(order_id, reason, cancelled_by)
            # Send notification
            if getattr(order, 'status') != 'draft':
                self.notificator.send_cancellation(
                    {**order, "id": order_id},
                    getattr(order, 'customer_email', ''),
                    reason,
                )
            self.audit_writer.log("order.cancelled", "order", order_id, cancelled_by,
                                  details={"reason": reason})
            return {"id": order_id, "cancelled": True}

        return self._run_in_transaction(_do_cancel)
```

## Phase 5: Verification Strategy at Every Commit

### 5.1 The Four-Layer Gate

Every refactoring commit MUST pass this pipeline:

```
┌──────────────────────────────────────────────────────────┐
│  COMMIT HOOK: Pre-Merge Gate                             │
├──────────────────────────────────────────────────────────┤
│                                                          │
│  LAYER 1: Golden Tests                                   │
│  pytest tests/characterization/ -x                       │
│  → Must produce IDENTICAL outputs to baseline            │
│                                                          │
│  LAYER 2: Unit Tests (newly written per extraction)      │
│  pytest tests/unit/ -x                                   │
│  → Each extracted module tested in isolation             │
│                                                          │
│  LAYER 3: Integration Tests (end-to-end flow)            │
│  pytest tests/integration/ -x                            │
│  → Validates new architecture produces correct results   │
│                                                          │
│  LAYER 4: Static Analysis                                │
│  mypy orders/ && ruff check orders/ && bandit orders/    │
│  → No type regressions, no lint violations, no secrets   │
│                                                          │
│  If ANY layer fails → merge blocked                      │
└──────────────────────────────────────────────────────────┘
```

### 5.2 Behavioral Diff Tool

Create `scripts/behavior_diff.py` to compare old vs new implementation outputs:

```python
"""
Compare outputs between legacy OrderService and new modular version.
Run post-refactor to prove byte-identical behavior.
"""
import json, subprocess, sys


def run_legacy_test(fixture_path: str) -> dict:
    """Runs golden test against current (legacy) code."""
    result = subprocess.run(
        ["pytest", fixture_path, "-q", "--tb=no", "-k", "golden"],
        capture_output=True, text=True
    )
    return {"status": result.returncode, "output": result.stdout}


def load_golden_baseline(fixture_path: str) -> dict:
    """Load committed golden output file."""
    with open(fixture_path) as f:
        return json.load(f)


def compare(name: str, expected: dict, actual: dict) -> bool:
    """Deep-compare two dicts, report differences."""
    if expected == actual:
        print(f"  ✓ {name}")
        return True
    print(f"  ✗ {name} — MISMATCH")
    # Report field-level diffs
    for key in set(list(expected.keys()) + list(actual.keys())):
        if expected.get(key) != actual.get(key):
            print(f"    [{key}]: expected={expected.get(key)!r} got={actual.get(key)!r}")
    return False


if __name__ == "__main__":
    baseline_file = sys.argv[1]
    expected = load_golden_baseline(baseline_file)
    all_passed = True
    for name, value in expected.items():
        actual = run_behavior(name)  # Implement per-scenario
        all_passed &= compare(name, value, actual)
    sys.exit(0 if all_passed else 1)
```

## Key Principles Applied

1. **Characterize before changing** — Golden tests are the safety net. Without them, we'd be guessing whether behavior changed.

2. **Extract lowest-coupling seams first** — Validator (no I/O) → Pricer (no I/O) → Tax (config dependency) → Persistence (DB) → Notifications (network). Each step builds on stabilized previous steps.

3. **Shippable increments** — Every extraction is its own PR with passing golden tests. No multi-week feature branches sitting unmerged.

4. **Strangler pattern via facade** — The public `OrderService` stays byte-compatible. Callers never know the internals changed. Gradual decommission of old inline code proceeds alongside.

5. **Behavior preservation proof** — Four-layer gate (golden + unit + integration + static analysis) at every commit. The golden suite is the behavioral contract — it either passes or the commit doesn't land.

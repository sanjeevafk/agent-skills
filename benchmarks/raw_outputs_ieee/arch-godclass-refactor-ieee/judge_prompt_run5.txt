class SqlCaptureConnection:
    """Wraps the existing DB connection; forwards everything, logs .execute() calls."""
    def __init__(self, delegate):
        self._delegate = delegate
        self.queries = []

    def execute(self, sql, params=None):
        self.queries.append({"sql": sql, "params": params})
        return self._delegate.execute(sql, params)

    # proxy every other method via __getattr__
    def __getattr__(self, name):
        return getattr(self._delegate, name)

### Step 0b — Snapshot database mutations (golden DB snapshots)

```python
import json
import hashlib
from contextlib import contextmanager

@contextmanager
def snapshot_db(session_factory, operations):
    """Run `operations` inside a transaction and return a diff-safe snapshot."""
    conn = session_factory()
    tx = conn.begin()
    try:
        # Capture pre-state for tables touched by the operation
        yield conn
        conn.commit()

        # After commit, serialize the affected tables
        rows = {}
        for table_name in _tables_touched(operations):
            query = f"SELECT * FROM {table_name} ORDER BY id"
            rows[table_name] = [dict(r._mapping) for r in conn.execute(query)]

        # Deterministic hash + JSON blob
        payload = json.dumps(rows, sort_keys=True, default=str).encode()
        snapshot = {"hash": hashlib.sha256(payload).hexdigest(), "payload": json.loads(json.dumps(rows, default=str))}
    finally:
        tx.rollback()
    return snapshot
```

Each public method under test produces a `{method_name -> snapshot}` entry written to `tests/golden/order_snapshots_v1.json`. If behavior changes later, the diff triggers a CI failure.

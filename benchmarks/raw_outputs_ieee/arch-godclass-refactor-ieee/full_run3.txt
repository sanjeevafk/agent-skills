@dataclass(frozen=True)
class DbRow:
    """Deterministic representation of a single table row."""
    table_name: str
    primary_key_value: str
    normalized_columns: dict  # values converted to strings, sorted keys


class DatabaseSnapshot:
    """Captures a point-in-time view of order-related tables."""

    ORDER_TABLES = {
        'orders', 'order_items', 'order_tax_lines',
        'order_payments', 'order_audit_log'
    }

    def __init__(self, connection):
        self._conn = connection

    def take(self) -> dict:
        """Return row counts + deterministic checksums per table."""
        snap = {}
        cur = self._conn.cursor()
        for table in self.ORDER_TABLES:
            try:
                # Get primary key column name (assumes standard naming)
                pk_map = {'orders': 'id', 'order_items': 'id',
                          'order_tax_lines': 'id', 'order_payments': 'id',
                          'order_audit_log': 'id'}
                pk_col = pk_map.get(table, 'id')

                # Row count
                cur.execute(f'SELECT COUNT(*) FROM {table}')
                snap[table] = {'row_count': cur.fetchone()[0], 'checksums': []}

                # Checksum of all data (excluding volatile fields)
                cur.execute(f'SELECT {pk_col} FROM {table} ORDER BY {pk_col}')
                pks = [str(r[0]) for r in cur.fetchall()]

                for pk in pks:
                    cur.execute(f'SELECT * FROM {table} WHERE {pk_col} = %s ORDER BY column_name', (pk,))
                    rows = cur.fetchall()
                    if rows:
                        col_names = [d[0] for d in cur.description]
                        # Normalize: exclude timestamps, convert bytes, sort
                        normalized = {}
                        for row in rows:
                            for i, val in enumerate(row):
                                col = col_names[i]
                                if val is None:
                                    normalized[col] = ''
                                elif isinstance(val, (datetime,)):
                                    normalized[col] = val.isoformat()  # deterministic
                                elif isinstance(val, (int, float, bool, str)):
                                    normalized[col] = str(val)
                                else:
                                    normalized[col] = repr(val)

                        ck = hashlib.sha256(
                            json.dumps(normalized, sort_keys=True).encode()
                        ).hexdigest()
                        snap[table]['checksums'].append({
                            'pk': str(pk),
                            'checksum': ck
                        })
            except Exception:
                continue  # Table may not exist in test schema
        return snap

    def describe_diff(self, before: dict, after: dict) -> list[str]:
        """Produce human-readable diffs between two snapshots."""
        diffs = []
        for table in self.ORDER_TABLES:
            b = before.get(table, {})
            a = after.get(table, {})

            bc = b.get('row_count', 0)
            ac = a.get('row_count', 0)
            if bc != ac:
                diffs.append(f'{table}: row_count changed {bc}→{ac}')

            b_ck = {x['pk']: x['checksum'] for x in b.get('checksums', [])}
            a_ck = {x['pk']: x['checksum'] for x in a.get('checksums', [])}

            for pk in set(b_ck) | set(a_ck):
                if b_ck.get(pk) != a_ck.get(pk):
                    diffs.append(f'{table}[pk={pk}]: content changed')
        return diffs

class FakeSmtpTransport:
    """Wraps the real SMTP client with a capture buffer.

    we also record every call for assertion. For truly offline tests,
    override send_email to use this transport instead of the real one.
    """

    def __init__(self, real_transport=None):
        self._real = real_transport
        self._emails: list[CapturedEmail] = []

    def send(self, sender: str, recipient: str,
             subject: str, body: str) -> str:
        msg = CapturedEmail(sender, recipient, subject, body,
                           datetime.now(timezone.utc).isoformat())
        self._emails.append(msg)
        if self._real:
            self._real.send(sender, recipient, subject, body)
        return msg.timestamp  # return whatever the real transport returns

    def get_sent_emails(self) -> list[CapturedEmail]:
        return list(self._emails)

    def reset(self):
        self._emails.clear()

def _normalize(value) -> str:
    """Convert any value to a deterministic string representation."""
    if isinstance(value, (int, float, bool, str, type(None))):
        return str(value)
    if isinstance(value, datetime):
        return value.replace(tzinfo=timezone.utc).isoformat()
    if isinstance(value, (list, tuple)):
        return '[' + ','.join(_normalize(v) for v in value) + ']'
    if isinstance(value, dict):
        return '{' + ','.join(
            f'{_normalize(k)}:{_normalize(v)}' for k, v in sorted(value.items())
        ) + '}'
    return repr(value)

                 if not any(skip in k.lower() for skip in
                           ['created_at', 'updated_at', 'id', 'guid', 'uuid'])}
        return json.dumps(clean, sort_keys=True, default=_normalize)
    return _normalize(result)

### Part 2: Seam Extraction Order

#### 2.1 Coupling Analysis Matrix

| Collaborator | Methods Calling It | Cyclomatic Complexity | Extraction Difficulty | Risk of Breakage |
|---|---|---|---|---|
| **Database/Persistence** | ~25+ (all CRUD methods) | Very High | Medium (lots of SQL) | **Low** — interface is clear-cut |
| **Pricing Logic** | CreateOrder, Update, Recalculate (~10) | Very High | High (complex nesting) | Medium |
| **Email** | CreateOrder (confirm), CancelOrder (refund), StatusChange (~5) | Low | Trivial | Low |
| **Audit Logging** | All mutating methods (~20) | Low | Trivial | Low |
| **Tax Rate Resolver** | Inside calculate_total, create_order (~3) | Low | Trivial | Low |
| **Cache** | Reads on lookup (~5) | Low | Easy | Low |

#### 2.2 Recommended Extraction Order

```
Phase 0: Characterization tests (current step)
Phase 1: Repository interface       ← FIRST — deepest coupling, clearest boundary
Phase 2: Pricing engine              ← SECOND — highest complexity, large blast radius
Phase 3: Tax resolver                ← THIRD — small, but pricing needs it decoupled
Phase 4: Validator                   ← FOURTH — domain rules, moderate size
Phase 5: Email & Audit (parallel)    ← FIFTH — trivial, can be done simultaneously
Phase 6: Orchestration cleanup       ← LAST — remove delegation, finalize
```

#### 2.3 Why Repository First?

Three reasons:

1. **Maximum coupling reduction**: Raw SQL touches nearly every method. Isolating it eliminates ~40% of the class's lines immediately and gives you a clean dependency inversion.

2. **Enables unit testing**: Once the repository is an interface, you can mock it and test pricing/tax/validation logic in isolation. Without this seam, every test hits the DB.

3. **Blast-radius containment**: If the repository extraction breaks, only data-access methods are affected. The pricing, validation, and notification logic remains untouched, making rollbacks surgical.

Counter-intuitively, extracting the most-coupled concern first is safer than the easiest — because the resulting interface lets you test everything else in isolation, reducing uncertainty for subsequent extractions.

class OrderValidator:
    def validate_create_request(self, request: CreateOrderRequest) -> None:
        """All validation from OrderService.CreateOrder combined."""
        if not request.user_id or len(request.user_id) > 64:
            raise ValidationError('user_id', 'Invalid length')
        
        if not request.items or len(request.items) == 0:
            raise ValidationError('items', 'Must have at least one item')

        for item in request.items:
            if item.qty <= 0:
                raise ValidationError(f'items[{item.sku}]', 'Quantity must be positive')
            if item.qty > 999:
                raise ValidationError(f'items[{item.sku}]', 'Exceeds maximum')

        # Shipping address validation
        addr = request.shipping_address
        if not addr or not all([addr.line1, addr.city, addr.state, addr.zip]):
            raise ValidationError('shipping_address', 'Incomplete address')

        # Promo code format
        if request.promo_code and not re.match(r'^[A-Z0-9]{4,10}$', request.promo_code):
            raise ValidationError('promo_code', 'Invalid format')

    def validate_cancel_request(self, order: OrderEntity, reason: str) -> None:
        if order.status in ('shipped', 'delivered'):
            raise ValidationError('order.status', 'Cannot cancel shipped order')
        if not reason or len(reason.strip()) < 3:
            raise ValidationError('reason', 'Reason required (3+ chars)')

### Part 5: Proving Behavior Preservation at Every Commit

At each commit checkpoint, run this **preservation checklist**:

```python
"""
commit_check.py — Automated preservation verification.
Run at every commit. Fails the pipeline if any check fails.
"""

import subprocess
import json
from pathlib import Path


def run_characterization_tests() -> bool:
    """Execute the full golden-master test suite."""
    result = subprocess.run(
        ['pytest', 'tests/test_characterization/', '-v', '--tb=short'],
        capture_output=True, text=True
    )
    if result.returncode != 0:
        print(f"CHARACTERIZATION FAIL:\n{result.stdout}\n{result.stderr}")
        return False
    print("✓ All golden-master tests passed")
    return True


def check_sql_operation_counts() -> bool:
    """Instrumentation-based: does the refactored code execute
    the same number of SQL operations per method call as before?"""
    # This requires an instrumentation hook added in Step 1.0:
    # OrderService now tracks ops_per_call: dict[str, int]
    from src.orders.service import OrderService
    svc = build_service()  # fresh instance

    # Benchmark CreateOrder with known input
    sql_counts_before_svc = svc.ops_per_call  # captured by instrumentation
    svc.CreateOrder(make_request('SKU-A', qty=3))

    sql_counts_after = dict(svc.ops_per_call)

    # Compare against known baseline
    baseline = load_baseline()
    if sql_counts_after != baseline['CreateOrder']:
        print(f"SQL operation count changed for CreateOrder:")
        print(f"  Baseline: {baseline['CreateOrder']}")
        print(f"  Actual:   {sql_counts_after}")
        return False
    print("✓ SQL operation counts match baseline")
    return True


def check_side_effect_counts() -> bool:
    """Do email and audit operations occur in the same quantities?"""
    svc = build_service()

    result = svc.CreateOrder(make_request('SKU-A', qty=3))
    email_count = svc._email_capture.count_sent()
    audit_count = svc._audit_capture.count_new_entries()

    baseline = load_baseline()
    if email_count != baseline['CreateOrder']['emails_sent']:
        print(f"Email count mismatch: got {email_count}, expected {baseline['CreateOrder']['emails_sent']}")
        return False
    if audit_count != baseline['CreateOrder']['audit_entries']:
        print(f"Audit entry count mismatch: got {audit_count}, expected {baseline['CreateOrder']['audit_entries']}")
        return False
    print("✓ Side effect counts match baseline")
    return True


def check_public_api_signature() -> bool:
    """Ensure no public method signatures changed."""
    import inspect
    from src.orders.service import OrderService

    expected = load_expected_signatures()
    actual = {}
    for name, method in inspect.getmembers(OrderService, predicate=inspect.isfunction):
        if name.startswith('_'):
            continue  # Skip private
        sig = inspect.signature(method)
        actual[name] = str(sig)

    mismatches = []
    for name, expected_sig in expected.items():
        if actual.get(name) != expected_sig:
            mismatches.append(f"  {name}: expected={expected_sig}, actual={actual.get(name)}")

    if mismatches:
        print(f"PUBLIC API SIGNATURE CHANGES:\n" + "\n".join(mismatches))
        return False
    print("✓ All public API signatures preserved")
    return True


def check_code_complexity_reduction() -> bool:
    """Optional: confirm cyclomatic complexity is trending down."""
    from radon.complexity import cc_visit
    from src.orders.service import OrderService

    source = inspect.getsource(OrderService)
    complexities = [cc.complexity for cc in cc_visit(source)]
    avg_cc = sum(complexities) / len(complexities) if complexities else 0

    print(f"  Average cyclomatic complexity: {avg_cc:.1f}")
    print(f"  Max cyclomatic complexity: {max(complexities)}")
    print(f"  Methods: {len(complexities)}")
    # Hard threshold: average CC should be below 5
    if avg_cc > 5:
        print("  ⚠ Warning: Average CC still elevated")
    print("✓ Complexity tracked")
    return True


def main():
    checks = [
        run_characterization_tests,
        check_sql_operation_counts,
        check_side_effect_counts,
        check_public_api_signature,
        check_code_complexity_reduction,
    ]

    all_passed = True
    for check in checks:
        if not check():
            all_passed = False

    if all_passed:
        print("\n✅ PRESERVATION CHECKLIST PASSED — safe to commit")
    else:
        print("\n❌ PRESERVATION CHECKLIST FAILED — DO NOT COMMIT")
        exit(1)


if __name__ == '__main__':
    main()
```

**Pipeline gate:** All checks must pass. If any fail, the commit is aborted and the team investigates the delta.
